   157→   157→// ── Helpers ──
   158→   158→function formatDate(d: string) {
   159→   159→  if (!d) return "—";
   160→   160→  try { return new Date(d + "T00:00:00").toLocaleDateString("en-US", { month: "short", day: "numeric" }); } catch { return d; }
   161→   161→}
   162→   162→
   163→   163→function formatCurrency(v: number) {
   164→   164→  return new Intl.NumberFormat("en-US", { style: "currency", currency: "ETB", maximumFractionDigits: 0 }).format(v);
   165→   165→}
   166→   166→
   167→   167→function todayStr() { return new Date().toISOString().split("T")[0]; }
   168→   168→
   169→   169→function addDays(d: string, n: number) {
   170→   170→  const dt = new Date(d + "T00:00:00");
   171→   171→  dt.setDate(dt.getDate() + n);
   172→   172→  return dt.toISOString().split("T")[0];
   173→   173→}
   174→   174→
   175→   175→function getFloorFromNumber(num: string): number | null {
   176→   176→  const match = num.match(/^\d/);
   177→   177→  return match ? parseInt(match[0], 10) : null;
   178→   178→}
   179→   179→
   180→   180→function parseAmenities(amenitiesStr: string | null | undefined): string[] {
   181→   181→  if (!amenitiesStr) return [];
   182→   182→  try {
   183→   183→    const parsed = JSON.parse(amenitiesStr);
   184→   184→    return Array.isArray(parsed) ? parsed : [];
   185→   185→  } catch {
   186→   186→    return amenitiesStr.split(",").map((s) => s.trim()).filter(Boolean);
   187→   187→  }
   188→   188→}
