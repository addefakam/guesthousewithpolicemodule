   570→/** Execute SQL via raw pg Client */
   571→async function execViaPg(sql: string, params?: unknown[]): Promise<void> {
   572→  const client = new Client({ connectionString: process.env.DATABASE_URL! });
   573→  try {
   574→    await client.connect();
   575→    await client.query(sql, params);
   576→  } finally {
   577→    await client.end().catch(() => {});
   578→  }
   579→}

   581→/** Execute DDL via Prisma $executeRawUnsafe — splits multi-statement SQL
   582→ *  because Prisma's driver does NOT support multiple statements per call. */
   583→async function execViaPrisma(sql: string): Promise<void> {
   584→  // Split on `DO $$` blocks and standalone statements
   585→  // Each `DO $$ ... END $$;` is one executable unit
   586→  const statements = sql
   587→    .split(/(?=DO \$\$)/)
   588→    .map(s => s.trim())
   589→    .filter(s => s.length > 0);
   590→
   591→  // If no DO blocks found, split by semicolons (for CREATE INDEX etc.)
   592→  const parts = statements.length > 1
   593→    ? statements
   594→    : sql.split(';').map(s => s.trim()).filter(s => s.length > 0);
   595→
   596→  const prisma = new PrismaClient({
   597→    log: ["warn", "error"],
   598→  });
   599→  try {
   600→    for (const part of parts) {
   601→      const stmt = part.endsWith(';') ? part : part + ';';
   602→      try {
   603→        await prisma.$executeRawUnsafe(stmt);
   604→      } catch (err) {
   605→        // Individual statement might fail (e.g. duplicate_object) — log but continue
   606→        console.error("[execViaPrisma] Statement failed (continuing):", stmt.slice(0, 80), err instanceof Error ? err.message : String(err));
   607→      }
   608→    }
   609→  } finally {
   610→    await prisma.$disconnect().catch(() => {});
   611→  }
   612→}

   827→        await prisma2.$executeRawUnsafe(
   828→          `INSERT INTO "SubscriptionPlan" ("id","name","cycle","price","isActive","createdAt","updatedAt")
   829→           SELECT 'plan-semi-annual-001','Semi-Annual','SEMI_ANNUAL',2600,true,NOW(),NOW()
   830→           WHERE NOT EXISTS (SELECT 1 FROM "SubscriptionPlan" WHERE "id"='plan-semi-annual-001')`
   831→        );
   832→        await prisma2.$executeRawUnsafe(
   833→          `INSERT INTO "SubscriptionPlan" ("id","name","cycle","price","isActive","createdAt","updatedAt")
   834→           SELECT 'plan-yearly-001','Annual','YEARLY',4800,true,NOW(),NOW()
   835→           WHERE NOT EXISTS (SELECT 1 FROM "SubscriptionPlan" WHERE "id"='plan-yearly-001')`
   836→        );
   837→        console.log("[init-db] SUPERUSER and plans seeded via Prisma.");
   838→      } finally {
   839→        await prisma2.$disconnect().catch(() => {});
   840→      }
   841→
   842→      console.log("[init-db] Database initialization complete via Prisma fallback.");
   843→      _initDone = true;
   844→    } catch (prismaErr) {

   846→      _initDone = false;
   847→      _initPromise = null; // Allow retry
   848→      throw new Error(
   849→        `Database initialization failed. pg error: ${pgError instanceof Error ? pgError.message : String(pgError)} | Prisma error: ${prismaErr instanceof Error ? prismaErr.message : String(prismaErr)}`
   850→      );
   851→    }
   852→  })();
   853→
   854→  return _initPromise;
   855→}
