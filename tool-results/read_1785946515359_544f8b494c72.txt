   560→/** Execute SQL via raw pg Client */
   561→async function execViaPg(sql: string, params?: unknown[]): Promise<void> {
   562→  const client = new Client({ connectionString: process.env.DATABASE_URL! });
   563→  try {
   564→    await client.connect();
   565→    await client.query(sql, params);
   566→  } finally {
   567→    await client.end().catch(() => {});
   568→  }
   569→}

   571→/** Execute DDL via Prisma $executeRawUnsafe — splits multi-statement SQL
   572→ *  because Prisma's driver does NOT support multiple statements per call. */
   573→async function execViaPrisma(sql: string): Promise<void> {
   574→  // Split on `DO $$` blocks and standalone statements
   575→  // Each `DO $$ ... END $$;` is one executable unit
   576→  const statements = sql
   577→    .split(/(?=DO \$\$)/)
   578→    .map(s => s.trim())
   579→    .filter(s => s.length > 0);
   580→
   581→  // If no DO blocks found, split by semicolons (for CREATE INDEX etc.)
   582→  const parts = statements.length > 1
   583→    ? statements
   584→    : sql.split(';').map(s => s.trim()).filter(s => s.length > 0);
   585→
   586→  const prisma = new PrismaClient({
   587→    log: ["warn", "error"],
   588→  });
   589→  try {
   590→    for (const part of parts) {
   591→      const stmt = part.endsWith(';') ? part : part + ';';
   592→      try {
   593→        await prisma.$executeRawUnsafe(stmt);
   594→      } catch (err) {
   595→        // Individual statement might fail (e.g. duplicate_object) — log but continue
   596→        console.error("[execViaPrisma] Statement failed (continuing):", stmt.slice(0, 80), err instanceof Error ? err.message : String(err));
   597→      }
   598→    }
   599→  } finally {
   600→    await prisma.$disconnect().catch(() => {});
   601→  }
   602→}

   623→export async function ensureDatabase(): Promise<void> {
   624→  if (_initDone) {
   625→    // Always run migrations synchronously to ensure columns exist
   626→    // before any query executes. Migrations are idempotent.
   627→    if (!_migrationsRan) {
   628→      _migrationsRan = true;
   629→      await runMigrationsOnly();
   630→    }
   631→    return;
   632→  }
   633→  if (_initPromise) return _initPromise;
   634→
   635→  if (!process.env.DATABASE_URL) {
   636→    throw new Error("DATABASE_URL environment variable is not set.");
   637→  }
   638→
   639→  _initPromise = (async () => {
   640→    console.log("[init-db] Starting database initialization...");
   641→
   642→    let usePg = true;
   643→    let pgError: unknown = null;
   644→
   645→    // Strategy 1: Try using pg Client directly
   646→    try {
   647→      const client = new Client({ connectionString: process.env.DATABASE_URL });
   648→      await client.connect();
   649→      console.log("[init-db] pg Client connected successfully.");
   650→
   651→      // Check if User table already exists
   652→      const res = await client.query(
   653→        `SELECT EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema='public' AND table_name='User') as ok`
   654→      );
   655→      const tableExists = res.rows[0]?.ok === true;
   656→      console.log("[init-db] User table exists:", tableExists);
   657→
   658→      if (tableExists) {
   659→        console.log("[init-db] Tables already exist, running migrations to add any missing columns...");
   660→        try {
   661→          await client.query(MIGRATIONS_SQL);
   662→          console.log("[init-db] Migrations applied to existing tables.");
   663→          await client.query(INDEXES_SQL);
   664→          console.log("[init-db] Indexes verified.");
   665→        } catch (migrateErr) {
   666→          console.error("[init-db] Migration on existing tables failed:", migrateErr instanceof Error ? migrateErr.message : String(migrateErr));
   667→          // Don't throw — the tables exist, migrations are best-effort
   668→        }
   669→        await client.end().catch(() => {});
   670→        _initDone = true;
   671→        return;
   672→      }
   673→
   674→      // Create enums
   675→      console.log("[init-db] Creating enum types...");
   676→      await client.query(ENUMS_SQL);
   677→
   678→      // Create tables
   679→      console.log("[init-db] Creating tables...");
   680→      await client.query(TABLES_SQL);
   681→
   682→      // Verify User table was created
   683→      const verify = await client.query(
   684→        `SELECT EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema='public' AND table_name='User') as ok`
   685→      );
   686→      if (!verify.rows[0]?.ok) {
   687→        throw new Error("DDL executed but User table still not found!");
   688→      }
   689→      console.log("[init-db] Tables verified.");
   690→
   691→      // Create foreign keys
   692→      console.log("[init-db] Creating foreign keys...");
   693→      await client.query(FKS_SQL);
   694→
   695→      // Run migrations (add new columns to existing tables)
   696→      console.log("[init-db] Running migrations...");
   697→      await client.query(MIGRATIONS_SQL);
   698→
   699→      // Create indexes
   700→      console.log("[init-db] Creating indexes...");
   701→      await client.query(INDEXES_SQL);
   702→
   703→      // Seed SUPERUSER
   704→      console.log("[init-db] Seeding SUPERUSER...");
   705→      const hashed = await bcrypt.hash("Admin@2024", 12);
   706→      await client.query(
   707→        `INSERT INTO "User" ("id","username","password","name","role","permissions","policeRank","createdAt","updatedAt")
   708→         SELECT 'su-admin-001','admin',$1,'System Administrator','SUPERUSER','[]','',NOW(),NOW()
   709→         WHERE NOT EXISTS (SELECT 1 FROM "User" WHERE "username"='admin')`,
   710→        [hashed]
   711→      );
   712→
   713→      // Seed PoliceAlertConfig
   714→      await client.query(
   715→        `INSERT INTO "PoliceAlertConfig" ("id","createdAt","updatedAt")
   716→         SELECT 'default-alert-config',NOW(),NOW()
   717→         WHERE NOT EXISTS (SELECT 1 FROM "PoliceAlertConfig" WHERE "id"='default-alert-config')`
   718→      );
   719→
   720→      // Seed default SubscriptionPlans
   721→      await client.query(`
   722→        INSERT INTO "SubscriptionPlan" ("id","name","cycle","price","isActive","createdAt","updatedAt")
   723→        SELECT 'plan-monthly-001','Monthly','MONTHLY',500,true,NOW(),NOW()
   724→        WHERE NOT EXISTS (SELECT 1 FROM "SubscriptionPlan" WHERE "id"='plan-monthly-001');
   725→        INSERT INTO "SubscriptionPlan" ("id","name","cycle","price","isActive","createdAt","updatedAt")
   726→        SELECT 'plan-quarterly-001','Quarterly','QUARTERLY',1400,true,NOW(),NOW()
   727→        WHERE NOT EXISTS (SELECT 1 FROM "SubscriptionPlan" WHERE "id"='plan-quarterly-001');
   728→        INSERT INTO "SubscriptionPlan" ("id","name","cycle","price","isActive","createdAt","updatedAt")
   729→        SELECT 'plan-semi-annual-001','Semi-Annual','SEMI_ANNUAL',2600,true,NOW(),NOW()
   730→        WHERE NOT EXISTS (SELECT 1 FROM "SubscriptionPlan" WHERE "id"='plan-semi-annual-001');
   731→        INSERT INTO "SubscriptionPlan" ("id","name","cycle","price","isActive","createdAt","updatedAt")
   732→        SELECT 'plan-yearly-001','Annual','YEARLY',4800,true,NOW(),NOW()
   733→        WHERE NOT EXISTS (SELECT 1 FROM "SubscriptionPlan" WHERE "id"='plan-yearly-001');
   734→      `);
   735→
   736→      console.log("[init-db] Database initialization complete via pg.");
   737→      await client.end().catch(() => {});
   738→      _initDone = true;
   739→      return;
   740→    } catch (err) {
   741→      pgError = err;
   742→      console.error("[init-db] pg approach failed:", err instanceof Error ? err.message : String(err));
   743→      usePg = false;
   744→    }
   745→
   746→    // Strategy 2: Fallback — use Prisma $executeRawUnsafe
   747→    console.log("[init-db] Falling back to Prisma $executeRawUnsafe...");
   748→    try {
   749→      // First check if tables exist using Prisma raw query
   750→      const prisma = new PrismaClient({ log: ["warn", "error"] });
   751→      try {
   752→        const res: Array<{ ok: boolean }> = await prisma.$queryRawUnsafe(
   753→          `SELECT EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema='public' AND table_name='User') as ok`
   754→        );
   755→        if (res[0]?.ok) {
   756→          console.log("[init-db] Tables already exist (verified via Prisma), running migrations...");
   757→          try {
   758→            await execViaPrisma(MIGRATIONS_SQL);
   759→            console.log("[init-db] Migrations applied via Prisma.");
   760→            await execViaPrisma(INDEXES_SQL);
   761→            console.log("[init-db] Indexes verified via Prisma.");
   762→          } catch (migrateErr) {
   763→            console.error("[init-db] Prisma migration on existing tables failed:", migrateErr instanceof Error ? migrateErr.message : String(migrateErr));
   764→          }
   765→          _initDone = true;
   766→          return;
   767→        }
   768→      } catch {
   769→        // Table might not exist, which is expected
   770→        console.log("[init-db] Tables do not exist yet, creating via Prisma...");
   771→      } finally {
   772→        await prisma.$disconnect().catch(() => {});
   773→      }
   774→
   775→      // Execute DDL via Prisma (split into parts to avoid query size issues)
   776→      await execViaPrisma(ENUMS_SQL);
   777→      console.log("[init-db] Enums created via Prisma.");
   778→
   779→      await execViaPrisma(TABLES_SQL);
   780→      console.log("[init-db] Tables created via Prisma.");
   781→
   782→      await execViaPrisma(FKS_SQL);
   783→      console.log("[init-db] Foreign keys created via Prisma.");
   784→
   785→      await execViaPrisma(MIGRATIONS_SQL);
   786→      console.log("[init-db] Migrations ran via Prisma.");
   787→
   788→      await execViaPrisma(INDEXES_SQL);
   789→      console.log("[init-db] Indexes created via Prisma.");
   790→
   791→      // Seed SUPERUSER via Prisma
   792→      const prisma2 = new PrismaClient({ log: ["warn", "error"] });
   793→      try {
   794→        const hashed = await bcrypt.hash("Admin@2024", 12);
   795→        await prisma2.$executeRawUnsafe(
   796→          `INSERT INTO "User" ("id","username","password","name","role","permissions","policeRank","createdAt","updatedAt")
   797→           SELECT 'su-admin-001','admin',$1,'System Administrator','SUPERUSER','[]','',NOW(),NOW()

   801→        await prisma2.$executeRawUnsafe(
   802→          `INSERT INTO "PoliceAlertConfig" ("id","createdAt","updatedAt")
   803→           SELECT 'default-alert-config',NOW(),NOW()
   804→           WHERE NOT EXISTS (SELECT 1 FROM "PoliceAlertConfig" WHERE "id"='default-alert-config')`
   805→        );
   806→        // Seed default SubscriptionPlans
   807→        await prisma2.$executeRawUnsafe(
   808→          `INSERT INTO "SubscriptionPlan" ("id","name","cycle","price","isActive","createdAt","updatedAt")
   809→           SELECT 'plan-monthly-001','Monthly','MONTHLY',500,true,NOW(),NOW()
   810→           WHERE NOT EXISTS (SELECT 1 FROM "SubscriptionPlan" WHERE "id"='plan-monthly-001')`
   811→        );
   812→        await prisma2.$executeRawUnsafe(
   813→          `INSERT INTO "SubscriptionPlan" ("id","name","cycle","price","isActive","createdAt","updatedAt")
   814→           SELECT 'plan-quarterly-001','Quarterly','QUARTERLY',1400,true,NOW(),NOW()
   815→           WHERE NOT EXISTS (SELECT 1 FROM "SubscriptionPlan" WHERE "id"='plan-quarterly-001')`
   816→        );
   817→        await prisma2.$executeRawUnsafe(
   818→          `INSERT INTO "SubscriptionPlan" ("id","name","cycle","price","isActive","createdAt","updatedAt")
   819→           SELECT 'plan-semi-annual-001','Semi-Annual','SEMI_ANNUAL',2600,true,NOW(),NOW()
   820→           WHERE NOT EXISTS (SELECT 1 FROM "SubscriptionPlan" WHERE "id"='plan-semi-annual-001')`
   821→        );
   822→        await prisma2.$executeRawUnsafe(
   823→          `INSERT INTO "SubscriptionPlan" ("id","name","cycle","price","isActive","createdAt","updatedAt")
   824→           SELECT 'plan-yearly-001','Annual','YEARLY',4800,true,NOW(),NOW()
   825→           WHERE NOT EXISTS (SELECT 1 FROM "SubscriptionPlan" WHERE "id"='plan-yearly-001')`

   827→        console.log("[init-db] SUPERUSER and plans seeded via Prisma.");
   828→      } finally {
   829→        await prisma2.$disconnect().catch(() => {});
   830→      }
   831→
   832→      console.log("[init-db] Database initialization complete via Prisma fallback.");
   833→      _initDone = true;
   834→    } catch (prismaErr) {
   835→      console.error("[init-db] Prisma fallback also failed:", prismaErr instanceof Error ? prismaErr.message : String(prismaErr));
   836→      _initDone = false;
   837→      _initPromise = null; // Allow retry
   838→      throw new Error(
   839→        `Database initialization failed. pg error: ${pgError instanceof Error ? pgError.message : String(pgError)} | Prisma error: ${prismaErr instanceof Error ? prismaErr.message : String(prismaErr)}`
   840→      );
   841→    }
   842→  })();

   847→/**
   848→ * Force-run just the migrations + indexes on the live database.
   849→ * Useful after deploying schema changes to add new columns without full re-init.
   850→ * All migrations use EXCEPTION WHEN duplicate_column so they are safe to re-run.
   851→ */
   852→export async function forceMigrate(): Promise<void> {
   853→  console.log("[forceMigrate] Running migrations on existing database...");
   854→  try {
   855→    await execViaPrisma(MIGRATIONS_SQL);
   856→    console.log("[forceMigrate] Migrations applied.");
   857→  } catch (err) {
   858→    console.error("[forceMigrate] Migration failed:", err instanceof Error ? err.message : String(err));
   859→    throw err;
   860→  }
   861→  try {
   862→    await execViaPrisma(INDEXES_SQL);
   863→    console.log("[forceMigrate] Indexes verified.");
   864→  } catch (err) {
   865→    console.error("[forceMigrate] Index creation failed:", err instanceof Error ? err.message : String(err));
   866→    // Don't throw — indexes are best-effort
   867→  }
   868→  // Reset init flag so next ensureDatabase() will do a full check
   869→  _initDone = false;
   870→  _initPromise = null;
   871→}
