   522→/** Execute SQL via raw pg Client */
   523→async function execViaPg(sql: string, params?: unknown[]): Promise<void> {
   524→  const client = new Client({ connectionString: process.env.DATABASE_URL! });
   525→  try {
   526→    await client.connect();
   527→    await client.query(sql, params);
   528→  } finally {
   529→    await client.end().catch(() => {});
   530→  }
   531→}

   533→/** Execute DDL via Prisma $executeRawUnsafe (bypasses Prisma's table-existence check) */
   534→async function execViaPrisma(sql: string): Promise<void> {
   535→  const prisma = new PrismaClient({
   536→    log: ["warn", "error"],
   537→  });
   538→  try {
   539→    // $executeRawUnsafe sends SQL directly without schema validation
   540→    await prisma.$executeRawUnsafe(sql);
   541→  } finally {
   542→    await prisma.$disconnect().catch(() => {});
   543→  }
   544→}

   565→export async function ensureDatabase(): Promise<void> {
   566→  if (_initDone) {
   567→    // Even if init was done, always run migrations on cold start
   568→    // to catch any new columns added in latest deploy.
   569→    // Migrations are idempotent (EXCEPTION WHEN duplicate_column).
   570→    if (!_migrationsRan) {
   571→      _migrationsRan = true;
   572→      runMigrationsOnly().catch((err) => {
   573→        console.error("[init-db] Background migration failed:", err instanceof Error ? err.message : String(err));
   574→        _migrationsRan = false; // Allow retry
   575→      });
   576→    }
   577→    return;
   578→  }
   579→  if (_initPromise) return _initPromise;
   580→
   581→  if (!process.env.DATABASE_URL) {
   582→    throw new Error("DATABASE_URL environment variable is not set.");
   583→  }
   584→
   585→  _initPromise = (async () => {
   586→    console.log("[init-db] Starting database initialization...");
   587→
   588→    let usePg = true;
   589→    let pgError: unknown = null;
   590→
   591→    // Strategy 1: Try using pg Client directly
   592→    try {
   593→      const client = new Client({ connectionString: process.env.DATABASE_URL });
   594→      await client.connect();
   595→      console.log("[init-db] pg Client connected successfully.");
   596→
   597→      // Check if User table already exists
   598→      const res = await client.query(
   599→        `SELECT EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema='public' AND table_name='User') as ok`
   600→      );
   601→      const tableExists = res.rows[0]?.ok === true;
   602→      console.log("[init-db] User table exists:", tableExists);
   603→
   604→      if (tableExists) {
   605→        console.log("[init-db] Tables already exist, running migrations to add any missing columns...");
   606→        try {
   607→          await client.query(MIGRATIONS_SQL);
   608→          console.log("[init-db] Migrations applied to existing tables.");
   609→          await client.query(INDEXES_SQL);
   610→          console.log("[init-db] Indexes verified.");
   611→        } catch (migrateErr) {
   612→          console.error("[init-db] Migration on existing tables failed:", migrateErr instanceof Error ? migrateErr.message : String(migrateErr));
   613→          // Don't throw — the tables exist, migrations are best-effort
   614→        }
   615→        await client.end().catch(() => {});
   616→        _initDone = true;
   617→        return;
   618→      }
   619→
   620→      // Create enums
   621→      console.log("[init-db] Creating enum types...");
   622→      await client.query(ENUMS_SQL);
   623→
   624→      // Create tables
   625→      console.log("[init-db] Creating tables...");
   626→      await client.query(TABLES_SQL);
   627→
   628→      // Verify User table was created
   629→      const verify = await client.query(
   630→        `SELECT EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema='public' AND table_name='User') as ok`
   631→      );
   632→      if (!verify.rows[0]?.ok) {
   633→        throw new Error("DDL executed but User table still not found!");
   634→      }
   635→      console.log("[init-db] Tables verified.");
   636→
   637→      // Create foreign keys
   638→      console.log("[init-db] Creating foreign keys...");
   639→      await client.query(FKS_SQL);
   640→
   641→      // Run migrations (add new columns to existing tables)
   642→      console.log("[init-db] Running migrations...");
   643→      await client.query(MIGRATIONS_SQL);
   644→
   645→      // Create indexes
   646→      console.log("[init-db] Creating indexes...");
   647→      await client.query(INDEXES_SQL);
   648→
   649→      // Seed SUPERUSER
   650→      console.log("[init-db] Seeding SUPERUSER...");
   651→      const hashed = await bcrypt.hash("Admin@2024", 12);
   652→      await client.query(
   653→        `INSERT INTO "User" ("id","username","password","name","role","permissions","policeRank","createdAt","updatedAt")
   654→         SELECT 'su-admin-001','admin',$1,'System Administrator','SUPERUSER','[]','',NOW(),NOW()
   655→         WHERE NOT EXISTS (SELECT 1 FROM "User" WHERE "username"='admin')`,
   656→        [hashed]
   657→      );
   658→
   659→      // Seed PoliceAlertConfig
   660→      await client.query(
   661→        `INSERT INTO "PoliceAlertConfig" ("id","createdAt","updatedAt")
   662→         SELECT 'default-alert-config',NOW(),NOW()
   663→         WHERE NOT EXISTS (SELECT 1 FROM "PoliceAlertConfig" WHERE "id"='default-alert-config')`
   664→      );
   665→
   666→      console.log("[init-db] Database initialization complete via pg.");
   667→      await client.end().catch(() => {});
   668→      _initDone = true;
   669→      return;
   670→    } catch (err) {
   671→      pgError = err;
   672→      console.error("[init-db] pg approach failed:", err instanceof Error ? err.message : String(err));
   673→      usePg = false;
   674→    }
   675→
   676→    // Strategy 2: Fallback — use Prisma $executeRawUnsafe
   677→    console.log("[init-db] Falling back to Prisma $executeRawUnsafe...");
   678→    try {
   679→      // First check if tables exist using Prisma raw query
   680→      const prisma = new PrismaClient({ log: ["warn", "error"] });
   681→      try {
   682→        const res: Array<{ ok: boolean }> = await prisma.$queryRawUnsafe(
   683→          `SELECT EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema='public' AND table_name='User') as ok`
   684→        );
   685→        if (res[0]?.ok) {
   686→          console.log("[init-db] Tables already exist (verified via Prisma), running migrations...");
   687→          try {
   688→            await execViaPrisma(MIGRATIONS_SQL);
   689→            console.log("[init-db] Migrations applied via Prisma.");
   690→            await execViaPrisma(INDEXES_SQL);
   691→            console.log("[init-db] Indexes verified via Prisma.");
   692→          } catch (migrateErr) {
   693→            console.error("[init-db] Prisma migration on existing tables failed:", migrateErr instanceof Error ? migrateErr.message : String(migrateErr));
   694→          }
   695→          _initDone = true;
   696→          return;
   697→        }
   698→      } catch {
   699→        // Table might not exist, which is expected
   700→        console.log("[init-db] Tables do not exist yet, creating via Prisma...");
   701→      } finally {
   702→        await prisma.$disconnect().catch(() => {});
   703→      }
   704→
   705→      // Execute DDL via Prisma (split into parts to avoid query size issues)
   706→      await execViaPrisma(ENUMS_SQL);
   707→      console.log("[init-db] Enums created via Prisma.");
   708→
   709→      await execViaPrisma(TABLES_SQL);
   710→      console.log("[init-db] Tables created via Prisma.");
   711→
   712→      await execViaPrisma(FKS_SQL);
   713→      console.log("[init-db] Foreign keys created via Prisma.");
   714→
   715→      await execViaPrisma(MIGRATIONS_SQL);
   716→      console.log("[init-db] Migrations ran via Prisma.");
   717→
   718→      await execViaPrisma(INDEXES_SQL);
   719→      console.log("[init-db] Indexes created via Prisma.");
   720→
   721→      // Seed SUPERUSER via Prisma
   722→      const prisma2 = new PrismaClient({ log: ["warn", "error"] });
   723→      try {
   724→        const hashed = await bcrypt.hash("Admin@2024", 12);
   725→        await prisma2.$executeRawUnsafe(
   726→          `INSERT INTO "User" ("id","username","password","name","role","permissions","policeRank","createdAt","updatedAt")
   727→           SELECT 'su-admin-001','admin',$1,'System Administrator','SUPERUSER','[]','',NOW(),NOW()
   728→           WHERE NOT EXISTS (SELECT 1 FROM "User" WHERE "username"='admin')`,
   729→          hashed
   730→        );
   731→        await prisma2.$executeRawUnsafe(
   732→          `INSERT INTO "PoliceAlertConfig" ("id","createdAt","updatedAt")
   733→           SELECT 'default-alert-config',NOW(),NOW()
   734→           WHERE NOT EXISTS (SELECT 1 FROM "PoliceAlertConfig" WHERE "id"='default-alert-config')`
   735→        );
   736→        console.log("[init-db] SUPERUSER seeded via Prisma.");
   737→      } finally {
   738→        await prisma2.$disconnect().catch(() => {});
   739→      }
   740→
   741→      console.log("[init-db] Database initialization complete via Prisma fallback.");
   742→      _initDone = true;
   743→    } catch (prismaErr) {
   744→      console.error("[init-db] Prisma fallback also failed:", prismaErr instanceof Error ? prismaErr.message : String(prismaErr));
   745→      _initDone = false;
   746→      _initPromise = null; // Allow retry
   747→      throw new Error(
   748→        `Database initialization failed. pg error: ${pgError instanceof Error ? pgError.message : String(pgError)} | Prisma error: ${prismaErr instanceof Error ? prismaErr.message : String(prismaErr)}`
   749→      );
   750→    }
   751→  })();
   752→
   753→  return _initPromise;
   754→}

   778→  _initDone = false;

   780→}
