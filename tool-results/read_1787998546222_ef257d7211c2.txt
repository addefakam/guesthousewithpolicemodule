   107→// ── Suspicious provider detection ──
   108→interface SuspicionFlag {
   109→  type: "high_maintenance" | "zero_available" | "all_non_available";
   110→  message: string;
   111→  severity: "HIGH" | "MEDIUM";
   112→}
   113→
   114→function detectSuspicion(p: ProviderStats, t: (key: string, opts?: Record<string, unknown>) => string): SuspicionFlag[] {
   115→  const flags: SuspicionFlag[] = [];
   116→
   117→  if (p.total === 0) return flags;
   118→
   119→  // More than 30% rooms in maintenance = suspicious hiding
   120→  const maintPct = (p.maintenance / p.total) * 100;
   121→  if (maintPct >= 30 && p.maintenance > 0) {
   122→    flags.push({
   123→      type: "high_maintenance",
   124→      message: t('flagRoomHiding', { maintenance: p.maintenance, total: p.total, pct: Math.round(maintPct) }),
   125→      severity: maintPct >= 50 ? "HIGH" : "MEDIUM",
   126→    });
   127→  }
   128→
   129→  // All rooms occupied/reserved/maintenance with zero available — suspicious during festivals
   130→  if (p.available === 0 && p.total > 0) {
   131→    flags.push({
   132→      type: "zero_available",
   133→      message: t('flagNoAvailable'),
   134→      severity: "MEDIUM",
   135→    });
   136→  }
   137→
   138→  return flags;
   139→}

   141→// ── Helpers ──
   142→function formatRoomType(type: string): string {
   143→  return type.charAt(0) + type.slice(1).toLowerCase().replace("_", "-");
   144→}
   145→
   146→function getRoomStatusColor(status: string): string {
   147→  switch (status) {
   148→    case "AVAILABLE": return "bg-emerald-100 text-emerald-800 border-emerald-200";
   149→    case "OCCUPIED": return "bg-blue-100 text-blue-800 border-blue-200";
   150→    case "RESERVED": return "bg-amber-100 text-amber-800 border-amber-200";
   151→    case "MAINTENANCE": return "bg-slate-200 text-slate-600 border-slate-300";
   152→    default: return "bg-slate-100 text-slate-600 border-slate-200";
   153→  }
   154→}
   155→
   156→function getUtilizationColor(rate: number): string {
   157→  if (rate >= 90) return "text-rose-600";
   158→  if (rate >= 70) return "text-amber-600";
   159→  return "text-emerald-600";
   160→}
   161→
   162→function getUtilizationBarColor(rate: number): string {
   163→  if (rate >= 90) return "bg-rose-500";
   164→  if (rate >= 70) return "bg-amber-500";
   165→  return "bg-emerald-500";
   166→}
