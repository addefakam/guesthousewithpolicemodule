   104→  return new Intl.NumberFormat("en-US", {

   108→    maximumFractionDigits: 0,

   112→export default function ReportsPage() {
   113→  const { refreshKey } = useAppStore();
   114→  const today = new Date().toISOString().split("T")[0];
   115→  const monthAgo = new Date(Date.now() - 30 * 86400000).toISOString().split("T")[0];
   116→
   117→  const [from, setFrom] = useState(monthAgo);
   118→  const [to, setTo] = useState(today);
   119→  const [data, setData] = useState<ReportData | null>(null);
   120→  const [loading, setLoading] = useState(true);
   121→  const [showGuests, setShowGuests] = useState(false);
   122→  const [showActiveGuests, setShowActiveGuests] = useState(false);
   123→  const [activeGuestSearch, setActiveGuestSearch] = useState();
   124→  const [expandedActiveIdx, setExpandedActiveIdx] = useState<number | null>(null);
   125→  const [expandedGuestIdx, setExpandedGuestIdx] = useState<number | null>(null);
   126→  const [guestSearch, setGuestSearch] = useState("");
   127→
   128→  const fetchReports = useCallback(async () => {
   129→    setLoading(true);
   130→    try {
   131→      const params = new URLSearchParams();
   132→      if (from) params.set("from", from);
   133→      if (to) params.set("to", to);
   134→      const result = await apiGetReports(params.toString());
   135→      setData(result);
   136→    } catch {
   137→      toast.error("Failed to load reports");
   138→    } finally {
   139→      setLoading(false);
   140→    }
   141→  }, [from, to]);
   142→
   143→  useEffect(() => {
   144→    fetchReports();
   145→  }, [fetchReports, refreshKey]);
   146→
   147→  const statusBreakdown = useMemo(() => {
   148→    if (!data?.reservations) return [];
   149→    const map = new Map<string, { count: number; revenue: number }>();
   150→    for (const r of data.reservations) {
   151→      const existing = map.get(r.status) || { count: 0, revenue: 0 };
   152→      existing.count += 1;
   153→      existing.revenue += r.paidAmount;
   154→      map.set(r.status, existing);
   155→    }
   156→    return Array.from(map.entries()).map(([status, info]) => ({
   157→      status,
   158→      ...info,
   159→    }));
   160→  }, [data]);
   161→
   162→  const maxDailyRevenue = useMemo(() => {
   163→    if (!data?.dailyRevenue?.length) return 0;
   164→    return Math.max(...data.dailyRevenue.map((d) => d.amount), 1);
   165→  }, [data]);
   166→
   167→  const maxExpenseCategory = useMemo(() => {
   168→    if (!data?.expenseBreakdown?.length) return 0;
   169→    return Math.max(...data.expenseBreakdown.map((e) => e.amount), 1);
   170→  }, [data]);
   171→
   172→  // Unique guests served in the period
   173→  const servedGuests = useMemo(() => {
   174→    if (!data?.reservations?.length) return [];
   175→    const map = new Map<string, {
   176→      guestId: string;
   177→      name: string;
   178→      phone: string;
   179→      email: string;
   180→      idNumber: string;
   181→      idType: string;
   182→      nationality: string;
   183→      address: string;
   184→      notes: string;
   185→      vip: boolean;
   186→      guestSince: string;
   187→      visits: number;
   188→      totalSpent: number;
   189→      lastVisit: string;
   190→      rooms: { number: string; name: string; type: string; checkIn: string; checkOut: string; status: string }[];
   191→    }>();
   192→    for (const r of data.reservations.filter((r: { status: string }) => r.status === "COMPLETED")) {
   193→      const g = r.guest;
   194→      const key = g?.id || g?.name || r.guestId || "unknown";
   195→      const existing = map.get(key) || {
   196→        guestId: g?.id || "",
   197→        name: g?.name || "Unknown",
   198→        phone: g?.phone || "",
   199→        email: g?.email || "",
   200→        idNumber: g?.idNumber || "",
   201→        idType: g?.idType || "",
   202→        nationality: g?.nationality || "",
   203→        address: g?.address || "",
   204→        notes: g?.notes || "",
   205→        vip: g?.vip || false,
   206→        guestSince: g?.createdAt || "",
   207→        visits: 0,
   208→        totalSpent: 0,
   209→        lastVisit: "",
   210→        rooms: [],
   211→      };
   212→      existing.visits += 1;
   213→      existing.totalSpent += r.paidAmount;
   214→      if (r.checkIn > existing.lastVisit) existing.lastVisit = r.checkIn;
   215→      existing.rooms.push({
   216→        number: r.room?.number || "",
   217→        name: r.room?.name || "",
   218→        type: r.room?.type || "",
   219→        checkIn: r.checkIn || "",
   220→        checkOut: r.checkOut || "",
   221→        status: r.status || "",
   222→      });
   223→      map.set(key, existing);
   224→    }
   225→    return Array.from(map.values()).sort((a, b) => b.lastVisit.localeCompare(a.lastVisit));
   226→  }, [data]);
   227→  const filteredServedGuests = useMemo(() => {
   228→    if (!guestSearch) return servedGuests;
   229→    const q = guestSearch.toLowerCase();
   230→    return servedGuests.filter((g) =>
   231→      g.name.toLowerCase().includes(q) ||
   232→      g.phone.toLowerCase().includes(q) ||
   233→      g.idNumber.toLowerCase().includes(q) ||
   234→      g.email.toLowerCase().includes(q) ||
   235→      g.nationality.toLowerCase().includes(q)
   236→    );
   237→  }, [servedGuests, guestSearch]);
   238→  // Active guests (OCCUPIED + RESERVED) in the period
   239→  const activeGuests = useMemo(() => {
   240→    if (!data?.activeGuests?.length) return [];
   241→    const map = new Map();
   242→    for (const r of data.activeGuests) {
   243→      const g = r.guest;
   244→      map.set(r.id, {
   245→        reservationId: r.id,
   246→        guestId: g?.id || "",
   247→        name: g?.name || "Unknown",
   248→        phone: g?.phone || "",
   249→        email: g?.email || "",
   250→        idNumber: g?.idNumber || "",
   251→        idType: g?.idType || "",
   252→        nationality: g?.nationality || "",
   253→        address: g?.address || "",
   254→        notes: g?.notes || "",
   255→        vip: g?.vip || false,
   256→        roomNumber: r.room?.number || "",
   257→        roomName: r.room?.name || "",
   258→        roomType: r.room?.type || "",
   259→        status: r.status,
   260→        checkIn: r.checkIn || "",
   261→        checkOut: r.checkOut || "",
   262→        paidAmount: r.paidAmount || 0,
   263→      });
   264→    }
   265→    return Array.from(map.values()).sort((a, b) => {
   266→      if (a.status === "ACTIVE" && b.status !== "ACTIVE") return -1;
   267→      if (a.status !== "ACTIVE" && b.status === "ACTIVE") return 1;
   268→      return a.checkIn.localeCompare(b.checkIn);
   269→    });
   270→  }, [data]);
   271→  const filteredActiveGuests = useMemo(() => {
   272→    if (!activeGuestSearch) return activeGuests;
   273→    const q = activeGuestSearch.toLowerCase();
   274→    return activeGuests.filter((g) =>
   275→      g.name.toLowerCase().includes(q) ||
   276→      g.phone.toLowerCase().includes(q) ||
   277→      g.idNumber.toLowerCase().includes(q) ||
   278→      g.email.toLowerCase().includes(q) ||
   279→      g.roomNumber.toLowerCase().includes(q)
   280→    );
   281→  }, [activeGuests, activeGuestSearch]);
   282→  const handleExport = () => {
   283→    if (!data) return;
   284→    const json = JSON.stringify(data, null, 2);
   285→    const blob = new Blob([json], { type: "application/json" });
   286→    const url = URL.createObjectURL(blob);
   287→    const a = document.createElement("a");
   288→    a.href = url;
   289→    a.download = `report-${from}-to-${to}.json`;
   290→    a.click();
   291→    URL.revokeObjectURL(url);
   292→    toast.success("Report exported");
   293→  };
   294→
   295→  return (
   296→    <div className="space-y-6 p-4 md:p-6">
   297→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   298→        <div>
   299→          <h1 className="text-2xl font-bold tracking-tight">Reports</h1>
   300→          <p className="text-sm text-muted-foreground">
   301→            View financial summaries and analytics.
   302→          </p>
   303→        </div>
   304→        <div className="flex gap-2">
   305→          <Button variant={showActiveGuests ? "default" : "outline"} onClick={() => setShowActiveGuests(!showActiveGuests)} disabled={loading || !data}>
   306→            <UserCheck className="mr-2 h-4 w-4" />
   307→            Active Guests ({activeGuests.length})
   308→          </Button>
   309→          <Button variant={showGuests ? "default" : "outline"} onClick={() => setShowGuests(!showGuests)} disabled={loading || !data}>
   310→            <Users className="mr-2 h-4 w-4" />
   311→            Served Guests ({servedGuests.length})
   312→          </Button>
   313→          <Button variant="outline" onClick={handleExport} disabled={loading || !data}>
   314→            <Download className="mr-2 h-4 w-4" />
   315→            Export JSON
   316→          </Button>
   317→        </div>
   318→      </div>
   319→
   320→      {/* Date Range Picker */}
   321→      <Card>
   322→        <CardContent className="flex flex-col sm:flex-row items-end gap-4 pt-6">
   323→          <div className="grid gap-2 flex-1 w-full sm:w-auto">
   324→            <Label htmlFor="from">From</Label>
   325→            <Input
   326→              id="from"
   327→              type="date"
   328→              value={from}
   329→              onChange={(e) => setFrom(e.target.value)}
   330→            />
   331→          </div>
   332→          <div className="grid gap-2 flex-1 w-full sm:w-auto">
   333→            <Label htmlFor="to">To</Label>
   334→            <Input
   335→              id="to"
   336→              type="date"
   337→              value={to}
   338→              onChange={(e) => setTo(e.target.value)}
   339→            />
   340→          </div>
   341→          <Button onClick={fetchReports} disabled={loading}>
   342→            <CalendarDays className="mr-2 h-4 w-4" />
   343→            Apply
   344→          </Button>
   345→        </CardContent>
   346→      </Card>
   347→
   348→      {loading ? (
   349→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   350→          {Array.from({ length: 4 }).map((_, i) => (
   351→            <Skeleton key={i} className="h-28 rounded-xl" />
   352→          ))}
   353→        </div>
   354→      ) : data ? (
   355→        <>
   356→          {/* Active Guests Section */}
   357→          {showActiveGuests && (
   358→            <Card>
   359→              <CardHeader>
   360→                <CardTitle className="flex items-center gap-2">
   361→                  <UserCheck className="h-5 w-5" />
   362→                  Active & Upcoming Guests
   363→                </CardTitle>
   364→                <div className="relative mt-2">
   365→                  <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-gray-400" />
   366→                  <Input
   367→                    placeholder="Search by name, ID, phone, room..."
   368→                    value={activeGuestSearch}
   369→                    onChange={(e) => setActiveGuestSearch(e.target.value)}
   370→                    className="pl-9"
   371→                  />
   372→                </div>
   373→              </CardHeader>
   374→              <CardContent>
   375→                {filteredActiveGuests.length === 0 ? (
   376→                  <p className="text-sm text-muted-foreground text-center py-8">
   377→                    No active or upcoming guests in this period.
   378→                  </p>
   379→                ) : (
   380→                  <>
   381→                    {/* Mobile cards */}
   382→                    <div className="md:hidden space-y-3">
   383→                      {filteredActiveGuests.map((g, i) => (
   384→                        <div key={g.reservationId}>
   385→                          <div
   386→                            className="rounded-lg border p-3 space-y-2 cursor-pointer"
   387→                            onClick={() => setExpandedActiveIdx(expandedActiveIdx === i ? null : i)}
   388→                          >
   389→                            <div className="flex items-center justify-between">
   390→                              <div className="flex items-center gap-2">
   391→                                <div className={"flex h-8 w-8 items-center justify-center rounded-full font-medium text-sm shrink-0 " + (g.status === "ACTIVE" ? "bg-emerald-100 text-emerald-600" : "bg-blue-100 text-blue-600")}>
   392→                                  {g.name.charAt(0).toUpperCase()}
   393→                                </div>
   394→                                <div>
   395→                                  <div className="flex items-center gap-1.5">
   396→                                    <p className="font-semibold text-sm">{g.name}</p>
   397→                                    {g.vip && <Star className="inline h-3.5 w-3.5 fill-amber-400 text-amber-400" />}
   398→                                  </div>
   399→                                  {g.phone && <p className="text-xs text-muted-foreground">{g.phone}</p>}
   400→                                </div>
   401→                              </div>
   402→                              <div className="flex items-center gap-2">
   403→                                <Badge variant="outline" className={STATUS_COLORS[g.status] || ""}>{g.status}</Badge>
   404→                                {expandedActiveIdx === i ? <ChevronUp className="h-4 w-4 text-gray-400" /> : <ChevronDown className="h-4 w-4 text-gray-400" />}
   405→                              </div>
   406→                            </div>
   407→                            <div className="flex items-center justify-between text-xs">
   408→                              <span className="text-muted-foreground">Room {g.roomNumber}{g.roomName ? ' ' + g.roomName : ''}</span>
   409→                              <span className="font-medium">{g.checkIn} → {g.checkOut}</span>
   410→                            </div>
   411→                          </div>
   412→                          {expandedActiveIdx === i && (
   413→                            <div className="mt-2 ml-4 rounded-lg border border-sky-100 bg-sky-50 p-3 space-y-2">
   414→                              {g.email && (
   415→                                <div className="flex items-center gap-2 text-xs"><Mail className="h-3.5 w-3.5 text-gray-400" /><span className="text-gray-600">{g.email}</span></div>
   416→                              )}
   417→                              {(g.idType || g.idNumber) && (
   418→                                <div className="flex items-center gap-2 text-xs"><CreditCard className="h-3.5 w-3.5 text-gray-400" /><span className="text-gray-600">{g.idType}{g.idNumber ? ' · ' + g.idNumber : ""}</span></div>
   419→                              )}
   420→                              {g.nationality && (
   421→                                <div className="flex items-center gap-2 text-xs"><Globe className="h-3.5 w-3.5 text-gray-400" /><span className="text-gray-600">{g.nationality}</span></div>
   422→                              )}
   423→                              {g.address && (
   424→                                <div className="flex items-center gap-2 text-xs"><MapPin className="h-3.5 w-3.5 text-gray-400" /><span className="text-gray-600">{g.address}</span></div>
   425→                              )}
   426→                              {g.notes && (
   427→                                <div className="flex items-start gap-2 text-xs mt-1"><FileText className="h-3.5 w-3.5 text-sky-500 mt-0.5 shrink-0" /><span className="text-sky-700">{g.notes}</span></div>
   428→                              )}
   429→                              {g.paidAmount > 0 && (
   430→                                <div className="flex items-center gap-2 text-xs mt-1"><DollarSign className="h-3.5 w-3.5 text-gray-400" /><span className="text-gray-600">Paid: {formatCurrency(g.paidAmount)}</span></div>
   431→                              )}
   432→                            </div>
   433→                          )}
   434→                        </div>
   435→                      ))}
   436→                    </div>
   437→                    {/* Desktop table */}
   438→                    <div className="hidden md:block rounded-xl border bg-white max-h-[500px] overflow-y-auto">
   439→                      <Table>
   440→                        <TableHeader>
   441→                          <TableRow className="bg-gray-50/80">
   442→                            <TableHead>Guest Name</TableHead>
   443→                            <TableHead>Phone</TableHead>
   444→                            <TableHead>Room</TableHead>
   445→                            <TableHead>Status</TableHead>
   446→                            <TableHead>Check-In</TableHead>
   447→                            <TableHead>Check-Out</TableHead>
   448→                            <TableHead className="text-right">Paid</TableHead>
   449→                            <TableHead></TableHead>
   450→                          </TableRow>
   451→                        </TableHeader>
   452→                        <TableBody>
   453→                          {filteredActiveGuests.map((g, i) => (
   454→                            <Fragment key={g.reservationId}>
   455→                              <TableRow className="cursor-pointer" onClick={() => setExpandedActiveIdx(expandedActiveIdx === i ? null : i)}>
   456→                                <TableCell>
   457→                                  <div className="flex items-center gap-2">
   458→                                    <div className={"flex h-8 w-8 items-center justify-center rounded-full font-medium text-sm " + (g.status === "ACTIVE" ? "bg-emerald-100 text-emerald-600" : "bg-blue-100 text-blue-600")}>
   459→                                      {g.name.charAt(0).toUpperCase()}
   460→                                    </div>
   461→                                    <div>
   462→                                      <span className="font-medium text-gray-900">{g.name}</span>
   463→                                      {g.vip && <span className="ml-1.5"><Star className="inline h-3.5 w-3.5 fill-amber-400 text-amber-400" /></span>}
   464→                                    </div>
   465→                                  </div>
   466→                                </TableCell>
   467→                                <TableCell className="text-gray-600">{g.phone || "—"}</TableCell>

   470→                                <TableCell>{g.checkIn}</TableCell>
   471→                                <TableCell>{g.checkOut}</TableCell>
   472→                                <TableCell className="text-right font-medium">{formatCurrency(g.paidAmount)}</TableCell>

   474→                              </TableRow>
   475→                              {expandedActiveIdx === i && (

   477→                                  <TableCell colSpan={8}>
   478→                                    <div className="grid grid-cols-1 gap-4 py-4 sm:grid-cols-2 lg:grid-cols-3">
   479→                                      {g.email && (
   480→                                        <div className="flex items-start gap-2">
   481→                                          <Mail className="mt-0.5 h-4 w-4 text-gray-400 shrink-0" />
   482→                                          <div><p className="text-xs text-gray-500">Email</p><p className="text-sm text-gray-900">{g.email}</p></div>
   483→                                        </div>
   484→                                      )}
   485→                                      {(g.idType || g.idNumber) && (
   486→                                        <div className="flex items-start gap-2">
   487→                                          <CreditCard className="mt-0.5 h-4 w-4 text-gray-400 shrink-0" />

   489→                                        </div>
   490→                                      )}
   491→                                      {g.nationality && (
   492→                                        <div className="flex items-start gap-2">
   493→                                          <Globe className="mt-0.5 h-4 w-4 text-gray-400 shrink-0" />
   494→                                          <div><p className="text-xs text-gray-500">Nationality</p><p className="text-sm text-gray-900">{g.nationality}</p></div>
   495→                                        </div>
   496→                                      )}
   497→                                    </div>
   498→                                    {g.address && (
