   169→// Derive floor from first digit of room number (e.g. "102" → 1, "201" → 2)
   170→const getFloorFromNumber = (num: string): number | null => {
   171→  const match = num.match(/^\d/);
   172→  return match ? parseInt(match[0], 10) : null;

   215→  const [deleteDialog, setDeleteDialog] = useState<Room | null>(null);
   216→  const [deleting, setDeleting] = useState(false);
   217→
   218→  // room info detail
   219→  const [infoRoom, setInfoRoom] = useState<Room | null>(null);
   220→  const [roomReservations, setRoomReservations] = useState<RoomReservation[]>([]);
   221→  const [roomResLoading, setRoomResLoading] = useState(false);
   222→
   223→  // excel import
   224→  const [importDialogOpen, setImportDialogOpen] = useState(false);

   227→  const [importResults, setImportResults] = useState<ImportResult[] | null>(null);
   228→  const importFileRef = useRef<HTMLInputElement>(null);
   229→
   230→  const fetchRooms = useCallback(async () => {
   231→    try {
   232→      setLoading(true);
   233→      const data = await apiGetRooms(search);
   234→      const list = data.rooms || [];
   235→      // Sort: AVAILABLE first, then RESERVED, then others; within each group by floor
   236→      const statusOrder: Record<string, number> = { AVAILABLE: 0, RESERVED: 1, OCCUPIED: 2, MAINTENANCE: 3 };
   237→      list.sort((a, b) => {
   238→        const oa = statusOrder[a.status] ?? 9;
   239→        const ob = statusOrder[b.status] ?? 9;
   240→        if (oa !== ob) return oa - ob;
   241→        const fa = getFloorFromNumber(a.number) ?? a.floor ?? 0;
   242→        const fb = getFloorFromNumber(b.number) ?? b.floor ?? 0;
   243→        return fa - fb;
   244→      });
   245→      setRooms(list);
   246→    } catch (err: unknown) {
   247→      const message = err instanceof Error ? err.message : "Failed to load rooms";
   248→      toast.error(message);
   249→    } finally {
   250→      setLoading(false);
   251→    }
   252→  }, [search]);
   253→
   254→  useEffect(() => {
   255→    const timer = setTimeout(() => fetchRooms(), 300);
   256→    return () => clearTimeout(timer);
   257→  }, [fetchRooms, refreshKey]);
   258→
   259→  // Fetch reservations for the room when info dialog opens
   260→  useEffect(() => {
   261→    if (!infoRoom) {
   262→      setRoomReservations([]);
   263→      return;

   266→      try {
   267→        setRoomResLoading(true);
   268→        const data = await apiGetReservations(`roomId=${infoRoom.id}`);
   269→        setRoomReservations(Array.isArray(data) ? data : []);
   270→      } catch {
   271→        setRoomReservations([]);
   272→      } finally {
   273→        setRoomResLoading(false);
   274→      }
   275→    })();
   276→  }, [infoRoom]);
   277→
   278→  const openCreate = () => {
   279→    setEditingRoom(null);
   280→    setForm(emptyForm);
   281→    setDialogOpen(true);
   282→  };
   283→
   284→  const openEdit = (room: Room) => {

   286→    setForm({
   287→      number: room.number,
   288→      type: room.type,
   289→      pricePerNight: String(room.pricePerNight),
   290→      floor: String(room.floor),
   291→      capacity: String(room.capacity),
   292→      amenities: room.amenities,
   293→      description: room.description,
   294→    });
   295→    setDialogOpen(true);
   296→  };
   297→
   298→  const handleSave = async () => {
   299→    if (!form.number || !form.pricePerNight || !form.floor || !form.capacity) {
   300→      toast.error("Please fill in all required fields");
   301→      return;
   302→    }
   303→
   304→    try {
   305→      setSaving(true);
   306→      const payload = {
   307→        number: form.number,
   308→        type: form.type,
   309→        pricePerNight: Number(form.pricePerNight),
   310→        floor: Number(form.floor),
   311→        capacity: Number(form.capacity),
   312→        amenities: form.amenities || "[]",
   313→        description: form.description,
   314→      };
   315→
   316→      if (editingRoom) {
   317→        await apiUpdateRoom(editingRoom.id, payload);
   318→        toast.success("Room updated successfully");
   319→      } else {
   320→        await apiCreateRoom(payload);
   321→        toast.success("Room created successfully");
   322→      }
   323→
   324→      setDialogOpen(false);
   325→      triggerRefresh();
   326→    } catch (err: unknown) {
   327→      const message = err instanceof Error ? err.message : "Failed to save room";
   328→      toast.error(message);
   329→    } finally {
   330→      setSaving(false);
   331→    }
   332→  };
   333→
   334→  const handleDelete = async () => {

   337→      setDeleting(true);
   338→      await apiDeleteRoom(deleteDialog.id);
   339→      toast.success("Room deleted successfully");
   340→      setDeleteDialog(null);
   341→      triggerRefresh();
   342→    } catch (err: unknown) {
   343→      const message = err instanceof Error ? err.message : "Failed to delete room";
   344→      toast.error(message);
   345→    } finally {
   346→      setDeleting(false);
   347→    }
   348→  };
   349→
   350→  const handleStatusChange = async (room: Room, newStatus: string) => {
   351→    try {
   352→      await apiUpdateRoomStatus(room.id, newStatus);
   353→      toast.success(`Room ${room.number} status changed to ${newStatus}`);

   355→    } catch (err: unknown) {
   356→      const message = err instanceof Error ? err.message : "Failed to update status";
   357→      toast.error(message);
   358→    }

   361→  const handleReserveFromRoom = (room: Room) => {

   368→      pricePerNight: room.pricePerNight,

   370→    setCurrentPage("reservations");

   372→
   373→  // ── Excel import ────────────────────────────────────────────────────────────
   374→  const handleImportFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {

   376→    setImportResults(null);
   377→  };
   378→
   379→  const handleImport = async () => {
   380→    if (!importFile) { toast.error("Please select an Excel file first"); return; }
   381→    setImportLoading(true);
   382→    try {
   383→      const buffer = await importFile.arrayBuffer();
   384→      const wb = XLSX.read(buffer, { type: "array" });
   385→      const ws = wb.Sheets[wb.SheetNames[0]];
   386→      const rows = XLSX.utils.sheet_to_json<Record<string, unknown>>(ws);
   387→
   388→      if (rows.length === 0) { toast.error("No data rows found in the file"); return; }
   389→
   390→      const result = await apiImportRooms(rows as Record<string, unknown>[]);
   391→      setImportResults(result.results);
   392→      toast.success(`Import complete: ${result.imported} created, ${result.skipped} skipped`);
   393→      triggerRefresh();
   394→    } catch (err: unknown) {
   395→      const message = err instanceof Error ? err.message : "Import failed";
   396→      toast.error(message);
   397→    } finally {
   398→      setImportLoading(false);
   399→    }
   400→  };
   401→
   402→  const closeImportDialog = () => {
   403→    setImportDialogOpen(false);
   404→    setImportFile(null);
   405→    setImportResults(null);

   408→

   410→    try {
   411→      const parsed = JSON.parse(amenitiesStr);
   412→      return Array.isArray(parsed) ? parsed : [];

   434→      room.name.toLowerCase().includes(q) ||
   435→      room.type.toLowerCase().includes(q)

   460→  }
   461→
   462→  return (
   463→    <div className="space-y-6 p-4 md:p-6">
   464→      {/* Header */}
   465→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">

   467→          <h1 className="text-2xl font-bold text-gray-900">Rooms</h1>

   469→            Manage your {rooms.length} room{rooms.length !== 1 ? "s" : ""}

   534→              </DropdownMenuTrigger>

   536→                {floors.slice(3).map((f) => (

   538→                    key={f}
   539→                    className="text-xs"
