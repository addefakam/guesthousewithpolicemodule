   159→export default function OwnerAccountsPage() {
   160→  const { currentUser, refreshKey } = useAppStore();
   161→  const [providers, setProviders] = useState<ProviderWithOwner[]>([]);
   162→  const [policeUsers, setPoliceUsers] = useState<AccountUser[]>([]);
   163→  const [loading, setLoading] = useState(true);
   164→  const [search, setSearch] = useState("");
   165→  const [activeTab, setActiveTab] = useState<TabType>("overview");
   166→
   167→  // Determine if current user can manage police
   168→  const canManagePolice =
   169→    currentUser?.role === "SUPERUSER" ||
   170→    (currentUser?.role === "POLICE" &&
   171→      (RANK_ORDER[currentUser?.policeRank || ""] || 0) >= RANK_ORDER.DETECTIVE);
   172→
   173→  // Which ranks can this user create?
   174→  const maxCreatableRank =
   175→    currentUser?.role === "SUPERUSER"
   176→      ? "ADMIN"
   177→      : currentUser?.role === "POLICE"
   178→        ? currentUser?.policeRank === "ADMIN"
   179→          ? "DETECTIVE"
   180→          : currentUser?.policeRank === "DETECTIVE"
   181→            ? "OFFICER"
   182→            : ""
   183→        : "";
   184→
   185→  // ── Reset credentials dialog ──
   186→  const [resetOpen, setResetOpen] = useState(false);
   187→  const [resetUserId, setResetUserId] = useState("");
   188→  const [resetUsername, setResetUsername] = useState("");
   189→  const [resetPassword, setResetPassword] = useState("");
   190→  const [showPassword, setShowPassword] = useState(false);
   191→  const [saving, setSaving] = useState(false);
   192→  const [resetLabel, setResetLabel] = useState("");
   193→  const [resetSublabel, setResetSublabel] = useState("");
   194→
   195→  // ── Add/Edit police dialog ──
   196→  const [policeDialogOpen, setPoliceDialogOpen] = useState(false);
   197→  const [editMode, setEditMode] = useState(false);
   198→  const [editPoliceId, setEditPoliceId] = useState("");
   199→  const [policeName, setPoliceName] = useState("");
   200→  const [policeUsername, setPoliceUsername] = useState("");
   201→  const [policePassword, setPolicePassword] = useState("");
   202→  const [policeRank, setPoliceRank] = useState("OFFICER");
   203→  const [policeSaving, setPoliceSaving] = useState(false);
   204→  const [showPolicePassword, setShowPolicePassword] = useState(false);
   205→
   206→  // ── Delete confirmation ──
   207→  const [deleteOpen, setDeleteOpen] = useState(false);
   208→  const [deleteTarget, setDeleteTarget] = useState<AccountUser | null>(null);
   209→  const [deleting, setDeleting] = useState(false);
   210→
   211→  const fetchAccounts = useCallback(async () => {
   212→    setLoading(true);
   213→    try {
   214→      const data: ApiResponse = await apiGetOwnerAccounts();
   215→      setProviders(data.providers);
   216→      setPoliceUsers(data.policeUsers);
   217→    } catch {
   218→      toast.error("Failed to load accounts");
   219→    } finally {
   220→      setLoading(false);
   221→    }
   222→  }, []);
   223→
   224→  const fetchPoliceOnly = useCallback(async () => {
   225→    try {
   226→      const data = await apiPoliceOfficers();
   227→      setPoliceUsers(data);
   228→    } catch {
   229→      // silent
   230→    }
   231→  }, []);
   232→
   233→  useEffect(() => {
   234→    fetchAccounts();
   235→  }, [fetchAccounts, refreshKey]);
   236→
   237→  // ── Owner reset handlers ──
   238→  const openOwnerReset = (provider: ProviderWithOwner) => {
   239→    const ownerUser = provider.users[0];
   240→    setResetUserId(ownerUser?.id || "");
   241→    setResetUsername(ownerUser?.username || "");
   242→    setResetPassword("");
   243→    setShowPassword(false);
   244→    setResetLabel(provider.name);
   245→    setResetSublabel(`${provider.ownerName}  ·  ${provider.phone}`);
   246→    setResetOpen(true);
   247→  };
   248→
   249→  const openPoliceReset = (police: AccountUser) => {
   250→    setResetUserId(police.id);
   251→    setResetUsername(police.username);
   252→    setResetPassword("");
   253→    setShowPassword(false);
   254→    setResetLabel("Police Account");
   255→    setResetSublabel(police.name);
   256→    setResetOpen(true);
   257→  };
   258→
   259→  const handleReset = async (e: React.FormEvent) => {
   260→    e.preventDefault();
   261→    if (!resetUserId) return;
   262→    if (!resetUsername.trim()) {
   263→      toast.error("Username is required");
   264→      return;
   265→    }
   266→    setSaving(true);
   267→    try {
   268→      const updateData: Record<string, unknown> = {};
   269→      if (resetUsername.trim()) updateData.username = resetUsername.trim();
   270→      if (resetPassword.trim()) updateData.password = resetPassword.trim();
   271→      if (Object.keys(updateData).length === 0) {
   272→        toast.error("No changes to save");
   273→        setSaving(false);
   274→        return;
   275→      }
   276→      await apiUpdateOwnerAccount(resetUserId, updateData);
   277→      toast.success("Credentials updated successfully");
   278→      setResetOpen(false);
   279→      fetchAccounts();
   280→    } catch (err: unknown) {
   281→      const msg = err instanceof Error ? err.message : "Failed to update credentials";
   282→      toast.error(msg);
   283→    } finally {
   284→      setSaving(false);
   285→    }
   286→  };
   287→
   288→  // ── Police CRUD handlers ──
   289→  const openAddPolice = () => {
   290→    setEditMode(false);
   291→    setEditPoliceId("");
   292→    setPoliceName("");
   293→    setPoliceUsername("");
   294→    setPolicePassword("");
   295→    setPoliceRank(maxCreatableRank === "DETECTIVE" ? "OFFICER" : "VIEWER");
   296→    setShowPolicePassword(false);
   297→    setPoliceDialogOpen(true);
   298→  };
   299→
   300→  const openEditPolice = (police: AccountUser) => {
   301→    if (!canManagePolice) return;
   302→    // Cannot edit self
   303→    if (currentUser?.role === "POLICE" && police.id === currentUser.id) {
   304→      toast.error("Cannot edit your own account");
   305→      return;
   306→    }
   307→    // Check if we can edit this rank
   308→    if (currentUser?.role === "POLICE") {
   309→      const targetLevel = RANK_ORDER[police.policeRank || "OFFICER"] || 0;
   310→      const myLevel = RANK_ORDER[currentUser.policeRank || "OFFICER"] || 0;
   311→      if (targetLevel >= myLevel) {
   312→        toast.error(`Cannot edit a ${POLICE_RANK_LABEL[police.policeRank || ""] || "officer"}`);
   313→        return;
   314→      }
   315→    }
   316→    setEditMode(true);
   317→    setEditPoliceId(police.id);
   318→    setPoliceName(police.name);
   319→    setPoliceUsername(police.username);
   320→    setPolicePassword("");
   321→    setPoliceRank(police.policeRank || "OFFICER");
   322→    setShowPolicePassword(false);
   323→    setPoliceDialogOpen(true);
   324→  };
   325→
   326→  const openDeletePolice = (police: AccountUser) => {
   327→    if (!canManagePolice) return;
   328→    if (currentUser?.role === "POLICE" && police.id === currentUser.id) {
   329→      toast.error("Cannot delete your own account");
   330→      return;
   331→    }
   332→    setDeleteTarget(police);
   333→    setDeleteOpen(true);
   334→  };
   335→
   336→  const handleSavePolice = async (e: React.FormEvent) => {
   337→    e.preventDefault();
   338→    if (!policeName.trim() || !policeUsername.trim()) {
   339→      toast.error("Name and username are required");
   340→      return;
   341→    }
   342→    if (!editMode && !policePassword.trim()) {
   343→      toast.error("Password is required for new accounts");
   344→      return;
   345→    }
   346→    setPoliceSaving(true);
   347→    try {
   348→      if (editMode) {
   349→        const data: Record<string, unknown> = { name: policeName.trim(), policeRank };
   350→        if (policePassword.trim()) data.password = policePassword.trim();
   351→        await apiPoliceUpdateOfficer(editPoliceId, data);
   352→        toast.success("Police officer updated");
   353→      } else {
   354→        await apiPoliceCreateOfficer({
   355→          name: policeName.trim(),
   356→          username: policeUsername.trim(),
   357→          password: policePassword.trim(),
   358→          policeRank,
   359→        });
   360→        toast.success("Police officer created");
   361→      }
   362→      setPoliceDialogOpen(false);
   363→      fetchPoliceOnly();
   364→    } catch (err: unknown) {
   365→      const msg = err instanceof Error ? err.message : "Failed to save police officer";
   366→      toast.error(msg);
   367→    } finally {
   368→      setPoliceSaving(false);
   369→    }
   370→  };
   371→
   372→  const handleDeletePolice = async () => {
   373→    if (!deleteTarget) return;
   374→    setDeleting(true);
   375→    try {
   376→      await apiPoliceDeleteOfficer(deleteTarget.id);
   377→      toast.success("Police officer deleted");
   378→      setDeleteOpen(false);
   379→      setDeleteTarget(null);
   380→      fetchPoliceOnly();
   381→    } catch (err: unknown) {
   382→      const msg = err instanceof Error ? err.message : "Failed to delete";
   383→      toast.error(msg);
   384→    } finally {
   385→      setDeleting(false);
   386→    }
   387→  };
   388→
   389→  // ── Filtering ──
   390→  const filteredProviders = providers.filter(
   391→    (p) =>
   392→      p.name.toLowerCase().includes(search.toLowerCase()) ||
   393→      p.ownerName.toLowerCase().includes(search.toLowerCase()) ||
   394→      p.phone.includes(search) ||
   395→      p.users.some((u) => u.username.toLowerCase().includes(search.toLowerCase()))
   396→  );
   397→
   398→  const filteredPolice = policeUsers.filter(
   399→    (u) =>
   400→      u.name.toLowerCase().includes(search.toLowerCase()) ||
   401→      u.username.toLowerCase().includes(search.toLowerCase())
   402→  );
   403→
   404→  const hasNoResults =
   405→    (activeTab === "overview" && filteredProviders.length === 0 && filteredPolice.length === 0) ||
   406→    (activeTab === "owners" && filteredProviders.length === 0) ||
   407→    (activeTab === "police" && filteredPolice.length === 0);
   408→
   409→  const totalProviders = providers.length;
   410→  const approvedProviders = providers.filter((p) => p.status === "APPROVED").length;
   411→  const pendingProviders = providers.filter((p) => p.status === "PENDING").length;
   412→  const totalPolice = policeUsers.length;
   413→
   414→  if (loading) {
   415→    return (
   416→      <div className="space-y-6 p-4 md:p-6">
   417→        <Skeleton className="h-8 w-64" />
   418→        <Skeleton className="h-10 w-full max-w-sm" />
   419→        <div className="space-y-3">
   420→          {Array.from({ length: 4 }).map((_, i) => (
   421→            <Skeleton key={i} className="h-24 rounded-lg" />
   422→          ))}
   423→        </div>
   424→      </div>
   425→    );
   426→  }
   427→
   428→  // ── Render Police Table Row ──
   429→  const renderPoliceRow = (police: AccountUser, isMobile = false) => {
   430→    const rank = police.policeRank || "OFFICER";
   431→    const isSelf = currentUser?.role === "POLICE" && police.id === currentUser.id;
   432→    const canEdit = canManagePolice && !isSelf;
   433→    const canDelete =
   434→      canManagePolice &&
   435→      !isSelf &&
   436→      (currentUser?.role === "SUPERUSER" ||
   437→        (RANK_ORDER[currentUser?.policeRank || ""] || 0) > (RANK_ORDER[rank] || 0));
   438→
   439→    if (isMobile) {
   440→      return (
   441→        <div key={police.id} className="rounded-lg border p-4 space-y-3">
   442→          <div className="flex items-center gap-2">
   443→            <div className="flex h-9 w-9 shrink-0 items-center justify-center rounded-lg bg-rose-100">
   444→              <Shield className="h-4 w-4 text-rose-600" />
   445→            </div>
   446→            <div className="min-w-0 flex-1">
   447→              <div className="flex items-center gap-2 flex-wrap">
   448→                <p className="font-semibold text-sm truncate">{police.name}</p>
   449→                <Badge variant="outline" className={POLICE_RANK_BADGE[rank] || ""}>
   450→                  {rank}
   451→                </Badge>
   452→                {isSelf && (
   453→                  <Badge variant="outline" className="bg-blue-100 text-blue-700 border-blue-200">
   454→                    You
   455→                  </Badge>
   456→                )}
   457→              </div>
   458→              <code className="rounded bg-muted px-1.5 py-0.5 text-xs font-mono">{police.username}</code>
   459→            </div>
   460→          </div>
   461→          <p className="text-xs text-muted-foreground">
   462→            Created: {new Date(police.createdAt).toLocaleDateString()}

   465→            <Button variant="outline" size="sm" className="flex-1 gap-1.5" onClick={() => openPoliceReset(police)}>
   466→              <KeyRound className="h-3.5 w-3.5" />

   469→            {canEdit && (
   470→              <Button variant="outline" size="sm" className="gap-1.5" onClick={() => openEditPolice(police)}>
   471→                <Pencil className="h-3.5 w-3.5" />
   472→              </Button>

   474→            {canDelete && (
   475→              <Button variant="outline" size="sm" className="gap-1.5 text-destructive hover:bg-destructive/10" onClick={() => openDeletePolice(police)}>
   476→                <Trash2 className="h-3.5 w-3.5" />
   477→              </Button>
   478→            )}
   479→          </div>
   480→        </div>
   481→      );
   482→    }
   483→
   484→    return (
   485→      <tr key={police.id} className="border-b last:border-0 hover:bg-muted/30 transition-colors">
   486→        <td className="px-4 py-3">
   487→          <div className="flex items-center gap-2">
   488→            <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-rose-100">
   489→              <Shield className="h-4 w-4 text-rose-600" />
   490→            </div>
   491→            <div>
   492→              <div className="flex items-center gap-2">
   493→                <span className="font-medium">{police.name}</span>
   494→                {isSelf && (
   495→                  <Badge variant="outline" className="text-[9px] bg-blue-100 text-blue-700 border-blue-200">
   496→                    You
   497→                  </Badge>
   498→                )}
   499→              </div>
   500→            </div>
   501→          </div>
   502→        </td>
   503→        <td className="px-4 py-3">
   504→          <code className="rounded bg-muted px-1.5 py-0.5 text-xs font-mono">{police.username}</code>
   505→        </td>
   506→        <td className="px-4 py-3">
   507→          <Badge variant="outline" className={POLICE_RANK_BADGE[rank] || ""}>
   508→            {rank}
   509→          </Badge>
   510→        </td>
   511→        <td className="px-4 py-3 text-muted-foreground text-xs">
   512→          {new Date(police.createdAt).toLocaleDateString()}
   513→        </td>
   514→        <td className="px-4 py-3 text-right">
   515→          <div className="flex items-center justify-end gap-1.5">
   516→            <Button variant="outline" size="sm" className="gap-1.5" onClick={() => openPoliceReset(police)}>
   517→              <KeyRound className="h-3.5 w-3.5" />
   518→              Reset
   519→            </Button>
   520→            {canEdit && (
   521→              <Button variant="outline" size="sm" className="gap-1.5" onClick={() => openEditPolice(police)}>
   522→                <Pencil className="h-3.5 w-3.5" />
   523→              </Button>
   524→            )}
   525→            {canDelete && (
   526→              <Button variant="outline" size="sm" className="gap-1.5 text-destructive hover:bg-destructive/10" onClick={() => openDeletePolice(police)}>
   527→                <Trash2 className="h-3.5 w-3.5" />
   528→              </Button>
   529→            )}
   530→          </div>
   531→        </td>
   532→      </tr>
   533→    );
   534→  };
   535→
   536→  return (
   537→    <div className="space-y-6 p-4 md:p-6">
   538→      {/* Header */}
   539→      <div className="flex flex-col gap-2.5 sm:flex-row sm:items-center sm:justify-between">
   540→        <div>
   541→          <h1 className="text-2xl font-bold tracking-tight">User Management</h1>
   542→          <p className="text-sm text-muted-foreground">
   543→            Manage user accounts, roles, credentials, and provider approvals.
   544→          </p>
   545→        </div>
   546→      </div>
   547→
   548→      {/* ── RBAC Role Reference Cards ── */}
   549→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-3">
   550→        {Object.entries(ROLE_CONFIG).map(([role, config]) => {
   551→          const Icon = config.icon;
   552→          return (
   553→            <Card key={role} className="border shadow-none">
   554→              <CardContent className="p-4">
   555→                <div className="flex items-center gap-3">
   556→                  <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg bg-primary/10">
   557→                    <Icon className="size-5 text-primary" />
   558→                  </div>
