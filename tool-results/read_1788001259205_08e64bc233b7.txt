   182→
   183→function ComposeTab({
   184→  isViewer,
   185→  isPolice,
   186→  isAdmin,
   187→}: {
   188→  isViewer: boolean;
   189→  isPolice: boolean;
   190→  isAdmin: boolean;
   191→}) {
   192→  const { t } = useTranslation("notificationDispatch");
   193→
   194→  const getChannelLabel = (val: string) => {
   195→    const map: Record<string, string> = {
   196→      IN_APP: t('channelsInApp'),
   197→      SMS: t('channelsSms'),
   198→      WHATSAPP: t('channelsWhatsapp'),
   199→      TELEGRAM: t('channelsTelegram'),
   200→    };
   201→    return map[val] || val;
   202→  };
   203→  const getPriorityLabel = (val: string) => {
   204→    const map: Record<string, string> = {
   205→      LOW: t('prioritiesLow'),
   206→      NORMAL: t('prioritiesNormal'),
   207→      HIGH: t('prioritiesHigh'),
   208→      URGENT: t('prioritiesUrgent'),
   209→    };
   210→    return map[val] || val;
   211→  };
   212→
   213→  const [title, setTitle] = useState("");
   214→  const [message, setMessage] = useState("");
   215→  const [channel, setChannel] = useState("IN_APP");
   216→  const [priority, setPriority] = useState("NORMAL");
   217→  const [targetType, setTargetType] = useState<"ALL_PROVIDERS" | "SELECTED_PROVIDERS">("ALL_PROVIDERS");
   218→  const [providers, setProviders] = useState<BroadcastProvider[]>([]);
   219→  const [selectedProviders, setSelectedProviders] = useState<Set<string>>(new Set());
   220→  const [loading, setLoading] = useState(false);
   221→  const [sending, setSending] = useState(false);
   222→  const [search, setSearch] = useState("");
   223→
   224→  const fetchProviders = useCallback(async () => {
   225→    setLoading(true);
   226→    try {
   227→      const data = await apiGetBroadcastProviders();
   228→      const arr = Array.isArray(data) ? data : [];
   229→      setProviders(arr);
   230→    } catch (err) {
   231→      console.error("Failed to load providers:", err);
   232→      toast.error(t('errorLoadProviders'));
   233→    } finally {
   234→      setLoading(false);
   235→    }
   236→  }, [t]);
   237→
   238→  useEffect(() => {
   239→    fetchProviders();
   240→  }, [fetchProviders]);
   241→
   242→  // Filter providers by search
   243→  const filteredProviders = providers.filter(
   244→    (p) =>
   245→      p.name.toLowerCase().includes(search.toLowerCase()) ||
   246→      p.ownerName.toLowerCase().includes(search.toLowerCase()) ||
   247→      p.address.toLowerCase().includes(search.toLowerCase())
   248→  );
   249→
   250→  const toggleProvider = (id: string) => {
   251→    setSelectedProviders((prev) => {
   252→      const next = new Set(prev);
   253→      if (next.has(id)) next.delete(id);
   254→      else next.add(id);
   255→      return next;
   256→    });
   257→  };
   258→
   259→  const toggleAll = () => {
   260→    if (selectedProviders.size === filteredProviders.length) {
   261→      setSelectedProviders(new Set());
   262→    } else {
   263→      setSelectedProviders(new Set(filteredProviders.map((p) => p.id)));
   264→    }
   265→  };
   266→
   267→  const canSend =
   268→    !isViewer && title.trim() && message.trim() &&
   269→    (targetType === "ALL_PROVIDERS" || selectedProviders.size > 0);
   270→
   271→  const handleSend = async () => {
   272→    if (!canSend) return;
   273→    setSending(true);
   274→    try {
   275→      const result = await apiSendBroadcast({
   276→        title: title.trim(),
   277→        message: message.trim(),
   278→        channel,
   279→        priority,
   280→        targetType,
   281→        providerIds: targetType === "SELECTED_PROVIDERS" ? Array.from(selectedProviders) : [],
   282→      });
   283→      toast.success(
   284→        t('successSent', { sent: result.sent, failed: result.failed }),
   285→        { description: t('successSentDesc', { channel: getChannelLabel(channel), priority: getPriorityLabel(priority) }) }
   286→      );
   287→      // Reset form
   288→      setTitle("");
   289→      setMessage("");
   290→      setPriority("NORMAL");
   291→      setSelectedProviders(new Set());
   292→      setTargetType("ALL_PROVIDERS");
   293→    } catch (err) {
   294→      toast.error(err instanceof Error ? err.message : t('errorSendBroadcast'));
   295→    } finally {
   296→      setSending(false);
   297→    }
   298→  };
   299→
   300→  // Check if selected channel is available for selected providers
   301→  const getChannelAvailability = () => {
   302→    if (targetType === "ALL_PROVIDERS") {
   303→      const targetList = filteredProviders;
   304→      if (channel === "SMS" || channel === "WHATSAPP") {
   305→        const withPhone = targetList.filter((p) => p.hasPhone).length;
   306→        return { available: withPhone, total: targetList.length, label: t('availabilityPhone') };
   307→      }
   308→      if (channel === "TELEGRAM") {
   309→        const withTelegram = targetList.filter((p) => p.hasTelegram).length;
   310→        return { available: withTelegram, total: targetList.length, label: t('availabilityTelegram') };
   311→      }
   312→      return { available: targetList.length, total: targetList.length, label: "" };
   313→    }
   314→    const selected = providers.filter((p) => selectedProviders.has(p.id));
   315→    if (channel === "SMS" || channel === "WHATSAPP") {
   316→      const withPhone = selected.filter((p) => p.hasPhone).length;
   317→      return { available: withPhone, total: selected.length, label: t('availabilityPhone') };
   318→    }
   319→    if (channel === "TELEGRAM") {
   320→      const withTelegram = selected.filter((p) => p.hasTelegram).length;
   321→      return { available: withTelegram, total: selected.length, label: t('availabilityTelegram') };
   322→    }
   323→    return { available: selected.length, total: selected.length, label: "" };
   324→  };
   325→
   326→  const availability = getChannelAvailability();
   327→
   328→  return (
   329→    <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
   330→      {/* Left column — Compose form */}
   331→      <div className="lg:col-span-2 space-y-6">
   332→        <Card>
   333→          <CardHeader>
   334→            <CardTitle className="flex items-center gap-2">
   335→              <Radio className="size-5 text-primary" />
   336→              {t('composeTitle')}
   337→            </CardTitle>
   338→            <CardDescription>
   339→              {t('composeDescription')}
   340→              {isViewer && (
   341→                <span className="block mt-1 text-amber-600 font-medium">
   342→                  {t('viewerWarning')}
   343→                </span>
   344→              )}
   345→            </CardDescription>
   346→          </CardHeader>
   347→          <CardContent className="space-y-5">
   348→            {/* Title */}
   349→            <div className="space-y-2">
   350→              <Label htmlFor="title" className="text-sm font-medium">
   351→                {t('lblTitle')} <span className="text-rose-500">*</span>
   352→              </Label>
   353→              <Input
   354→                id="title"
   355→                placeholder={t('placeholderTitle')}
   356→                value={title}
   357→                onChange={(e) => setTitle(e.target.value)}
   358→                disabled={isViewer}
   359→              />
   360→            </div>
   361→
   362→            {/* Message */}
   363→            <div className="space-y-2">
   364→              <Label htmlFor="message" className="text-sm font-medium">
   365→                {t('lblMessage')} <span className="text-rose-500">*</span>
   366→              </Label>
   367→              <Textarea
   368→                id="message"
   369→                placeholder={t('placeholderMessage')}
   370→                value={message}
   371→                onChange={(e) => setMessage(e.target.value)}
   372→                rows={6}
   373→                disabled={isViewer}
   374→                className="resize-y"
   375→              />
   376→              <p className="text-xs text-slate-400">{message.length} {t('characters')}</p>
   377→            </div>
   378→
   379→            {/* Channel & Priority row */}
   380→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   381→              <div className="space-y-2">
   382→                <Label>{t('lbldeliveryChannel', 'Delivery Channel')}</Label>
   383→                <Select value={channel} onValueChange={setChannel} disabled={isViewer}>
   384→                  <SelectTrigger>
   385→                    <SelectValue />
   386→                  </SelectTrigger>
   387→                  <SelectContent>
   388→                    {CHANNELS.map((ch) => {
   389→                      const Icon = ch.icon;
   390→                      return (
   391→                        <SelectItem key={ch.value} value={ch.value}>
   392→                          <span className="flex items-center gap-2">
   393→                            <Icon className={`size-4 ${ch.color}`} />
   394→                            {getChannelLabel(ch.value)}
   395→                          </span>
   396→                        </SelectItem>
   397→                      );
   398→                    })}
   399→                  </SelectContent>
   400→                </Select>
   401→                {/* Channel availability warning */}
   402→                {channel !== "IN_APP" && availability.label && (
   403→                  <div className={`text-xs mt-1.5 flex items-center gap-1.5 ${
   404→                    availability.available === 0
   405→                      ? "text-rose-600"
   406→                      : availability.available < availability.total
   407→                      ? "text-amber-600"
   408→                      : "text-emerald-600"
   409→                  }`}>
   410→                    {availability.available === 0 ? (
   411→                      <XCircle className="size-3" />
   412→                    ) : availability.available < availability.total ? (
   413→                      <AlertTriangle className="size-3" />
   414→                    ) : (
   415→                      <CheckCircle2 className="size-3" />
   416→                    )}
   417→                    {t('availabilityWarning', { available: availability.available, total: availability.total, label: availability.label })}
   418→                  </div>
   419→                )}
   420→              </div>
   421→              <div className="space-y-2">
   422→                <Label>{t('lblpriorityLevel', 'Priority Level')}</Label>
   423→                <Select value={priority} onValueChange={setPriority} disabled={isViewer}>
   424→                  <SelectTrigger>
   425→                    <SelectValue />
   426→                  </SelectTrigger>
   427→                  <SelectContent>
   428→                    {PRIORITIES.map((pr) => (
   429→                      <SelectItem key={pr.value} value={pr.value}>
   430→                        <span className="flex items-center gap-2">
   431→                          <span className={`inline-block h-2 w-2 rounded-full ${
   432→                            pr.value === "URGENT"
   433→                              ? "bg-rose-500"
   434→                              : pr.value === "HIGH"
   435→                              ? "bg-amber-500"
   436→                              : pr.value === "NORMAL"
   437→                              ? "bg-blue-500"
   438→                              : "bg-slate-400"
   439→                          }`} />
   440→                          {getPriorityLabel(pr.value)}
   441→                        </span>
   442→                      </SelectItem>
   443→                    ))}
   444→                  </SelectContent>
   445→                </Select>
   446→              </div>
   447→            </div>
   448→
   449→            {/* Target Type */}
   450→            <div className="space-y-2">
   451→              <Label>{t('lbltargetAudience', 'Target Audience')}</Label>
   452→              <div className="flex items-center gap-4">
   453→                <label className="flex items-center gap-2 cursor-pointer text-sm">
   454→                  <input
   455→                    type="radio"
   456→                    name="targetType"
   457→                    value="ALL_PROVIDERS"
   458→                    checked={targetType === "ALL_PROVIDERS"}
   459→                    onChange={() => setTargetType("ALL_PROVIDERS")}
   460→                    disabled={isViewer}
   461→                    className="accent-primary"
   462→                  />
   463→                  {t('allApprovedProviders')}
   464→                </label>
   465→                <label className="flex items-center gap-2 cursor-pointer text-sm">
   466→                  <input
   467→                    type="radio"
   468→                    name="targetType"
   469→                    value="SELECTED_PROVIDERS"
   470→                    checked={targetType === "SELECTED_PROVIDERS"}
   471→                    onChange={() => setTargetType("SELECTED_PROVIDERS")}
   472→                    disabled={isViewer}
   473→                    className="accent-primary"
   474→                  />
   475→                  {t('selectedProvidersOnly')}
   476→                </label>
   477→              </div>
   478→            </div>
   479→
   480→            {/* Send button */}
   481→            <div className="flex items-center gap-3 pt-2">
   482→              <Button
   483→                onClick={handleSend}
   484→                disabled={!canSend || sending}
   485→                className="bg-gradient-to-r from-rose-500 to-amber-500 hover:from-rose-600 hover:to-amber-600 text-white shadow-md"
   486→              >
   487→                {sending ? (
   488→                  <>
   489→                    <span className="mr-2 inline-block h-4 w-4 animate-spin rounded-full border-2 border-white border-t-transparent" />
   490→                    {t('dispatching')}
   491→                  </>
   492→                ) : (
   493→                  <>
   494→                    <Send className="mr-2 size-4" />
   495→                    {t('dispatchTo')}{" "}
   496→                    {targetType === "ALL_PROVIDERS"
   497→                      ? t('allCount', { count: providers.length })
   498→                      : t('selectedCount', { count: selectedProviders.size })}{" "}
   499→                    {t('providers')}
   500→                  </>
   501→                )}
   502→              </Button>
   503→              {priority === "URGENT" && (
   504→                <Badge variant="outline" className="bg-rose-50 text-rose-600 border-rose-200 animate-pulse">
   505→                  <AlertTriangle className="mr-1 size-3" />
   506→                  {t('urgentBadge')}
   507→                </Badge>
   508→              )}
   509→            </div>
   510→          </CardContent>
   511→        </Card>
   512→      </div>
   513→
   514→      {/* Right column — Provider selection */}
   515→      <div className="space-y-4">
   516→        <Card>
   517→          <CardHeader className="pb-3">
   518→            <CardTitle className="text-base flex items-center justify-between">
   519→              <span className="flex items-center gap-2">
   520→                <Building2 className="size-4 text-primary" />
   521→                {t('providers')}
   522→              </span>
   523→              <Badge variant="secondary" className="text-xs">
   524→                {targetType === "ALL_PROVIDERS"
   525→                  ? providers.length
   526→                  : selectedProviders.size} / {providers.length}
   527→              </Badge>
   528→            </CardTitle>
   529→            {targetType === "SELECTED_PROVIDERS" && (
   530→              <>
   531→                <Input
   532→                  placeholder={t('searchProviders')}
   533→                  value={search}
   534→                  onChange={(e) => setSearch(e.target.value)}
   535→                  className="mt-2"
   536→                />
   537→                <div className="flex items-center justify-between mt-2">
   538→                  <label className="flex items-center gap-2 cursor-pointer text-xs text-slate-500">
   539→                    <Checkbox
   540→                      checked={
   541→                        filteredProviders.length > 0 &&
   542→                        selectedProviders.size === filteredProviders.length
   543→                      }
   544→                      onCheckedChange={toggleAll}
   545→                      disabled={isViewer}
   546→                    />
   547→                    {t('selectAll')}
   548→                  </label>
   549→                </div>
   550→              </>
   551→            )}
   552→          </CardHeader>
   553→          <CardContent className="p-0">
   554→            {loading ? (
   555→              <div className="space-y-3 p-4">
   556→                {Array.from({ length: 5 }).map((_, i) => (
   557→                  <div key={i} className="flex items-center gap-3">
   558→                    <Skeleton className="h-4 w-4 rounded" />
   559→                    <div className="flex-1 space-y-1.5">
   560→                      <Skeleton className="h-3.5 w-3/4" />
   561→                      <Skeleton className="h-3 w-1/2" />
   562→                    </div>
   563→                  </div>
   564→                ))}
   565→              </div>
   566→            ) : providers.length === 0 ? (
   567→              <div className="flex flex-col items-center gap-2 py-8 text-slate-400">
   568→                <Building2 className="size-8" />
   569→                <p className="text-sm">{t('emptyProviders')}</p>
   570→              </div>
   571→            ) : (
   572→              <ScrollArea className="h-[420px]">
   573→                <div className="px-4 pb-4 space-y-1">
   574→                  {(targetType === "ALL_PROVIDERS" ? providers : filteredProviders).map(
   575→                    (provider) => {

   666→      URGENT: t('prioritiesUrgent'),

   677→    setLoading(true);
