    82→function getActionIcon(action: string) {
    83→  const lower = action.toLowerCase();
    84→  if (lower === "login" || lower === "login_failed") return <LogIn className="h-3.5 w-3.5" />;
    85→  if (lower === "logout") return <LogOut className="h-3.5 w-3.5" />;
    86→  if (lower === "create") return <Plus className="h-3.5 w-3.5" />;
    87→  if (lower === "update") return <Pencil className="h-3.5 w-3.5" />;
    88→  if (lower === "delete") return <Trash2 className="h-3.5 w-3.5" />;
    89→  if (lower === "view") return <Eye className="h-3.5 w-3.5" />;
    90→  if (lower === "export" || lower === "backup") return <FileDown className="h-3.5 w-3.5" />;
    91→  return <Activity className="h-3.5 w-3.5" />;
    92→}

    94→function getActionColor(action: string): string {
    95→  const lower = action.toLowerCase();
    96→  if (lower === "login") return "bg-emerald-100 text-emerald-700 border-emerald-200";
    97→  if (lower === "logout") return "bg-slate-100 text-slate-600 border-slate-200";
    98→  if (lower === "create" || lower === "approve") return "bg-sky-100 text-sky-700 border-sky-200";
    99→  if (lower === "update") return "bg-amber-100 text-amber-700 border-amber-200";
   100→  if (lower === "delete" || lower === "suspend") return "bg-rose-100 text-rose-700 border-rose-200";
   101→  if (lower === "view") return "bg-violet-100 text-violet-600 border-violet-200";
   102→  if (lower === "export" || lower === "backup") return "bg-cyan-100 text-cyan-700 border-cyan-200";

   105→  if (lower === "config_change") return "bg-pink-100 text-pink-700 border-pink-200";

   109→function formatTimestamp(dateStr: string): string {
   110→  try {
   111→    const date = new Date(dateStr);
   112→    return date.toLocaleDateString("en-US", {
   113→      year: "numeric",
   114→      month: "short",
   115→      day: "numeric",
   116→      hour: "2-digit",
   117→      minute: "2-digit",
   118→      second: "2-digit",
   119→      hour12: true,
   120→    });
   121→  } catch {
   122→    return dateStr;
   123→  }
   124→}
   125→
   126→function isToday(dateStr: string): boolean {
   127→  try {
   128→    const date = new Date(dateStr);
   129→    const now = new Date();
   130→    return (
   131→      date.getDate() === now.getDate() &&
   132→      date.getMonth() === now.getMonth() &&
   133→      date.getFullYear() === now.getFullYear()
   134→    );
   135→  } catch {
   136→    return false;
   137→  }
   138→}
   139→
   140→function isThisWeek(dateStr: string): boolean {
   141→  try {
   142→    const date = new Date(dateStr);
   143→    const now = new Date();
   144→    const startOfWeek = new Date(now);
   145→    startOfWeek.setDate(now.getDate() - now.getDay());
   146→    startOfWeek.setHours(0, 0, 0, 0);
   147→    return date >= startOfWeek;
   148→  } catch {
   149→    return false;
   150→  }
   151→}
