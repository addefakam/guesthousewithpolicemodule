   217→// ── Helper: get nav items based on role ──
   218→function getNavItems(user: CurrentUser, disabledPages: string[] = []): NavItem[] {
   219→  const disabledSet = new Set(disabledPages);
   220→
   221→  switch (user.role) {
   222→    case "POLICE": {
   223→      const rank = (user.policeRank || "OFFICER") as PoliceRank;
   224→      const allowedPages = POLICE_RANK_PERMISSIONS[rank] || POLICE_RANK_PERMISSIONS.OFFICER;
   225→      return POLICE_NAV_ITEMS.filter((item) => allowedPages.includes(item.page));
   226→    }
   227→
   228→    case "SUPERUSER": {
   229→      // Guest house owners (SUPERUSER with providerId) see the provider dashboard.
   230→      // System admins (SUPERUSER without providerId) see the admin dashboard.
   231→      if (user.providerId) {
   232→        return ALL_NAV_ITEMS.filter(
   233→          (item) => !OPERATOR_EXCLUDED.has(item.page) && !disabledSet.has(item.page)
   234→        );
   235→      }
   236→      return SUPERUSER_NAV_ITEMS;
   237→    }
   238→
   239→    case "OPERATOR":
   240→      return ALL_NAV_ITEMS.filter(
   241→        (item) => !OPERATOR_EXCLUDED.has(item.page) && !disabledSet.has(item.page)
   242→      );
   243→
   244→    case "STAFF": {
   245→      const items: NavItem[] = [];
   246→      // Always give staff access to dashboard
   247→      if (!disabledSet.has("dashboard")) {
   248→        items.push({
   249→          page: "dashboard",
   250→          label: "Dashboard",
   251→          icon: LayoutDashboard,
   252→        });
   253→      }
   254→      // Add items based on permissions (deduplicate by page key)
   255→      const seen = new Set<string>();
   256→      for (const perm of user.permissions) {
   257→        const mapped = PERMISSION_PAGE_MAP[perm];
   258→        if (mapped && !seen.has(mapped.page) && !disabledSet.has(mapped.page)) {
   259→          seen.add(mapped.page);
   260→          items.push(mapped);
   261→        }
   262→      }
   263→      return items;
   264→    }
   265→
   266→    default:
   267→      return [
   268→        { page: "dashboard", label: "Dashboard", icon: LayoutDashboard },
   269→      ];
   270→  }
   271→}

   273→// ── Helper: get initials from name ──
   274→function getInitials(name: string | undefined | null): string {
   275→  if (!name) return "?";
   276→  return name
   277→    .split(" ")
   278→    .map((w) => w[0])
   279→    .join("")
   280→    .toUpperCase()
   281→    .slice(0, 2);
   282→}

   800→            className="absolute -right-3 top-5 flex h-6 w-6 items-center justify-center rounded-full border border-slate-200 bg-white shadow-sm transition-colors hover:bg-slate-50"
