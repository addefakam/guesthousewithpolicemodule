   207→// ── Subscription Status Badge (for Rooms page header) ──
   208→function SubscriptionBadge() {
   209→  const { t } = useTranslation("rooms");
   210→  const { currentUser, subscription } = useAppStore();
   211→
   212→  // Only show for OPERATOR/STAFF
   213→  if (!currentUser || (currentUser.role !== "OPERATOR" && currentUser.role !== "STAFF")) return null;
   214→  if (!subscription) return null;
   215→
   216→  const isActive = subscription.status === "ACTIVE";
   217→  const isWarning = subscription.status === "WARNING";
   218→  const isExpired = subscription.status === "EXPIRED";
   219→  const isSuspended = subscription.status === "SUSPENDED";
   220→
   221→  const badgeClass = isActive
   222→    ? "border-emerald-300 bg-emerald-50 text-emerald-700"
   223→    : isSuspended || isExpired
   224→    ? "border-rose-300 bg-rose-50 text-rose-700"
   225→    : "border-amber-300 bg-amber-50 text-amber-700";
   226→
   227→  const dotClass = isActive
   228→    ? "bg-emerald-500"
   229→    : isSuspended || isExpired
   230→    ? "bg-rose-500"
   231→    : "bg-amber-500";
   232→
   233→  const label = isActive
   234→    ? t("subActive")
   235→    : isSuspended
   236→    ? t("subSuspended")
   237→    : isExpired
   238→    ? t("subExpired")
   239→    : t("subExpiring");
   240→
   241→  return (
   242→    <div className={`inline-flex items-center gap-1.5 rounded-full border px-2.5 py-1 text-xs font-semibold ${badgeClass} animate-subtle-pulse`}>
   243→      <span className={`relative flex h-2 w-2`}>
   244→        <span className={`absolute inline-flex h-full w-full animate-ping rounded-full opacity-75 ${dotClass}`} />
   245→        <span className={`relative inline-flex h-2 w-2 rounded-full ${dotClass}`} />
   246→      </span>
   247→      <span>{label}</span>
   248→      <span className="font-mono text-[11px] opacity-70">{Math.abs(subscription.daysRemaining)}d</span>
   249→    </div>
   250→  );
   251→}
