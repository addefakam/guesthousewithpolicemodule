   570→/** Execute SQL via raw pg Client */
   571→async function execViaPg(sql: string, params?: unknown[]): Promise<void> {
   572→  const client = new Client({ connectionString: process.env.DATABASE_URL! });
   573→  try {
   574→    await client.connect();
   575→    await client.query(sql, params);
   576→  } finally {
   577→    await client.end().catch(() => {});
   578→  }
   579→}

   581→/** Execute DDL via Prisma $executeRawUnsafe — splits multi-statement SQL
   582→ *  because Prisma's driver does NOT support multiple statements per call. */
   583→async function execViaPrisma(sql: string): Promise<void> {
   584→  // Split on `DO $$` blocks and standalone statements
   585→  // Each `DO $$ ... END $$;` is one executable unit
   586→  const statements = sql
   587→    .split(/(?=DO \$\$)/)
   588→    .map(s => s.trim())
   589→    .filter(s => s.length > 0);
   590→
   591→  // If no DO blocks found, split by semicolons (for CREATE INDEX etc.)
   592→  const parts = statements.length > 1
   593→    ? statements
   594→    : sql.split(';').map(s => s.trim()).filter(s => s.length > 0);
   595→
   596→  const prisma = new PrismaClient({
   597→    log: ["warn", "error"],
   598→  });
   599→  try {
   600→    for (const part of parts) {
   601→      const stmt = part.endsWith(';') ? part : part + ';';
   602→      try {
   603→        await prisma.$executeRawUnsafe(stmt);
   604→      } catch (err) {
   605→        // Individual statement might fail (e.g. duplicate_object) — log but continue
   606→        console.error("[execViaPrisma] Statement failed (continuing):", stmt.slice(0, 80), err instanceof Error ? err.message : String(err));
   607→      }
   608→    }
   609→  } finally {
   610→    await prisma.$disconnect().catch(() => {});
   611→  }
   612→}

   618→async function runMigrationsOnly(): Promise<void> {
   619→  console.log("[init-db] Running idempotent migrations on existing DB...");
   620→  try {
   621→    await execViaPrisma(MIGRATIONS_SQL);
   622→    console.log("[init-db] Background migrations applied.");
   623→  } catch (err) {
   624→    throw err;
   625→  }
   626→  try {
   627→    await execViaPrisma(INDEXES_SQL);
   628→  } catch {
   629→    // Indexes are best-effort
   630→  }
   631→}

   633→export async function ensureDatabase(): Promise<void> {
   634→  if (_initDone) {
   635→    // Always run migrations synchronously to ensure columns exist
   636→    // before any query executes. Migrations are idempotent.
   637→    if (!_migrationsRan) {
   638→      _migrationsRan = true;
   639→      await runMigrationsOnly();
   640→    }
   641→    return;
   642→  }
   643→  if (_initPromise) return _initPromise;
   644→
   645→  if (!process.env.DATABASE_URL) {
   646→    throw new Error("DATABASE_URL environment variable is not set.");
   647→  }
   648→
   649→  _initPromise = (async () => {
   650→    console.log("[init-db] Starting database initialization...");
   651→
   652→    let usePg = true;
   653→    let pgError: unknown = null;
   654→
   655→    // Strategy 1: Try using pg Client directly
   656→    try {
   657→      const client = new Client({ connectionString: process.env.DATABASE_URL });
   658→      await client.connect();
   659→      console.log("[init-db] pg Client connected successfully.");
   660→
   661→      // Check if User table already exists
   662→      const res = await client.query(
   663→        `SELECT EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema='public' AND table_name='User') as ok`
   664→      );
   665→      const tableExists = res.rows[0]?.ok === true;
   666→      console.log("[init-db] User table exists:", tableExists);
   667→
   668→      if (tableExists) {
   669→        console.log("[init-db] Tables already exist, running migrations to add any missing columns...");
   670→        try {
   671→          await client.query(MIGRATIONS_SQL);
   672→          console.log("[init-db] Migrations applied to existing tables.");
   673→          await client.query(INDEXES_SQL);
   674→          console.log("[init-db] Indexes verified.");
   675→        } catch (migrateErr) {
   676→          console.error("[init-db] Migration on existing tables failed:", migrateErr instanceof Error ? migrateErr.message : String(migrateErr));
   677→          // Don't throw — the tables exist, migrations are best-effort
   678→        }
   679→        await client.end().catch(() => {});
   680→        _initDone = true;
   681→        return;
   682→      }
   683→
   684→      // Create enums
   685→      console.log("[init-db] Creating enum types...");
   686→      await client.query(ENUMS_SQL);
   687→
   688→      // Create tables
   689→      console.log("[init-db] Creating tables...");
   690→      await client.query(TABLES_SQL);
   691→
   692→      // Verify User table was created
   693→      const verify = await client.query(
   694→        `SELECT EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema='public' AND table_name='User') as ok`
   695→      );
   696→      if (!verify.rows[0]?.ok) {
   697→        throw new Error("DDL executed but User table still not found!");
   698→      }
   699→      console.log("[init-db] Tables verified.");
   700→
   701→      // Create foreign keys
   702→      console.log("[init-db] Creating foreign keys...");
   703→      await client.query(FKS_SQL);
   704→
   705→      // Run migrations (add new columns to existing tables)
   706→      console.log("[init-db] Running migrations...");
   707→      await client.query(MIGRATIONS_SQL);
   708→
   709→      // Create indexes
   710→      console.log("[init-db] Creating indexes...");
   711→      await client.query(INDEXES_SQL);

   761→      try {

   763→          `SELECT EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema='public' AND table_name='User') as ok`
   764→        );
   765→        if (res[0]?.ok) {
   766→          console.log("[init-db] Tables already exist (verified via Prisma), running migrations...");
   767→          try {
   768→            await execViaPrisma(MIGRATIONS_SQL);
   769→            console.log("[init-db] Migrations applied via Prisma.");
   770→            await execViaPrisma(INDEXES_SQL);
   771→            console.log("[init-db] Indexes verified via Prisma.");
   772→          } catch (migrateErr) {
   773→            console.error("[init-db] Prisma migration on existing tables failed:", migrateErr instanceof Error ? migrateErr.message : String(migrateErr));
   774→          }
   775→          _initDone = true;
   776→          return;
   777→        }
   778→      } catch {
   779→        // Table might not exist, which is expected
   780→        console.log("[init-db] Tables do not exist yet, creating via Prisma...");

   791→
   792→      await execViaPrisma(FKS_SQL);

   794→
   795→      await execViaPrisma(MIGRATIONS_SQL);
   796→      console.log("[init-db] Migrations ran via Prisma.");
   797→
   798→      await execViaPrisma(INDEXES_SQL);
   799→      console.log("[init-db] Indexes created via Prisma.");
   800→
   801→      // Seed SUPERUSER via Prisma
   802→      const prisma2 = new PrismaClient({ log: ["warn", "error"] });
