   194→// ── Helper: get nav items based on role ──
   195→function getNavItems(user: CurrentUser): NavItem[] {
   196→  switch (user.role) {
   197→    case "POLICE": {
   198→      const rank = (user.policeRank || "OFFICER") as PoliceRank;
   199→      const allowedPages = POLICE_RANK_PERMISSIONS[rank] || POLICE_RANK_PERMISSIONS.OFFICER;
   200→      return POLICE_NAV_ITEMS.filter((item) => allowedPages.includes(item.page));
   201→    }
   202→
   203→    case "SUPERUSER":
   204→      return SUPERUSER_NAV_ITEMS;
   205→
   206→    case "OPERATOR":
   207→      return ALL_NAV_ITEMS.filter((item) => !OPERATOR_EXCLUDED.has(item.page));
   208→
   209→    case "STAFF": {
   210→      const items: NavItem[] = [];
   211→      // Always give staff access to dashboard
   212→      items.push({
   213→        page: "dashboard",
   214→        label: "Dashboard",
   215→        icon: LayoutDashboard,
   216→      });
   217→      // Add items based on permissions
   218→      for (const perm of user.permissions) {
   219→        const mapped = PERMISSION_PAGE_MAP[perm];
   220→        if (mapped) {
   221→          items.push(mapped);
   222→        }
   223→      }
   224→      return items;
   225→    }
   226→
   227→    default:
   228→      return [
   229→        { page: "dashboard", label: "Dashboard", icon: LayoutDashboard },
   230→      ];
   231→  }
   232→}

   234→// ── Helper: get initials from name ──
   235→function getInitials(name: string): string {
   236→  return name
   237→    .split(" ")
   238→    .map((w) => w[0])
   239→    .join("")
   240→    .toUpperCase()
   241→    .slice(0, 2);
   242→}

   321→// ── Anomaly Detection Toggle Widget (sidebar inline) ──
   322→function AnomalyToggle({ collapsed }: { collapsed: boolean }) {
   323→  const currentUser = useAppStore((s) => s.currentUser);
   324→  const isAdmin = currentUser?.role === "POLICE" && currentUser?.policeRank === "ADMIN";
   325→  const [enabled, setEnabled] = useState(false);
   326→  const [toggling, setToggling] = useState(false);
   327→
   328→  const fetchStatus = useCallback(async () => {
   329→    try {
   330→      const data = await apiGetAnomalies("pageSize=1");
   331→      if (data.enabled !== undefined) setEnabled(data.enabled);
   332→    } catch { /* non-critical */ }
   333→  }, []);
   334→
   335→  useEffect(() => {
   336→    fetchStatus();
   337→  }, [fetchStatus]);
   338→
   339→  const handleToggle = async () => {
   340→    if (!isAdmin || toggling) return;
   341→    try {
   342→      setToggling(true);
   343→      const result = await apiToggleAnomalyDetection(!enabled);
   344→      setEnabled(result.enabled);
   345→    } catch { /* non-critical */ }
   346→    finally { setToggling(false); }
   347→  };
   348→
   349→  // Collapsed: small indicator dot
   350→  if (collapsed) {
   351→    return (
   352→      <div className="flex justify-center py-1" title={enabled ? "Anomaly Detection: Active" : "Anomaly Detection: Inactive"}>
   353→        <button
   354→          onClick={handleToggle}
   355→          disabled={!isAdmin}
   356→          className={`h-3 w-3 rounded-full transition-colors ${
   357→            enabled ? "bg-violet-500 shadow-sm shadow-violet-200" : "bg-slate-300"
   358→          } ${isAdmin ? "cursor-pointer hover:ring-2 hover:ring-violet-300" : "cursor-not-allowed"}`}
   359→        />
   360→      </div>
   361→    );
   362→  }
   363→
   364→  // Expanded: inline toggle row
   365→  return (
   366→    <button
   367→      onClick={handleToggle}
   368→      disabled={!isAdmin || toggling}
   369→      className={`group flex w-full items-center gap-2.5 rounded-lg px-3 py-2 text-xs font-medium transition-colors outline-none ${
   370→        isAdmin
   371→          ? enabled
   372→            ? "text-violet-700 hover:bg-violet-50"
   373→            : "text-slate-500 hover:bg-slate-100"
   374→          : "text-slate-400 cursor-not-allowed"
   375→      }`}
   376→      title={!isAdmin ? "Only ADMIN can toggle detection" : enabled ? "Click to disable" : "Click to enable"}
   377→    >
   378→      {toggling ? (
   379→        <Loader2 className="size-3.5 animate-spin shrink-0" />
   380→      ) : enabled ? (
   381→        <ToggleRight className="size-3.5 shrink-0 text-violet-500" />
   382→      ) : (
   383→        <ToggleLeft className="size-3.5 shrink-0" />
   384→      )}
   385→      <span className="truncate">
   386→        {enabled ? "Detection Active" : "Detection Inactive"}
   387→      </span>
   388→      {/* Mini toggle switch */}
   389→      <span
   390→        className={`ml-auto inline-flex h-5 w-9 shrink-0 items-center rounded-full transition-colors duration-200 ${
   391→          enabled ? "bg-violet-500" : "bg-slate-300"
   392→        }`}
   393→      >
   394→        <span
   395→          className={`inline-block h-3.5 w-3.5 transform rounded-full bg-white shadow-sm transition-transform duration-200 ${
   396→            enabled ? "translate-x-4" : "translate-x-0.5"
   397→          }`}
   398→        />
   399→      </span>
   400→    </button>
   401→  );
   402→}

   406→function SubscriptionStatusCard({ collapsed }: { collapsed: boolean }) {
   407→  const subscription = useAppStore((s) => s.subscription);
   408→
   409→  if (!subscription) return null;
   410→
   411→  const isActive = subscription.status === "ACTIVE";
   412→  const isWarning = subscription.status === "WARNING";
   413→  const isGrace = subscription.status === "GRACE";
   414→  const isSuspended = subscription.status === "SUSPENDED";
   415→
   416→  // Collapsed mode — icon indicator
   417→  if (collapsed) {
   418→    const bgClass = isActive
   419→      ? "bg-emerald-100"
   420→      : isSuspended || isGrace
   421→      ? "bg-rose-100"
   422→      : "bg-amber-100";
   423→    const iconClass = isActive
   424→      ? "text-emerald-600"
   425→      : isSuspended || isGrace
   426→      ? "text-rose-600"
   427→      : "text-amber-600";
   428→    return (
   429→      <div className="flex flex-col items-center gap-1 px-2 py-2">
   430→        <div className={`flex h-8 w-8 items-center justify-center rounded-full ${bgClass} animate-pulse shadow-sm`}> 
   431→          <Clock className={`size-4 ${iconClass}`} />
   432→        </div>
   433→      </div>
   434→    );
   435→  }
   436→
   437→  // Expanded card
   438→  const containerClass = isActive
   439→    ? "border-emerald-200 bg-emerald-50"
   440→    : isSuspended || isGrace
   441→    ? "border-rose-200 bg-rose-50"
   442→    : "border-amber-200 bg-amber-50";
   443→
   444→  const iconColorClass = isActive
   445→    ? "text-emerald-600"
   446→    : isSuspended || isGrace
   447→    ? "text-rose-600"
   448→    : "text-amber-600";
   449→
   450→  const titleClass = isActive
   451→    ? "text-emerald-800"
   452→    : isSuspended || isGrace
   453→    ? "text-rose-800"
   454→    : "text-amber-800";
   455→
   456→  const subtextClass = isActive
   457→    ? "text-emerald-700"
   458→    : isSuspended || isGrace
   459→    ? "text-rose-700"
   460→    : "text-amber-700";
   461→

   463→    ? "bg-emerald-200/60 text-emerald-800"
   464→    : isSuspended || isGrace
   465→    ? "bg-rose-200/60 text-rose-800"
   466→    : "bg-amber-200/60 text-amber-800";
   467→
   468→  const Icon = isSuspended || isGrace ? AlertTriangle : Clock;
   469→
   470→  return (
   471→    <div className={`rounded-lg border p-3 ${containerClass} animate-subtle-pulse shadow-sm`}> 
   472→      <div className="flex items-center gap-2">
   473→        <Icon className={`h-4 w-4 shrink-0 ${iconColorClass}`} />

   475→          {isActive
   476→            ? "Subscription Active"
   477→            : isSuspended
   478→            ? "Service Suspended"
   479→            : isGrace
   480→            ? "Subscription Expired"
   481→            : "Expiring Soon"}
   482→        </p>
   483→      </div>
   484→      <div className={`mt-1.5 flex items-center justify-between text-[11px] ${subtextClass}`}>
   485→        <span>{formatDaysRemaining(subscription.daysRemaining)}</span>

   487→          <Clock className="h-3 w-3" />
   488→          {Math.abs(subscription.daysRemaining)}d
   489→        </span>
   490→      </div>

   492→        <div className={`mt-1 text-[11px] ${subtextClass}`}>
   493→          Due: {subscription.price.toLocaleString()} ETB / {formatCycle(subscription.cycle)}

   495→      ) : (
   496→        <div className={`mt-1 text-[11px] ${subtextClass}`}>
   497→          {formatCycle(subscription.cycle)} cycle — Free trial
   498→        </div>
   499→      )}
   500→    </div>
   501→  );
   502→}

   579→                      <LogOut className="size-4" />
   580→                      <span>Sign Out</span>
   581→                    </DropdownMenuItem>
   582→                  </DropdownMenuContent>
   583→                </DropdownMenu>
   584→              ) : (
   585→                <div className="flex items-center justify-between gap-1">
   586→                  <p className="truncate text-sm font-semibold text-slate-900">
   587→                    {user.name}
   588→                  </p>
   589→                  <button
   590→                    onClick={onLogout}
   591→                    className="rounded-md p-1 text-slate-400 hover:bg-slate-100 hover:text-rose-600 transition-colors"
   592→                    title="Logout"
   593→                  >
   594→                    <LogOut className="size-4 shrink-0" />
   595→                  </button>
   596→                </div>
   597→              )}
   598→              <div className="mt-0.5 flex items-center gap-2">
   599→                <Badge
   600→                  variant="outline"
   601→                  className={`text-[10px] font-semibold leading-none px-1.5 py-0.5 ${roleDisplay.className}`}
   602→                >
   603→                  <Shield className="mr-1 size-2.5" />
   604→                  {roleDisplay.label}
   605→                </Badge>
   606→                {user.role === "POLICE" && user.policeRank && (
   607→                  <Badge
   608→                    variant="outline"
   609→                    className={`text-[9px] font-semibold leading-none px-1.5 py-0.5 ${RANK_BADGE_CLASSES[(user.policeRank as PoliceRank)] || ""}`}
   610→                  >
   611→                    {RANK_LABELS[(user.policeRank as PoliceRank)] || user.policeRank}
   612→                  </Badge>
   613→                )}
   614→                {user.providerName && (
   615→                  <span className="truncate text-[11px] text-slate-400">
   616→                    {user.providerName}
   617→                  </span>
   618→                )}
   619→              </div>
   620→            </div>
   621→          )}
   622→        </div>
   623→      </div>
   624→
   625→      <Separator className="bg-slate-200/60" />
   626→
   627→      {/* ── Navigation links ── */}
   628→      <ScrollArea className="flex-1 px-3 py-3">
   629→        <nav className="flex flex-col gap-1" aria-label="Main navigation">
   630→          {navItems.map((item) => (
   631→            <React.Fragment key={item.page}>
   632→              <NavItemButton
   633→                item={item}
   634→                currentPage={currentPage}
   635→                onClick={() => onNavigate(item.page)}
   636→              />

   638→              {item.page === "anomaly-detection" && user.role === "POLICE" && (
   639→                <AnomalyToggle collapsed={collapsed} />
   640→              )}
   641→            </React.Fragment>
   642→          ))}
   643→
   644→          {/* Joint Operations — only shown during active joint session */}
   645→          {jointSession.active && (
   646→            <>
   647→              <Separator className="my-2 bg-slate-200/60" />
   648→              <NavItemButton
   649→                item={{ page: "joint-operations", label: "Joint Operations", icon: ShieldCheck }}
   650→                currentPage={currentPage}
   651→                onClick={() => onNavigate("joint-operations")}
   652→              />
   653→            </>
   654→          )}
   655→
   656→          {/* Start Joint Session button — shown for SUPERUSER/POLICE ADMIN when no joint session */}
   657→          {canStartJoint && !jointSession.active && !collapsed && (
   658→            <>
   659→              <Separator className="my-2 bg-slate-200/60" />
   660→              <button
   661→                onClick={() => setJointLoginDialogOpen(true)}
   662→                className="flex w-full items-center gap-3 rounded-lg border border-dashed border-amber-300 bg-amber-50/50 px-3 py-2.5 text-sm font-medium text-amber-700 transition-all hover:bg-amber-50 hover:border-amber-400 hover:text-amber-800 outline-none focus-visible:ring-2 focus-visible:ring-amber-400"
   663→              >
   664→                <ShieldCheck className="size-[18px] shrink-0" />
   665→                <span>Start Joint Session</span>
   666→              </button>
   667→            </>
   668→          )}
   669→          {canStartJoint && !jointSession.active && collapsed && (
   670→            <button
   671→              onClick={() => setJointLoginDialogOpen(true)}
   672→              className="flex items-center justify-center rounded-lg border border-dashed border-amber-300 p-2 text-amber-600 hover:bg-amber-50 transition-colors outline-none"
   673→              title="Start Joint Session"
   674→            >
   675→              <ShieldCheck className="size-[18px]" />
   676→            </button>
   677→          )}
   678→        </nav>
   679→      </ScrollArea>
   680→
   681→      {/* ── Subscription status for providers ── */}
   682→      {(user.role === "OPERATOR" || user.role === "STAFF") && (
   683→        <div className="px-3 py-2">
   684→          <SubscriptionStatusCard collapsed={collapsed} />
   685→        </div>
   686→      )}
   687→
   688→      <Separator className="bg-slate-200/60" />
   689→
   690→      {/* ── Bottom section: collapse toggle + logout ── */}
   691→      <div className="p-3">
   692→        {/* Desktop collapse toggle */}
   693→        {!collapsed && (
   694→          <button
   695→            onClick={onToggleCollapse}
   696→            className="mb-1 flex w-full items-center gap-3 rounded-lg px-3 py-2 text-sm font-medium text-slate-500 transition-colors hover:bg-slate-100 hover:text-slate-700 outline-none focus-visible:ring-2 focus-visible:ring-ring"
   697→          >
   698→            <ChevronLeft className="size-[18px] text-slate-400" />
   699→            <span>Collapse</span>
   700→          </button>
   701→        )}
   702→
   703→        {/* Logout — hidden for SUPERUSER (moved to name dropdown) */}
   704→        {user.role !== "SUPERUSER" && (
   705→          <button
   706→            onClick={onLogout}
   707→            className="flex w-full items-center gap-3 rounded-lg px-3 py-2.5 text-sm font-medium text-rose-600 transition-colors hover:bg-rose-50 outline-none focus-visible:ring-2 focus-visible:ring-ring"
   708→          >
   709→            <LogOut className="size-[18px]" />
   710→            {!collapsed && <span>Logout</span>}
   711→          </button>
   712→        )}
   713→      </div>
   714→    </div>
   715→  );
   716→}
