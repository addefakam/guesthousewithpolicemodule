    51→
    52→interface SettingsData {
    53→  id: string | null;
    54→  guestHouseName: string;
    55→  ownerName: string;
    56→  address: string;
    57→  phone: string;
    58→  email: string;
    59→  currency: string;
    60→  taxRate: number;
    61→  language: string;
    62→  checkInTime: string;
    63→  checkOutTime: string;
    64→}
    65→
    66→const DEFAULTS: SettingsData = {
    67→  id: null,
    68→  guestHouseName: "Guest House",
    69→  ownerName: "",
    70→  address: "",
    71→  phone: "",
    72→  email: "",
    73→  currency: "ETB",
    74→  taxRate: 0,
    75→  language: "en",
    76→  checkInTime: "14:00",
    77→  checkOutTime: "12:00",
    78→};
    79→
    80→function ProviderSettings() {
    81→  const { refreshKey, triggerRefresh } = useAppStore();
    82→  const [settings, setSettings] = useState<SettingsData>(DEFAULTS);
    83→  const [loading, setLoading] = useState(true);
    84→  const [saving, setSaving] = useState(false);
    85→  const [exporting, setExporting] = useState(false);
    86→  const [importing, setImporting] = useState(false);
    87→  const fileRef = useRef<HTMLInputElement>(null);
    88→
    89→  const fetchSettings = useCallback(async () => {
    90→    setLoading(true);
    91→    try {
    92→      const data = await apiGetSettings();
    93→      setSettings({
    94→        id: data.id || null,
    95→        guestHouseName: data.guestHouseName ?? DEFAULTS.guestHouseName,
    96→        ownerName: data.ownerName ?? "",
    97→        address: data.address ?? "",
    98→        phone: data.phone ?? "",
    99→        email: data.email ?? "",
   100→        currency: data.currency ?? "ETB",
   101→        taxRate: data.taxRate ?? 0,
   102→        language: data.language ?? "en",
   103→        checkInTime: data.checkInTime ?? "14:00",
   104→        checkOutTime: data.checkOutTime ?? "12:00",
   105→      });
   106→    } catch {
   107→      toast.error("Failed to load settings");
   108→    } finally {
   109→      setLoading(false);
   110→    }
   111→  }, []);
   112→
   113→  useEffect(() => {
   114→    fetchSettings();
   115→  }, [fetchSettings, refreshKey]);
   116→
   117→  const update = (key: keyof SettingsData, value: string | number) => {
   118→    setSettings((prev) => ({ ...prev, [key]: value }));
   119→  };
   120→
   121→  const handleSave = async () => {
   122→    setSaving(true);
   123→    try {
   124→      await apiUpdateSettings({
   125→        guestHouseName: settings.guestHouseName,
   126→        ownerName: settings.ownerName,
   127→        address: settings.address,
   128→        phone: settings.phone,
   129→        email: settings.email,
   130→        currency: settings.currency,
   131→        taxRate: settings.taxRate,
   132→        language: settings.language,
   133→        checkInTime: settings.checkInTime,
   134→        checkOutTime: settings.checkOutTime,
   135→      });
   136→      toast.success("Settings saved");
   137→      triggerRefresh();
   138→    } catch (err: unknown) {
   139→      const msg = err instanceof Error ? err.message : "Failed to save settings";
   140→      toast.error(msg);
   141→    } finally {
   142→      setSaving(false);
   143→    }
   144→  };
   145→
   146→  const handleExport = async () => {
   147→    setExporting(true);
   148→    try {
   149→      const data = await apiExportData();
   150→      const json = JSON.stringify(data, null, 2);
   151→      const blob = new Blob([json], { type: "application/json" });
   152→      const url = URL.createObjectURL(blob);
   153→      const a = document.createElement("a");
   154→      a.href = url;
   155→      a.download = `ghms-export-${new Date().toISOString().split("T")[0]}.json`;
   156→      a.click();
   157→      URL.revokeObjectURL(url);
   158→      toast.success("Data exported successfully");
   159→    } catch (err: unknown) {
   160→      const msg = err instanceof Error ? err.message : "Export failed";
   161→      toast.error(msg);
   162→    } finally {
   163→      setExporting(false);
   164→    }
   165→  };
   166→
   167→  const handleImport = async (e: React.ChangeEvent<HTMLInputElement>) => {
   168→    const file = e.target.files?.[0];
   169→    if (!file) return;
   170→
   171→    setImporting(true);
   172→    try {
   173→      const text = await file.text();
   174→      const data = JSON.parse(text);
   175→      await apiImportData(data);
   176→      toast.success("Data imported successfully");
   177→      triggerRefresh();
   178→    } catch (err: unknown) {
   179→      const msg = err instanceof Error ? err.message : "Import failed";
   180→      toast.error(msg);
   181→    } finally {
   182→      setImporting(false);
   183→      if (fileRef.current) fileRef.current.value = "";
   184→    }
   185→  };
   186→
   187→  if (loading) {
   188→    return (
   189→      <div className="space-y-6 p-4 md:p-6">
   190→        <Skeleton className="h-8 w-40" />
   191→        <Skeleton className="h-96 rounded-xl" />
   192→      </div>
   193→    );
   194→  }
   195→
   196→  return (
   197→    <div className="space-y-6 max-w-2xl p-4 md:p-6">
   198→      <div>
   199→        <h1 className="text-2xl font-bold tracking-tight">Settings</h1>
   200→        <p className="text-sm text-muted-foreground">
   201→          Configure your guest house preferences.
   202→        </p>
   203→      </div>
   204→
   205→      {/* General Settings */}
   206→      <Card>
   207→        <CardHeader>
   208→          <CardTitle className="flex items-center gap-2">
   209→            <Settings className="h-5 w-5" />
   210→            General Settings
   211→          </CardTitle>
   212→          <CardDescription>
   213→            Basic information about your guest house.
   214→          </CardDescription>
   215→        </CardHeader>
   216→        <CardContent className="space-y-4">
   217→          <div className="grid gap-2">
   218→            <Label htmlFor="ghName">Guest House Name</Label>
   219→            <Input
   220→              id="ghName"
   221→              value={settings.guestHouseName}
   222→              onChange={(e) => update("guestHouseName", e.target.value)}
   223→              placeholder="My Guest House"
   224→            />
   225→          </div>
   226→          <div className="grid gap-2">
   227→            <Label htmlFor="ownerName">Owner Name</Label>
   228→            <Input
   229→              id="ownerName"
   230→              value={settings.ownerName}
   231→              onChange={(e) => update("ownerName", e.target.value)}
   232→              placeholder="Owner full name"
   233→            />
   234→          </div>
   235→          <div className="grid gap-2">
   236→            <Label htmlFor="address">Address</Label>
   237→            <Input
   238→              id="address"
   239→              value={settings.address}
   240→              onChange={(e) => update("address", e.target.value)}
   241→              placeholder="Street, City, Country"
   242→            />
   243→          </div>
   244→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   245→            <div className="grid gap-2">
   246→              <Label htmlFor="phone">Phone</Label>
   247→              <Input
   248→                id="phone"
   249→                value={settings.phone}
   250→                onChange={(e) => update("phone", e.target.value)}
   251→                placeholder="+251..."
   252→              />
   253→            </div>
   254→            <div className="grid gap-2">
   255→              <Label htmlFor="email">Email</Label>
   256→              <Input
   257→                id="email"
   258→                type="email"
   259→                value={settings.email}
   260→                onChange={(e) => update("email", e.target.value)}
   261→                placeholder="info@example.com"
   262→              />
   263→            </div>
   264→          </div>
   265→          <div className="flex justify-end pt-2">
   266→            <Button onClick={handleSave} disabled={saving}>
   267→              {saving ? (
   268→                <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   269→              ) : (
   270→                <Save className="mr-2 h-4 w-4" />
   271→              )}
   272→              Save Settings
   273→            </Button>
   274→          </div>
   275→        </CardContent>
   276→      </Card>
   277→
   278→      {/* Data Management */}
   279→      <Card>
   280→        <CardHeader>
   281→          <CardTitle>Data Management</CardTitle>
   282→          <CardDescription>
   283→            Export or import all guest house data as JSON.
   284→          </CardDescription>
   285→        </CardHeader>
   286→        <CardContent className="space-y-4">
   287→          <div className="flex flex-col sm:flex-row gap-3">
   288→            <Button
   289→              variant="outline"
   290→              className="flex-1"
   291→              onClick={handleExport}
   292→              disabled={exporting}
   293→            >
   294→              {exporting ? (
   295→                <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   296→              ) : (
   297→                <Download className="mr-2 h-4 w-4" />
   298→              )}
   299→              Export All Data
   300→            </Button>
   301→
   302→            <input
   303→              ref={fileRef}
   304→              type="file"
   305→              accept=".json"
   306→              className="hidden"
   307→              onChange={handleImport}
   308→            />
   309→            <Button
   310→              variant="outline"
   311→              className="flex-1"
   312→              onClick={() => fileRef.current?.click()}
   313→              disabled={importing}
   314→            >
   315→              {importing ? (
   316→                <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   317→              ) : (
   318→                <Upload className="mr-2 h-4 w-4" />
   319→              )}
   320→              Import Data
   321→            </Button>
   322→          </div>
   323→          <p className="text-xs text-muted-foreground">
   324→            Export downloads a JSON file with all your data. Importing will merge
   325→            data by upserting records by ID.
   326→          </p>
   327→        </CardContent>
   328→      </Card>
   329→    </div>
   330→  );
   331→}

   334→// SUPERUSER Profile Settings
   335→// ═══════════════════════════════════════════════════════
   336→
   337→function SuperuserSettings() {
   338→  const { currentUser, setCurrentUser, triggerRefresh } = useAppStore();
   339→  const [loading, setLoading] = useState(false);
   340→
   341→  // Profile form
   342→  const [name, setName] = useState(currentUser?.name || "");
   343→  const [username, setUsername] = useState(currentUser?.username || "");
   344→  const [email, setEmail] = useState("");
   345→  const [phone, setPhone] = useState("");
   346→  const [saving, setSaving] = useState(false);
   347→
   348→  // Password form
   349→  const [currentPassword, setCurrentPassword] = useState("");
   350→  const [newPassword, setNewPassword] = useState("");
   351→  const [confirmPassword, setConfirmPassword] = useState("");
   352→  const [showCurrent, setShowCurrent] = useState(false);
   353→  const [showNew, setShowNew] = useState(false);
   354→  const [changingPassword, setChangingPassword] = useState(false);
   355→
   356→  // System settings form
   357→  const [appName, setAppName] = useState("GHMS");
   358→  const [defaultCurrency, setDefaultCurrency] = useState("ETB");
   359→  const [defaultLanguage, setDefaultLanguage] = useState("en");
   360→  const [savingSystem, setSavingSystem] = useState(false);
   361→
   362→  useEffect(() => {
   363→    if (currentUser) {
   364→      setName(currentUser.name);
   365→      setUsername(currentUser.username);
   366→    }
   367→  }, [currentUser]);
   368→
   369→  const handleProfileSave = async () => {
   370→    if (!currentUser) return;
   371→    if (!name.trim()) {
   372→      toast.error("Name is required");
   373→      return;
   374→    }
   375→    if (!username.trim()) {
   376→      toast.error("Username is required");
   377→      return;
   378→    }
   379→    if (username.length < 3) {
   380→      toast.error("Username must be at least 3 characters");
   381→      return;
   382→    }
   383→
   384→    setSaving(true);
   385→    try {
   386→      const data: Record<string, unknown> = { name: name.trim(), username: username.trim() };
   387→      if (email.trim()) data.email = email.trim();
   388→      if (phone.trim()) data.phone = phone.trim();
   389→
   390→      await apiUpdateUser(currentUser.id, data);
   391→
   392→      // Update local session
   393→      setCurrentUser({
   394→        ...currentUser,
   395→        name: name.trim(),
   396→        username: username.trim(),
   397→      });
   398→
   399→      toast.success("Profile updated successfully");
   400→      triggerRefresh();
   401→    } catch (err: unknown) {
   402→      const msg = err instanceof Error ? err.message : "Failed to update profile";
   403→      toast.error(msg);
   404→    } finally {
   405→      setSaving(false);
   406→    }
   407→  };
   408→
   409→  const handlePasswordChange = async () => {
   410→    if (!currentUser) return;
   411→    if (!currentPassword) {
   412→      toast.error("Current password is required");
   413→      return;
   414→    }
   415→    if (!newPassword) {
   416→      toast.error("New password is required");
   417→      return;
   418→    }
   419→    if (newPassword.length < 6) {
   420→      toast.error("New password must be at least 6 characters");
   421→      return;
   422→    }
   423→    if (newPassword !== confirmPassword) {
   424→      toast.error("New passwords do not match");
   425→      return;
   426→    }
   427→    if (currentPassword === newPassword) {
   428→      toast.error("New password must be different from current password");
   429→      return;
   430→    }
   431→
   432→    setChangingPassword(true);
   433→    try {
   434→      await apiUpdateUser(currentUser.id, {
   435→        password: newPassword,
   436→      });
   437→      toast.success("Password changed successfully");
   438→      setCurrentPassword("");
   439→      setNewPassword("");
   440→      setConfirmPassword("");
   441→      setShowCurrent(false);
   442→      setShowNew(false);
   443→    } catch (err: unknown) {
   444→      const msg = err instanceof Error ? err.message : "Failed to change password";
   445→      toast.error(msg);
   446→    } finally {
   447→      setChangingPassword(false);
   448→    }
   449→  };
   450→
   451→  const handleSystemSave = async () => {
   452→    setSavingSystem(true);
   453→    try {
   454→      await apiUpdateSettings({
   455→        guestHouseName: appName,
   456→        currency: defaultCurrency,
   457→        language: defaultLanguage,
   458→      });
   459→      toast.success("System settings saved");
   460→      triggerRefresh();
   461→    } catch (err: unknown) {
   462→      const msg = err instanceof Error ? err.message : "Failed to save system settings";
   463→      toast.error(msg);
   464→    } finally {
   465→      setSavingSystem(false);
   466→    }
   467→  };
   468→
   469→  if (loading) {
   470→    return (
   471→      <div className="space-y-6 p-4 md:p-6">
   472→        <Skeleton className="h-8 w-40" />
   473→        <Skeleton className="h-96 rounded-xl" />
   474→      </div>
   475→    );
   476→  }
   477→
   478→  const getInitials = (n: string) =>
   479→    n.split(" ").map((w) => w[0]).join("").toUpperCase().slice(0, 2);
   480→
   481→  return (
   482→    <div className="flex justify-center p-4 md:p-6">
   483→      <div className="w-full max-w-2xl space-y-6">
   484→      <div>
   485→        <h1 className="text-2xl font-bold tracking-tight">My Profile</h1>
   486→        <p className="text-sm text-muted-foreground">
   487→          Manage your System Admin account and preferences.
   488→        </p>
   489→      </div>
   490→
   491→      {/* ── Profile Card ── */}
   492→      <Card>
   493→        <CardHeader>
   494→          <CardTitle className="flex items-center gap-2">
   495→            <User className="h-5 w-5" />
   496→            Profile Information
   497→          </CardTitle>
   498→          <CardDescription>
   499→            Update your personal details and contact information.
   500→          </CardDescription>
   501→        </CardHeader>
   502→        <CardContent className="space-y-5">
   503→          {/* Avatar + Role Badge */}
   504→          <div className="flex items-center gap-4">
   505→            <Avatar className="h-16 w-16 ring-2 ring-primary/20">
   506→              <AvatarFallback className="bg-primary/10 text-lg font-bold text-primary">
   507→                {getInitials(name || "SA")}
   508→              </AvatarFallback>
   509→            </Avatar>
   510→            <div>
   511→              <p className="font-semibold text-slate-900">{name || "System Admin"}</p>
   512→              <div className="mt-1 flex items-center gap-2">
   513→                <Badge
   514→                  variant="outline"
   515→                  className="bg-amber-100 text-amber-700 border-amber-200 text-xs font-semibold"
   516→                >
   517→                  <Shield className="mr-1 size-3" />
   518→                  Superuser
   519→                </Badge>
   520→                <Badge variant="outline" className="text-xs">
   521→                  <BadgeCheck className="mr-1 size-3 text-emerald-500" />
   522→                  System Administrator
   523→                </Badge>
   524→              </div>
   525→              <p className="mt-1 text-xs text-slate-400">
   526→                @username: {currentUser?.username}
   527→              </p>
   528→            </div>
   529→          </div>
   530→
   531→          <Separator />
   532→
   533→          {/* Name & Username */}
   534→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   535→            <div className="grid gap-2">
   536→              <Label htmlFor="su-name">
   537→                <User className="inline mr-1.5 size-3.5" />
   538→                Full Name
   539→              </Label>
   540→              <Input
   541→                id="su-name"
   542→                value={name}
   543→                onChange={(e) => setName(e.target.value)}
   544→                placeholder="Your full name"
   545→              />
   546→            </div>
   547→            <div className="grid gap-2">
   548→              <Label htmlFor="su-username">
   549→                <BadgeCheck className="inline mr-1.5 size-3.5" />
   550→                Username
   551→              </Label>
   552→              <Input
   553→                id="su-username"
   554→                value={username}
   555→                onChange={(e) => setUsername(e.target.value)}
   556→                placeholder="Username"
   557→              />
   558→            </div>
   559→          </div>
   560→
   561→          {/* Email & Phone */}
   562→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   563→            <div className="grid gap-2">
   564→              <Label htmlFor="su-email">
   565→                <Mail className="inline mr-1.5 size-3.5" />
   566→                Email Address
   567→              </Label>
   568→              <Input
   569→                id="su-email"
   570→                type="email"
   571→                value={email}
   572→                onChange={(e) => setEmail(e.target.value)}
   573→                placeholder="admin@example.com"
   574→              />
   575→            </div>
   576→            <div className="grid gap-2">
   577→              <Label htmlFor="su-phone">
   578→                <Phone className="inline mr-1.5 size-3.5" />
   579→                Phone Number
   580→              </Label>
   581→              <Input
   582→                id="su-phone"
   583→                type="tel"
   584→                value={phone}
   585→                onChange={(e) => setPhone(e.target.value)}
   586→                placeholder="+251..."
   587→              />
   588→            </div>
   589→          </div>
   590→
   591→          <div className="flex justify-end pt-1">
   592→            <Button onClick={handleProfileSave} disabled={saving}>
   593→              {saving ? (
   594→                <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   595→              ) : (
   596→                <Save className="mr-2 h-4 w-4" />
   597→              )}
   598→              Save Profile
   599→            </Button>
   600→          </div>
   601→        </CardContent>
   602→      </Card>
   603→
   604→      {/* ── Security Card ── */}
   605→      <Card>
   606→        <CardHeader>
   607→          <CardTitle className="flex items-center gap-2">
   608→            <Lock className="h-5 w-5" />
   609→            Change Password
   610→          </CardTitle>
   611→          <CardDescription>
   612→            Update your password to keep your account secure.
   613→          </CardDescription>
   614→        </CardHeader>
   615→        <CardContent className="space-y-4">
   616→          <div className="grid gap-2">
   617→            <Label htmlFor="su-current-pw">Current Password</Label>
   618→            <div className="relative">
   619→              <Input
   620→                id="su-current-pw"
   621→                type={showCurrent ? "text" : "password"}
   622→                value={currentPassword}
   623→                onChange={(e) => setCurrentPassword(e.target.value)}
   624→                placeholder="Enter current password"
   625→                className="pr-10"
   626→              />
   627→              <button
   628→                type="button"
   629→                onClick={() => setShowCurrent(!showCurrent)}
   630→                className="absolute right-3 top-1/2 -translate-y-1/2 text-slate-400 hover:text-slate-600"
   631→              >
   632→                {showCurrent ? <EyeOff className="size-4" /> : <Eye className="size-4" />}
   633→              </button>
   634→            </div>
   635→          </div>
   636→
   637→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   638→            <div className="grid gap-2">

   643→                  type={showNew ? "text" : "password"}

   666→              />
   667→            </div>

   670→          {newPassword && confirmPassword && newPassword !== confirmPassword && (

   672→              Passwords do not match
   673→            </p>
   674→          )}
   675→
   676→          {newPassword && newPassword.length > 0 && newPassword.length < 6 && (

   678→              Password must be at least 6 characters
   679→            </p>
   680→          )}

   682→          <div className="flex justify-end pt-1">
   683→            <Button
   684→              variant="outline"
   685→              onClick={handlePasswordChange}
   686→              disabled={changingPassword || !currentPassword || !newPassword || !confirmPassword}
   687→            >
   688→              {changingPassword ? (
   689→                <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   690→              ) : (
   691→                <Lock className="mr-2 h-4 w-4" />
   692→              )}
   693→              Update Password
   694→            </Button>
   695→          </div>
   696→        </CardContent>
   697→      </Card>
   698→
   699→      {/* ── System Preferences ── */}
   700→      <Card>
   701→        <CardHeader>
   702→          <CardTitle className="flex items-center gap-2">
   703→            <Globe className="h-5 w-5" />
   704→            System Preferences
   705→          </CardTitle>
   706→          <CardDescription>
   707→            Configure default settings for the entire system.
   708→          </CardDescription>
   709→        </CardHeader>
   710→        <CardContent className="space-y-4">
   711→          <div className="grid gap-2">
   712→            <Label htmlFor="su-appname">
   713→              <Building2 className="inline mr-1.5 size-3.5" />

   719→              onChange={(e) => setAppName(e.target.value)}
   720→              placeholder="GHMS"
   721→            />
   722→            <p className="text-xs text-slate-400">
   723→              Displayed in the sidebar and browser tab.
   724→            </p>
   725→          </div>
   726→
   727→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   728→            <div className="grid gap-2">
   729→              <Label htmlFor="su-currency">
   730→                <Clock className="inline mr-1.5 size-3.5" />
   731→                Default Currency
   732→              </Label>
   733→              <Select
   734→                value={defaultCurrency}
   735→                onValueChange={setDefaultCurrency}
   736→              >

   738→                  <SelectValue />

   740→                <SelectContent>
   741→                  <SelectItem value="ETB">ETB (Birr)</SelectItem>
   742→                  <SelectItem value="USD">USD (Dollar)</SelectItem>
   743→                  <SelectItem value="EUR">EUR (Euro)</SelectItem>
   744→                </SelectContent>
   745→              </Select>
   746→            </div>
   747→            <div className="grid gap-2">
   748→              <Label htmlFor="su-language">
   749→                <Globe className="inline mr-1.5 size-3.5" />
   750→                Default Language
   751→              </Label>
   752→              <Select
   753→                value={defaultLanguage}

   755→              >

   759→                <SelectContent>
   760→                  <SelectItem value="en">English</SelectItem>
   761→                  <SelectItem value="am">አማርኛ</SelectItem>
   762→                </SelectContent>
   763→              </Select>
   764→            </div>
   765→          </div>
   766→
   767→          <div className="flex justify-end pt-1">
   768→            <Button onClick={handleSystemSave} disabled={savingSystem}>
   769→              {savingSystem ? (
   770→                <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   771→              ) : (
   772→                <Save className="mr-2 h-4 w-4" />
   773→              )}
   774→              Save System Settings
   775→            </Button>
   776→          </div>
   777→        </CardContent>
   778→      </Card>
   779→      </div>
   780→    </div>
   781→  );
   782→}
