   112→function StatusBadge({ status }: { status: string }) {
   113→  const cfg = STATUS_CONFIG[status] || { label: status, variant: "secondary" as const };

   287→        toast.error("Could not find coordinates for this address");
   288→      }
   289→    } catch {
   290→      toast.error("Geocoding failed");
   291→    } finally {
   292→      setGeocodingOne(false);
   293→    }

   341→      setActioning(false);

   349→        await req(`/api/providers/${provider.id}/reactivate`, { method: "POST" });

   386→      await apiSuperCreateProvider({
   387→        ...registerForm,
