    99→const STATUS_BADGE_CLASS: Record<string, string> = {
   100→  PENDING: "bg-yellow-100 text-yellow-800 hover:bg-yellow-100 border-yellow-200",
   101→  APPROVED: "bg-emerald-100 text-emerald-800 hover:bg-emerald-100 border-emerald-200",
   102→  REJECTED: "bg-red-100 text-red-800 hover:bg-red-100 border-red-200",

   104→};
   105→
   106→function StatusBadge({ status }: { status: string }) {
   107→  const cfg = STATUS_CONFIG[status] || { label: status, variant: "secondary" as const };
   108→  return (
   109→    <Badge variant={cfg.variant} className={STATUS_BADGE_CLASS[status] || ""}>
   110→      {cfg.label}
   111→    </Badge>
   112→  );
   113→}

   147→export default function ProvidersPage() {
   148→  const { refreshKey, triggerRefresh, currentUser } = useAppStore();
   149→  const isSuperuser = currentUser?.role === "SUPERUSER";
   150→  const [providers, setProviders] = useState<Provider[]>([]);
   151→  const [loading, setLoading] = useState(true);
   152→  const [selectedProvider, setSelectedProvider] = useState<Provider | null>(null);
   153→  const [detailOpen, setDetailOpen] = useState(false);
   154→
   155→  // Registration form
   156→  const [registerOpen, setRegisterOpen] = useState(false);
   157→  const [registerForm, setRegisterForm] = useState<RegisterForm>(emptyRegisterForm);
   158→  const [registering, setRegistering] = useState(false);
   159→
   160→  // Coordinate editing
   161→  const [coordProvider, setCoordProvider] = useState<Provider | null>(null);
   162→  const [coordOpen, setCoordOpen] = useState(false);
   163→  const [editLat, setEditLat] = useState(9.02);
   164→  const [editLng, setEditLng] = useState(38.75);
   165→  const [savingCoord, setSavingCoord] = useState(false);
   166→  const [geocodingOne, setGeocodingOne] = useState(false);
   167→
   168→  // Batch geocode
   169→  const [geocodingAll, setGeocodingAll] = useState(false);
   170→  const [batchResult, setBatchResult] = useState<{ updated: number; failed: number; total: number; results?: any[] } | null>(null);
   171→  const [batchOpen, setBatchOpen] = useState(false);
   172→
   173→  // Action dialogs
   174→  const [rejectDialog, setRejectDialog] = useState<Provider | null>(null);
   175→  const [rejectReason, setRejectReason] = useState("");
   176→  const [confirmAction, setConfirmAction] = useState<{ provider: Provider; action: string } | null>(null);
   177→  const [actioning, setActioning] = useState(false);
   178→
   179→  // Suspension dialog (with reason + notification)
   180→  const [suspendDialog, setSuspendDialog] = useState<Provider | null>(null);
   181→  const [suspensionReason, setSuspensionReason] = useState("");
   182→  const [providerMessage, setProviderMessage] = useState("");
   183→  const [suspending, setSuspending] = useState(false);
   184→
   185→  const pagination = usePagination({ totalItems: providers.length, initialPageSize: 5, pageSizeOptions: [5, 10, 20, 50] });
   186→  const paginatedProviders = useMemo(() => pagination.paginate(providers), [providers, pagination]);
   187→
   188→  const fetchProviders = useCallback(async () => {
   189→    try {
   190→      setLoading(true);
   191→      const data = await apiGetProviders();
   192→      setProviders(Array.isArray(data) ? data : []);
   193→    } catch (err: unknown) {
   194→      const message = err instanceof Error ? err.message : "Failed to load providers";
   195→      toast.error(message);
   196→    } finally {
   197→      setLoading(false);
   198→    }
   199→  }, []);
   200→
   201→  useEffect(() => {
   202→    fetchProviders();
   203→  }, [fetchProviders, refreshKey]);
   204→
   205→  const openDetail = (provider: Provider) => {
   206→    setSelectedProvider(provider);
   207→    setDetailOpen(true);
   208→  };
   209→
   210→  const openReject = (provider: Provider) => {
   211→    setRejectDialog(provider);
   212→    setRejectReason("");
   213→  };
   214→
   215→  const openCoordPicker = (provider: Provider) => {
   216→    setCoordProvider(provider);
   217→    setEditLat(provider.latitude || 9.02);
   218→    setEditLng(provider.longitude || 38.75);
   219→    setCoordOpen(true);
   220→  };
   221→
   222→  const geocodeSingle = async () => {
   223→    if (!coordProvider || !coordProvider.address) {
   224→      toast.error("Provider has no address to geocode");
   225→      return;
   226→    }
   227→    try {
   228→      setGeocodingOne(true);
   229→      const result: any = await apiGeocodeAddress(coordProvider.address);
   230→      if (result.lat && result.lng) {
   231→        setEditLat(result.lat);
   232→        setEditLng(result.lng);
   233→        toast.success(`Found: ${result.lat.toFixed(4)}, ${result.lng.toFixed(4)}`);
   234→      } else {
   235→        toast.error("Could not find coordinates for this address");
   236→      }
   237→    } catch {
   238→      toast.error("Geocoding failed");
   239→    } finally {
   240→      setGeocodingOne(false);
   241→    }
   242→  };
   243→
   244→  const saveCoordinates = async () => {
   245→    if (!coordProvider) return;
   246→    try {
   247→      setSavingCoord(true);
   248→      await apiUpdateProvider(coordProvider.id, { status: coordProvider.status, latitude: editLat, longitude: editLng });
   249→      toast.success("Coordinates updated");
   250→      setCoordOpen(false);
   251→      triggerRefresh();
   252→    } catch {
   253→      toast.error("Failed to save coordinates");
   254→    } finally {
   255→      setSavingCoord(false);
   256→    }
   257→  };
   258→
   259→  const batchGeocode = async () => {
   260→    try {
   261→      setGeocodingAll(true);
   262→      const result: any = await apiGeocodeBatch();
   263→      setBatchResult(result);
   264→      toast.success(result.message || `Geocoded ${result.updated} providers`);
   265→      triggerRefresh();
   266→    } catch {
   267→      toast.error("Batch geocoding failed");
   268→    } finally {
   269→      setGeocodingAll(false);
   270→    }
   271→  };
   272→
   273→  const handleReject = async () => {
   274→    if (!rejectDialog || !rejectReason.trim()) {
   275→      toast.error("Rejection reason is required");
   276→      return;
   277→    }
   278→    try {
   279→      setActioning(true);
   280→      await apiUpdateProvider(rejectDialog.id, { status: "REJECTED", rejectionReason: rejectReason.trim() });
   281→      toast.success("Provider rejected");
   282→      setRejectDialog(null);
   283→      setRejectReason("");
   284→      triggerRefresh();
   285→    } catch (err: unknown) {
   286→      const message = err instanceof Error ? err.message : "Failed to reject provider";
   287→      toast.error(message);
   288→    } finally {
   289→      setActioning(false);
   290→    }
   291→  };
   292→
   293→  const handleStatusAction = async (provider: Provider, status: string) => {
   294→    try {
   295→      setActioning(true);
   296→      await apiUpdateProvider(provider.id, { status });
   297→      toast.success(`Provider ${status.toLowerCase()}`);
   298→      setConfirmAction(null);
   299→      triggerRefresh();
   300→    } catch (err: unknown) {
   301→      const message = err instanceof Error ? err.message : "Failed to update provider";
   302→      toast.error(message);
   303→    } finally {
   304→      setActioning(false);
   305→    }
   306→  };
   307→
   308→  const openSuspend = (provider: Provider) => {
   309→    setSuspendDialog(provider);
   310→    setSuspensionReason("");
   311→    setProviderMessage("");
   312→  };
   313→
   314→  const handleRegister = async () => {
   315→    if (!registerForm.name.trim() || !registerForm.ownerName.trim() || !registerForm.phone.trim()) {
   316→      toast.error("Guesthouse name, owner name, and phone are required");
   317→      return;
   318→    }
   319→    if (!registerForm.username.trim() || !registerForm.password.trim()) {
   320→      toast.error("Operator username and password are required");
   321→      return;
   322→    }
   323→    if (registerForm.password.trim().length < 4) {
   324→      toast.error("Password must be at least 4 characters");
   325→      return;
   326→    }
   327→    try {
   328→      setRegistering(true);
   329→      await apiSuperCreateProvider({
   330→        ...registerForm,
   331→      });
   332→      toast.success(`"${registerForm.name}" registered and approved successfully`);
   333→      setRegisterOpen(false);
   334→      setRegisterForm(emptyRegisterForm);
   335→      triggerRefresh();
   336→    } catch (err: unknown) {
   337→      toast.error(err instanceof Error ? err.message : "Failed to register guesthouse");
   338→    } finally {
   339→      setRegistering(false);
   340→    }
   341→  };
   342→
   343→  const handleSuspend = async () => {
   344→    if (!suspendDialog) return;
   345→    if (!suspensionReason.trim()) {
   346→      toast.error("Please provide a reason for suspension");
   347→      return;
   348→    }
   349→    try {
   350→      setSuspending(true);
   351→      await apiPoliceSuspendProvider({
   352→        providerId: suspendDialog.id,
   353→        suspensionReason: suspensionReason.trim(),
   354→        providerMessage: providerMessage.trim() || undefined,
   355→      });
   356→      toast.success(`"${suspendDialog.name}" has been suspended. Notification sent to provider.`);
   357→      setSuspendDialog(null);
   358→      setSuspensionReason("");
   359→      setProviderMessage("");
   360→      triggerRefresh();
   361→    } catch (err: unknown) {
   362→      toast.error(err instanceof Error ? err.message : "Failed to suspend provider");
   363→    } finally {
   364→      setSuspending(false);
   365→    }
   366→  };
   367→
   368→  const formatDate = (dateStr: string | null) => {
   369→    if (!dateStr) return "—";
   370→    return new Date(dateStr).toLocaleDateString("en-US", {
   371→      year: "numeric",
   372→      month: "short",
   373→      day: "numeric",
   374→    });
   375→  };
   376→
   377→  const approvedCount = providers.filter((p) => p.status === "APPROVED").length;
   378→  const pendingCount = providers.filter((p) => p.status === "PENDING").length;
   379→  const withCoords = providers.filter(
   380→    (p) => p.latitude !== 9.02 || p.longitude !== 38.75,
   381→  ).length;
   382→
   383→  return (
   384→    <div className="space-y-4 p-3 sm:p-4 md:p-6">
   385→      {/* Summary Cards */}
   386→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-2 sm:gap-3">
   387→        <div className="rounded-xl border bg-card p-3 sm:p-4 shadow-sm">
   388→          <div className="flex items-center gap-2 sm:gap-3">
   389→            <div className="flex h-8 w-8 sm:h-10 sm:w-10 items-center justify-center rounded-lg bg-slate-100">
   390→              <Building2 className="h-4 w-4 sm:h-5 sm:w-5 text-slate-600" />
   391→            </div>
   392→            <div>
   393→              <p className="text-[10px] sm:text-sm text-muted-foreground">Total</p>
   394→              <p className="text-lg sm:text-2xl font-bold">{providers.length}</p>
   395→            </div>
   396→          </div>
   397→        </div>
   398→        <div className="rounded-xl border bg-card p-3 sm:p-4 shadow-sm">
   399→          <div className="flex items-center gap-2 sm:gap-3">
   400→            <div className="flex h-8 w-8 sm:h-10 sm:w-10 items-center justify-center rounded-lg bg-emerald-50">
   401→              <CheckCircle2 className="h-4 w-4 sm:h-5 sm:w-5 text-emerald-600" />
   402→            </div>
   403→            <div>
   404→              <p className="text-[10px] sm:text-sm text-muted-foreground">Approved</p>
   405→              <p className="text-lg sm:text-2xl font-bold text-emerald-600">{approvedCount}</p>
   406→            </div>
   407→          </div>
   408→        </div>
   409→        <div className="rounded-xl border bg-card p-3 sm:p-4 shadow-sm">
   410→          <div className="flex items-center gap-2 sm:gap-3">
   411→            <div className="flex h-8 w-8 sm:h-10 sm:w-10 items-center justify-center rounded-lg bg-yellow-50">
   412→              <ShieldCheck className="h-4 w-4 sm:h-5 sm:w-5 text-yellow-600" />
   413→            </div>
   414→            <div>
   415→              <p className="text-[10px] sm:text-sm text-muted-foreground">Pending</p>
   416→              <p className="text-lg sm:text-2xl font-bold text-yellow-600">{pendingCount}</p>
   417→            </div>
   418→          </div>
   419→        </div>
   420→        <div className="rounded-xl border bg-card p-3 sm:p-4 shadow-sm">
   421→          <div className="flex items-center gap-2 sm:gap-3">
   422→            <div className="flex h-8 w-8 sm:h-10 sm:w-10 items-center justify-center rounded-lg bg-blue-50">
   423→              <MapPinned className="h-4 w-4 sm:h-5 sm:w-5 text-blue-600" />
   424→            </div>
   425→            <div>
   426→              <p className="text-[10px] sm:text-sm text-muted-foreground">With Coords</p>
   427→              <p className="text-lg sm:text-2xl font-bold text-blue-600">{withCoords}/{providers.length}</p>

   429→          </div>
   430→        </div>
   431→      </div>
   432→
   433→      {/* Geocode All Button */}
   434→      <div className="flex items-center justify-between">
   435→        <div>

   437→            {isSuperuser ? "Guesthouses" : "Provider Applications"}
   438→          </h2>
   439→          <p className="text-xs sm:text-sm text-muted-foreground">
   440→            {isSuperuser
   441→              ? "View registered guesthouses and suspend any that violate policy. Approval / rejection of pending registrations is handled by the Police module."
   442→              : "Manage registrations, licensing, and map locations"}
   443→          </p>
   444→        </div>
   445→        <div className="flex gap-2">
   446→          {!isSuperuser && (
   447→            <Button variant="outline" size="sm" onClick={() => setBatchOpen(true)} disabled={geocodingAll}>
   448→              <Globe className={`mr-1 h-3.5 w-3.5 ${geocodingAll ? "animate-spin" : ""}`} />
   449→              {geocodingAll ? "Geocoding..." : "Geocode All"}
   450→            </Button>
   451→          )}
   452→        </div>
   453→      </div>
   454→
   455→      {/* Providers Table/Cards */}
   456→      <div className="rounded-xl border bg-card shadow-sm">
   457→        {loading ? (
   458→          <div className="space-y-3 p-4 sm:p-6">
   459→            {Array.from({ length: 5 }).map((_, i) => (
   460→              <Skeleton key={i} className="h-20 w-full sm:h-12" />
   461→            ))}
   462→          </div>
   463→        ) : providers.length === 0 ? (
   464→          <div className="flex flex-col items-center justify-center py-12 sm:py-16 text-center">
   465→            <Building2 className="mb-3 h-10 w-10 sm:h-12 sm:w-12 text-muted-foreground/40" />
   466→            <p className="text-xs sm:text-sm text-muted-foreground">No providers registered yet</p>
   467→          </div>
   468→        ) : (
   469→          <>
   470→            {/* Mobile/Tablet: Card layout */}
   471→            <div className="divide-y lg:hidden">
   472→              {paginatedProviders.map((provider) => (
   473→                <div key={provider.id} className="p-3 sm:p-4">
   474→                  <div className="flex items-start justify-between gap-2">
   475→                    <button className="min-w-0 flex-1 text-left" onClick={() => openDetail(provider)}>
   476→                      <div className="flex items-center gap-2">
   477→                        <p className="truncate text-sm font-medium">{provider.name}</p>
   478→                        <StatusBadge status={provider.status} />
   479→                      </div>

   482→                        <span className="flex items-center gap-1"><Phone className="h-3 w-3" /> {provider.phone}</span>

   484→                      {provider.address && (
   485→                        <div className="mt-1 flex items-center gap-1 text-[10px] text-muted-foreground">
   486→                          <MapPin className="h-3 w-3" /> {provider.address}

   488→                      )}
   489→                      <div className="mt-1 flex items-center gap-2 text-[10px]">

   491→                        {(provider.latitude !== 9.02 || provider.longitude !== 38.75) ? (
   492→                          <span className="flex items-center gap-0.5 text-emerald-600"><Crosshair className="h-2.5 w-2.5" /> {provider.latitude.toFixed(4)}, {provider.longitude.toFixed(4)}</span>
   493→                        ) : (
   494→                          <span className="text-muted-foreground">No coordinates</span>
   495→                        )}
   496→                      </div>
   497→                    </button>
   498→                  </div>
   499→
   500→                  <div className="mt-2.5 flex items-center gap-1.5 border-t pt-2.5 flex-wrap">
   501→                    <button className="flex items-center gap-1 rounded-md px-2.5 py-1.5 text-xs font-medium text-slate-600 hover:bg-slate-100" onClick={() => openDetail(provider)}>
   502→                      <Eye className="h-3.5 w-3.5" /> Details
   503→                    </button>
   504→                    {!isSuperuser && (
   505→                      <button className="flex items-center gap-1 rounded-md px-2.5 py-1.5 text-xs font-medium text-blue-600 hover:bg-blue-50" onClick={() => openCoordPicker(provider)}>
