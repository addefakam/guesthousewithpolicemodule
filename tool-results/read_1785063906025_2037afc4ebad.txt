   716→/**

   722→export async function GET(req: NextRequest) {
   723→  try {
   724→    const auth = await getAuthContext(req);
   725→
   726→    // SUPERUSER can view police accounts too

  1484→        ...(severity !== undefined ? { severity: severity } : {}),
  1485→        ...(is_active !== undefined ? { is_active: is_active } : {}),
  1486→      },
  1487→      include: {

  1490→    });
  1491→
  1492→    return NextResponse.json(person);
