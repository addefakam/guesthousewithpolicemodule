   137→export default function MySubscriptionPage() {
   138→  const { t } = useTranslation("subscription");
   139→  // ── i18n lookup maps ──
   140→  const PAYMENT_METHOD_LABELS: Record<string, string> = {
   141→    CHAPA: t("payMethodCHAPA"),
   142→    CASH: t("payMethodCASH"),
   143→    BANK_TRANSFER: t("payMethodBANK_TRANSFER"),
   144→    TELEBIRR: t("payMethodTELEBIRR"),
   145→    CBE_BIRR: t("payMethodCBE_BIRR"),
   146→    OTHER: t("payMethodOTHER"),
   147→  };
   148→  const CYCLE_LABELS: Record<string, string> = {
   149→    MONTHLY: t("cycleMONTHLY"),
   150→    QUARTERLY: t("cycleQUARTERLY"),
   151→    SEMI_ANNUAL: t("cycleSEMI_ANNUAL"),
   152→    YEARLY: t("cycleYEARLY"),
   153→  };
   154→  const STATUS_LABELS: Record<string, string> = {
   155→    ACTIVE: t("statusACTIVE"),
   156→    WARNING: t("statusWARNING"),
   157→    EXPIRED: t("statusEXPIRED"),
   158→    SUSPENDED: t("statusSUSPENDED"),
   159→  };
   160→  function fmtDaysRemaining(days: number): string {
   161→    if (days > 0) return t(days === 1 ? "daysRemaining_one" : "daysRemaining_other", { count: days });
   162→    if (days === 0) return t("expiresToday");
   163→    const abs = Math.abs(days);
   164→    return t(abs === 1 ? "expiredAgo_one" : "expiredAgo_other", { count: abs });
   165→  }
   166→  function fmtCycle(cycle: string): string {
   167→    return CYCLE_LABELS[cycle] || cycle;
   168→  }
   169→
   170→  const [data, setData] = useState<SubData | null>(null);
   171→  const [loading, setLoading] = useState(true);
   172→  const [showPayDialog, setShowPayDialog] = useState(false);
   173→  const [selectedPlan, setSelectedPlan] = useState<SubData["plans"][0] | null>(null);
   174→  const [submitting, setSubmitting] = useState(false);
   175→  const [payMethod, setPayMethod] = useState("");
   176→  const [payRef, setPayRef] = useState("");
   177→  const [payNotes, setPayNotes] = useState("");
   178→  const [payAmount, setPayAmount] = useState("");
   179→  const [showHistory, setShowHistory] = useState(false);
   180→  const [chapaVerifying, setChapaVerifying] = useState(false);
   181→  const [chapaVerifyResult, setChapaVerifyResult] = useState<string | null>(null);
   182→
   183→  // Chapa result is passed via sessionStorage by the main page.tsx
   184→  // (which reads URL params from Chapa redirect and stores them here)
   185→  const [chapaResult, setChapaResult] = useState<string | null>(null);
   186→  useEffect(() => {
   187→    const stored = sessionStorage.getItem("chapa_result");
   188→    if (stored) {
   189→      setChapaResult(stored);
   190→      sessionStorage.removeItem("chapa_result");
   191→      sessionStorage.removeItem("chapa_sub");
   192→      sessionStorage.removeItem("chapa_timestamp");
   193→    }
   194→  }, []);
   195→
   196→  const fetchData = useCallback(async () => {
   197→    try {
   198→      setLoading(true);
   199→      const res = await apiMySubscription();
   200→      setData(res as SubData);
   201→    } catch (err) {
   202→      const msg = err instanceof Error ? err.message : t("toastFailedLoad");
   203→      toast.error(`t("toastFailedLoad") ${msg}`);
   204→    } finally {
   205→      setLoading(false);
   206→    }
   207→  }, []);
   208→
   209→  useEffect(() => {
   210→    fetchData();
   211→  }, [fetchData]);
   212→
   213→  // Handle Chapa payment callback — actively verify with Chapa API
   214→  useEffect(() => {
   215→    if (chapaResult === "success") {
   216→      toast.success(t("toastChapaCompleted"));
   217→      setChapaVerifying(true);
   218→      setChapaVerifyResult(null);
   219→
   220→      // Wait a moment for Chapa to process, then actively verify
   221→      const timer = setTimeout(async () => {
   222→        try {
   223→          const res = await apiChapaClientVerify() as any;
   224→          if (res.verified > 0) {
   225→            setChapaVerifyResult("success");
   226→            toast.success(`${t("toastPaymentVerified")}`);
   227→          } else if (res.alreadyVerified) {
   228→            setChapaVerifyResult("success");
   229→          } else {
   230→            setChapaVerifyResult("pending");
   231→            toast.info(res.results?.[0] || t("toastChapaPending"));
   232→          }
   233→        } catch (err) {
   234→          setChapaVerifyResult("error");
   235→          toast.error(err instanceof Error ? err.message : t("toastVerifyFailed"));
   236→        } finally {
   237→          setChapaVerifying(false);
   238→          fetchData();
   239→        }
   240→      }, 3000);
   241→
   242→      return () => clearTimeout(timer);
   243→    }
   244→  }, [chapaResult, fetchData]);
   245→
   246→  const handleSelectPlan = (plan: SubData["plans"][0]) => {
   247→    setSelectedPlan(plan);
   248→    setPayAmount(String(plan.price));
   249→    setPayMethod("");
   250→    setPayRef("");
   251→    setPayNotes("");
   252→    setShowPayDialog(true);
   253→  };
   254→
   255→  const isChapaMethod = payMethod === "CHAPA";
   256→
   257→  const handleChapaPayment = async () => {
   258→    if (!selectedPlan || !payAmount || Number(payAmount) <= 0) {
   259→      toast.error(t("toastInvalidPayment"));
   260→      return;
   261→    }
   262→    setSubmitting(true);
   263→    try {
   264→      const res = await apiInitiateChapaPayment({
   265→        cycle: selectedPlan.cycle,
   266→        amount: Number(payAmount),
   267→        planId: selectedPlan.id,
   268→      });
   269→      if (res.checkoutUrl) {
   270→        // Redirect to Chapa checkout page
   271→        window.location.href = res.checkoutUrl;
   272→      } else {
   273→        toast.error(t("toastNoChapaUrl"));
   274→      }
   275→    } catch (e: unknown) {
   276→      toast.error(e instanceof Error ? e.message : t("toastChapaInitFailed"));
   277→    } finally {
   278→      setSubmitting(false);
   279→    }
   280→  };
   281→
   282→  const handleSubmitPayment = async () => {
   283→    // If Chapa is selected, redirect to Chapa checkout
   284→    if (isChapaMethod) {
   285→      await handleChapaPayment();
   286→      return;
   287→    }
   288→
   289→    if (!selectedPlan || !payMethod || !payAmount || Number(payAmount) <= 0) {
   290→      toast.error(t("toastFillRequired"));
   291→      return;
   292→    }
   293→    if (!isChapaMethod && !payRef.trim()) {
   294→      toast.error(t("toastRefRequired"));
   295→      return;
   296→    }
   297→    setSubmitting(true);
   298→    try {
   299→      await apiSubmitPayment({
   300→        planId: selectedPlan.id,
   301→        cycle: selectedPlan.cycle,
   302→        amount: Number(payAmount),
   303→        paymentMethod: payMethod,
   304→        referenceNo: payRef,
   305→        notes: payNotes,
   306→      });
   307→      toast.success(t("toastPaymentSubmitted"));
   308→      setShowPayDialog(false);
   309→      fetchData();
   310→    } catch (e: unknown) {
   311→      toast.error(e instanceof Error ? e.message : t("toastSubmitFailed"));
   312→    } finally {
   313→      setSubmitting(false);
   314→    }
   315→  };
   316→
   317→  const sub = data?.subscription;
   318→  const cur = data?.config.currencySymbol || "Br";
   319→  const totalBeds = data?.totalBeds || 0;
   320→  const pricePerBed = data?.config.pricePerBedPerDay || 0;
   321→  const pricingLocked = data?.config.pricingEnabled === false;
   322→
   323→  if (loading) return <LoadingSkeleton />;
   324→  if (!data || !sub) {
   325→    return (
   326→      <div className="flex flex-col items-center justify-center py-20 text-center">
   327→        <CreditCard className="w-12 h-12 text-slate-300 mb-3" />
   328→        <p className="text-sm font-medium text-slate-500">{t("noSubFound")}</p>
   329→        <p className="text-xs text-slate-400 mt-1">{t("noSubContact")}</p>
   330→      </div>
   331→    );
   332→  }
   333→
   334→  const isTrial = sub.price === 0 && data.payments.length === 0;
   335→  const isExpired = sub.status === "EXPIRED";
   336→  const penaltyPercent = data.config.latePaymentPenalty || 10;
   337→  const baseAmount = sub.price || 0;
   338→  const penaltyAmount = isExpired && baseAmount > 0
   339→    ? Math.round(baseAmount * (1 + penaltyPercent / 100))
   340→    : 0;
   341→  const statusColor = getStatusBadgeClasses(sub.status as "ACTIVE" | "WARNING" | "EXPIRED" | "SUSPENDED");
   342→
   343→  return (
   344→    <div className="space-y-4 p-3 sm:p-4 md:p-6 max-w-4xl mx-auto">
   345→      {/* Header */}
   346→      <div className="flex items-center justify-between">
   347→        <div>
   348→          <h2 className="text-base sm:text-lg font-semibold">{t("pageTitle")}</h2>
   349→          <p className="text-xs sm:text-sm text-muted-foreground">
   350→            t("pageSubtitle")
   351→          </p>
   352→        </div>
   353→        <Button
   354→          variant="outline"
   355→          size="sm"
   356→          onClick={fetchData}
   357→          className="h-8 text-xs"
   358→        >
   359→          <RefreshCw className="mr-1 h-3 w-3" />
   360→          {t("btnRefresh")}
   361→        </Button>
   362→      </div>
   363→
   364→      {/* Pricing Info Banner */}
   365→      <div className="flex items-center gap-2 p-3 bg-slate-50 rounded-xl border border-slate-200">
   366→        <div className="flex items-center justify-center w-8 h-8 rounded-lg bg-blue-100 shrink-0">
   367→          <Building2 className="w-4 h-4 text-blue-600" />
   368→        </div>
   369→        <div className="flex-1 min-w-0">
   370→          <p className="text-xs font-medium text-slate-700">
   371→            {t("pricingCalc", { cur, price: pricePerBed, beds: totalBeds, bedLabel: totalBeds === 1 ? t("bed_one") : t("bed_other"), total: (pricePerBed * totalBeds).toLocaleString() })}
   372→          </p>
   373→          <p className="text-[10px] text-muted-foreground">
   374→            {totalBeds === 0
   375→              ? t("noRoomsConfigured")
   376→              : `${t("bedsCalcDescription", { beds: totalBeds })}`}
   377→          </p>
   378→        </div>
   379→        {pricingLocked && (
   380→          <Badge variant="outline" className="text-[10px] border-slate-300 text-slate-500 bg-slate-100 shrink-0">
   381→            <Lock className="w-3 h-3 mr-1" />
   382→            t("ratesLocked")
   383→          </Badge>
   384→        )}
   385→      </div>
   386→
   387→      {/* ═══ Current Status Card ═══ */}
   388→      <Card className={`overflow-hidden ${
   389→        sub.status === "WARNING" ? "border-amber-300 bg-amber-50/30" :
   390→        sub.status === "EXPIRED" ? "border-2 border-rose-400 bg-rose-50/30" :
   391→        sub.status === "SUSPENDED" ? "border-slate-300" : ""
   392→      }`}>
   393→        <CardContent className="p-0">
   394→          {/* Status bar */}
   395→          <div className={`px-4 py-3 flex items-center justify-between ${
   396→            sub.status === "ACTIVE" ? "bg-emerald-50" :
   397→            sub.status === "WARNING" ? "bg-amber-50" :
   398→            sub.status === "EXPIRED" ? "bg-rose-100" :
   399→            "bg-slate-50"
   400→          }`}>
   401→            <div className="flex items-center gap-2">
   402→              {sub.status === "ACTIVE" ? (
   403→                <ShieldCheck className="w-5 h-5 text-emerald-600" />
   404→              ) : sub.status === "WARNING" ? (
   405→                <AlertTriangle className="w-5 h-5 text-amber-600" />
   406→              ) : sub.status === "EXPIRED" ? (
   407→                <XCircle className="w-5 h-5 text-rose-600" />
   408→              ) : (
   409→                <XCircle className="w-5 h-5 text-slate-400" />
   410→              )}
   411→              <span className={`text-sm font-semibold px-2.5 py-0.5 rounded-full border ${statusColor}`}>
   412→                {STATUS_LABELS[sub.status] || sub.status}
   413→              </span>
   414→              {isTrial && (
   415→                <Badge variant="outline" className="text-xs border-blue-200 text-blue-600 bg-blue-50">
   416→                  <Sparkles className="w-3 h-3 mr-1" />
   417→                  Free Trial
   418→                </Badge>
   419→              )}
   420→            </div>
   421→            <span className="text-xs font-medium text-muted-foreground">
   422→              {fmtDaysRemaining(sub.daysRemaining)}
   423→            </span>
   424→          </div>
   425→
   426→          {/* Details grid */}
   427→          <div className="p-4 grid grid-cols-2 sm:grid-cols-4 gap-4">
   428→            <div>
   429→              <p className="text-[10px] uppercase tracking-wider text-muted-foreground font-medium">{t("lblCurrentPlan")}</p>
   430→              <p className="text-sm font-semibold mt-0.5">
   431→                {isTrial ? t("freeTrial") : fmtCycle(sub.cycle)}
   432→              </p>
   433→            </div>
   434→            <div>
   435→              <p className="text-[10px] uppercase tracking-wider text-muted-foreground font-medium">{t("lblAmount")}</p>
   436→              <p className="text-sm font-semibold mt-0.5">
   437→                {isTrial ? t("freeTrial") : `${Number(sub.price).toLocaleString()} ${cur}`}
   438→              </p>
   439→            </div>
   440→            <div>
   441→              <p className="text-[10px] uppercase tracking-wider text-muted-foreground font-medium">{t("lblExpiresOn")}</p>
   442→              <p className="text-sm font-semibold mt-0.5">
   443→                {new Date(sub.endDate).toLocaleDateString("en-GB", {
   444→                  day: "numeric",
   445→                  month: "short",
   446→                  year: "numeric",
   447→                })}
   448→              </p>
   449→            </div>
   450→            <div>
   451→              <p className="text-[10px] uppercase tracking-wider text-muted-foreground font-medium">{t("lblProvider")}</p>
   452→              <p className="text-sm font-semibold mt-0.5 truncate">
   453→                {data.provider?.name || "—"}
   454→              </p>
   455→            </div>
   456→          </div>
   457→
   458→          {/* Payment instructions */}
   459→          {data.config.paymentInstructions && (
   460→            <div className="px-4 pb-4">
   461→              <div className="flex items-start gap-2 p-3 bg-blue-50 border border-blue-200 rounded-lg">
   462→                <Info className="w-4 h-4 text-blue-600 mt-0.5 shrink-0" />
   463→                <p className="text-xs text-blue-800 whitespace-pre-wrap leading-relaxed">
   464→                  {data.config.paymentInstructions}
   465→                </p>
   466→              </div>
   467→            </div>
   468→          )}
   469→
   470→          {/* Penalty alarm for EXPIRED */}
   471→          {isExpired && baseAmount > 0 && (
   472→            <div className="px-4 pb-4">
   473→              <div className="flex items-start gap-3 p-4 rounded-xl border-2 border-rose-300 bg-rose-50 animate-pulse">
   474→                <div className="shrink-0 mt-0.5 animate-bounce">
   475→                  <AlertTriangle className="w-5 h-5 text-rose-600" />
   476→                </div>
   477→                <div className="flex-1">
   478→                  <p className="text-sm font-bold text-rose-800">{t("penaltyTitle")}</p>
   479→                  <p className="text-xs text-rose-700 mt-0.5">
   480→                    {t("penaltyDesc", { days: Math.abs(sub.daysRemaining), percent: penaltyPercent })}
   481→                    
   482→                  </p>
   483→                  <div className="mt-2 flex items-center gap-2 flex-wrap">
   484→                    <span className="text-xs text-rose-600">{t("penaltyBase")}: <strong>{baseAmount.toLocaleString()} {cur}</strong></span>
   485→                    <span className="text-rose-300">+</span>
   486→                    <span className="text-xs text-rose-600">{t("penaltyLabel")}: <strong>{(penaltyAmount - baseAmount).toLocaleString()} {cur}</strong></span>
   487→                    <span className="text-rose-300">=</span>
   488→                    <span className="text-sm font-bold text-rose-900 bg-rose-100 border border-rose-300 px-2.5 py-1 rounded-lg">
   489→                      {t("penaltyTotalDue")}: {penaltyAmount.toLocaleString()} {cur}
   490→                    </span>
   491→                  </div>
   492→                </div>
   493→              </div>
   494→            </div>
   495→          )}
   496→
   497→          {/* Warning message for WARNING status */}
   498→          {sub.status === "WARNING" && (
   499→            <div className="px-4 pb-4">
   500→              <div className="flex items-start gap-2 p-3 rounded-lg border bg-amber-50 border-amber-200">
   501→                <AlertTriangle className="w-4 h-4 mt-0.5 shrink-0 text-amber-600" />
   502→                <div>
   503→                  <p className="text-xs font-semibold text-amber-800">{t("warningTitle")}</p>
   504→                  <p className="text-xs mt-1 text-amber-700">{t("warningDesc")}</p>
   505→                </div>
   506→              </div>
   507→            </div>
   508→          )}
   509→        </CardContent>
   510→      </Card>
   511→
   512→      {/* ═══ Available Plans ═══ */}
   513→      <div>
   514→        <div className="flex items-center justify-between mb-3">
   515→          <h3 className="text-sm font-semibold flex items-center gap-2">
   516→            <IndianRupee className="w-4 h-4" />
   517→            {t("availablePlans")}
   518→          </h3>
   519→          <span className="text-[10px] text-muted-foreground">
   520→            {t(data.plans.length === 1 ? "plansCount_one" : "plansCount_other", { count: data.plans.length })}
   521→          </span>
   522→        </div>
   523→
   524→        {data.plans.length === 0 ? (
   525→          <Card>
