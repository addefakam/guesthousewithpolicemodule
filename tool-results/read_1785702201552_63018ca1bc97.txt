   526→   526→/** Execute SQL via raw pg Client */
   527→   527→async function execViaPg(sql: string, params?: unknown[]): Promise<void> {
   528→   528→  const client = new Client({ connectionString: process.env.DATABASE_URL! });
   529→   529→  try {
   530→   530→    await client.connect();
   531→   531→    await client.query(sql, params);
   532→   532→  } finally {
   533→   533→    await client.end().catch(() => {});
   534→   534→  }
   535→   535→}

   537→   537→/** Execute DDL via Prisma $executeRawUnsafe — splits multi-statement SQL
   538→   538→ *  because Prisma's driver does NOT support multiple statements per call. */
   539→   539→async function execViaPrisma(sql: string): Promise<void> {
   540→   540→  // Split on `DO $$` blocks and standalone statements
   541→   541→  // Each `DO $$ ... END $$;` is one executable unit
   542→   542→  const statements = sql
   543→   543→    .split(/(?=DO \$\$)/)
   544→   544→    .map(s => s.trim())
   545→   545→    .filter(s => s.length > 0);
   546→   546→
   547→   547→  // If no DO blocks found, split by semicolons (for CREATE INDEX etc.)
   548→   548→  const parts = statements.length > 1
   549→   549→    ? statements
   550→   550→    : sql.split(';').map(s => s.trim()).filter(s => s.length > 0);
   551→   551→
   552→   552→  const prisma = new PrismaClient({
   553→   553→    log: ["warn", "error"],
   554→   554→  });
   555→   555→  try {
   556→   556→    for (const part of parts) {
   557→   557→      const stmt = part.endsWith(';') ? part : part + ';';
   558→   558→      try {
   559→   559→        await prisma.$executeRawUnsafe(stmt);
   560→   560→      } catch (err) {
   561→   561→        // Individual statement might fail (e.g. duplicate_object) — log but continue
   562→   562→        console.error("[execViaPrisma] Statement failed (continuing):", stmt.slice(0, 80), err instanceof Error ? err.message : String(err));
   563→   563→      }
   564→   564→    }
   565→   565→  } finally {
   566→   566→    await prisma.$disconnect().catch(() => {});
   567→   567→  }
   568→   568→}

   572→   572→ * Used as a background check when _initDone is already true.

   574→   574→async function runMigrationsOnly(): Promise<void> {

   589→   589→export async function ensureDatabase(): Promise<void> {
   590→   590→  if (_initDone) {
   591→   591→    // Always run migrations synchronously to ensure columns exist
   592→   592→    // before any query executes. Migrations are idempotent.
   593→   593→    if (!_migrationsRan) {
   594→   594→      _migrationsRan = true;
   595→   595→      await runMigrationsOnly();
   596→   596→    }
   597→   597→    return;
   598→   598→  }
   599→   599→  if (_initPromise) return _initPromise;
   600→   600→
   601→   601→  if (!process.env.DATABASE_URL) {
   602→   602→    throw new Error("DATABASE_URL environment variable is not set.");

   605→   605→  _initPromise = (async () => {

   648→   648→      // Verify User table was created

   704→   704→        );
