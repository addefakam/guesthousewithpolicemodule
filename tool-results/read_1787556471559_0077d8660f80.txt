   172→// ─── Component ──────────────────────────────────────────────────────────────
   173→
   174→export default function GuestCommunicationPage() {
   175→  const { t } = useTranslation();
   176→  const { refreshKey } = useAppStore();
   177→  const [activeTab, setActiveTab] = useState("templates");
   178→
   179→  // ── Templates State ──
   180→  const [templates, setTemplates] = useState<MessageTemplate[]>([]);
   181→  const [tplLoading, setTplLoading] = useState(true);
   182→  const [tplDialogOpen, setTplDialogOpen] = useState(false);
   183→  const [editingTpl, setEditingTpl] = useState<MessageTemplate | null>(null);
   184→  const [tplForm, setTplForm] = useState(emptyTemplateForm);
   185→  const [tplSaving, setTplSaving] = useState(false);
   186→  const [tplDeleteTarget, setTplDeleteTarget] = useState<MessageTemplate | null>(null);
   187→  const [tplDeleting, setTplDeleting] = useState(false);
   188→  const [togglingId, setTogglingId] = useState<string | null>(null);
   189→
   190→  // ── Single Message State ──
   191→  const [singleForm, setSingleForm] = useState(emptySingleForm);
   192→  const [sendingSingle, setSendingSingle] = useState(false);
   193→
   194→  // ── Bulk Send State ──
   195→  const [bulkForm, setBulkForm] = useState(emptyBulkForm);
   196→  const [sendingBulk, setSendingBulk] = useState(false);
   197→  const [bulkConfirmOpen, setBulkConfirmOpen] = useState(false);
   198→
   199→  // ── Message History State ──
   200→  const [logs, setLogs] = useState<MessageLog[]>([]);
   201→  const [logsLoading, setLogsLoading] = useState(true);
   202→  const [logsTotal, setLogsTotal] = useState(0);
   203→  const [logsPage, setLogsPage] = useState(1);
   204→  const [logsTotalPages, setLogsTotalPages] = useState(0);
   205→  const [filterStatus, setFilterStatus] = useState<string>("ALL");
   206→  const [filterChannel, setFilterChannel] = useState<string>("ALL");
   207→
   208→  // ─── Data Fetching ───────────────────────────────────────────────────────
   209→
   210→  const fetchTemplates = useCallback(async () => {
   211→    setTplLoading(true);
   212→    try {
   213→      const data = await apiGetMessageTemplates();
   214→      setTemplates(Array.isArray(data) ? data : []);
   215→    } catch {
   216→      toast.error("Failed to load message templates");
   217→    } finally {
   218→      setTplLoading(false);
   219→    }
   220→  }, []);
   221→
   222→  const fetchLogs = useCallback(async () => {
   223→    setLogsLoading(true);
   224→    try {
   225→      const params: Record<string, string | number> = { page: logsPage, limit: 10 };
   226→      if (filterStatus !== "ALL") params.status = filterStatus;
   227→      if (filterChannel !== "ALL") params.channel = filterChannel;
   228→      const result: PaginatedLogs = await apiGetMessageLogs(Object.entries(params).map(([k, v]) => `${k}=${v}`).join("&"));
   229→      setLogs(result.data ?? []);
   230→      setLogsTotal(result.total ?? 0);
   231→      setLogsPage(result.page ?? 1);
   232→      setLogsTotalPages(result.totalPages ?? 1);
   233→    } catch {
   234→      toast.error("Failed to load message history");
   235→    } finally {
   236→      setLogsLoading(false);
   237→    }
   238→  }, [logsPage, filterStatus, filterChannel]);
   239→
   240→  useEffect(() => {
   241→    fetchTemplates();
   242→  }, [fetchTemplates, refreshKey]);
   243→
   244→  useEffect(() => {
   245→    fetchLogs();
   246→  }, [fetchLogs, refreshKey]);
   247→
   248→  // Reset to page 1 when filters change
   249→  useEffect(() => {
   250→    setLogsPage(1);
   251→  }, [filterStatus, filterChannel]);
   252→
   253→  // ─── Template Helpers ─────────────────────────────────────────────────────
   254→
   255→  const openCreateTpl = () => {
   256→    setEditingTpl(null);
   257→    setTplForm(emptyTemplateForm);
   258→    setTplDialogOpen(true);
   259→  };
   260→
   261→  const openEditTpl = (tpl: MessageTemplate) => {
   262→    setEditingTpl(tpl);
   263→    setTplForm({
   264→      name: tpl.name,
   265→      type: tpl.type,
   266→      channel: tpl.channel,
   267→      body: tpl.body,
   268→    });
   269→    setTplDialogOpen(true);
   270→  };
   271→
   272→  const handleSaveTpl = async () => {
   273→    if (!tplForm.name.trim() || !tplForm.body.trim()) {
   274→      toast.error("Template name and body are required");
   275→      return;
   276→    }
   277→    setTplSaving(true);
   278→    try {
   279→      if (editingTpl) {
   280→        await apiUpdateMessageTemplate(editingTpl.id, tplForm);
   281→        toast.success("Template updated");
   282→      } else {
   283→        await apiCreateMessageTemplate(tplForm);
   284→        toast.success("Template created");
   285→      }
   286→      setTplDialogOpen(false);
   287→      fetchTemplates();
   288→    } catch (err: unknown) {
   289→      toast.error(err instanceof Error ? err.message : "Failed to save template");
   290→    } finally {
   291→      setTplSaving(false);
   292→    }
   293→  };
   294→
   295→  const handleDeleteTpl = async () => {
   296→    if (!tplDeleteTarget) return;
   297→    setTplDeleting(true);
   298→    try {
   299→      await apiDeleteMessageTemplate(tplDeleteTarget.id);
   300→      toast.success("Template deleted");
   301→      setTplDeleteTarget(null);
   302→      fetchTemplates();
   303→    } catch (err: unknown) {
   304→      toast.error(err instanceof Error ? err.message : "Failed to delete template");
   305→    } finally {
   306→      setTplDeleting(false);
   307→    }
   308→  };
   309→
   310→  const handleToggleActive = async (tpl: MessageTemplate) => {
   311→    const next = !tpl.isActive;
   312→    setTogglingId(tpl.id);
   313→    setTemplates((prev) =>
   314→      prev.map((x) => (x.id === tpl.id ? { ...x, isActive: next } : x))
   315→    );
   316→    try {
   317→      await apiUpdateMessageTemplate(tpl.id, { isActive: next });
   318→      toast.success(next ? "Template activated" : "Template deactivated");
   319→    } catch {
   320→      toast.error("Failed to toggle template");
   321→      setTemplates((prev) =>
   322→        prev.map((x) => (x.id === tpl.id ? { ...x, isActive: !next } : x))
   323→      );
   324→    } finally {
   325→      setTogglingId(null);
   326→    }
   327→  };
   328→
   329→  // ─── Send Message Helpers ─────────────────────────────────────────────────
   330→
   331→  const selectedTemplateForSingle = templates.find(
   332→    (x) => x.id === singleForm.templateId
   333→  );
   334→
   335→  const handleSingleTemplateChange = (templateId: string) => {
   336→    const tpl = templates.find((x) => x.id === templateId);
   337→    setSingleForm((prev) => ({
   338→      ...prev,
   339→      templateId,
   340→      message: tpl ? tpl.body : "",
   341→    }));
   342→  };
   343→
   344→  const handleSendSingle = async () => {
   345→    if (!singleForm.recipient.trim()) {
   346→      toast.error("Recipient phone number is required");
   347→      return;
   348→    }
   349→    if (!singleForm.message.trim()) {
   350→      toast.error("Message body is required");
   351→      return;
   352→    }
   353→    setSendingSingle(true);
   354→    try {
   355→      const payload: Record<string, string> = {
   356→        recipient: singleForm.recipient.trim(),
   357→        channel: singleForm.channel,
   358→        message: singleForm.message,
   359→      };
   360→      if (singleForm.templateId) payload.templateId = singleForm.templateId;
   361→      await apiSendMessage(payload);
   362→      toast.success("Message sent successfully");
   363→      setSingleForm(emptySingleForm);
   364→      fetchLogs();
   365→    } catch (err: unknown) {
   366→      toast.error(err instanceof Error ? err.message : "Failed to send message");
   367→    } finally {
   368→      setSendingSingle(false);
   369→    }
   370→  };
   371→
   372→  const handleBulkSend = async () => {
   373→    if (!bulkForm.templateId) {
   374→      toast.error("Please select a template");
   375→      return;
   376→    }
   377→    setSendingBulk(true);
   378→    try {
   379→      const result = await apiBulkSendMessages({
   380→        templateId: bulkForm.templateId,
   381→        status: bulkForm.status,
   382→        channel: bulkForm.channel,
   383→      });
   384→      const sent = (result as Record<string, number>)?.sent ?? 0;
   385→      const failed = (result as Record<string, number>)?.failed ?? 0;
   386→      toast.success(`${sent} messages sent, ${failed} failed`);
   387→      setBulkConfirmOpen(false);
   388→      setBulkForm(emptyBulkForm);
   389→      fetchLogs();
   390→    } catch (err: unknown) {
   391→      toast.error(err instanceof Error ? err.message : "Bulk send failed");
   392→    } finally {
   393→      setSendingBulk(false);
   394→    }
   395→  };
   396→
   397→  // ─── Render ───────────────────────────────────────────────────────────────
   398→
   399→  return (
   400→    <div className="space-y-6 p-4 md:p-6">
   401→      {/* Header */}
   402→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   403→        <div>
   404→          <h1 className="text-2xl font-bold tracking-tight">
   405→            Guest Communication
   406→          </h1>
   407→          <p className="mt-1 text-sm text-muted-foreground">
   408→            Send SMS/WhatsApp messages to guests
   409→          </p>
   410→        </div>
   411→        {activeTab === "templates" && (
   412→          <Button onClick={openCreateTpl} className="gap-2 shrink-0">
   413→            <Plus className="h-4 w-4" />
   414→            New Template
   415→          </Button>
   416→        )}
   417→      </div>
   418→
   419→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   420→        <TabsList>
   421→          <TabsTrigger value="templates" className="gap-2">
   422→            <MessageSquare className="h-4 w-4" />
   423→            Templates ({templates.length})
   424→          </TabsTrigger>
   425→          <TabsTrigger value="send" className="gap-2">
   426→            <Send className="h-4 w-4" />
   427→            Send Message
   428→          </TabsTrigger>
   429→          <TabsTrigger value="history" className="gap-2">
   430→            <History className="h-4 w-4" />
   431→            Message History
   432→          </TabsTrigger>
   433→        </TabsList>
   434→
   435→        {/* ─── Tab 1: Templates ────────────────────────────────────────────── */}
   436→        <TabsContent value="templates">
   437→          {tplLoading ? (
   438→            <div className="grid grid-cols-1 gap-4 md:grid-cols-2 lg:grid-cols-3">
   439→              {Array.from({ length: 6 }).map((_, i) => (
   440→                <Skeleton key={i} className="h-52 rounded-xl" />
   441→              ))}
   442→            </div>
   443→          ) : templates.length === 0 ? (
   444→            <div className="flex flex-col items-center justify-center rounded-xl border border-dashed py-16">
   445→              <MessageSquare className="h-12 w-12 text-gray-300 mb-3" />
   446→              <p className="text-lg font-medium text-gray-500">
   447→                No message templates
   448→              </p>
   449→              <p className="mt-1 text-sm text-gray-400">
   450→                Create your first template to get started
   451→              </p>
   452→              <Button
   453→                onClick={openCreateTpl}
   454→                variant="outline"
   455→                className="mt-4 gap-2"
   456→              >
   457→                <Plus className="h-4 w-4" /> New Template
   458→              </Button>
   459→            </div>
   460→          ) : (
   461→            <div className="grid grid-cols-1 gap-4 md:grid-cols-2 lg:grid-cols-3">
   462→              {templates.map((tpl) => {
   463→                const channelColor =
   464→                  CHANNEL_BADGE[tpl.channel] ?? CHANNEL_BADGE.SMS;
   465→                const typeColor = TYPE_BADGE[tpl.type] ?? TYPE_BADGE.CUSTOM;
   466→                return (
   467→                  <Card
   468→                    key={tpl.id}
   469→                    className={`gap-0 overflow-hidden py-0 transition-shadow hover:shadow-md ${
   470→                      !tpl.isActive ? "opacity-60" : ""
   471→                    }`}
   472→                  >
   473→                    <CardContent className="p-4">
   474→                      <div className="flex items-start justify-between mb-3">
   475→                        <div className="flex-1 min-w-0">
   476→                          <h3 className="font-semibold text-gray-900 leading-tight truncate">
   477→                            {tpl.name}
   478→                          </h3>
   479→                          <div className="flex items-center gap-1.5 mt-1.5 flex-wrap">
   480→                            <Badge
   481→                              variant="outline"
   482→                              className={`text-xs ${typeColor}`}
   483→                            >
   484→                              {TYPE_LABELS[tpl.type] ?? tpl.type}
   485→                            </Badge>
   486→                            <Badge
   487→                              variant="outline"
   488→                              className={`text-xs ${channelColor}`}
   489→                            >
   490→                              {tpl.channel}
   491→                            </Badge>
   492→                            {tpl.isDefault && (
   493→                              <Badge
   494→                                variant="outline"
   495→                                className="text-xs bg-amber-50 text-amber-700 border-amber-200"
   496→                              >
   497→                                Default
   498→                              </Badge>
   499→                            )}
   500→                          </div>
   501→                        </div>
   502→                      </div>
   503→
   504→                      <p className="text-sm text-gray-600 line-clamp-2 mb-3 whitespace-pre-line">
   505→                        {tpl.body}
   506→                      </p>
   507→
   508→                      <div className="flex items-center justify-between">
   509→                        <div className="flex items-center gap-2">
   510→                          <Switch
   511→                            checked={tpl.isActive}
   512→                            disabled={togglingId === tpl.id}
   513→                            onCheckedChange={() => handleToggleActive(tpl)}
   514→                          />
   515→                          <span className="text-xs text-muted-foreground">
   516→                            {tpl.isActive ? "Active" : "Inactive"}
   517→                          </span>
   518→                        </div>
   519→                        <div className="flex items-center gap-1">
   520→                          <Button
   521→                            variant="ghost"
   522→                            size="icon"
   523→                            className="h-8 w-8"
   524→                            onClick={() => openEditTpl(tpl)}
   525→                          >
   526→                            <Pencil className="h-4 w-4" />
   527→                          </Button>
   528→                          <Button
   529→                            variant="ghost"
   530→                            size="icon"
   531→                            className="h-8 w-8 text-muted-foreground hover:text-destructive"
   532→                            disabled={tpl.isDefault}
   533→                            onClick={() => setTplDeleteTarget(tpl)}
   534→                          >
   535→                            <Trash2 className="h-4 w-4" />
   536→                          </Button>
   537→                        </div>
   538→                      </div>
   539→                    </CardContent>
   540→                  </Card>
   541→                );
   542→              })}
   543→            </div>
   544→          )}
   545→        </TabsContent>
   546→
   547→        {/* ─── Tab 2: Send Message ─────────────────────────────────────────── */}
   548→        <TabsContent value="send">
   549→          <div className="grid grid-cols-1 gap-6 lg:grid-cols-2">
   550→            {/* Single Message */}
   551→            <Card>
   552→              <CardHeader>
   553→                <CardTitle className="flex items-center gap-2 text-lg">
   554→                  <Send className="h-5 w-5 text-primary" />
   555→                  Single Message
   556→                </CardTitle>
   557→              </CardHeader>
   558→              <CardContent className="space-y-4">
   559→                <div className="space-y-2">
   560→                  <Label htmlFor="single-recipient">
   561→                    Recipient Phone <span className="text-rose-500">*</span>
   562→                  </Label>
   563→                  <Input
   564→                    id="single-recipient"
   565→                    placeholder="+251912345678"
   566→                    value={singleForm.recipient}
   567→                    onChange={(e) =>
   568→                      setSingleForm((p) => ({
   569→                        ...p,
   570→                        recipient: e.target.value,
   571→                      }))
   572→                    }
   573→                  />
   574→                </div>
   575→
   576→                <div className="space-y-2">
   577→                  <Label>{t('lblchannel', 'Channel')}</Label>
   578→                  <Select
   579→                    value={singleForm.channel}
   580→                    onValueChange={(v) =>
   581→                      setSingleForm((p) => ({ ...p, channel: v }))
   582→                    }
   583→                  >
   584→                    <SelectTrigger id="single-channel">
   585→                      <SelectValue />
   586→                    </SelectTrigger>
   587→                    <SelectContent>
   588→                      <SelectItem value="SMS">SMS</SelectItem>
   589→                      <SelectItem value="WHATSAPP">WhatsApp</SelectItem>
   590→                    </SelectContent>
   591→                  </Select>
   592→                </div>
   593→
   594→                <div className="space-y-2">
   595→                  <Label>{t('lbltemplateOptional', 'Template (Optional)')}</Label>
   596→                  <Select
   597→                    value={singleForm.templateId}
