     3→
     4→import { useState, useEffect, useCallback } from "react";
     5→import { useAppStore } from "@/lib/store";
     6→import {
     7→  apiGetDaytimeServices,
     8→  apiCreateDaytimeService,
     9→  apiUpdateDaytimeService,
    10→  apiDeleteDaytimeService,
    11→  apiGetDaytimeBookings,

    13→  apiUpdateDaytimeBooking,
    14→  apiDeleteDaytimeBooking,
    15→  apiCreatePayment,
    16→} from "@/lib/api";
    17→import { toast } from "sonner";
    18→import { isValidPhone } from "@/lib/utils";
    19→import { Card, CardContent } from "@/components/ui/card";
    20→import { Button } from "@/components/ui/button";
    21→import { Badge } from "@/components/ui/badge";
    22→import { Input } from "@/components/ui/input";
    23→import { Label } from "@/components/ui/label";
    24→import { Textarea } from "@/components/ui/textarea";
    25→import { Skeleton } from "@/components/ui/skeleton";
    26→import { Switch } from "@/components/ui/switch";
    27→import { Tabs, TabsContent, TabsList, TabsTrigger } from "@/components/ui/tabs";
    28→import {
    29→  Dialog,
    30→  DialogContent,
    31→  DialogHeader,
    32→  DialogTitle,
    33→  DialogDescription,
    34→  DialogFooter,
    35→} from "@/components/ui/dialog";
    36→import {
    37→  AlertDialog,
    38→  AlertDialogContent,
    39→  AlertDialogHeader,
    40→  AlertDialogTitle,
    41→  AlertDialogDescription,
    42→  AlertDialogFooter,
    43→  AlertDialogAction,
    44→  AlertDialogCancel,
    45→} from "@/components/ui/alert-dialog";
    46→import {
    47→  Select,
    48→  SelectContent,
    49→  SelectItem,
    50→  SelectTrigger,
    51→  SelectValue,
    52→} from "@/components/ui/select";
    53→import {
    54→  Table,
    55→  TableBody,
    56→  TableCell,
    57→  TableHead,
    58→  TableHeader,
    59→  TableRow,
    60→} from "@/components/ui/table";
    61→import {
    62→  DropdownMenu,
    63→  DropdownMenuContent,
    64→  DropdownMenuItem,
    65→  DropdownMenuSeparator,
    66→  DropdownMenuTrigger,
    67→} from "@/components/ui/dropdown-menu";
    68→import {
    69→  Sun,
    70→  Plus,
    71→  MoreVertical,
    72→  Pencil,
    73→  Trash2,
    74→  DollarSign,
    75→  Clock,
    76→  CreditCard,
    77→  CalendarDays,
    78→  Tag,
    79→} from "lucide-react";

    81→// ─── Types ─────────────────────────────────────────────────────────────────
    82→
    83→interface Service {
    84→  id: string;
    85→  name: string;
    86→  price: number;
    87→  category: string;
    88→  duration: string;
    89→  description: string;
    90→  active: boolean;
    91→}
    92→
    93→interface Booking {
    94→  id: string;
    95→  serviceId: string;
    96→  guestName: string;
    97→  guestPhone: string;
    98→  date: string;
    99→  time: string;
   100→  quantity: number;
   101→  unitPrice: number;
   102→  totalCost: number;
   103→  paidAmount: number;
   104→  paymentStatus: string;
   105→  paymentMethod: string | null;
   106→  notes: string;
   107→  service?: { id: string; name: string; category: string };
   108→}
   109→
   110→// ─── Helpers ───────────────────────────────────────────────────────────────
   111→
   112→const formatPrice = (price: number) =>
   113→  new Intl.NumberFormat("en-US", { style: "currency", currency: "ETB", maximumFractionDigits: 0 }).format(price);
   114→
   115→const PAYMENT_STYLES: Record<string, string> = {
   116→  PAID: "bg-emerald-100 text-emerald-800 border-emerald-200",
   117→  PARTIAL: "bg-amber-100 text-amber-800 border-amber-200",
   118→  PENDING: "bg-rose-100 text-rose-800 border-rose-200",
   119→};
   120→
   121→const CATEGORY_COLORS: Record<string, string> = {
   122→  SPA: "bg-purple-50 text-purple-700 border-purple-200",
   123→  FOOD: "bg-orange-50 text-orange-700 border-orange-200",
   124→  LAUNDRY: "bg-sky-50 text-sky-700 border-sky-200",
   125→  TOUR: "bg-teal-50 text-teal-700 border-teal-200",
   126→  TRANSPORT: "bg-slate-50 text-slate-700 border-slate-200",
   127→  GYM: "bg-rose-50 text-rose-700 border-rose-200",
   128→  POOL: "bg-cyan-50 text-cyan-700 border-cyan-200",
   129→  EVENT: "bg-amber-50 text-amber-700 border-amber-200",
   130→  OTHER: "bg-gray-50 text-gray-700 border-gray-200",
   131→};
   132→
   133→// ─── Component ─────────────────────────────────────────────────────────────
   134→
   135→export default function DaytimePage() {
   136→  const { t } = useTranslation(["daytime", "common"]);
   137→  const { refreshKey, triggerRefresh } = useAppStore();
   138→  const [activeTab, setActiveTab] = useState("services");
   139→
   140→  // Services state
   141→  const [services, setServices] = useState<Service[]>([]);
   142→  const [svcLoading, setSvcLoading] = useState(true);
   143→  const [svcDialogOpen, setSvcDialogOpen] = useState(false);
   144→  const [editingSvc, setEditingSvc] = useState<Service | null>(null);
   145→  const [svcForm, setSvcForm] = useState({ name: "", price: "", category: "", duration: "", description: "" });
   146→  const [svcSaving, setSvcSaving] = useState(false);
   147→  const [svcDeleteTarget, setSvcDeleteTarget] = useState<Service | null>(null);
   148→  const [svcDeleting, setSvcDeleting] = useState(false);
   149→
   150→  // Bookings state
   151→  const [bookings, setBookings] = useState<Booking[]>([]);
   152→  const [bkLoading, setBkLoading] = useState(true);
   153→  const [bkDialogOpen, setBkDialogOpen] = useState(false);
   154→  const [editingBk, setEditingBk] = useState<Booking | null>(null);
   155→  const [bkForm, setBkForm] = useState({
   156→    serviceId: "", guestName: "", guestPhone: "", date: "", time: "", quantity: "1",
   157→  });
   158→  const [bkSaving, setBkSaving] = useState(false);
   159→  const [bkDeleteTarget, setBkDeleteTarget] = useState<Booking | null>(null);
   160→  const [bkDeleting, setBkDeleting] = useState(false);
   161→
   162→  // Payment dialog
   163→  const [payDialogOpen, setPayDialogOpen] = useState(false);
   164→  const [payTarget, setPayTarget] = useState<Booking | null>(null);
   165→  const [payForm, setPayForm] = useState({ amount: "", method: "CASH" });
   166→  const [paySaving, setPaySaving] = useState(false);
   167→
   168→  // ─── Data Fetching ────────────────────────────────────────────────────────
   169→
   170→  const fetchServices = useCallback(async () => {
   171→    try {
   172→      setSvcLoading(true);
   173→      const data = await apiGetDaytimeServices();
   174→      setServices(Array.isArray(data) ? data : []);
   175→    } catch (err: unknown) {
   176→      const msg = err instanceof Error ? err.message : t("toastFailedLoadServices");
   177→      toast.error(msg);
   178→    } finally {
   179→      setSvcLoading(false);
   180→    }
   181→  }, []);
   182→
   183→  const fetchBookings = useCallback(async () => {
   184→    try {
   185→      setBkLoading(true);
   186→      const data = await apiGetDaytimeBookings();
   187→      setBookings(Array.isArray(data) ? data : []);
   188→    } catch (err: unknown) {
   189→      const msg = err instanceof Error ? err.message : t("toastFailedLoadBookings");
   190→      toast.error(msg);
   191→    } finally {
   192→      setBkLoading(false);
   193→    }
   194→  }, []);
   195→
   196→  useEffect(() => { fetchServices(); }, [fetchServices, refreshKey]);
   197→  useEffect(() => { fetchBookings(); }, [fetchBookings, refreshKey]);
   198→
   199→  // ─── Service CRUD ─────────────────────────────────────────────────────────
   200→
   201→  const openCreateSvc = () => {
   202→    setEditingSvc(null);
   203→    setSvcForm({ name: "", price: "", category: "", duration: "", description: "" });
   204→    setSvcDialogOpen(true);
   205→  };
   206→
   207→  const openEditSvc = (svc: Service) => {
   208→    setEditingSvc(svc);
   209→    setSvcForm({
   210→      name: svc.name,
   211→      price: String(svc.price),
   212→      category: svc.category,
   213→      duration: svc.duration,
   214→      description: svc.description,
   215→    });
   216→    setSvcDialogOpen(true);
   217→  };
   218→
   219→  const handleSaveSvc = async () => {
   220→    if (!svcForm.name || !svcForm.price) {
   221→      toast.error(t("valNamePriceRequired"));
   222→      return;
   223→    }
   224→    try {
   225→      setSvcSaving(true);
   226→      const payload = {
   227→        name: svcForm.name,
   228→        price: Number(svcForm.price),
   229→        category: svcForm.category,
   230→        duration: svcForm.duration,
   231→        description: svcForm.description,
   232→        active: editingSvc ? editingSvc.active : true,
   233→      };
   234→      if (editingSvc) {
   235→        await apiUpdateDaytimeService(editingSvc.id, payload);
   236→        toast.success(t("toastServiceUpdated"));
   237→      } else {
   238→        await apiCreateDaytimeService(payload);
   239→        toast.success(t("toastServiceCreated"));
   240→      }
   241→      setSvcDialogOpen(false);
   242→      triggerRefresh();
   243→    } catch (err: unknown) {
   244→      toast.error(err instanceof Error ? err.message : t("toastFailedSaveService"));
   245→    } finally {
   246→      setSvcSaving(false);
   247→    }
   248→  };
   249→
   250→  const handleDeleteSvc = async () => {
   251→    if (!svcDeleteTarget) return;
   252→    try {
   253→      setSvcDeleting(true);
   254→      await apiDeleteDaytimeService(svcDeleteTarget.id);
   255→      toast.success(t("toastServiceDeleted"));
   256→      setSvcDeleteTarget(null);
   257→      triggerRefresh();
   258→    } catch (err: unknown) {
   259→      toast.error(err instanceof Error ? err.message : t("toastFailedDeleteService"));
   260→    } finally {
   261→      setSvcDeleting(false);
   262→    }
   263→  };
   264→
   265→  const handleToggleActive = async (svc: Service) => {
   266→    try {
   267→      await apiUpdateDaytimeService(svc.id, { active: !svc.active });
   268→      toast.success(svc.active ? t("toastServiceDeactivated") : t("toastServiceActivated"));
   269→      triggerRefresh();
   270→    } catch (err: unknown) {
   271→      toast.error(err instanceof Error ? err.message : t("toastFailedToggleService"));
   272→    }
   273→  };
   274→
   275→  // ─── Booking CRUD ─────────────────────────────────────────────────────────
   276→
   277→  const selectedService = services.find((s) => s.id === bkForm.serviceId);
   278→
   279→  const openCreateBk = () => {
   280→    setEditingBk(null);
   281→    setBkForm({ serviceId: "", guestName: "", guestPhone: "", date: "", time: "", quantity: "1" });
   282→    setBkDialogOpen(true);
   283→  };
   284→
   285→  const openEditBk = (bk: Booking) => {
   286→    setEditingBk(bk);
   287→    setBkForm({
   288→      serviceId: bk.serviceId,
   289→      guestName: bk.guestName,
   290→      guestPhone: bk.guestPhone,
   291→      date: bk.date,
   292→      time: bk.time,
   293→      quantity: String(bk.quantity),
   294→    });
   295→    setBkDialogOpen(true);
   296→  };
   297→
   298→  const handleSaveBk = async () => {
   299→    if (!bkForm.serviceId || !bkForm.guestName || !bkForm.date || !bkForm.time) {
   300→      toast.error(t("valBookingRequired"));
   301→      return;
   302→    }
   303→    const svc = services.find((s) => s.id === bkForm.serviceId);
   304→    if (!svc) { toast.error(t("valServiceNotFound")); return; }
   305→    if (bkForm.guestPhone.trim() && !isValidPhone(bkForm.guestPhone)) {
   306→      toast.error(t("valInvalidPhone"));
   307→      return;
   308→    }
   309→
   310→    try {
   311→      setBkSaving(true);
   312→      const qty = Number(bkForm.quantity) || 1;
   313→      const unitPrice = editingBk ? editingBk.unitPrice : svc.price;
   314→      const payload = {
   315→        serviceId: bkForm.serviceId,
   316→        guestName: bkForm.guestName,
   317→        guestPhone: bkForm.guestPhone,
   318→        date: bkForm.date,
   319→        time: bkForm.time,
   320→        quantity: qty,
   321→        unitPrice,
   322→        totalCost: unitPrice * qty,
   323→      };
   324→      if (editingBk) {
   325→        await apiUpdateDaytimeBooking(editingBk.id, payload);
   326→        toast.success(t("toastBookingUpdated"));
   327→      } else {
   328→        await apiCreateDaytimeBooking(payload);
   329→        toast.success(t("toastBookingCreated"));
   330→      }
   331→      setBkDialogOpen(false);
   332→      triggerRefresh();
   333→    } catch (err: unknown) {
   334→      toast.error(err instanceof Error ? err.message : t("toastFailedSaveBooking"));
   335→    } finally {
   336→      setBkSaving(false);
   337→    }
   338→  };
   339→
   340→  const handleDeleteBk = async () => {
   341→    if (!bkDeleteTarget) return;
   342→    try {
   343→      setBkDeleting(true);
   344→      await apiDeleteDaytimeBooking(bkDeleteTarget.id);
   345→      toast.success(t("toastBookingDeleted"));
   346→      setBkDeleteTarget(null);
   347→      triggerRefresh();
   348→    } catch (err: unknown) {
   349→      toast.error(err instanceof Error ? err.message : t("toastFailedDeleteBooking"));
   350→    } finally {
   351→      setBkDeleting(false);
   352→    }
   353→  };
   354→
   355→  // ─── Payment Recording ────────────────────────────────────────────────────
   356→
   357→  const openPayDialog = (bk: Booking) => {
   358→    setPayTarget(bk);
   359→    const remaining = bk.totalCost - bk.paidAmount;
   360→    setPayForm({ amount: String(remaining), method: "CASH" });
   361→    setPayDialogOpen(true);
   362→  };
   363→
   364→  const handleRecordPayment = async () => {
   365→    if (!payTarget || !payForm.amount || Number(payForm.amount) <= 0) {
   366→      toast.error(t("valValidAmount"));
   367→      return;
   368→    }
   369→    try {
   370→      setPaySaving(true);
   371→      const amount = Number(payForm.amount);
   372→      const newPaid = payTarget.paidAmount + amount;
   373→      const newStatus = newPaid >= payTarget.totalCost ? "PAID" : "PARTIAL";
   374→
   375→      await apiCreatePayment({
   376→        daytimeBookingId: payTarget.id,
   377→        amount,
   378→        method: payForm.method,
   379→      });
   380→
   381→      await apiUpdateDaytimeBooking(payTarget.id, {
   382→        paidAmount: newPaid,
   383→        paymentStatus: newStatus,
   384→        paymentMethod: payForm.method,
   385→      });
   386→
   387→      toast.success(t("toastPaymentRecorded", { amount: formatPrice(amount) }));
   388→      setPayDialogOpen(false);
   389→      setPayTarget(null);
   390→      triggerRefresh();
   391→    } catch (err: unknown) {
   392→      toast.error(err instanceof Error ? err.message : t("toastFailedRecordPayment"));
   393→    } finally {
   394→      setPaySaving(false);
   395→    }
   396→  };
   397→
   398→  // ─── Render ───────────────────────────────────────────────────────────────
   399→
   400→  return (
   401→    <div className="space-y-6 p-4 md:p-6">
   402→      {/* Header */}
   403→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   404→        <div>
   405→          <h1 className="text-2xl font-bold text-gray-900">{t("pageTitle"})</h1>
   406→          <p className="mt-1 text-sm text-gray-500">
   407→            t("pageSubtitle")
   408→          </p>
   409→        </div>
   410→        <Button
   411→          onClick={activeTab === "services" ? openCreateSvc : openCreateBk}
   412→          className="gap-2"
   413→        >
   414→          <Plus className="h-4 w-4" />
   415→          {activeTab === "services" ? t("btnAddService") : t("btnNewBooking"})
   416→        </Button>
   417→      </div>
   418→
   419→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   420→        <TabsList>
   421→          <TabsTrigger value="services" className="gap-2">
   422→            <Sun className="h-4 w-4" />
   423→            {t("tabServices"}) ({services.length})
   424→          </TabsTrigger>
   425→          <TabsTrigger value="bookings" className="gap-2">
   426→            <CalendarDays className="h-4 w-4" />
   427→            {t("tabBookings"}) ({bookings.length})
   428→          </TabsTrigger>
   429→        </TabsList>
   430→
   431→        {/* ─── Services Tab ────────────────────────────────────────────────── */}
   432→        <TabsContent value="services">
   433→          {svcLoading ? (
   434→            <div className="grid grid-cols-1 gap-4 md:grid-cols-2 lg:grid-cols-3">
   435→              {Array.from({ length: 6 }).map((_, i) => (
   436→                <Skeleton key={i} className="h-52 rounded-xl" />
   437→              )})
   438→            </div>
   439→          ) : services.length === 0 ? (
   440→            <div className="flex flex-col items-center justify-center rounded-xl border border-dashed py-16">
   441→              <Sun className="h-12 w-12 text-gray-300 mb-3" />
   442→              <p className="text-lg font-medium text-gray-500">{t("noServicesYet"})</p>
   443→              <p className="mt-1 text-sm text-gray-400">{t("noServicesYetDesc"})</p>
   444→              <Button onClick={openCreateSvc} variant="outline" className="mt-4 gap-2">
   445→                <Plus className="h-4 w-4" /> Add Service
   446→              </Button>
   447→            </div>
   448→          ) : (
   449→            <div className="grid grid-cols-1 gap-4 md:grid-cols-2 lg:grid-cols-3">
   450→              {services.map((svc) => {
   451→                const catColor = CATEGORY_COLORS[svc.category.toUpperCase()] || CATEGORY_COLORS.OTHER;
   452→                return (
   453→                  <Card key={svc.id} className={`gap-0 overflow-hidden py-0 transition-shadow hover:shadow-md ${!svc.active ? "opacity-60" : ""}`}>
   454→                    <CardContent className="p-4">
   455→                      <div className="flex items-start justify-between mb-3">
   456→                        <div className="flex items-center gap-2.5">
   457→                          <div className={`flex h-10 w-10 items-center justify-center rounded-lg ${catColor}`}>
   458→                            <Sun className="h-5 w-5" />
   459→                          </div>
   460→                          <div>
   461→                            <h3 className="font-semibold text-gray-900 leading-tight">{svc.name}</h3>
   462→                            {svc.category && (
   463→                              <Badge variant="outline" className={`mt-1 text-xs ${catColor}`}>{svc.category}</Badge>
   464→                            })
   465→                          </div>
   466→                        </div>
   467→                        <DropdownMenu>
   468→                          <DropdownMenuTrigger asChild>
   469→                            <Button variant="ghost" size="icon" className="h-8 w-8">
   470→                              <MoreVertical className="h-4 w-4" />
   471→                            </Button>
   472→                          </DropdownMenuTrigger>
   473→                          <DropdownMenuContent align="end">
   474→                            <DropdownMenuItem onClick={() => openEditSvc(svc})>
   475→                              <Pencil className="mr-2 h-4 w-4" /> {t("menuEdit"})
   476→                            </DropdownMenuItem>
   477→                            <DropdownMenuItem onClick={() => handleToggleActive(svc})>
   478→                              {svc.active ? t("menuDeactivate") : t("menuActivate"})
   479→                            </DropdownMenuItem>
   480→                            <DropdownMenuSeparator />
   481→                            <DropdownMenuItem
   482→                              className="text-rose-600 focus:text-rose-600"
   483→                              onClick={() => setSvcDeleteTarget(svc})
   484→                            >
   485→                              <Trash2 className="mr-2 h-4 w-4" /> {t("menuDelete"})
   486→                            </DropdownMenuItem>
   487→                          </DropdownMenuContent>
   488→                        </DropdownMenu>
   489→                      </div>
   490→
   491→                      <div className="grid grid-cols-2 gap-2 text-sm mb-3">
   492→                        <div className="flex items-center gap-1.5 text-gray-600">
   493→                          <DollarSign className="h-3.5 w-3.5 text-gray-400" />
   494→                          {formatPrice(svc.price})
   495→                        </div>
   496→                        {svc.duration && (
   497→                          <div className="flex items-center gap-1.5 text-gray-600">
   498→                            <Clock className="h-3.5 w-3.5 text-gray-400" />
   499→                            {svc.duration}
   500→                          </div>
   501→                        })
   502→                      </div>
   503→
   504→                      {svc.description && (
   505→                        <p className="text-xs text-gray-500 line-clamp-2">{svc.description}</p>
   506→                      })
   507→
   508→                      <div className="mt-3 flex items-center justify-between">
   509→                        <Badge
   510→                          variant="outline"
   511→                          className={svc.active ? "bg-emerald-50 text-emerald-700 border-emerald-200" : "bg-gray-100 text-gray-500 border-gray-200"}
   512→                        >
   513→                          {svc.active ? t("statusActive") : t("statusInactive"})
   514→                        </Badge>
   515→                        <Switch
   516→                          checked={svc.active}
   517→                          onCheckedChange={() => handleToggleActive(svc})
   518→                        />
   519→                      </div>
   520→                    </CardContent>
   521→                  </Card>
   522→                );
   523→              }})
   524→            </div>
   525→          })
   526→        </TabsContent>
   527→
   528→        {/* ─── Bookings Tab ────────────────────────────────────────────────── */}
   529→        <TabsContent value="bookings">
   530→          {bkLoading ? (
   531→            <div className="space-y-3">
   532→              {Array.from({ length: 5 }).map((_, i) => <Skeleton key={i} className="h-14 w-full rounded-lg" />})
   533→            </div>
   534→          ) : bookings.length === 0 ? (
   535→            <div className="flex flex-col items-center justify-center rounded-xl border border-dashed py-16">
   536→              <CalendarDays className="h-12 w-12 text-gray-300 mb-3" />
   537→              <p className="text-lg font-medium text-gray-500">{t("noBookingsYet"})</p>
   538→              <p className="mt-1 text-sm text-gray-400">{t("noBookingsYetDesc"})</p>
   539→              <Button onClick={openCreateBk} variant="outline" className="mt-4 gap-2">
   540→                <Plus className="h-4 w-4" /> New Booking
   541→              </Button>
   542→            </div>
   543→          ) : (
   544→            <div className="rounded-xl border bg-white overflow-hidden">
   545→              <div className="max-h-[480px] overflow-y-auto">
   546→                <Table>
   547→                  <TableHeader>
   548→                    <TableRow>
