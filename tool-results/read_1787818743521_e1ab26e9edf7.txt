   206→// ── Subscription Status Badge (for Rooms page header) ──
   207→function SubscriptionBadge() {
   208→  const { currentUser, subscription } = useAppStore();
   209→
   210→  // Only show for OPERATOR/STAFF
   211→  if (!currentUser || (currentUser.role !== "OPERATOR" && currentUser.role !== "STAFF")) return null;
   212→  if (!subscription) return null;
   213→
   214→  const isActive = subscription.status === "ACTIVE";
   215→  const isWarning = subscription.status === "WARNING";
   216→  const isExpired = subscription.status === "EXPIRED";
   217→  const isSuspended = subscription.status === "SUSPENDED";
   218→
   219→  const badgeClass = isActive
   220→    ? "border-emerald-300 bg-emerald-50 text-emerald-700"
   221→    : isSuspended || isExpired
   222→    ? "border-rose-300 bg-rose-50 text-rose-700"
   223→    : "border-amber-300 bg-amber-50 text-amber-700";
   224→
   225→  const dotClass = isActive
   226→    ? "bg-emerald-500"
   227→    : isSuspended || isExpired
   228→    ? "bg-rose-500"
   229→    : "bg-amber-500";
   230→
   231→  const label = isActive
   232→    ? "Active"
   233→    : isSuspended
   234→    ? "Suspended"
   235→    : isExpired
   236→    ? "Expired"
   237→    : "Expiring";
   238→
   239→  return (
   240→    <div className={`inline-flex items-center gap-1.5 rounded-full border px-2.5 py-1 text-xs font-semibold ${badgeClass} animate-subtle-pulse`}>
   241→      <span className={`relative flex h-2 w-2`}>
   242→        <span className={`absolute inline-flex h-full w-full animate-ping rounded-full opacity-75 ${dotClass}`} />
   243→        <span className={`relative inline-flex h-2 w-2 rounded-full ${dotClass}`} />
   244→      </span>
   245→      <span>{label}</span>
   246→      <span className="font-mono text-[11px] opacity-70">{Math.abs(subscription.daysRemaining)}d</span>
   247→    </div>
   248→  );
   249→}
