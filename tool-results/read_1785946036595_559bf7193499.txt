   192→// ── Helper: get nav items based on role ──
   193→function getNavItems(user: CurrentUser, isJointActive: boolean): NavItem[] {
   194→  switch (user.role) {
   195→    case "POLICE": {
   196→      const rank = (user.policeRank || "OFFICER") as PoliceRank;
   197→      const allowedPages = POLICE_RANK_PERMISSIONS[rank] || POLICE_RANK_PERMISSIONS.OFFICER;
   198→      const base = POLICE_NAV_ITEMS.filter((item) => allowedPages.includes(item.page));
   199→      if (isJointActive) {
   200→        const jointItems = POLICE_JOINT_ONLY_ITEMS.filter((item) => allowedPages.includes(item.page));
   201→        return [...base, ...jointItems];
   202→      }
   203→      return base;
   204→    }
   205→
   206→    case "SUPERUSER":
   207→      return SUPERUSER_NAV_ITEMS;
   208→
   209→    case "OPERATOR":
   210→      return ALL_NAV_ITEMS.filter((item) => !OPERATOR_EXCLUDED.has(item.page));
   211→
   212→    case "STAFF": {
   213→      const items: NavItem[] = [];
   214→      // Always give staff access to dashboard
   215→      items.push({
   216→        page: "dashboard",
   217→        label: "Dashboard",
   218→        icon: LayoutDashboard,
   219→      });
   220→      // Add items based on permissions (deduplicate by page key)
   221→      const seen = new Set<string>();
   222→      for (const perm of user.permissions) {
   223→        const mapped = PERMISSION_PAGE_MAP[perm];
   224→        if (mapped && !seen.has(mapped.page)) {
   225→          seen.add(mapped.page);
   226→          items.push(mapped);
   227→        }
   228→      }
   229→      return items;
   230→    }
   231→
   232→    default:
   233→      return [
   234→        { page: "dashboard", label: "Dashboard", icon: LayoutDashboard },
   235→      ];
   236→  }
   237→}

   239→// ── Helper: get initials from name ──
   240→function getInitials(name: string): string {
   241→  return name
   242→    .split(" ")
   243→    .map((w) => w[0])
   244→    .join("")
   245→    .toUpperCase()
   246→    .slice(0, 2);
   247→}

   336→  const isSuspended = subscription.status === "SUSPENDED";
   337→
   338→  // Collapsed mode — icon indicator
   339→  if (collapsed) {
   340→    const bgClass = isActive
   341→      ? "bg-emerald-100"
   342→      : isSuspended || isGrace
   343→      ? "bg-rose-100"
   344→      : "bg-amber-100";
   345→    const iconClass = isActive
   346→      ? "text-emerald-600"
   347→      : isSuspended || isGrace

   357→  }
   358→
   359→  // Expanded card
   360→  const containerClass = isActive
   361→    ? "border-emerald-200 bg-emerald-50"
   362→    : isSuspended || isGrace
   363→    ? "border-rose-200 bg-rose-50"
   364→    : "border-amber-200 bg-amber-50";

   366→  const iconColorClass = isActive
   367→    ? "text-emerald-600"
   368→    : isSuspended || isGrace

   370→    : "text-amber-600";

   373→    ? "text-emerald-800"
   374→    : isSuspended || isGrace
   375→    ? "text-rose-800"
   376→    : "text-amber-800";

   379→    ? "text-emerald-700"
   380→    : isSuspended || isGrace
   381→    ? "text-rose-700"
   382→    : "text-amber-700";

   384→  const badgeClass = isActive
   385→    ? "bg-emerald-200/60 text-emerald-800"
   386→    : isSuspended || isGrace
   387→    ? "bg-rose-200/60 text-rose-800"

   390→  const Icon = isSuspended || isGrace ? AlertTriangle : Clock;
   391→

   394→      <div className="flex items-center gap-2">
   395→        <Icon className={`h-4 w-4 shrink-0 ${iconColorClass}`} />
   396→        <p className={`text-xs font-semibold ${titleClass}`}>
   397→          {isActive
   398→            ? "Subscription Active"
   399→            : isSuspended
   400→            ? "Service Suspended"
   401→            : isGrace
   402→            ? "Subscription Expired"
   403→            : "Expiring Soon"}

   411→        </span>
   412→      </div>
   413→      {subscription.price > 0 ? (
   414→        <div className={`mt-1 text-[11px] ${subtextClass}`}>
   415→          Due: {subscription.price.toLocaleString()} ETB / {formatCycle(subscription.cycle)}

   418→        <div className={`mt-1 text-[11px] ${subtextClass}`}>
   419→          {formatCycle(subscription.cycle)} cycle — Free trial

   423→  );

   618→      {/* ── Subscription status for providers ── */}

   630→        {!collapsed && (

   636→            <span>Collapse</span>
   637→          </button>
   638→        )}
   639→      </div>
   640→    </div>
   641→  );
