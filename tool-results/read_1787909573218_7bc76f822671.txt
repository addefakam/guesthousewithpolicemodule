   141→export default function MySubscriptionPage() {
   142→  const { t } = useTranslation("subscription");
   143→  // ── i18n lookup maps ──
   144→  const PAYMENT_METHOD_LABELS: Record<string, string> = {
   145→    CHAPA: t("payMethodCHAPA"),
   146→    CASH: t("payMethodCASH"),
   147→    BANK_TRANSFER: t("payMethodBANK_TRANSFER"),
   148→    TELEBIRR: t("payMethodTELEBIRR"),
   149→    CBE_BIRR: t("payMethodCBE_BIRR"),
   150→    OTHER: t("payMethodOTHER"),
   151→  };
   152→  const CYCLE_LABELS: Record<string, string> = {
   153→    MONTHLY: t("cycleMONTHLY"),
   154→    QUARTERLY: t("cycleQUARTERLY"),
   155→    SEMI_ANNUAL: t("cycleSEMI_ANNUAL"),
   156→    YEARLY: t("cycleYEARLY"),
   157→  };
   158→  const STATUS_LABELS: Record<string, string> = {
   159→    ACTIVE: t("statusACTIVE"),
   160→    WARNING: t("statusWARNING"),
   161→    EXPIRED: t("statusEXPIRED"),
   162→    SUSPENDED: t("statusSUSPENDED"),
   163→  };
   164→  function fmtDaysRemaining(days: number): string {
   165→    if (days > 0) return t(days === 1 ? "daysRemaining_one" : "daysRemaining_other", { count: days });
   166→    if (days === 0) return t("expiresToday");
   167→    const abs = Math.abs(days);
   168→    return t(abs === 1 ? "expiredAgo_one" : "expiredAgo_other", { count: abs });
   169→  }
   170→  function fmtCycle(cycle: string): string {
   171→    return CYCLE_LABELS[cycle] || cycle;
   172→  }
   173→
   174→  const [data, setData] = useState<SubData | null>(null);
   175→  const [loading, setLoading] = useState(true);
   176→  const [showPayDialog, setShowPayDialog] = useState(false);
   177→  const [selectedPlan, setSelectedPlan] = useState<SubData["plans"][0] | null>(null);
   178→  const [submitting, setSubmitting] = useState(false);
   179→  const [payMethod, setPayMethod] = useState("");
   180→  const [payRef, setPayRef] = useState("");
   181→  const [payNotes, setPayNotes] = useState("");
   182→  const [payAmount, setPayAmount] = useState("");
   183→  const [showHistory, setShowHistory] = useState(false);
   184→  const [chapaVerifying, setChapaVerifying] = useState(false);
   185→  const [chapaVerifyResult, setChapaVerifyResult] = useState<string | null>(null);
   186→
   187→  // Chapa result is passed via sessionStorage by the main page.tsx
   188→  // (which reads URL params from Chapa redirect and stores them here)
   189→  const [chapaResult, setChapaResult] = useState<string | null>(null);
   190→  useEffect(() => {
   191→    const stored = sessionStorage.getItem("chapa_result");
   192→    if (stored) {
   193→      setChapaResult(stored);
   194→      sessionStorage.removeItem("chapa_result");
   195→      sessionStorage.removeItem("chapa_sub");
   196→      sessionStorage.removeItem("chapa_timestamp");
   197→    }
   198→  }, []);
   199→
   200→  const fetchData = useCallback(async () => {
   201→    try {
   202→      setLoading(true);
   203→      const res = await apiMySubscription();
   204→      setData(res as SubData);
   205→    } catch (err) {
   206→      const msg = err instanceof Error ? err.message : t("toastFailedLoad");
   207→      toast.error(`t("toastFailedLoad") ${msg}`);
   208→    } finally {
   209→      setLoading(false);
   210→    }
   211→  }, []);
   212→
   213→  useEffect(() => {
   214→    fetchData();
   215→  }, [fetchData]);
   216→
   217→  // Handle Chapa payment callback — actively verify with Chapa API
   218→  useEffect(() => {
   219→    if (chapaResult === "success") {
   220→      toast.success(t("toastChapaCompleted"));
   221→      setChapaVerifying(true);
   222→      setChapaVerifyResult(null);
   223→
   224→      // Wait a moment for Chapa to process, then actively verify
   225→      const timer = setTimeout(async () => {
   226→        try {
   227→          const res = await apiChapaClientVerify() as any;
   228→          if (res.verified > 0) {
   229→            setChapaVerifyResult("success");
   230→            toast.success(`${t("toastPaymentVerified")}`);
   231→          } else if (res.alreadyVerified) {
   232→            setChapaVerifyResult("success");
   233→          } else {
   234→            setChapaVerifyResult("pending");
   235→            toast.info(res.results?.[0] || t("toastChapaPending"));
   236→          }
   237→        } catch (err) {
   238→          setChapaVerifyResult("error");
   239→          toast.error(err instanceof Error ? err.message : t("toastVerifyFailed"));
   240→        } finally {
   241→          setChapaVerifying(false);
   242→          fetchData();
   243→        }
   244→      }, 3000);
   245→
   246→      return () => clearTimeout(timer);
   247→    }
   248→  }, [chapaResult, fetchData]);
   249→
   250→  const handleSelectPlan = (plan: SubData["plans"][0]) => {
   251→    setSelectedPlan(plan);
   252→    setPayAmount(String(plan.price));
   253→    setPayMethod("");
   254→    setPayRef("");
   255→    setPayNotes("");
   256→    setShowPayDialog(true);
   257→  };
   258→
   259→  const isChapaMethod = payMethod === "CHAPA";
   260→
   261→  const handleChapaPayment = async () => {
   262→    if (!selectedPlan || !payAmount || Number(payAmount) <= 0) {
   263→      toast.error(t("toastInvalidPayment"));
   264→      return;
   265→    }
   266→    setSubmitting(true);
   267→    try {
   268→      const res = await apiInitiateChapaPayment({
   269→        cycle: selectedPlan.cycle,
   270→        amount: Number(payAmount),
   271→        planId: selectedPlan.id,
   272→      });
   273→      if (res.checkoutUrl) {
   274→        // Redirect to Chapa checkout page
   275→        window.location.href = res.checkoutUrl;
   276→      } else {
   277→        toast.error(t("toastNoChapaUrl"));
   278→      }
   279→    } catch (e: unknown) {
   280→      toast.error(e instanceof Error ? e.message : t("toastChapaInitFailed"));
   281→    } finally {
   282→      setSubmitting(false);
   283→    }
   284→  };
   285→
   286→  const handleSubmitPayment = async () => {
   287→    // If Chapa is selected, redirect to Chapa checkout
   288→    if (isChapaMethod) {
   289→      await handleChapaPayment();
   290→      return;
   291→    }
   292→
   293→    if (!selectedPlan || !payMethod || !payAmount || Number(payAmount) <= 0) {
   294→      toast.error(t("toastFillRequired"));
   295→      return;
   296→    }
   297→    if (!isChapaMethod && !payRef.trim()) {
   298→      toast.error(t("toastRefRequired"));
   299→      return;
   300→    }
   301→    setSubmitting(true);
   302→    try {
   303→      await apiSubmitPayment({
   304→        planId: selectedPlan.id,
   305→        cycle: selectedPlan.cycle,
   306→        amount: Number(payAmount),
   307→        paymentMethod: payMethod,
   308→        referenceNo: payRef,
   309→        notes: payNotes,
   310→      });
   311→      toast.success(t("toastPaymentSubmitted"));
   312→      setShowPayDialog(false);
   313→      fetchData();
   314→    } catch (e: unknown) {
   315→      toast.error(e instanceof Error ? e.message : t("toastSubmitFailed"));
   316→    } finally {
   317→      setSubmitting(false);
   318→    }
   319→  };
   320→
   321→  const sub = data?.subscription;
   322→  const cur = data?.config.currencySymbol || "Br";
   323→  const totalBeds = data?.totalBeds || 0;
   324→  const pricePerBed = data?.config.pricePerBedPerDay || 0;
   325→  const pricingLocked = data?.config.pricingEnabled === false;
   326→
   327→  if (loading) return <LoadingSkeleton />;
   328→  if (!data || !sub) {
   329→    return (
   330→      <div className="flex flex-col items-center justify-center py-20 text-center">
   331→        <CreditCard className="w-12 h-12 text-slate-300 mb-3" />
   332→        <p className="text-sm font-medium text-slate-500">{t("noSubFound")}</p>
   333→        <p className="text-xs text-slate-400 mt-1">{t("noSubContact")}</p>
   334→      </div>
   335→    );
   336→  }
   337→
   338→  const isTrial = sub.price === 0 && data.payments.length === 0;
   339→  const isExpired = sub.status === "EXPIRED";
   340→  const penaltyPercent = data.config.latePaymentPenalty || 10;
   341→  const baseAmount = sub.price || 0;
   342→  const penaltyAmount = isExpired && baseAmount > 0
   343→    ? Math.round(baseAmount * (1 + penaltyPercent / 100))
   344→    : 0;
   345→  const statusColor = getStatusBadgeClasses(sub.status as "ACTIVE" | "WARNING" | "EXPIRED" | "SUSPENDED");
   346→
   347→  return (
   348→    <div className="space-y-4 p-3 sm:p-4 md:p-6 max-w-4xl mx-auto">
   349→      {/* Header */}
   350→      <div className="flex items-center justify-between">
   351→        <div>
   352→          <h2 className="text-base sm:text-lg font-semibold">{t("pageTitle")}</h2>
   353→          <p className="text-xs sm:text-sm text-muted-foreground">
   354→            t("pageSubtitle")
   355→          </p>
   356→        </div>
   357→        <Button
   358→          variant="outline"
   359→          size="sm"
   360→          onClick={fetchData}
   361→          className="h-8 text-xs"
   362→        >
   363→          <RefreshCw className="mr-1 h-3 w-3" />
   364→          {t("btnRefresh")}
   365→        </Button>
   366→      </div>
   367→
   368→      {/* Pricing Info Banner */}
   369→      <div className="flex items-center gap-2 p-3 bg-slate-50 rounded-xl border border-slate-200">
   370→        <div className="flex items-center justify-center w-8 h-8 rounded-lg bg-blue-100 shrink-0">
   371→          <Building2 className="w-4 h-4 text-blue-600" />
   372→        </div>
   373→        <div className="flex-1 min-w-0">
   374→          <p className="text-xs font-medium text-slate-700">
   375→            {t("pricingCalc", { cur, price: pricePerBed, beds: totalBeds, bedLabel: totalBeds === 1 ? t("bed_one") : t("bed_other"), total: (pricePerBed * totalBeds).toLocaleString() })}
   376→          </p>
   377→          <p className="text-[10px] text-muted-foreground">
   378→            {totalBeds === 0
   379→              ? t("noRoomsConfigured")
   380→              : `${t("bedsCalcDescription", { beds: totalBeds })}`}
   381→          </p>
   382→        </div>
   383→        {pricingLocked && (
   384→          <Badge variant="outline" className="text-[10px] border-slate-300 text-slate-500 bg-slate-100 shrink-0">
   385→            <Lock className="w-3 h-3 mr-1" />
   386→            t("ratesLocked")
   387→          </Badge>
   388→        )}
   389→      </div>
   390→
   391→      {/* ═══ Current Status Card ═══ */}
   392→      <Card className={`overflow-hidden ${
   393→        sub.status === "WARNING" ? "border-amber-300 bg-amber-50/30" :
   394→        sub.status === "EXPIRED" ? "border-2 border-rose-400 bg-rose-50/30" :
   395→        sub.status === "SUSPENDED" ? "border-slate-300" : ""
   396→      }`}>
   397→        <CardContent className="p-0">
   398→          {/* Status bar */}
   399→          <div className={`px-4 py-3 flex items-center justify-between ${
   400→            sub.status === "ACTIVE" ? "bg-emerald-50" :
   401→            sub.status === "WARNING" ? "bg-amber-50" :
   402→            sub.status === "EXPIRED" ? "bg-rose-100" :
   403→            "bg-slate-50"
   404→          }`}>
   405→            <div className="flex items-center gap-2">
   406→              {sub.status === "ACTIVE" ? (
   407→                <ShieldCheck className="w-5 h-5 text-emerald-600" />
   408→              ) : sub.status === "WARNING" ? (
   409→                <AlertTriangle className="w-5 h-5 text-amber-600" />
   410→              ) : sub.status === "EXPIRED" ? (
   411→                <XCircle className="w-5 h-5 text-rose-600" />
   412→              ) : (
   413→                <XCircle className="w-5 h-5 text-slate-400" />
   414→              )}
   415→              <span className={`text-sm font-semibold px-2.5 py-0.5 rounded-full border ${statusColor}`}>
   416→                {STATUS_LABELS[sub.status] || sub.status}
   417→              </span>
   418→              {isTrial && (
   419→                <Badge variant="outline" className="text-xs border-blue-200 text-blue-600 bg-blue-50">
   420→                  <Sparkles className="w-3 h-3 mr-1" />
   421→                  Free Trial
   422→                </Badge>
   423→              )}
   424→            </div>
   425→            <span className="text-xs font-medium text-muted-foreground">
   426→              {fmtDaysRemaining(sub.daysRemaining)}
   427→            </span>
   428→          </div>
   429→
   430→          {/* Details grid */}
   431→          <div className="p-4 grid grid-cols-2 sm:grid-cols-4 gap-4">
   432→            <div>
   433→              <p className="text-[10px] uppercase tracking-wider text-muted-foreground font-medium">{t("lblCurrentPlan")}</p>
   434→              <p className="text-sm font-semibold mt-0.5">
   435→                {isTrial ? t("freeTrial") : fmtCycle(sub.cycle)}
   436→              </p>
   437→            </div>
   438→            <div>
   439→              <p className="text-[10px] uppercase tracking-wider text-muted-foreground font-medium">{t("lblAmount")}</p>
   440→              <p className="text-sm font-semibold mt-0.5">
   441→                {isTrial ? t("freeTrial") : `${Number(sub.price).toLocaleString()} ${cur}`}
   442→              </p>
   443→            </div>
   444→            <div>
   445→              <p className="text-[10px] uppercase tracking-wider text-muted-foreground font-medium">{t("lblExpiresOn")}</p>
   446→              <p className="text-sm font-semibold mt-0.5">
   447→                {new Date(sub.endDate).toLocaleDateString("en-GB", {
   448→                  day: "numeric",
   449→                  month: "short",
   450→                  year: "numeric",
   451→                })}
   452→              </p>
   453→            </div>
   454→            <div>
   455→              <p className="text-[10px] uppercase tracking-wider text-muted-foreground font-medium">{t("lblProvider")}</p>
   456→              <p className="text-sm font-semibold mt-0.5 truncate">
   457→                {data.provider?.name || "—"}
   458→              </p>
   459→            </div>
   460→          </div>
   461→
   462→          {/* Payment instructions */}
   463→          {data.config.paymentInstructions && (
   464→            <div className="px-4 pb-4">
   465→              <div className="flex items-start gap-2 p-3 bg-blue-50 border border-blue-200 rounded-lg">
   466→                <Info className="w-4 h-4 text-blue-600 mt-0.5 shrink-0" />
   467→                <p className="text-xs text-blue-800 whitespace-pre-wrap leading-relaxed">
   468→                  {data.config.paymentInstructions}
   469→                </p>
   470→              </div>
   471→            </div>
   472→          )}
   473→
   474→          {/* Penalty alarm for EXPIRED */}
   475→          {isExpired && baseAmount > 0 && (
   476→            <div className="px-4 pb-4">
   477→              <div className="flex items-start gap-3 p-4 rounded-xl border-2 border-rose-300 bg-rose-50 animate-pulse">
   478→                <div className="shrink-0 mt-0.5 animate-bounce">
   479→                  <AlertTriangle className="w-5 h-5 text-rose-600" />
   480→                </div>
   481→                <div className="flex-1">
   482→                  <p className="text-sm font-bold text-rose-800">{t("penaltyTitle")}</p>
   483→                  <p className="text-xs text-rose-700 mt-0.5">
   484→                    {t("penaltyDesc", { days: Math.abs(sub.daysRemaining), percent: penaltyPercent })}
   485→                    
   486→                  </p>
   487→                  <div className="mt-2 flex items-center gap-2 flex-wrap">
   488→                    <span className="text-xs text-rose-600">{t("penaltyBase")}: <strong>{baseAmount.toLocaleString()} {cur}</strong></span>
   489→                    <span className="text-rose-300">+</span>
   490→                    <span className="text-xs text-rose-600">{t("penaltyLabel")}: <strong>{(penaltyAmount - baseAmount).toLocaleString()} {cur}</strong></span>
   491→                    <span className="text-rose-300">=</span>
   492→                    <span className="text-sm font-bold text-rose-900 bg-rose-100 border border-rose-300 px-2.5 py-1 rounded-lg">
   493→                      {t("penaltyTotalDue")}: {penaltyAmount.toLocaleString()} {cur}
   494→                    </span>
   495→                  </div>
   496→                </div>
   497→              </div>
   498→            </div>
   499→          )}
   500→
   501→          {/* Warning message for WARNING status */}
   502→          {sub.status === "WARNING" && (
   503→            <div className="px-4 pb-4">
   504→              <div className="flex items-start gap-2 p-3 rounded-lg border bg-amber-50 border-amber-200">
   505→                <AlertTriangle className="w-4 h-4 mt-0.5 shrink-0 text-amber-600" />
   506→                <div>
   507→                  <p className="text-xs font-semibold text-amber-800">{t("warningTitle")}</p>
   508→                  <p className="text-xs mt-1 text-amber-700">{t("warningDesc")}</p>
   509→                </div>
   510→              </div>
   511→            </div>
   512→          )}
   513→        </CardContent>
   514→      </Card>
   515→
   516→      {/* ═══ Available Plans ═══ */}
   517→      <div>
   518→        <div className="flex items-center justify-between mb-3">
   519→          <h3 className="text-sm font-semibold flex items-center gap-2">
   520→            <IndianRupee className="w-4 h-4" />
   521→            {t("availablePlans")}
   522→          </h3>
   523→          <span className="text-[10px] text-muted-foreground">
   524→            {t(data.plans.length === 1 ? "plansCount_one" : "plansCount_other", { count: data.plans.length })}
   525→          </span>
   526→        </div>
   527→
   528→        {data.plans.length === 0 ? (
