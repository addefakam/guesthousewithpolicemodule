   452→   452→/** Execute SQL via raw pg Client */
   453→   453→async function execViaPg(sql: string, params?: unknown[]): Promise<void> {
   454→   454→  const client = new Client({ connectionString: process.env.DATABASE_URL! });
   455→   455→  try {
   456→   456→    await client.connect();
   457→   457→    await client.query(sql, params);
   458→   458→  } finally {
   459→   459→    await client.end().catch(() => {});
   460→   460→  }
   461→   461→}

   463→   463→/** Execute DDL via Prisma $executeRawUnsafe (bypasses Prisma's table-existence check) */
   464→   464→async function execViaPrisma(sql: string): Promise<void> {
   465→   465→  const prisma = new PrismaClient({
   466→   466→    log: ["warn", "error"],
   467→   467→  });
   468→   468→  try {
   469→   469→    // $executeRawUnsafe sends SQL directly without schema validation
   470→   470→    await prisma.$executeRawUnsafe(sql);
   471→   471→  } finally {
   472→   472→    await prisma.$disconnect().catch(() => {});
   473→   473→  }
   474→   474→}
