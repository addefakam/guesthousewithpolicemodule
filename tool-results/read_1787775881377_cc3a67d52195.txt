   216→// ── Helper: get nav items based on role ──
   217→function getNavItems(user: CurrentUser): NavItem[] {
   218→  switch (user.role) {
   219→    case "POLICE": {
   220→      const rank = (user.policeRank || "OFFICER") as PoliceRank;
   221→      const allowedPages = POLICE_RANK_PERMISSIONS[rank] || POLICE_RANK_PERMISSIONS.OFFICER;
   222→      return POLICE_NAV_ITEMS.filter((item) => allowedPages.includes(item.page));
   223→    }
   224→
   225→    case "SUPERUSER": {
   226→      // Guest house owners (SUPERUSER with providerId) see the provider dashboard.
   227→      // System admins (SUPERUSER without providerId) see the admin dashboard.
   228→      if (user.providerId) {
   229→        return ALL_NAV_ITEMS.filter((item) => !OPERATOR_EXCLUDED.has(item.page));
   230→      }
   231→      return SUPERUSER_NAV_ITEMS;
   232→    }
   233→
   234→    case "OPERATOR":
   235→      return ALL_NAV_ITEMS.filter((item) => !OPERATOR_EXCLUDED.has(item.page));
   236→
   237→    case "STAFF": {
   238→      const items: NavItem[] = [];
   239→      // Always give staff access to dashboard
   240→      items.push({
   241→        page: "dashboard",
   242→        label: "Dashboard",
   243→        icon: LayoutDashboard,
   244→      });
   245→      // Add items based on permissions (deduplicate by page key)
   246→      const seen = new Set<string>();
   247→      for (const perm of user.permissions) {
   248→        const mapped = PERMISSION_PAGE_MAP[perm];
   249→        if (mapped && !seen.has(mapped.page)) {
   250→          seen.add(mapped.page);
   251→          items.push(mapped);
   252→        }
   253→      }
   254→      return items;
   255→    }
   256→
   257→    default:
   258→      return [
   259→        { page: "dashboard", label: "Dashboard", icon: LayoutDashboard },
   260→      ];
   261→  }
   262→}

   264→// ── Helper: get initials from name ──
   265→function getInitials(name: string | undefined | null): string {
   266→  if (!name) return "?";
   267→  return name
   268→    .split(" ")
   269→    .map((w) => w[0])
   270→    .join("")
   271→    .toUpperCase()
   272→    .slice(0, 2);
   273→}

   355→function SubscriptionStatusCard({ collapsed }: { collapsed: boolean }) {
   356→  const subscription = useAppStore((s) => s.subscription);
   357→
   358→  if (!subscription) return null;
   359→
   360→  const isActive = subscription.status === "ACTIVE";
   361→  const isWarning = subscription.status === "WARNING";
   362→  const isExpired = subscription.status === "EXPIRED";
   363→  const isSuspended = subscription.status === "SUSPENDED";
   364→
   365→  // Collapsed mode — icon indicator
   366→  if (collapsed) {
   367→    const bgClass = isActive
   368→      ? "bg-emerald-100"
   369→      : isSuspended || isExpired
   370→      ? "bg-rose-100"
   371→      : "bg-amber-100";
   372→    const iconClass = isActive
   373→      ? "text-emerald-600"
   374→      : isSuspended || isExpired
   375→      ? "text-rose-600"
   376→      : "text-amber-600";
   377→    return (
   378→      <div className="flex flex-col items-center gap-1 px-2 py-2">
   379→        <div className={`flex h-8 w-8 items-center justify-center rounded-full ${bgClass} animate-pulse shadow-sm`}> 
   380→          <Clock className={`size-4 ${iconClass}`} />
   381→        </div>
   382→      </div>
   383→    );
   384→  }
   385→
   386→  // Expanded card
   387→  const containerClass = isActive
   388→    ? "border-emerald-200 bg-emerald-50"
   389→    : isSuspended || isExpired
   390→    ? "border-rose-200 bg-rose-50"
   391→    : "border-amber-200 bg-amber-50";
   392→
   393→  const iconColorClass = isActive
   394→    ? "text-emerald-600"
   395→    : isSuspended || isExpired
   396→    ? "text-rose-600"
   397→    : "text-amber-600";
   398→
   399→  const titleClass = isActive
   400→    ? "text-emerald-800"
   401→    : isSuspended || isExpired
   402→    ? "text-rose-800"
   403→    : "text-amber-800";
   404→
   405→  const subtextClass = isActive
   406→    ? "text-emerald-700"
   407→    : isSuspended || isExpired
   408→    ? "text-rose-700"
   409→    : "text-amber-700";
   410→
   411→  const badgeClass = isActive
   412→    ? "bg-emerald-200/60 text-emerald-800"
   413→    : isSuspended || isExpired
   414→    ? "bg-rose-200/60 text-rose-800"
   415→    : "bg-amber-200/60 text-amber-800";
   416→
   417→  const Icon = isSuspended || isExpired ? AlertTriangle : Clock;
   418→
   419→  return (
   420→    <div className={`rounded-lg border p-3 ${containerClass} animate-subtle-pulse shadow-sm`}> 
   421→      <div className="flex items-center gap-2">
   422→        <Icon className={`h-4 w-4 shrink-0 ${iconColorClass}`} />
   423→        <p className={`text-xs font-semibold ${titleClass}`}>
   424→          {isActive
   425→            ? "Subscription Active"
   426→            : isSuspended
   427→            ? "Service Suspended"
   428→            : isExpired
   429→            ? "Subscription Expired"
   430→            : "Expiring Soon"}
   431→        </p>
   432→      </div>
   433→      <div className={`mt-1.5 flex items-center justify-between text-[11px] ${subtextClass}`}>
   434→        <span>{formatDaysRemaining(subscription.daysRemaining)}</span>
   435→        <span className={`inline-flex items-center gap-1 rounded-full px-1.5 py-0.5 font-semibold ${badgeClass}`}>
   436→          <Clock className="h-3 w-3" />
   437→          {Math.abs(subscription.daysRemaining)}d
   438→        </span>
   439→      </div>
   440→      {subscription.price > 0 ? (
   441→        <div className={`mt-1 text-[11px] ${subtextClass}`}>
   442→          Due: {subscription.price.toLocaleString()} ETB / {formatCycle(subscription.cycle)}
   443→        </div>
   444→      ) : (
   445→        <div className={`mt-1 text-[11px] ${subtextClass}`}>
   446→          {formatCycle(subscription.cycle)} cycle — Free trial
   447→        </div>
   448→      )}
   449→    </div>
   450→  );
   451→}

   675→      </div>
