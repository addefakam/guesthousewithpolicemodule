   194→// ── Helper: get nav items based on role ──
   195→function getNavItems(user: CurrentUser): NavItem[] {
   196→  switch (user.role) {
   197→    case "POLICE": {
   198→      const rank = (user.policeRank || "OFFICER") as PoliceRank;
   199→      const allowedPages = POLICE_RANK_PERMISSIONS[rank] || POLICE_RANK_PERMISSIONS.OFFICER;
   200→      return POLICE_NAV_ITEMS.filter((item) => allowedPages.includes(item.page));
   201→    }
   202→
   203→    case "SUPERUSER":
   204→      return SUPERUSER_NAV_ITEMS;
   205→
   206→    case "OPERATOR":
   207→      return ALL_NAV_ITEMS.filter((item) => !OPERATOR_EXCLUDED.has(item.page));
   208→
   209→    case "STAFF": {
   210→      const items: NavItem[] = [];
   211→      // Always give staff access to dashboard
   212→      items.push({
   213→        page: "dashboard",
   214→        label: "Dashboard",
   215→        icon: LayoutDashboard,
   216→      });
   217→      // Add items based on permissions
   218→      for (const perm of user.permissions) {
   219→        const mapped = PERMISSION_PAGE_MAP[perm];
   220→        if (mapped) {
   221→          items.push(mapped);
   222→        }
   223→      }
   224→      return items;
   225→    }
   226→
   227→    default:
   228→      return [
   229→        { page: "dashboard", label: "Dashboard", icon: LayoutDashboard },
   230→      ];
   231→  }
   232→}

   239→    .join("")
   240→    .toUpperCase()
   241→    .slice(0, 2);

   321→// ── Anomaly Detection Toggle Widget (sidebar) ──
   322→function AnomalyToggle({ collapsed }: { collapsed: boolean }) {
   323→  const currentUser = useAppStore((s) => s.currentUser);
   324→  const isAdmin = currentUser?.role === "POLICE" && currentUser?.policeRank === "ADMIN";
   325→  const [enabled, setEnabled] = useState(false);
   326→  const [toggling, setToggling] = useState(false);
   327→
   328→  const fetchStatus = useCallback(async () => {
   329→    try {
   330→      const data = await apiGetAnomalies("pageSize=1");
   331→      if (data.enabled !== undefined) setEnabled(data.enabled);
   332→    } catch { /* non-critical */ }
   333→  }, []);
   334→
   335→  useEffect(() => {
   336→    fetchStatus();
   337→  }, [fetchStatus]);
   338→
   339→  const handleToggle = async () => {
   340→    if (!isAdmin || toggling) return;
   341→    try {
   342→      setToggling(true);
   343→      const result = await apiToggleAnomalyDetection(!enabled);
   344→      setEnabled(result.enabled);
   345→    } catch { /* non-critical */ }
   346→    finally { setToggling(false); }
   347→  };
   348→
   349→  // Collapsed: prominent colored bar indicator
   350→  if (collapsed) {
   351→    return (
   352→      <div className="flex justify-center py-1.5 px-1">
   353→        <button
   354→          onClick={handleToggle}
   355→          disabled={!isAdmin}
   356→          title={enabled ? "Anomaly Detection: ON — click to disable" : "Anomaly Detection: OFF — click to enable"}
   357→          className={`relative h-7 w-7 rounded-lg flex items-center justify-center transition-all duration-200 ${
   358→            enabled
   359→              ? "bg-violet-100 text-violet-600 shadow-sm shadow-violet-100 hover:bg-violet-200"
   360→              : "bg-slate-100 text-slate-400 hover:bg-slate-200"
   361→          } ${isAdmin ? "cursor-pointer" : "cursor-not-allowed opacity-50"}`}
   362→        >
   363→          {toggling ? (
   364→            <Loader2 className="size-3.5 animate-spin" />
   365→          ) : enabled ? (
   366→            <ToggleRight className="size-4" />
   367→          ) : (
   368→            <ToggleLeft className="size-4" />
   369→          )}
   370→        </button>
   371→      </div>
   372→    );
   373→  }
   374→
   375→  // Expanded: prominent card-style toggle widget
   376→  return (
   377→    <div
   378→      className={`my-1 rounded-xl border p-3 transition-all duration-200 ${
   379→        enabled
   380→          ? "border-violet-200 bg-gradient-to-r from-violet-50 to-fuchsia-50 shadow-sm"
   381→          : "border-slate-200 bg-slate-50"
   382→      }`}
   383→    >
   384→      <div className="flex items-center justify-between gap-2">
   385→        <div className="flex items-center gap-2.5 min-w-0">
   386→          <div
   387→            className={`flex h-8 w-8 shrink-0 items-center justify-center rounded-lg transition-colors ${
   388→              enabled ? "bg-violet-500 text-white shadow-sm shadow-violet-200" : "bg-slate-200 text-slate-400"
   389→            }`}
   390→          >
   391→            {toggling ? (
   392→              <Loader2 className="size-4 animate-spin" />
   393→            ) : enabled ? (
   394→              <ToggleRight className="size-4" />
   395→            ) : (
   396→              <ToggleLeft className="size-4" />
   397→            )}
   398→          </div>
   399→          <div className="min-w-0">
   400→            <p className={`text-xs font-bold leading-tight ${enabled ? "text-violet-800" : "text-slate-600"}`}>
   401→              Smart Detection
   402→            </p>
   403→            <p className={`text-[10px] leading-tight mt-0.5 ${enabled ? "text-violet-600" : "text-slate-400"}`}>
   404→              {enabled ? "Active — auto-scanning" : "Inactive — no scan"}
   405→            </p>
   406→          </div>
   407→        </div>
   408→        {/* Toggle switch */}
   409→        <button
   410→          onClick={handleToggle}
   411→          disabled={!isAdmin || toggling}
   412→          className={`relative inline-flex h-6 w-11 shrink-0 items-center rounded-full transition-colors duration-300 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-violet-400 focus-visible:ring-offset-1 ${
   413→            enabled ? "bg-violet-500" : "bg-slate-300"
   414→          } ${!isAdmin ? "opacity-50 cursor-not-allowed" : "cursor-pointer"}`}
   415→          title={!isAdmin ? "Only ADMIN can toggle" : enabled ? "Click to disable" : "Click to enable"}
   416→        >
   417→          <span
   418→            className={`inline-block h-4 w-4 transform rounded-full bg-white shadow-md transition-transform duration-300 ${
   419→              enabled ? "translate-x-6" : "translate-x-1"

   421→          />
   422→        </button>
   423→      </div>
   424→      {!isAdmin && (
   425→        <p className="mt-2 text-[10px] text-slate-400 italic">Only ADMIN can toggle this setting</p>

   429→}

   446→      ? "bg-emerald-100"

   448→      ? "bg-rose-100"
   449→      : "bg-amber-100";

   451→      ? "text-emerald-600"
   452→      : isSuspended || isGrace

   468→    ? "border-rose-200 bg-rose-50"

   473→    : isSuspended || isGrace
   474→    ? "text-rose-600"
   475→    : "text-amber-600";

   477→  const titleClass = isActive
   478→    ? "text-emerald-800"
   479→    : isSuspended || isGrace
   480→    ? "text-rose-800"
   481→    : "text-amber-800";

   492→    ? "bg-rose-200/60 text-rose-800"
   493→    : "bg-amber-200/60 text-amber-800";

   495→  const Icon = isSuspended || isGrace ? AlertTriangle : Clock;

   508→            : "Expiring Soon"}

   514→          <Clock className="h-3 w-3" />

   516→        </span>
   517→      </div>

   523→        <div className={`mt-1 text-[11px] ${subtextClass}`}>
   524→          {formatCycle(subscription.cycle)} cycle — Free trial
   525→        </div>
   526→      )}
   527→    </div>
   528→  );
   529→}

   686→              <Separator className="my-2 bg-slate-200/60" />

   700→              title="Start Joint Session"

   731→        {user.role !== "SUPERUSER" && (
   732→          <button
   733→            onClick={onLogout}
   734→            className="flex w-full items-center gap-3 rounded-lg px-3 py-2.5 text-sm font-medium text-rose-600 transition-colors hover:bg-rose-50 outline-none focus-visible:ring-2 focus-visible:ring-ring"

   736→            <LogOut className="size-[18px]" />
   737→            {!collapsed && <span>Logout</span>}
   738→          </button>
   739→        )}
   740→      </div>
   741→    </div>
   742→  );
   743→}
