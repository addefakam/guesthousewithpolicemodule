    90→const STATUS_CONFIG: Record<string, { label: string; variant: "default" | "secondary" | "destructive" | "outline" }> = {
    91→  PENDING: { label: "Pending", variant: "secondary" },
    92→  APPROVED: { label: "Approved", variant: "default" },
    93→  REJECTED: { label: "Rejected", variant: "destructive" },
    94→  SUSPENDED: { label: "Suspended", variant: "outline" },
    95→};
    96→
    97→const STATUS_BADGE_CLASS: Record<string, string> = {
    98→  PENDING: "bg-yellow-100 text-yellow-800 hover:bg-yellow-100 border-yellow-200",
    99→  APPROVED: "bg-emerald-100 text-emerald-800 hover:bg-emerald-100 border-emerald-200",
   100→  REJECTED: "bg-red-100 text-red-800 hover:bg-red-100 border-red-200",
   101→  SUSPENDED: "bg-orange-100 text-orange-800 hover:bg-orange-100 border-orange-200",
   102→};
   103→
   104→function StatusBadge({ status }: { status: string }) {
   105→  const cfg = STATUS_CONFIG[status] || { label: status, variant: "secondary" as const };
   106→  return (
   107→    <Badge variant={cfg.variant} className={STATUS_BADGE_CLASS[status] || ""}>
   108→      {cfg.label}
   109→    </Badge>
   110→  );
   111→}

   113→export default function ProvidersPage() {
   114→  const { refreshKey, triggerRefresh, currentUser } = useAppStore();
   115→  const isSuperuser = currentUser?.role === "SUPERUSER";
   116→  const [providers, setProviders] = useState<Provider[]>([]);
   117→  const [loading, setLoading] = useState(true);
   118→  const [selectedProvider, setSelectedProvider] = useState<Provider | null>(null);
   119→  const [detailOpen, setDetailOpen] = useState(false);
   120→
   121→  // Coordinate editing
   122→  const [coordProvider, setCoordProvider] = useState<Provider | null>(null);
   123→  const [coordOpen, setCoordOpen] = useState(false);
   124→  const [editLat, setEditLat] = useState(9.02);
   125→  const [editLng, setEditLng] = useState(38.75);
   126→  const [savingCoord, setSavingCoord] = useState(false);
   127→  const [geocodingOne, setGeocodingOne] = useState(false);
   128→
   129→  // Batch geocode
   130→  const [geocodingAll, setGeocodingAll] = useState(false);
   131→  const [batchResult, setBatchResult] = useState<{ updated: number; failed: number; total: number; results?: any[] } | null>(null);
   132→  const [batchOpen, setBatchOpen] = useState(false);
   133→
   134→  // Action dialogs
   135→  const [rejectDialog, setRejectDialog] = useState<Provider | null>(null);
   136→  const [rejectReason, setRejectReason] = useState("");
   137→  const [confirmAction, setConfirmAction] = useState<{ provider: Provider; action: string } | null>(null);
   138→  const [actioning, setActioning] = useState(false);
   139→
   140→  // Suspension dialog (with reason + notification)
   141→  const [suspendDialog, setSuspendDialog] = useState<Provider | null>(null);
   142→  const [suspensionReason, setSuspensionReason] = useState("");
   143→  const [providerMessage, setProviderMessage] = useState("");
   144→  const [suspending, setSuspending] = useState(false);
   145→
   146→  const pagination = usePagination({ totalItems: providers.length, initialPageSize: 5, pageSizeOptions: [5, 10, 20, 50] });
   147→  const paginatedProviders = useMemo(() => pagination.paginate(providers), [providers, pagination]);
   148→
   149→  const fetchProviders = useCallback(async () => {
   150→    try {
   151→      setLoading(true);
   152→      const data = await apiGetProviders();
   153→      setProviders(Array.isArray(data) ? data : []);
   154→    } catch (err: unknown) {
   155→      const message = err instanceof Error ? err.message : "Failed to load providers";
   156→      toast.error(message);
   157→    } finally {
   158→      setLoading(false);
   159→    }
   160→  }, []);
   161→
   162→  useEffect(() => {
   163→    fetchProviders();
   164→  }, [fetchProviders, refreshKey]);
   165→
   166→  const openDetail = (provider: Provider) => {
   167→    setSelectedProvider(provider);
   168→    setDetailOpen(true);
   169→  };
   170→
   171→  const openReject = (provider: Provider) => {
   172→    setRejectDialog(provider);
   173→    setRejectReason("");
   174→  };
   175→
   176→  const openCoordPicker = (provider: Provider) => {
   177→    setCoordProvider(provider);
   178→    setEditLat(provider.latitude || 9.02);
   179→    setEditLng(provider.longitude || 38.75);
   180→    setCoordOpen(true);
   181→  };
   182→
   183→  const geocodeSingle = async () => {
   184→    if (!coordProvider || !coordProvider.address) {
   185→      toast.error("Provider has no address to geocode");
   186→      return;
   187→    }
   188→    try {
   189→      setGeocodingOne(true);
   190→      const result: any = await apiGeocodeAddress(coordProvider.address);
   191→      if (result.lat && result.lng) {
   192→        setEditLat(result.lat);
   193→        setEditLng(result.lng);
   194→        toast.success(`Found: ${result.lat.toFixed(4)}, ${result.lng.toFixed(4)}`);
   195→      } else {
   196→        toast.error("Could not find coordinates for this address");
   197→      }
   198→    } catch {
   199→      toast.error("Geocoding failed");
   200→    } finally {
   201→      setGeocodingOne(false);
   202→    }
   203→  };
   204→
   205→  const saveCoordinates = async () => {
   206→    if (!coordProvider) return;
   207→    try {
   208→      setSavingCoord(true);
   209→      await apiUpdateProvider(coordProvider.id, { status: coordProvider.status, latitude: editLat, longitude: editLng });
   210→      toast.success("Coordinates updated");
   211→      setCoordOpen(false);
   212→      triggerRefresh();
   213→    } catch {
   214→      toast.error("Failed to save coordinates");
   215→    } finally {
   216→      setSavingCoord(false);
   217→    }
   218→  };
   219→
   220→  const batchGeocode = async () => {
   221→    try {
   222→      setGeocodingAll(true);
   223→      const result: any = await apiGeocodeBatch();
   224→      setBatchResult(result);
   225→      toast.success(result.message || `Geocoded ${result.updated} providers`);
   226→      triggerRefresh();
   227→    } catch {
   228→      toast.error("Batch geocoding failed");
   229→    } finally {
   230→      setGeocodingAll(false);
   231→    }
   232→  };
   233→
   234→  const handleReject = async () => {
   235→    if (!rejectDialog || !rejectReason.trim()) {
   236→      toast.error("Rejection reason is required");
   237→      return;
   238→    }
   239→    try {
   240→      setActioning(true);
   241→      await apiUpdateProvider(rejectDialog.id, { status: "REJECTED", rejectionReason: rejectReason.trim() });
   242→      toast.success("Provider rejected");
   243→      setRejectDialog(null);
   244→      setRejectReason("");
   245→      triggerRefresh();
   246→    } catch (err: unknown) {
   247→      const message = err instanceof Error ? err.message : "Failed to reject provider";
   248→      toast.error(message);
   249→    } finally {
   250→      setActioning(false);
   251→    }
   252→  };
   253→
   254→  const handleStatusAction = async (provider: Provider, status: string) => {
   255→    try {
   256→      setActioning(true);
   257→      await apiUpdateProvider(provider.id, { status });
   258→      toast.success(`Provider ${status.toLowerCase()}`);
   259→      setConfirmAction(null);
   260→      triggerRefresh();
   261→    } catch (err: unknown) {
   262→      const message = err instanceof Error ? err.message : "Failed to update provider";
   263→      toast.error(message);
   264→    } finally {
   265→      setActioning(false);
   266→    }
   267→  };
   268→
   269→  const openSuspend = (provider: Provider) => {
   270→    setSuspendDialog(provider);
   271→    setSuspensionReason("");
   272→    setProviderMessage("");
   273→  };
   274→
   275→  const handleSuspend = async () => {
   276→    if (!suspendDialog) return;
   277→    if (!suspensionReason.trim()) {
   278→      toast.error("Please provide a reason for suspension");
   279→      return;
   280→    }
   281→    try {
   282→      setSuspending(true);
   283→      await apiPoliceSuspendProvider({
   284→        providerId: suspendDialog.id,
   285→        suspensionReason: suspensionReason.trim(),
   286→        providerMessage: providerMessage.trim() || undefined,
   287→      });
   288→      toast.success(`"${suspendDialog.name}" has been suspended. Notification sent to provider.`);
   289→      setSuspendDialog(null);
   290→      setSuspensionReason("");
   291→      setProviderMessage("");
   292→      triggerRefresh();
   293→    } catch (err: unknown) {
   294→      toast.error(err instanceof Error ? err.message : "Failed to suspend provider");
   295→    } finally {
   296→      setSuspending(false);
   297→    }
   298→  };
   299→
   300→  const formatDate = (dateStr: string | null) => {
   301→    if (!dateStr) return "—";
   302→    return new Date(dateStr).toLocaleDateString("en-US", {
   303→      year: "numeric",
   304→      month: "short",
   305→      day: "numeric",
   306→    });
   307→  };
   308→
   309→  const approvedCount = providers.filter((p) => p.status === "APPROVED").length;
   310→  const pendingCount = providers.filter((p) => p.status === "PENDING").length;
   311→  const withCoords = providers.filter(
   312→    (p) => p.latitude !== 9.02 || p.longitude !== 38.75,
   313→  ).length;
   314→
   315→  return (
   316→    <div className="space-y-4 p-3 sm:p-4 md:p-6">
   317→      {/* Summary Cards */}
   318→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-2 sm:gap-3">
   319→        <div className="rounded-xl border bg-card p-3 sm:p-4 shadow-sm">
   320→          <div className="flex items-center gap-2 sm:gap-3">
   321→            <div className="flex h-8 w-8 sm:h-10 sm:w-10 items-center justify-center rounded-lg bg-slate-100">
   322→              <Building2 className="h-4 w-4 sm:h-5 sm:w-5 text-slate-600" />
   323→            </div>
   324→            <div>
   325→              <p className="text-[10px] sm:text-sm text-muted-foreground">Total</p>
   326→              <p className="text-lg sm:text-2xl font-bold">{providers.length}</p>
   327→            </div>
   328→          </div>
   329→        </div>
   330→        <div className="rounded-xl border bg-card p-3 sm:p-4 shadow-sm">
   331→          <div className="flex items-center gap-2 sm:gap-3">
   332→            <div className="flex h-8 w-8 sm:h-10 sm:w-10 items-center justify-center rounded-lg bg-emerald-50">
   333→              <CheckCircle2 className="h-4 w-4 sm:h-5 sm:w-5 text-emerald-600" />
   334→            </div>
   335→            <div>
   336→              <p className="text-[10px] sm:text-sm text-muted-foreground">Approved</p>
   337→              <p className="text-lg sm:text-2xl font-bold text-emerald-600">{approvedCount}</p>
   338→            </div>
   339→          </div>
   340→        </div>
   341→        <div className="rounded-xl border bg-card p-3 sm:p-4 shadow-sm">
   342→          <div className="flex items-center gap-2 sm:gap-3">
   343→            <div className="flex h-8 w-8 sm:h-10 sm:w-10 items-center justify-center rounded-lg bg-yellow-50">
   344→              <ShieldCheck className="h-4 w-4 sm:h-5 sm:w-5 text-yellow-600" />
   345→            </div>
   346→            <div>
   347→              <p className="text-[10px] sm:text-sm text-muted-foreground">Pending</p>
   348→              <p className="text-lg sm:text-2xl font-bold text-yellow-600">{pendingCount}</p>
   349→            </div>
   350→          </div>
   351→        </div>
   352→        <div className="rounded-xl border bg-card p-3 sm:p-4 shadow-sm">
   353→          <div className="flex items-center gap-2 sm:gap-3">
   354→            <div className="flex h-8 w-8 sm:h-10 sm:w-10 items-center justify-center rounded-lg bg-blue-50">
   355→              <MapPinned className="h-4 w-4 sm:h-5 sm:w-5 text-blue-600" />
   356→            </div>
   357→            <div>
   358→              <p className="text-[10px] sm:text-sm text-muted-foreground">With Coords</p>
   359→              <p className="text-lg sm:text-2xl font-bold text-blue-600">{withCoords}/{providers.length}</p>
   360→            </div>
   361→          </div>
   362→        </div>
   363→      </div>
   364→
   365→      {/* Geocode All Button */}
   366→      <div className="flex items-center justify-between">
   367→        <div>
   368→          <h2 className="text-base sm:text-lg font-semibold">
   369→            {isSuperuser ? "Guesthouses" : "Provider Applications"}
   370→          </h2>
   371→          <p className="text-xs sm:text-sm text-muted-foreground">
   372→            {isSuperuser
   373→              ? "View registered guesthouses and suspend any that violate policy. Approval / rejection of pending registrations is handled by the Police module."
   374→              : "Manage registrations, licensing, and map locations"}
   375→          </p>
   376→        </div>
   377→        <div className="flex gap-2">
   378→          {!isSuperuser && (
   379→            <Button variant="outline" size="sm" onClick={() => setBatchOpen(true)} disabled={geocodingAll}>
   380→              <Globe className={`mr-1 h-3.5 w-3.5 ${geocodingAll ? "animate-spin" : ""}`} />
   381→              {geocodingAll ? "Geocoding..." : "Geocode All"}
   382→            </Button>
   383→          )}
   384→        </div>
   385→      </div>
   386→
   387→      {/* Providers Table/Cards */}
   388→      <div className="rounded-xl border bg-card shadow-sm">
   389→        {loading ? (
   390→          <div className="space-y-3 p-4 sm:p-6">
   391→            {Array.from({ length: 5 }).map((_, i) => (
   392→              <Skeleton key={i} className="h-20 w-full sm:h-12" />
   393→            ))}
   394→          </div>
   395→        ) : providers.length === 0 ? (
   396→          <div className="flex flex-col items-center justify-center py-12 sm:py-16 text-center">
   397→            <Building2 className="mb-3 h-10 w-10 sm:h-12 sm:w-12 text-muted-foreground/40" />
   398→            <p className="text-xs sm:text-sm text-muted-foreground">No providers registered yet</p>
   399→          </div>
   400→        ) : (
   401→          <>
   402→            {/* Mobile/Tablet: Card layout */}
   403→            <div className="divide-y lg:hidden">
   404→              {paginatedProviders.map((provider) => (
   405→                <div key={provider.id} className="p-3 sm:p-4">
   406→                  <div className="flex items-start justify-between gap-2">
   407→                    <button className="min-w-0 flex-1 text-left" onClick={() => openDetail(provider)}>
   408→                      <div className="flex items-center gap-2">
   409→                        <p className="truncate text-sm font-medium">{provider.name}</p>
   410→                        <StatusBadge status={provider.status} />
   411→                      </div>
   412→                      <div className="mt-1.5 flex flex-wrap items-center gap-x-3 gap-y-1 text-xs text-muted-foreground">
   413→                        <span className="flex items-center gap-1"><User className="h-3 w-3" /> {provider.ownerName}</span>
   414→                        <span className="flex items-center gap-1"><Phone className="h-3 w-3" /> {provider.phone}</span>
   415→                      </div>
   416→                      {provider.address && (
   417→                        <div className="mt-1 flex items-center gap-1 text-[10px] text-muted-foreground">
   418→                          <MapPin className="h-3 w-3" /> {provider.address}
   419→                        </div>
   420→                      )}
   421→                      <div className="mt-1 flex items-center gap-2 text-[10px]">
   422→                        <span className="rounded bg-slate-100 px-1.5 py-0.5 font-medium">{provider.type.replace(/_/g, " ")}</span>
   423→                        {(provider.latitude !== 9.02 || provider.longitude !== 38.75) ? (
   424→                          <span className="flex items-center gap-0.5 text-emerald-600"><Crosshair className="h-2.5 w-2.5" /> {provider.latitude.toFixed(4)}, {provider.longitude.toFixed(4)}</span>
   425→                        ) : (
   426→                          <span className="text-muted-foreground">No coordinates</span>
   427→                        )}
   428→                      </div>
   429→                    </button>
   430→                  </div>
   431→
   432→                  <div className="mt-2.5 flex items-center gap-1.5 border-t pt-2.5 flex-wrap">
   433→                    <button className="flex items-center gap-1 rounded-md px-2.5 py-1.5 text-xs font-medium text-slate-600 hover:bg-slate-100" onClick={() => openDetail(provider)}>
   434→                      <Eye className="h-3.5 w-3.5" /> Details
   435→                    </button>
   436→                    {!isSuperuser && (
   437→                      <button className="flex items-center gap-1 rounded-md px-2.5 py-1.5 text-xs font-medium text-blue-600 hover:bg-blue-50" onClick={() => openCoordPicker(provider)}>
   438→                        <MapPinned className="h-3.5 w-3.5" /> Set Location
   439→                      </button>
   440→                    )}
   441→                    {!isSuperuser && provider.status !== "APPROVED" && (
   442→                      <button className="flex items-center gap-1 rounded-md px-2.5 py-1.5 text-xs font-medium text-emerald-600 hover:bg-emerald-50" onClick={() => setConfirmAction({ provider, action: "APPROVED" })}>
   443→                        <CheckCircle2 className="h-3.5 w-3.5" /> Approve
   444→                      </button>
   445→                    )}
   446→                    {!isSuperuser && provider.status !== "REJECTED" && (
   447→                      <button className="flex items-center gap-1 rounded-md px-2.5 py-1.5 text-xs font-medium text-red-600 hover:bg-red-50" onClick={() => openReject(provider)}>
   448→                        <XCircle className="h-3.5 w-3.5" /> Reject
   449→                      </button>
   450→                    )}
   451→                    {provider.status === "APPROVED" && (
   452→                      <button className="flex items-center gap-1 rounded-md px-2.5 py-1.5 text-xs font-medium text-orange-600 hover:bg-orange-50" onClick={() => openSuspend(provider)}>
   453→                        <Ban className="h-3.5 w-3.5" /> Suspend
   454→                      </button>
   455→                    )}
   456→                  </div>
   457→                </div>
   458→              ))}
   459→            </div>
   460→
   461→            {/* Desktop: Table layout */}
   462→            <div className="hidden lg:block overflow-x-auto">
   463→              <Table>
   464→                <TableHeader>
   465→                  <TableRow>
   466→                    <TableHead>Provider Name</TableHead>
   467→                    <TableHead>Address</TableHead>
   468→                    <TableHead>Owner</TableHead>
   469→                    <TableHead>Phone</TableHead>
   470→                    <TableHead>Coordinates</TableHead>
   471→                    <TableHead>Status</TableHead>
   472→                    <TableHead className="text-right">Actions</TableHead>
   473→                  </TableRow>
   474→                </TableHeader>
   475→                <TableBody>
   476→                  {paginatedProviders.map((provider) => (
   477→                    <TableRow key={provider.id} className="cursor-pointer hover:bg-muted/50" onClick={() => openDetail(provider)}>
   478→                      <TableCell className="font-medium">{provider.name}</TableCell>
   479→                      <TableCell className="max-w-[150px] truncate text-xs">{provider.address || "—"}</TableCell>
   480→                      <TableCell>{provider.ownerName}</TableCell>
   481→                      <TableCell>{provider.phone}</TableCell>
   482→                      <TableCell>
   483→                        {(provider.latitude !== 9.02 || provider.longitude !== 38.75) ? (
   484→                          <span className="flex items-center gap-1 text-[10px] font-mono text-emerald-600">
   485→                            <Crosshair className="h-3 w-3" /> {provider.latitude.toFixed(4)}, {provider.longitude.toFixed(4)}
   486→                          </span>

   488→                          <span className="text-[10px] text-muted-foreground">Not set</span>
   489→                        )}
   490→                      </TableCell>
   491→                      <TableCell><StatusBadge status={provider.status} /></TableCell>
   492→                      <TableCell className="text-right" onClick={(e) => e.stopPropagation()}>
   493→                        <div className="flex items-center justify-end gap-1">
   494→                          {!isSuperuser && (
   495→                            <Button size="sm" variant="ghost" className="h-8 text-blue-600 hover:bg-blue-50" onClick={() => openCoordPicker(provider)}>
   496→                              <MapPinned className="mr-1 h-3.5 w-3.5" /> Location
   497→                            </Button>
