   166→// ── Helpers ──
   167→function formatDate(d: string) {
   168→  if (!d) return "—";
   169→  try { return new Date(d + "T00:00:00").toLocaleDateString("en-US", { month: "short", day: "numeric" }); } catch { return d; }
   170→}
   171→
   172→function formatCurrency(v: number) {
   173→  return new Intl.NumberFormat("en-US", { style: "currency", currency: "ETB", maximumFractionDigits: 0 }).format(v);
   174→}
   175→
   176→function todayStr() { return new Date().toISOString().split("T")[0]; }
   177→
   178→function addDays(d: string, n: number) {
   179→  const dt = new Date(d + "T00:00:00");
   180→  dt.setDate(dt.getDate() + n);
   181→  return dt.toISOString().split("T")[0];
   182→}
   183→
   184→function getFloorFromNumber(num: string): number | null {
   185→  const match = num.match(/^\d/);
   186→  return match ? parseInt(match[0], 10) : null;
   187→}
   188→
   189→function parseAmenities(amenitiesStr: string | null | undefined): string[] {
   190→  if (!amenitiesStr) return [];
   191→  try {
   192→    const parsed = JSON.parse(amenitiesStr);
   193→    return Array.isArray(parsed) ? parsed : [];
   194→  } catch {
   195→    return amenitiesStr.split(",").map((s) => s.trim()).filter(Boolean);
   196→  }
   197→}
