   204→   204→  async function handleVerifyAction(action: "approve" | "reject") {

   213→   213→      if (currentPayIndex + 1 < pendingPayments.length) {
   214→   214→        toast.success(`${label} payment of ${currentPayment.amount.toLocaleString()} ETB — ${pendingPayments.length - currentPayIndex - 1} remaining`);

   216→   216→        // Don't increment index since we removed the current item
   217→   217→        setCurrentPayIndex(0);
   218→   218→        setVerifyReason("");
   219→   219→        setVerifyDeclineMode(false);
   220→   220→      } else {
   221→   221→        toast.success(`${label} payment for ${pendingRow?.providerName}`);

   226→   226→      toast.error(err instanceof Error ? err.message : `Failed to ${action} payment`);

   228→   228→      setVerifyAction(null);
