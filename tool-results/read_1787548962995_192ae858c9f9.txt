   104→function StatusBadge({ status }: { status: string }) {
   105→  const cfg = STATUS_CONFIG[status] || { label: status, variant: "secondary" as const };

   107→    <Badge variant={cfg.variant} className={STATUS_BADGE_CLASS[status] || ""}>

   149→export default function ProvidersPage() {
   150→  const { refreshKey, triggerRefresh, currentUser } = useAppStore();
   151→  const isSuperuser = currentUser?.role === "SUPERUSER";
   152→  const [providers, setProviders] = useState<Provider[]>([]);
   153→  const [loading, setLoading] = useState(true);
   154→  const [selectedProvider, setSelectedProvider] = useState<Provider | null>(null);
   155→  const [detailOpen, setDetailOpen] = useState(false);
   156→
   157→  // Registration form
   158→  const [registerOpen, setRegisterOpen] = useState(false);
   159→  const [registerForm, setRegisterForm] = useState<RegisterForm>(emptyRegisterForm);
   160→  const [registering, setRegistering] = useState(false);
   161→
   162→
   163→
   164→  // Action dialogs
   165→  const [rejectDialog, setRejectDialog] = useState<Provider | null>(null);
   166→  const [rejectReason, setRejectReason] = useState("");
   167→  const [confirmAction, setConfirmAction] = useState<{ provider: Provider; action: string } | null>(null);
   168→  const [actioning, setActioning] = useState(false);
   169→
   170→  // Suspension dialog (with reason + notification)
   171→  const [suspendDialog, setSuspendDialog] = useState<Provider | null>(null);
   172→  const [suspensionReason, setSuspensionReason] = useState("");
   173→  const [providerMessage, setProviderMessage] = useState("");
   174→  const [suspending, setSuspending] = useState(false);
   175→
   176→  const pagination = usePagination({ totalItems: providers.length, initialPageSize: 5, pageSizeOptions: [5, 10, 20, 50] });
   177→  const paginatedProviders = useMemo(() => pagination.paginate(providers), [providers, pagination]);
   178→
   179→  const fetchProviders = useCallback(async () => {
   180→    try {
   181→      setLoading(true);
   182→      const data = await apiGetProviders();
   183→      setProviders(Array.isArray(data) ? data : []);
   184→    } catch (err: unknown) {
   185→      const message = err instanceof Error ? err.message : "Failed to load providers";
   186→      toast.error(message);
   187→    } finally {
   188→      setLoading(false);
   189→    }
   190→  }, []);
   191→
   192→  useEffect(() => {
   193→    fetchProviders();
   194→  }, [fetchProviders, refreshKey]);
   195→
   196→  const openDetail = (provider: Provider) => {
   197→    setSelectedProvider(provider);
   198→    setDetailOpen(true);
   199→  };
   200→
   201→  const openReject = (provider: Provider) => {
   202→    setRejectDialog(provider);
   203→    setRejectReason("");
   204→  };
   205→
   206→
   207→
   208→  const handleReject = async () => {
   209→    if (!rejectDialog || !rejectReason.trim()) {
   210→      toast.error("Rejection reason is required");
   211→      return;
   212→    }
   213→    try {
   214→      setActioning(true);
   215→      await apiUpdateProvider(rejectDialog.id, { status: "REJECTED", rejectionReason: rejectReason.trim() });
   216→      toast.success("Provider rejected");
   217→      setRejectDialog(null);
   218→      setRejectReason("");
   219→      triggerRefresh();
   220→    } catch (err: unknown) {
   221→      const message = err instanceof Error ? err.message : "Failed to reject provider";
   222→      toast.error(message);
   223→    } finally {
   224→      setActioning(false);
   225→    }
   226→  };
   227→
   228→  const handleStatusAction = async (provider: Provider, status: string) => {
   229→    try {
   230→      setActioning(true);
   231→      if (status === "REACTIVATE") {
   232→        // Police reactivates: set back to APPROVED and clear suspension fields
   233→        await apiUpdateProvider(provider.id, { status: "APPROVED" });
   234→        toast.success(`"${provider.name}" has been reactivated`);
   235→      } else {
   236→        await apiUpdateProvider(provider.id, { status });
   237→        toast.success(`Provider ${status.toLowerCase()}`);
   238→      }
   239→      setConfirmAction(null);
   240→      triggerRefresh();
   241→    } catch (err: unknown) {
   242→      const message = err instanceof Error ? err.message : "Failed to update provider";
   243→      toast.error(message);
   244→    } finally {
   245→      setActioning(false);
   246→    }
   247→  };
   248→
   249→  const openSuspend = (provider: Provider) => {
   250→    setSuspendDialog(provider);
   251→    setSuspensionReason("");
   252→    setProviderMessage("");
   253→  };
   254→
   255→  const handleRegister = async () => {
   256→    if (!registerForm.name.trim() || !registerForm.ownerName.trim() || !registerForm.phone.trim()) {
   257→      toast.error("Guesthouse name, owner name, and phone are required");
   258→      return;
   259→    }
   260→    if (!registerForm.username.trim() || !registerForm.password.trim()) {
   261→      toast.error("Operator username and password are required");
   262→      return;
   263→    }
   264→    if (registerForm.password.trim().length < 4) {
   265→      toast.error("Password must be at least 4 characters");
   266→      return;
   267→    }
   268→    if (!isValidPhone(registerForm.phone)) {
   269→      toast.error("Invalid phone number. Use format like +251 9XX XXX XXX (7-15 digits)");
   270→      return;
   271→    }
   272→    if (registerForm.email.trim() && !isValidEmail(registerForm.email)) {
   273→      toast.error("Invalid email address format");
   274→      return;
   275→    }
   276→    try {
   277→      setRegistering(true);
   278→      await apiSuperCreateProvider({
   279→        ...registerForm,
   280→        licenseFile: registerForm.licenseFileData || undefined,
   281→      });
   282→      toast.success(`"${registerForm.name}" registered and approved successfully`);
   283→      setRegisterOpen(false);
   284→      setRegisterForm(emptyRegisterForm);
   285→      triggerRefresh();
   286→    } catch (err: unknown) {
   287→      toast.error(err instanceof Error ? err.message : "Failed to register guesthouse");
   288→    } finally {
   289→      setRegistering(false);
   290→    }
   291→  };
   292→
   293→  const handleSuspend = async () => {
   294→    if (!suspendDialog) return;

   296→      toast.error("Please provide a reason for suspension");

   302→        providerId: suspendDialog.id,
   303→        suspensionReason: suspensionReason.trim(),
   304→        providerMessage: providerMessage.trim() || undefined,
   305→      });
   306→      toast.success(`"${suspendDialog.name}" has been suspended. Notification sent to provider.`);
   307→      setSuspendDialog(null);
   308→      setSuspensionReason("");
   309→      setProviderMessage("");
   310→      triggerRefresh();
   311→    } catch (err: unknown) {
   312→      toast.error(err instanceof Error ? err.message : "Failed to suspend provider");
   313→    } finally {
   314→      setSuspending(false);
   315→    }
   316→  };

   322→      month: "short",
   323→      day: "numeric",
   324→    });
   325→  };

   329→
   330→  return (

   337→              <Building2 className="h-4 w-4 sm:h-5 sm:w-5 text-slate-600" />
   338→            </div>
   339→            <div>
   340→              <p className="text-[10px] sm:text-sm text-muted-foreground">Total</p>

   342→            </div>
   343→          </div>
   344→        </div>
   345→        <div className="rounded-xl border bg-card p-3 sm:p-4 shadow-sm">
   346→          <div className="flex items-center gap-2 sm:gap-3">

   349→            </div>

   353→            </div>
   354→          </div>

   356→        <div className="rounded-xl border bg-card p-3 sm:p-4 shadow-sm">
   357→          <div className="flex items-center gap-2 sm:gap-3">
   358→            <div className="flex h-8 w-8 sm:h-10 sm:w-10 items-center justify-center rounded-lg bg-yellow-50">

   388→      {/* Providers Table/Cards */}

   390→        {loading ? (
   391→          <div className="space-y-3 p-4 sm:p-6">

   397→          <div className="flex flex-col items-center justify-center py-12 sm:py-16 text-center">
   398→            <Building2 className="mb-3 h-10 w-10 sm:h-12 sm:w-12 text-muted-foreground/40" />

   402→          <>

   406→                <div key={provider.id} className={`p-3 sm:p-4 ${provider.status === "SUSPENDED" ? "bg-orange-50/60" : ""}`}>
   407→                  <div className="flex items-start justify-between gap-2">

   409→                      <div className="flex items-center gap-2">

   433→                      <button className="flex items-center gap-1 rounded-md px-2.5 py-1.5 text-xs font-medium text-emerald-600 hover:bg-emerald-50" onClick={() => setConfirmAction({ provider, action: "APPROVED" })}>

   439→                        <RotateCcw className="h-3.5 w-3.5" /> Reactivate

   456→
   457→            {/* Desktop: Table layout */}
   458→            <div className="hidden lg:block overflow-x-auto">
   459→              <Table>

   511→      <PaginationControls
