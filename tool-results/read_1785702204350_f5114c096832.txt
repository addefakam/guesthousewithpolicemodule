   526→/** Execute SQL via raw pg Client */
   527→async function execViaPg(sql: string, params?: unknown[]): Promise<void> {
   528→  const client = new Client({ connectionString: process.env.DATABASE_URL! });
   529→  try {
   530→    await client.connect();
   531→    await client.query(sql, params);
   532→  } finally {
   533→    await client.end().catch(() => {});
   534→  }
   535→}

   537→/** Execute DDL via Prisma $executeRawUnsafe — splits multi-statement SQL
   538→ *  because Prisma's driver does NOT support multiple statements per call. */
   539→async function execViaPrisma(sql: string): Promise<void> {
   540→  // Split on `DO $$` blocks and standalone statements
   541→  // Each `DO $$ ... END $$;` is one executable unit
   542→  const statements = sql
   543→    .split(/(?=DO \$\$)/)
   544→    .map(s => s.trim())
   545→    .filter(s => s.length > 0);
   546→
   547→  // If no DO blocks found, split by semicolons (for CREATE INDEX etc.)
   548→  const parts = statements.length > 1
   549→    ? statements
   550→    : sql.split(';').map(s => s.trim()).filter(s => s.length > 0);
   551→
   552→  const prisma = new PrismaClient({
   553→    log: ["warn", "error"],
   554→  });
   555→  try {
   556→    for (const part of parts) {
   557→      const stmt = part.endsWith(';') ? part : part + ';';
   558→      try {
   559→        await prisma.$executeRawUnsafe(stmt);
   560→      } catch (err) {
   561→        // Individual statement might fail (e.g. duplicate_object) — log but continue
   562→        console.error("[execViaPrisma] Statement failed (continuing):", stmt.slice(0, 80), err instanceof Error ? err.message : String(err));
   563→      }
   564→    }
   565→  } finally {
   566→    await prisma.$disconnect().catch(() => {});
   567→  }
   568→}

   584→  } catch {
   585→    // Indexes are best-effort
   586→  }

   589→export async function ensureDatabase(): Promise<void> {
   590→  if (_initDone) {
   591→    // Always run migrations synchronously to ensure columns exist
   592→    // before any query executes. Migrations are idempotent.
   593→    if (!_migrationsRan) {
   594→      _migrationsRan = true;
   595→      await runMigrationsOnly();
   596→    }
   597→    return;
   598→  }
   599→  if (_initPromise) return _initPromise;
   600→
   601→  if (!process.env.DATABASE_URL) {
   602→    throw new Error("DATABASE_URL environment variable is not set.");
   603→  }
   604→
   605→  _initPromise = (async () => {
   606→    console.log("[init-db] Starting database initialization...");
   607→
   608→    let usePg = true;
   609→    let pgError: unknown = null;
   610→
   611→    // Strategy 1: Try using pg Client directly
   612→    try {
   613→      const client = new Client({ connectionString: process.env.DATABASE_URL });
   614→      await client.connect();
   615→      console.log("[init-db] pg Client connected successfully.");
   616→
   617→      // Check if User table already exists
   618→      const res = await client.query(
   619→        `SELECT EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema='public' AND table_name='User') as ok`
   620→      );
   621→      const tableExists = res.rows[0]?.ok === true;
   622→      console.log("[init-db] User table exists:", tableExists);
   623→
   624→      if (tableExists) {
   625→        console.log("[init-db] Tables already exist, running migrations to add any missing columns...");
   626→        try {
   627→          await client.query(MIGRATIONS_SQL);
   628→          console.log("[init-db] Migrations applied to existing tables.");
   629→          await client.query(INDEXES_SQL);
   630→          console.log("[init-db] Indexes verified.");
   631→        } catch (migrateErr) {
   632→          console.error("[init-db] Migration on existing tables failed:", migrateErr instanceof Error ? migrateErr.message : String(migrateErr));
   633→          // Don't throw — the tables exist, migrations are best-effort
   634→        }
   635→        await client.end().catch(() => {});
   636→        _initDone = true;
   637→        return;
   638→      }
   639→
   640→      // Create enums
   641→      console.log("[init-db] Creating enum types...");
   642→      await client.query(ENUMS_SQL);
   643→
   644→      // Create tables
   645→      console.log("[init-db] Creating tables...");
   646→      await client.query(TABLES_SQL);
   647→
   648→      // Verify User table was created
   649→      const verify = await client.query(
   650→        `SELECT EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema='public' AND table_name='User') as ok`
   651→      );
   652→      if (!verify.rows[0]?.ok) {
   653→        throw new Error("DDL executed but User table still not found!");
   654→      }
   655→      console.log("[init-db] Tables verified.");
   656→
   657→      // Create foreign keys
   658→      console.log("[init-db] Creating foreign keys...");
   659→      await client.query(FKS_SQL);
   660→
   661→      // Run migrations (add new columns to existing tables)
   662→      console.log("[init-db] Running migrations...");
   663→      await client.query(MIGRATIONS_SQL);
   664→
   665→      // Create indexes
   666→      console.log("[init-db] Creating indexes...");
   667→      await client.query(INDEXES_SQL);
   668→
   669→      // Seed SUPERUSER
   670→      console.log("[init-db] Seeding SUPERUSER...");
   671→      const hashed = await bcrypt.hash("Admin@2024", 12);
   672→      await client.query(
   673→        `INSERT INTO "User" ("id","username","password","name","role","permissions","policeRank","createdAt","updatedAt")
   674→         SELECT 'su-admin-001','admin',$1,'System Administrator','SUPERUSER','[]','',NOW(),NOW()
   675→         WHERE NOT EXISTS (SELECT 1 FROM "User" WHERE "username"='admin')`,
   676→        [hashed]
   677→      );
   678→
   679→      // Seed PoliceAlertConfig
   680→      await client.query(
   681→        `INSERT INTO "PoliceAlertConfig" ("id","createdAt","updatedAt")
   682→         SELECT 'default-alert-config',NOW(),NOW()
   683→         WHERE NOT EXISTS (SELECT 1 FROM "PoliceAlertConfig" WHERE "id"='default-alert-config')`
   684→      );
   685→
   686→      console.log("[init-db] Database initialization complete via pg.");
   687→      await client.end().catch(() => {});
   688→      _initDone = true;
   689→      return;
   690→    } catch (err) {
   691→      pgError = err;
   692→      console.error("[init-db] pg approach failed:", err instanceof Error ? err.message : String(err));
   693→      usePg = false;
   694→    }
   695→
   696→    // Strategy 2: Fallback — use Prisma $executeRawUnsafe
   697→    console.log("[init-db] Falling back to Prisma $executeRawUnsafe...");
   698→    try {
   699→      // First check if tables exist using Prisma raw query
   700→      const prisma = new PrismaClient({ log: ["warn", "error"] });
   701→      try {
   702→        const res: Array<{ ok: boolean }> = await prisma.$queryRawUnsafe(
   703→          `SELECT EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema='public' AND table_name='User') as ok`
   704→        );
   705→        if (res[0]?.ok) {
   706→          console.log("[init-db] Tables already exist (verified via Prisma), running migrations...");
   707→          try {
   708→            await execViaPrisma(MIGRATIONS_SQL);
   709→            console.log("[init-db] Migrations applied via Prisma.");
   710→            await execViaPrisma(INDEXES_SQL);
   711→            console.log("[init-db] Indexes verified via Prisma.");
   712→          } catch (migrateErr) {
   713→            console.error("[init-db] Prisma migration on existing tables failed:", migrateErr instanceof Error ? migrateErr.message : String(migrateErr));
   714→          }
   715→          _initDone = true;
   716→          return;
   717→        }
   718→      } catch {
   719→        // Table might not exist, which is expected
   720→        console.log("[init-db] Tables do not exist yet, creating via Prisma...");
   721→      } finally {
   722→        await prisma.$disconnect().catch(() => {});
   723→      }
   724→
   725→      // Execute DDL via Prisma (split into parts to avoid query size issues)
   726→      await execViaPrisma(ENUMS_SQL);
   727→      console.log("[init-db] Enums created via Prisma.");
   728→
   729→      await execViaPrisma(TABLES_SQL);
   730→      console.log("[init-db] Tables created via Prisma.");
   731→
   732→      await execViaPrisma(FKS_SQL);
   733→      console.log("[init-db] Foreign keys created via Prisma.");
   734→
   735→      await execViaPrisma(MIGRATIONS_SQL);
   736→      console.log("[init-db] Migrations ran via Prisma.");
   737→
   738→      await execViaPrisma(INDEXES_SQL);
   739→      console.log("[init-db] Indexes created via Prisma.");
   740→
   741→      // Seed SUPERUSER via Prisma
   742→      const prisma2 = new PrismaClient({ log: ["warn", "error"] });
   743→      try {
   744→        const hashed = await bcrypt.hash("Admin@2024", 12);
   745→        await prisma2.$executeRawUnsafe(
   746→          `INSERT INTO "User" ("id","username","password","name","role","permissions","policeRank","createdAt","updatedAt")
   747→           SELECT 'su-admin-001','admin',$1,'System Administrator','SUPERUSER','[]','',NOW(),NOW()
   748→           WHERE NOT EXISTS (SELECT 1 FROM "User" WHERE "username"='admin')`,
   749→          hashed
   750→        );
   751→        await prisma2.$executeRawUnsafe(
   752→          `INSERT INTO "PoliceAlertConfig" ("id","createdAt","updatedAt")
   753→           SELECT 'default-alert-config',NOW(),NOW()
   754→           WHERE NOT EXISTS (SELECT 1 FROM "PoliceAlertConfig" WHERE "id"='default-alert-config')`
   755→        );
   756→        console.log("[init-db] SUPERUSER seeded via Prisma.");
   757→      } finally {
   758→        await prisma2.$disconnect().catch(() => {});
   759→      }
   760→
   761→      console.log("[init-db] Database initialization complete via Prisma fallback.");
   762→      _initDone = true;
   763→    } catch (prismaErr) {
   764→      console.error("[init-db] Prisma fallback also failed:", prismaErr instanceof Error ? prismaErr.message : String(prismaErr));
   765→      _initDone = false;
   766→      _initPromise = null; // Allow retry
   767→      throw new Error(
   768→        `Database initialization failed. pg error: ${pgError instanceof Error ? pgError.message : String(pgError)} | Prisma error: ${prismaErr instanceof Error ? prismaErr.message : String(prismaErr)}`
   769→      );
   770→    }
   771→  })();
   772→
   773→  return _initPromise;
   774→}
