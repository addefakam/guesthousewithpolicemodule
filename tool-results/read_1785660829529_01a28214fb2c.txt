   581→          )}
   582→
   583→          {/* Start Joint Session button — shown for SUPERUSER/POLICE ADMIN when no joint session */}
   584→          {canStartJoint && !jointSession.active && !collapsed && (
   585→            <>

   629→
   630→        {/* Logout — hidden for SUPERUSER (moved to name dropdown) */}
   631→        {user.role !== "SUPERUSER" && (
   632→          <button
   633→            onClick={onLogout}

   635→          >

   637→            {!collapsed && <span>Logout</span>}
   638→          </button>
   639→        )}
   640→      </div>
   641→    </div>
   642→  );
   643→}
