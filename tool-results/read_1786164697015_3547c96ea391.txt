   192→// ── Helper: get nav items based on role ──
   193→function getNavItems(user: CurrentUser, isJointActive: boolean): NavItem[] {
   194→  switch (user.role) {
   195→    case "POLICE": {
   196→      const rank = (user.policeRank || "OFFICER") as PoliceRank;
   197→      const allowedPages = POLICE_RANK_PERMISSIONS[rank] || POLICE_RANK_PERMISSIONS.OFFICER;
   198→      const base = POLICE_NAV_ITEMS.filter((item) => allowedPages.includes(item.page));
   199→      // Joint-only pages are only available to ADMIN rank during an active joint session
   200→      if (isJointActive && rank === "ADMIN") {
   201→        const jointItems = POLICE_JOINT_ONLY_ITEMS.filter((item) => allowedPages.includes(item.page));
   202→        return [...base, ...jointItems];
   203→      }
   204→      return base;
   205→    }
   206→
   207→    case "SUPERUSER":
   208→      return SUPERUSER_NAV_ITEMS;
   209→
   210→    case "OPERATOR":
   211→      return ALL_NAV_ITEMS.filter((item) => !OPERATOR_EXCLUDED.has(item.page));
   212→
   213→    case "STAFF": {
   214→      const items: NavItem[] = [];
   215→      // Always give staff access to dashboard
   216→      items.push({
   217→        page: "dashboard",
   218→        label: "Dashboard",
   219→        icon: LayoutDashboard,
   220→      });
   221→      // Add items based on permissions (deduplicate by page key)
   222→      const seen = new Set<string>();
   223→      for (const perm of user.permissions) {
   224→        const mapped = PERMISSION_PAGE_MAP[perm];
   225→        if (mapped && !seen.has(mapped.page)) {
   226→          seen.add(mapped.page);
   227→          items.push(mapped);
   228→        }
   229→      }
   230→      return items;
   231→    }
   232→
   233→    default:
   234→      return [
   235→        { page: "dashboard", label: "Dashboard", icon: LayoutDashboard },
   236→      ];
   237→  }
   238→}

   240→// ── Helper: get initials from name ──
   241→function getInitials(name: string): string {
   242→  return name
   243→    .split(" ")
   244→    .map((w) => w[0])
   245→    .join("")
   246→    .toUpperCase()
   247→    .slice(0, 2);
   248→}

   329→function SubscriptionStatusCard({ collapsed }: { collapsed: boolean }) {
   330→  const subscription = useAppStore((s) => s.subscription);
   331→
   332→  if (!subscription) return null;
   333→
   334→  const isActive = subscription.status === "ACTIVE";
   335→  const isWarning = subscription.status === "WARNING";
   336→  const isGrace = subscription.status === "GRACE";
   337→  const isSuspended = subscription.status === "SUSPENDED";
   338→
   339→  // Collapsed mode — icon indicator
   340→  if (collapsed) {
   341→    const bgClass = isActive
   342→      ? "bg-emerald-100"
   343→      : isSuspended || isGrace
   344→      ? "bg-rose-100"
   345→      : "bg-amber-100";
   346→    const iconClass = isActive
   347→      ? "text-emerald-600"
   348→      : isSuspended || isGrace
   349→      ? "text-rose-600"
   350→      : "text-amber-600";
   351→    return (
   352→      <div className="flex flex-col items-center gap-1 px-2 py-2">
   353→        <div className={`flex h-8 w-8 items-center justify-center rounded-full ${bgClass} animate-pulse shadow-sm`}> 
   354→          <Clock className={`size-4 ${iconClass}`} />
   355→        </div>
   356→      </div>
   357→    );
   358→  }
   359→
   360→  // Expanded card
   361→  const containerClass = isActive
   362→    ? "border-emerald-200 bg-emerald-50"
   363→    : isSuspended || isGrace
   364→    ? "border-rose-200 bg-rose-50"
   365→    : "border-amber-200 bg-amber-50";
   366→
   367→  const iconColorClass = isActive
   368→    ? "text-emerald-600"
   369→    : isSuspended || isGrace
   370→    ? "text-rose-600"
   371→    : "text-amber-600";
   372→
   373→  const titleClass = isActive
   374→    ? "text-emerald-800"
   375→    : isSuspended || isGrace
   376→    ? "text-rose-800"
   377→    : "text-amber-800";
   378→
   379→  const subtextClass = isActive
   380→    ? "text-emerald-700"
   381→    : isSuspended || isGrace
   382→    ? "text-rose-700"
   383→    : "text-amber-700";
   384→
   385→  const badgeClass = isActive
   386→    ? "bg-emerald-200/60 text-emerald-800"
   387→    : isSuspended || isGrace
   388→    ? "bg-rose-200/60 text-rose-800"
   389→    : "bg-amber-200/60 text-amber-800";
   390→
   391→  const Icon = isSuspended || isGrace ? AlertTriangle : Clock;
   392→
   393→  return (
   394→    <div className={`rounded-lg border p-3 ${containerClass} animate-subtle-pulse shadow-sm`}> 
   395→      <div className="flex items-center gap-2">
   396→        <Icon className={`h-4 w-4 shrink-0 ${iconColorClass}`} />
   397→        <p className={`text-xs font-semibold ${titleClass}`}>
   398→          {isActive
   399→            ? "Subscription Active"
   400→            : isSuspended
   401→            ? "Service Suspended"
   402→            : isGrace
   403→            ? "Subscription Expired"
   404→            : "Expiring Soon"}
   405→        </p>
   406→      </div>
   407→      <div className={`mt-1.5 flex items-center justify-between text-[11px] ${subtextClass}`}>
   408→        <span>{formatDaysRemaining(subscription.daysRemaining)}</span>
   409→        <span className={`inline-flex items-center gap-1 rounded-full px-1.5 py-0.5 font-semibold ${badgeClass}`}>
   410→          <Clock className="h-3 w-3" />
   411→          {Math.abs(subscription.daysRemaining)}d
   412→        </span>
   413→      </div>
   414→      {subscription.price > 0 ? (
   415→        <div className={`mt-1 text-[11px] ${subtextClass}`}>
   416→          Due: {subscription.price.toLocaleString()} ETB / {formatCycle(subscription.cycle)}
   417→        </div>
   418→      ) : (
   419→        <div className={`mt-1 text-[11px] ${subtextClass}`}>
   420→          {formatCycle(subscription.cycle)} cycle — Free trial
   421→        </div>
   422→      )}
   423→    </div>
   424→  );
   425→}

   579→            </>
   580→          )}
   581→
   582→          {/* Joint Operations — SUPERUSER during active joint session */}
   583→          {user.role === "SUPERUSER" && jointSession.active && (

   619→      {/* ── Subscription status for providers ── */}

   637→            <span>Collapse</span>
   638→          </button>
   639→        )}
   640→      </div>
   641→    </div>
   642→  );
   643→}
