    81→    81→function InlineBar({ value, max, color = "bg-primary" }: { value: number; max: number; color?: string }) {
    82→    82→  const pct = max > 0 ? Math.round((value / max) * 100) : 0;

    84→    84→    <div className="flex items-center gap-2">

    89→    89→    </div>

    91→    91→}
