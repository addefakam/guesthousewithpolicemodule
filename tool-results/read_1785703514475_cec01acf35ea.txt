   161→   161→  disabled = false,
   162→   162→}: {
   163→   163→  id: string;
   164→   164→  checked: boolean;
   165→   165→  onChange: (value: boolean) => void;
   166→   166→  disabled?: boolean;

   169→   169→    <button
   170→   170→      id={id}
   171→   171→      type="button"
   172→   172→      role="switch"
   173→   173→      aria-checked={checked}
   174→   174→      disabled={disabled}

   177→   177→        `relative inline-flex h-6 w-11 shrink-0 cursor-pointer items-center rounded-full transition-colors duration-200 ease-in-out focus:outline-none focus-visible:ring-2 focus-visible:ring-primary/40 focus-visible:ring-offset-2 ` +
   178→   178→        (checked

   182→   182→      }
