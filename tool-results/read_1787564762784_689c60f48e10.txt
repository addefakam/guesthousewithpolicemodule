    46→function getInitials(n: string | undefined | null) {
    47→  if (!n) return "?";
    48→  return n
    49→    .split(" ")
    50→    .map((w) => w[0])
    51→    .join("")
    52→    .toUpperCase()
    53→    .slice(0, 2);
    54→}
