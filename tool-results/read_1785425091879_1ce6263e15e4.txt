   194→   194→// ── Helper: get nav items based on role ──
   195→   195→function getNavItems(user: CurrentUser): NavItem[] {
   196→   196→  switch (user.role) {
   197→   197→    case "POLICE": {
   198→   198→      const rank = (user.policeRank || "OFFICER") as PoliceRank;
   199→   199→      const allowedPages = POLICE_RANK_PERMISSIONS[rank] || POLICE_RANK_PERMISSIONS.OFFICER;
   200→   200→      return POLICE_NAV_ITEMS.filter((item) => allowedPages.includes(item.page));
   201→   201→    }
   202→   202→
   203→   203→    case "SUPERUSER":
   204→   204→      return SUPERUSER_NAV_ITEMS;
   205→   205→
   206→   206→    case "OPERATOR":
   207→   207→      return ALL_NAV_ITEMS.filter((item) => !OPERATOR_EXCLUDED.has(item.page));
   208→   208→
   209→   209→    case "STAFF": {
   210→   210→      const items: NavItem[] = [];
   211→   211→      // Always give staff access to dashboard
   212→   212→      items.push({
   213→   213→        page: "dashboard",
   214→   214→        label: "Dashboard",
   215→   215→        icon: LayoutDashboard,
   216→   216→      });
   217→   217→      // Add items based on permissions
   218→   218→      for (const perm of user.permissions) {
   219→   219→        const mapped = PERMISSION_PAGE_MAP[perm];
   220→   220→        if (mapped) {
   221→   221→          items.push(mapped);
   222→   222→        }
   223→   223→      }
   224→   224→      return items;
   225→   225→    }
   226→   226→
   227→   227→    default:
   228→   228→      return [
   229→   229→        { page: "dashboard", label: "Dashboard", icon: LayoutDashboard },
   230→   230→      ];
   231→   231→  }
   232→   232→}

   321→   321→// ── Anomaly Detection Toggle Widget (sidebar inline) ──
   322→   322→function AnomalyToggle({ collapsed }: { collapsed: boolean }) {
   323→   323→  const currentUser = useAppStore((s) => s.currentUser);
   324→   324→  const isAdmin = currentUser?.role === "POLICE" && currentUser?.policeRank === "ADMIN";
   325→   325→  const [enabled, setEnabled] = useState(false);
   326→   326→  const [toggling, setToggling] = useState(false);
   327→   327→
   328→   328→  const fetchStatus = useCallback(async () => {
   329→   329→    try {
   330→   330→      const data = await apiGetAnomalies("pageSize=1");
   331→   331→      if (data.enabled !== undefined) setEnabled(data.enabled);
   332→   332→    } catch { /* non-critical */ }
   333→   333→  }, []);
   334→   334→
   335→   335→  useEffect(() => {
   336→   336→    fetchStatus();
   337→   337→  }, [fetchStatus]);
   338→   338→
   339→   339→  const handleToggle = async () => {
   340→   340→    if (!isAdmin || toggling) return;
   341→   341→    try {
   342→   342→      setToggling(true);
   343→   343→      const result = await apiToggleAnomalyDetection(!enabled);
   344→   344→      setEnabled(result.enabled);
   345→   345→    } catch { /* non-critical */ }
   346→   346→    finally { setToggling(false); }
   347→   347→  };
   348→   348→
   349→   349→  // Collapsed: small indicator dot
   350→   350→  if (collapsed) {
   351→   351→    return (
   352→   352→      <div className="flex justify-center py-1" title={enabled ? "Anomaly Detection: Active" : "Anomaly Detection: Inactive"}>
   353→   353→        <button
   354→   354→          onClick={handleToggle}
   355→   355→          disabled={!isAdmin}
   356→   356→          className={`h-3 w-3 rounded-full transition-colors ${
   357→   357→            enabled ? "bg-violet-500 shadow-sm shadow-violet-200" : "bg-slate-300"
   358→   358→          } ${isAdmin ? "cursor-pointer hover:ring-2 hover:ring-violet-300" : "cursor-not-allowed"}`}
   359→   359→        />
   360→   360→      </div>
   361→   361→    );
   362→   362→  }
   363→   363→
   364→   364→  // Expanded: inline toggle row
   365→   365→  return (
   366→   366→    <button
   367→   367→      onClick={handleToggle}
   368→   368→      disabled={!isAdmin || toggling}
   369→   369→      className={`group flex w-full items-center gap-2.5 rounded-lg px-3 py-2 text-xs font-medium transition-colors outline-none ${
   370→   370→        isAdmin
   371→   371→          ? enabled
   372→   372→            ? "text-violet-700 hover:bg-violet-50"
   373→   373→            : "text-slate-500 hover:bg-slate-100"
   374→   374→          : "text-slate-400 cursor-not-allowed"
   375→   375→      }`}
   376→   376→      title={!isAdmin ? "Only ADMIN can toggle detection" : enabled ? "Click to disable" : "Click to enable"}
   377→   377→    >
   378→   378→      {toggling ? (
   379→   379→        <Loader2 className="size-3.5 animate-spin shrink-0" />
   380→   380→      ) : enabled ? (
   381→   381→        <ToggleRight className="size-3.5 shrink-0 text-violet-500" />
   382→   382→      ) : (
   383→   383→        <ToggleLeft className="size-3.5 shrink-0" />
   384→   384→      )}
   385→   385→      <span className="truncate">
   386→   386→        {enabled ? "Detection Active" : "Detection Inactive"}
   387→   387→      </span>
   388→   388→      {/* Mini toggle switch */}
   389→   389→      <span
   390→   390→        className={`ml-auto inline-flex h-5 w-9 shrink-0 items-center rounded-full transition-colors duration-200 ${
   391→   391→          enabled ? "bg-violet-500" : "bg-slate-300"
   392→   392→        }`}
   393→   393→      >
   394→   394→        <span
   395→   395→          className={`inline-block h-3.5 w-3.5 transform rounded-full bg-white shadow-sm transition-transform duration-200 ${
   396→   396→            enabled ? "translate-x-4" : "translate-x-0.5"
   397→   397→          }`}
   398→   398→        />
   399→   399→      </span>
   400→   400→    </button>
   401→   401→  );
   402→   402→}
