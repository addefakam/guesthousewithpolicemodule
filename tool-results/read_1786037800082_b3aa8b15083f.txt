   105→  PENDING: "bg-yellow-100 text-yellow-800 hover:bg-yellow-100 border-yellow-200",
   106→  APPROVED: "bg-emerald-100 text-emerald-800 hover:bg-emerald-100 border-emerald-200",

   111→function StatusBadge({ status }: { status: string }) {
   112→  const cfg = STATUS_CONFIG[status] || { label: status, variant: "secondary" as const };

   114→    <Badge variant={cfg.variant} className={STATUS_BADGE_CLASS[status] || ""}>
   115→      {cfg.label}
   116→    </Badge>
   117→  );

   156→export default function ProvidersPage() {
   157→  const { refreshKey, triggerRefresh, currentUser } = useAppStore();
   158→  const isSuperuser = currentUser?.role === "SUPERUSER";
   159→  const [providers, setProviders] = useState<Provider[]>([]);
   160→  const [loading, setLoading] = useState(true);
   161→  const [selectedProvider, setSelectedProvider] = useState<Provider | null>(null);
   162→  const [detailOpen, setDetailOpen] = useState(false);
   163→
   164→  // Registration form
   165→  const [registerOpen, setRegisterOpen] = useState(false);
   166→  const [registerForm, setRegisterForm] = useState<RegisterForm>(emptyRegisterForm);
   167→  const [registering, setRegistering] = useState(false);
   168→
   169→  // Coordinate editing
   170→  const [coordProvider, setCoordProvider] = useState<Provider | null>(null);
   171→  const [coordOpen, setCoordOpen] = useState(false);
   172→  const [editLat, setEditLat] = useState(9.02);
   173→  const [editLng, setEditLng] = useState(38.75);
   174→  const [savingCoord, setSavingCoord] = useState(false);
   175→  const [geocodingOne, setGeocodingOne] = useState(false);
   176→
   177→  // Batch geocode
   178→  const [geocodingAll, setGeocodingAll] = useState(false);
   179→  const [batchResult, setBatchResult] = useState<{ updated: number; failed: number; total: number; results?: any[] } | null>(null);
   180→  const [batchOpen, setBatchOpen] = useState(false);
   181→
   182→  // Action dialogs
   183→  const [rejectDialog, setRejectDialog] = useState<Provider | null>(null);
   184→  const [rejectReason, setRejectReason] = useState("");
   185→  const [confirmAction, setConfirmAction] = useState<{ provider: Provider; action: string } | null>(null);
   186→  const [actioning, setActioning] = useState(false);
   187→
   188→  // Suspension dialog (with reason + notification)
   189→  const [suspendDialog, setSuspendDialog] = useState<Provider | null>(null);
   190→  const [suspensionReason, setSuspensionReason] = useState("");
   191→  const [providerMessage, setProviderMessage] = useState("");
   192→  const [suspending, setSuspending] = useState(false);
   193→
   194→  const pagination = usePagination({ totalItems: 0, initialPageSize: 50, pageSizeOptions: [10, 20, 50, 100] });
   195→  const [serverPage, setServerPage] = useState(1);
   196→  const [serverPageSize, setServerPageSize] = useState(50);

   199→    try {
   200→      setLoading(true);
   201→      const data = await apiGetProviders({ page: serverPage, pageSize: serverPageSize });
   202→      if (data && typeof data === 'object' && 'providers' in data) {
   203→        const resp = data as { providers: Provider[]; total: number; page: number; pageSize: number };
   204→        setProviders(resp.providers);
   205→        pagination.setTotalOnly(resp.total);
   206→      } else {
   207→        setProviders(Array.isArray(data) ? data : []);
   208→        pagination.setTotalOnly(Array.isArray(data) ? data.length : 0);
   209→      }
   210→    } catch (err: unknown) {
   211→      const message = err instanceof Error ? err.message : "Failed to load providers";
   212→      toast.error(message);
   213→    } finally {
   214→      setLoading(false);
   215→    }
   216→  }, [serverPage, serverPageSize, pagination.setTotalOnly]);
   217→
   218→  // Single effect: initial load + refresh + page/pageSize changes
   219→  useEffect(() => {
   220→    fetchProviders();
   221→  }, [fetchProviders, refreshKey]);
   222→
   223→  // Sync pagination controls → server fetch
   224→  const handlePageChange = useCallback((page: number) => {
   225→    setServerPage(page);
   226→  }, []);
   227→
   228→  const handlePageSizeChange = useCallback((size: number) => {
   229→    setServerPage(1);
   230→    setServerPageSize(size);
   231→  }, []);
   232→

   235→    setDetailOpen(true);
   236→  };
   237→
   238→  const openReject = (provider: Provider) => {

   240→    setRejectReason("");
   241→  };
   242→
   243→  const openCoordPicker = (provider: Provider) => {
   244→    setCoordProvider(provider);
   245→    setEditLat(provider.latitude || 9.02);
   246→    setEditLng(provider.longitude || 38.75);
   247→    setCoordOpen(true);
   248→  };
   249→
   250→  const geocodeSingle = async () => {
   251→    if (!coordProvider || !coordProvider.address) {
   252→      toast.error("Provider has no address to geocode");
   253→      return;
   254→    }
   255→    try {
   256→      setGeocodingOne(true);
   257→      const result: any = await apiGeocodeAddress(coordProvider.address);
   258→      if (result.lat && result.lng) {
   259→        setEditLat(result.lat);
   260→        setEditLng(result.lng);
   261→        toast.success(`Found: ${result.lat.toFixed(4)}, ${result.lng.toFixed(4)}`);
   262→      } else {
   263→        toast.error("Could not find coordinates for this address");
   264→      }
   265→    } catch {
   266→      toast.error("Geocoding failed");
   267→    } finally {
   268→      setGeocodingOne(false);
   269→    }
   270→  };
   271→
   272→  const saveCoordinates = async () => {
   273→    if (!coordProvider) return;
   274→    try {
   275→      setSavingCoord(true);
   276→      await apiUpdateProvider(coordProvider.id, { status: coordProvider.status, latitude: editLat, longitude: editLng });
   277→      toast.success("Coordinates updated");
   278→      setCoordOpen(false);
   279→      triggerRefresh();
   280→    } catch {
   281→      toast.error("Failed to save coordinates");
   282→    } finally {
   283→      setSavingCoord(false);
   284→    }
   285→  };
   286→
   287→  const batchGeocode = async () => {
   288→    try {
   289→      setGeocodingAll(true);
   290→      const result: any = await apiGeocodeBatch();
   291→      setBatchResult(result);
   292→      toast.success(result.message || `Geocoded ${result.updated} providers`);
   293→      triggerRefresh();
   294→    } catch {
   295→      toast.error("Batch geocoding failed");
   296→    } finally {
   297→      setGeocodingAll(false);
   298→    }
   299→  };
   300→
   301→  const handleReject = async () => {
   302→    if (!rejectDialog || !rejectReason.trim()) {
   303→      toast.error("Rejection reason is required");
   304→      return;
   305→    }
   306→    try {
   307→      setActioning(true);
   308→      await apiUpdateProvider(rejectDialog.id, { status: "REJECTED", rejectionReason: rejectReason.trim() });
   309→      toast.success("Provider rejected");
   310→      setRejectDialog(null);
   311→      setRejectReason("");
   312→      triggerRefresh();
   313→    } catch (err: unknown) {
   314→      const message = err instanceof Error ? err.message : "Failed to reject provider";
   315→      toast.error(message);
   316→    } finally {
   317→      setActioning(false);
   318→    }
   319→  };
   320→
   321→  const handleStatusAction = async (provider: Provider, status: string) => {
   322→    try {
   323→      setActioning(true);
   324→      if (status === "REACTIVATE") {
   325→        await req(`/api/providers/${provider.id}/reactivate`, { method: "POST" });
   326→        toast.success(`"${provider.name}" has been reactivated and is fully active`);
   327→      } else {
   328→        await apiUpdateProvider(provider.id, { status });
   329→        toast.success(`Provider ${status.toLowerCase()}`);
   330→      }
   331→      setConfirmAction(null);
   332→      triggerRefresh();
   333→    } catch (err: unknown) {
   334→      const message = err instanceof Error ? err.message : "Failed to update provider";
   335→      toast.error(message);
   336→    } finally {
   337→      setActioning(false);
   338→    }
   339→  };
   340→
   341→  const openSuspend = (provider: Provider) => {
   342→    setSuspendDialog(provider);
   343→    setSuspensionReason("");
   344→    setProviderMessage("");
   345→  };
   346→
   347→  const handleRegister = async () => {
   348→    if (!registerForm.name.trim() || !registerForm.ownerName.trim() || !registerForm.phone.trim()) {
   349→      toast.error("Guesthouse name, owner name, and phone are required");
   350→      return;
   351→    }
   352→    if (!registerForm.username.trim() || !registerForm.password.trim()) {
   353→      toast.error("Operator username and password are required");
   354→      return;
   355→    }
   356→    if (registerForm.password.trim().length < 4) {
   357→      toast.error("Password must be at least 4 characters");
   358→      return;
   359→    }
   360→    try {
   361→      setRegistering(true);
   362→      await apiSuperCreateProvider({
   363→        ...registerForm,
   364→        licenseFile: registerForm.licenseFileData || undefined,
   365→      });
   366→      toast.success(`"${registerForm.name}" registered and approved successfully`);
   367→      setRegisterOpen(false);
   368→      setRegisterForm(emptyRegisterForm);
   369→      triggerRefresh();
   370→    } catch (err: unknown) {
   371→      toast.error(err instanceof Error ? err.message : "Failed to register guesthouse");
   372→    } finally {
   373→      setRegistering(false);
   374→    }
   375→  };
   376→
   377→  const handleSuspend = async () => {
   378→    if (!suspendDialog) return;
   379→    if (!suspensionReason.trim()) {
   380→      toast.error("Please provide a reason for suspension");
   381→      return;
   382→    }

   385→      await apiPoliceSuspendProvider({
   386→        providerId: suspendDialog.id,
   387→        suspensionReason: suspensionReason.trim(),
   388→        providerMessage: providerMessage.trim() || undefined,
   389→      });

   391→      setSuspendDialog(null);
   392→      setSuspensionReason("");
   393→      setProviderMessage("");
   394→      triggerRefresh();
   395→    } catch (err: unknown) {
   396→      toast.error(err instanceof Error ? err.message : "Failed to suspend provider");
   397→    } finally {
   398→      setSuspending(false);
   399→    }
   400→  };
   401→
   402→  const formatDate = (dateStr: string | null) => {
   403→    if (!dateStr) return "—";

   405→      year: "numeric",
   406→      month: "short",
   407→      day: "numeric",
   408→    });

   412→  const pendingCount = providers.filter((p) => p.status === "PENDING").length;
   413→  const withCoords = providers.filter(
   414→    (p) => p.latitude !== 9.02 || p.longitude !== 38.75,

   419→      {/* Summary Cards */}

   421→        <div className="rounded-xl border bg-card p-3 sm:p-4 shadow-sm">
   422→          <div className="flex items-center gap-2 sm:gap-3">
   423→            <div className="flex h-8 w-8 sm:h-10 sm:w-10 items-center justify-center rounded-lg bg-slate-100">
   424→              <Building2 className="h-4 w-4 sm:h-5 sm:w-5 text-slate-600" />
   425→            </div>
   426→            <div>
   427→              <p className="text-[10px] sm:text-sm text-muted-foreground">Total</p>
   428→              <p className="text-lg sm:text-2xl font-bold">{providers.length}</p>
   429→            </div>
   430→          </div>
   431→        </div>
   432→        <div className="rounded-xl border bg-card p-3 sm:p-4 shadow-sm">
