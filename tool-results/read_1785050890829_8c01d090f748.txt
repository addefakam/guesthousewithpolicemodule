   104→  return new Date(dateStr).toLocaleDateString("en-US", {
   105→    month: "short",
   106→    day: "numeric",
   107→    year: "numeric",
   108→  });
   109→}
   110→
   111→function formatDateTime(dateStr: string) {
   112→  return new Date(dateStr).toLocaleString("en-US", {
   113→    month: "short",
   114→    day: "numeric",
   115→    hour: "2-digit",
   116→    minute: "2-digit",
   117→  });

   121→  name: "",
   122→  phone: "",
   123→  idNumber: "",

   131→export default function SuspectedPersonsPage() {
   132→  const { refreshKey } = useAppStore();
   133→  const [persons, setPersons] = useState<SuspectedPerson[]>([]);
   134→  const [search, setSearch] = useState("");
   135→  const [loading, setLoading] = useState(true);
   136→
   137→  // Dialogs
   138→  const [formOpen, setFormOpen] = useState(false);
   139→  const [editingId, setEditingId] = useState<string | null>(null);
   140→  const [form, setForm] = useState(emptyForm);
   141→  const [formLoading, setFormLoading] = useState(false);
   142→
   143→  const [detailOpen, setDetailOpen] = useState(false);
   144→  const [detailPerson, setDetailPerson] = useState<PersonWithHistory | null>(null);
   145→  const [detailLoading, setDetailLoading] = useState(false);
   146→
   147→  const [deleteOpen, setDeleteOpen] = useState(false);
   148→  const [deleteId, setDeleteId] = useState<string | null>(null);
   149→  const [deleteLoading, setDeleteLoading] = useState(false);
   150→
   151→  const pagination = usePagination({ totalItems: persons.length, initialPageSize: 10, pageSizeOptions: [5, 10, 20, 50] });
   152→
   153→  // Reset to page 1 when search changes
   154→  useEffect(() => { pagination.resetToFirst(); }, [search]);
   155→  const paginatedPersons = useMemo(() => pagination.paginate(persons), [persons, pagination]);
   156→
   157→  const fetchPersons = useCallback(async () => {
   158→    try {
   159→      setLoading(true);
   160→      const query = search ? `q=${encodeURIComponent(search)}` : "";
   161→      const data = await apiGetSuspectedPersons(query);
   162→      setPersons(Array.isArray(data) ? data : []);
   163→    } catch (err: unknown) {
   164→      const message = err instanceof Error ? err.message : "Failed to load suspected persons";
   165→      toast.error(message);
   166→    } finally {
   167→      setLoading(false);
   168→    }
   169→  }, [search]);
   170→
   171→  useEffect(() => {
   172→    const timer = setTimeout(() => fetchPersons(), 300);
   173→    return () => clearTimeout(timer);
   174→  }, [fetchPersons, refreshKey]);
   175→
   176→  const openAdd = () => {
   177→    setEditingId(null);
   178→    setForm(emptyForm);
   179→    setFormOpen(true);
   180→  };
   181→
   182→  const openEdit = (person: SuspectedPerson) => {
   183→    setEditingId(person.id);
   184→    setForm({
   185→      name: person.name,
   186→      phone: person.phone,
   187→      idNumber: person.idNumber,
   188→      idType: person.idType,
   189→      nationality: person.nationality,
   190→      address: person.address,
   191→      description: person.description,
   192→      severity: person.severity,
   193→    });
   194→    setFormOpen(true);
   195→  };
   196→
   197→  const handleSubmit = async (e: FormEvent) => {
   198→    e.preventDefault();
   199→    if (!form.name.trim()) {
   200→      toast.error("Name is required");
   201→      return;
   202→    }
   203→    setFormLoading(true);
   204→    try {
   205→      if (editingId) {
   206→        const updated = await apiUpdateSuspectedPerson(editingId, form);
   207→        setPersons((prev) =>
   208→          prev.map((p) => (p.id === editingId ? { ...p, ...updated } : p))
   209→        );
   210→        toast.success("Suspected person updated");
   211→      } else {
   212→        const created = await apiCreateSuspectedPerson(form);
   213→        setPersons((prev) => [created, ...prev]);
   214→        toast.success("Suspected person added");
   215→      }
   216→      setFormOpen(false);
   217→    } catch (err: unknown) {
   218→      const message = err instanceof Error ? err.message : "Operation failed";
   219→      toast.error(message);
   220→    } finally {
   221→      setFormLoading(false);
   222→    }
   223→  };
   224→
   225→  const openDetail = async (person: SuspectedPerson) => {
   226→    setDetailPerson(person);
   227→    setDetailOpen(true);
   228→    setDetailLoading(true);
   229→    try {
   230→      // Fetch full person with match history
   231→      const res = await fetch(`/api/suspected-persons/${person.id}`, {
   232→        headers: { "Content-Type": "application/json" },
   233→      });
   234→      if (res.ok) {
   235→        const data = await res.json();
   236→        setDetailPerson(data);
   237→      }
   238→    } catch {
   239→      // Keep the basic person data
   240→    } finally {
   241→      setDetailLoading(false);
   242→    }
   243→  };
   244→
   245→  const confirmDelete = (id: string) => {
   246→    setDeleteId(id);
   247→    setDeleteOpen(true);
   248→  };
   249→
   250→  const handleDelete = async () => {
   251→    if (!deleteId) return;
   252→    setDeleteLoading(true);
   253→    try {
   254→      await apiDeleteSuspectedPerson(deleteId);
   255→      setPersons((prev) => prev.filter((p) => p.id !== deleteId));
   256→      toast.success("Suspected person deleted");
   257→      setDeleteOpen(false);
   258→    } catch (err: unknown) {
   259→      toast.error("Failed to delete");
   260→    } finally {
   261→      setDeleteLoading(false);
   262→    }
   263→  };
   264→
   265→  const toggleActive = async (person: SuspectedPerson) => {

   267→      const updated = await apiUpdateSuspectedPerson(person.id, { is_active: !person.is_active });

   269→        prev.map((p) => (p.id === person.id ? { ...p, is_active: updated.is_active } : p))
   270→      );
   271→      toast.success(updated.is_active ? "Reactivated" : "Deactivated");
   272→    } catch {
   273→      toast.error("Failed to update status");
   274→    }
   275→  };

   290→            <Input
   291→              placeholder="Search name, phone, ID..."
   292→              value={search}
   293→              onChange={(e) => setSearch(e.target.value)}
   294→              className="pl-9 h-9 sm:h-10"
   295→            />

   297→          <Button onClick={openAdd} className="h-9 sm:h-10 shrink-0">
   298→            <Plus className="mr-1 h-4 w-4" />
   299→            <span className="hidden sm:inline">Add</span>
   300→          </Button>
   301→        </div>
   302→      </div>
   303→
   304→      {/* Count */}
   305→      {!loading && persons.length > 0 && (
   306→        <p className="text-xs text-muted-foreground px-1">
   307→          {persons.length} person{persons.length !== 1 ? "s" : ""} registered
   308→        </p>
   309→      )}
   310→
   311→      {/* List */}
   312→      <div className="rounded-xl border bg-card shadow-sm">
   313→        {loading ? (
   314→          <div className="space-y-3 p-4 sm:p-6">
   315→            {Array.from({ length: 4 }).map((_, i) => (
   316→              <Skeleton key={i} className="h-20 w-full sm:h-12" />
   317→            ))}
   318→          </div>
   319→        ) : persons.length === 0 ? (
   320→          <div className="flex flex-col items-center justify-center py-12 sm:py-16 text-center">
   321→            <UserX className="mb-3 h-10 w-10 sm:h-12 sm:w-12 text-muted-foreground/40" />
   322→            <p className="text-xs sm:text-sm text-muted-foreground">
   323→              {search ? "No matches found" : "No suspected persons registered yet"}
   324→            </p>
   325→            <p className="mt-1 text-[10px] sm:text-xs text-muted-foreground/70">
   326→              Add persons to monitor — system will alert when they make reservations
   327→            </p>
   328→          </div>
   329→        ) : (
   330→          <>
   331→            {/* Mobile: Card layout */}
   332→            <div className="divide-y md:hidden">
   333→              {paginatedPersons.map((person) => (
   334→                <div key={person.id} className="p-3 space-y-2">
   335→                  <div className="flex items-start gap-3">
   336→                    <div className={`flex h-9 w-9 shrink-0 items-center justify-center rounded-full ${
   337→                      !person.is_active ? "bg-slate-100" : person.severity === "CRITICAL" ? "bg-red-100" : person.severity === "HIGH" ? "bg-orange-100" : "bg-yellow-100"
   338→                    }`}>
   339→                      <UserX className={`h-4 w-4 ${
   340→                        !person.is_active ? "text-slate-400" : person.severity === "CRITICAL" ? "text-red-600" : person.severity === "HIGH" ? "text-orange-600" : "text-yellow-600"
   341→                      }`} />
   342→                    </div>
   343→                    <div className="min-w-0 flex-1">
   344→                      <div className="flex items-center gap-1.5">
   345→                        <p className={`truncate text-sm font-medium ${!person.is_active ? "text-muted-foreground line-through" : ""}`}>
   346→                          {person.name}
   347→                        </p>
   348→                        <Badge variant="outline" className={`shrink-0 text-[9px] ${SEVERITY_STYLES[person.severity] || ""}`}>
   349→                          {person.severity}
   350→                        </Badge>
   351→                        {!person.is_active && (
   352→                          <Badge variant="outline" className="shrink-0 text-[9px] bg-slate-50 text-slate-400 border-slate-200">
   353→                            Inactive
   354→                          </Badge>
   355→                        )}
   356→                      </div>
   357→                      {person.phone && (
   358→                        <p className="text-xs text-muted-foreground font-mono">{person.phone}</p>
   359→                      )}
   360→                      <div className="mt-1.5 flex flex-wrap items-center gap-x-2.5 gap-y-1">
   361→                        {person.idNumber && (
   362→                          <span className="text-[10px] text-muted-foreground">ID: {person.idNumber}</span>
   363→                        )}
   364→                        <span className="flex items-center gap-1 text-[10px] text-red-600 font-medium">
   365→                          <ShieldAlert className="h-2.5 w-2.5" />
   366→                          {person._count.matches} match{person._count.matches !== 1 ? "es" : ""}
   367→                        </span>
   368→                      </div>
   369→                    </div>
   370→                  </div>
   371→                  <div className="flex justify-end gap-1.5 pl-12">
   372→                    <Button variant="ghost" size="sm" className="h-7 text-xs px-2" onClick={() => openDetail(person)}>
   373→                      <Eye className="h-3.5 w-3.5" />
   374→                    </Button>
   375→                    <Button variant="ghost" size="sm" className="h-7 text-xs px-2" onClick={() => openEdit(person)}>
   376→                      <Pencil className="h-3.5 w-3.5" />
   377→                    </Button>
   378→                    <Button
   379→                      variant="ghost"
   380→                      size="sm"
   381→                      className="h-7 text-xs px-2"
   382→                      onClick={() => toggleActive(person)}
   383→                    >
   384→                      {person.is_active ? "Deactivate" : "Activate"}
   385→                    </Button>
   386→                    <Button
   387→                      variant="ghost"
   388→                      size="sm"
   389→                      className="h-7 text-xs px-2 text-red-600 hover:text-red-700 hover:bg-red-50"
   390→                      onClick={() => confirmDelete(person.id)}
   391→                    >
   392→                      <Trash2 className="h-3.5 w-3.5" />
   393→                    </Button>
   394→                  </div>
   395→                </div>
   396→              ))}
   397→            </div>
   398→
   399→            {/* Desktop: Table layout */}
   400→            <div className="hidden md:block overflow-x-auto">
   401→              <Table>
   402→                <TableHeader>
   403→                  <TableRow>
   404→                    <TableHead>Name</TableHead>
   405→                    <TableHead>Phone</TableHead>
   406→                    <TableHead>ID Number</TableHead>
   407→                    <TableHead>Nationality</TableHead>
   408→                    <TableHead>Severity</TableHead>
   409→                    <TableHead className="text-center">Matches</TableHead>
   410→                    <TableHead>Status</TableHead>
   411→                    <TableHead>Registered</TableHead>
   412→                    <TableHead className="text-right">Actions</TableHead>
   413→                  </TableRow>
   414→                </TableHeader>
   415→                <TableBody>
   416→                  {paginatedPersons.map((person) => (
   417→                    <TableRow key={person.id} className={!person.is_active ? "opacity-60" : ""}>
   418→                      <TableCell className="font-medium">{person.name}</TableCell>
   419→                      <TableCell className="font-mono text-sm">{person.phone || "—"}</TableCell>
   420→                      <TableCell className="font-mono text-sm">{person.idNumber || "—"}</TableCell>
   421→                      <TableCell>{person.nationality || "—"}</TableCell>
   422→                      <TableCell>
   423→                        <Badge variant="outline" className={SEVERITY_STYLES[person.severity] || ""}>
   424→                          {person.severity}
   425→                        </Badge>
   426→                      </TableCell>
   427→                      <TableCell className="text-center">
   428→                        {person._count.matches > 0 ? (
   429→                          <Badge className="bg-red-100 text-red-800 hover:bg-red-100 border-red-200 text-xs">
   430→                            {person._count.matches}
   431→                          </Badge>
   432→                        ) : (
   433→                          <span className="text-muted-foreground">0</span>
   434→                        )}
   435→                      </TableCell>
   436→                      <TableCell>
   437→                        <Badge variant={person.is_active ? "outline" : "secondary"} className={person.is_active ? "bg-emerald-50 text-emerald-700 border-emerald-200" : ""}>
   438→                          {person.is_active ? "Active" : "Inactive"}
   439→                        </Badge>
   440→                      </TableCell>
   441→                      <TableCell className="text-sm text-muted-foreground">{formatDate(person.createdAt)}</TableCell>
   442→                      <TableCell>
   443→                        <div className="flex justify-end gap-1">
   444→                          <Button variant="ghost" size="sm" className="h-8 w-8 p-0" onClick={() => openDetail(person)} title="View">
   445→                            <Eye className="h-3.5 w-3.5" />
   446→                          </Button>
   447→                          <Button variant="ghost" size="sm" className="h-8 w-8 p-0" onClick={() => openEdit(person)} title="Edit">
   448→                            <Pencil className="h-3.5 w-3.5" />
   449→                          </Button>
   450→                          <Button variant="ghost" size="sm" className="h-8 w-8 p-0" onClick={() => toggleActive(person)} title={person.is_active ? "Deactivate" : "Activate"}>
   451→                            <AlertTriangle className="h-3.5 w-3.5" />
   452→                          </Button>
   453→                          <Button variant="ghost" size="sm" className="h-8 w-8 p-0 text-red-600 hover:text-red-700 hover:bg-red-50" onClick={() => confirmDelete(person.id)} title="Delete">
   454→                            <Trash2 className="h-3.5 w-3.5" />
   455→                          </Button>
   456→                        </div>
   457→                      </TableCell>
   458→                    </TableRow>
   459→                  ))}
   460→                </TableBody>
   461→              </Table>
   462→            </div>
   463→          </>
   464→        )}
   465→      </div>
   466→
   467→      {/* Add/Edit Dialog */}
   468→      <Dialog open={formOpen} onOpenChange={setFormOpen}>
   469→        <DialogContent className="max-w-lg max-h-[85vh] overflow-y-auto mx-4 sm:mx-0 w-[calc(100%-2rem)] sm:w-full">
   470→          <DialogHeader>
   471→            <DialogTitle>{editingId ? "Edit Suspected Person" : "Register Suspected Person"}</DialogTitle>
   472→            <DialogDescription>
   473→              {editingId ? "Update suspect details" : "Add a new person to monitor across all service providers"}
   474→            </DialogDescription>
   475→          </DialogHeader>
   476→          <form onSubmit={handleSubmit} className="space-y-4">
   477→            <div className="grid gap-2">
   478→              <Label htmlFor="sp-name">Full Name *</Label>
   479→              <Input
   480→                id="sp-name"
   481→                placeholder="Full name of the suspected person"
   482→                value={form.name}
   483→                onChange={(e) => setForm((f) => ({ ...f, name: e.target.value }))}
   484→                autoFocus
   485→              />
   486→            </div>
   487→
   488→            <div className="grid grid-cols-2 gap-3">
   489→              <div className="grid gap-2">
   490→                <Label htmlFor="sp-phone">Phone</Label>
   491→                <Input
   492→                  id="sp-phone"
   493→                  placeholder="Phone number"
   494→                  value={form.phone}
   495→                  onChange={(e) => setForm((f) => ({ ...f, phone: e.target.value }))}
   496→                />
   497→              </div>
   498→              <div className="grid gap-2">
   499→                <Label htmlFor="sp-severity">Severity</Label>
   500→                <Select value={form.severity} onValueChange={(v) => setForm((f) => ({ ...f, severity: v }))}>
   501→                  <SelectTrigger id="sp-severity">
   502→                    <SelectValue />
   503→                  </SelectTrigger>
   504→                  <SelectContent>
   505→                    <SelectItem value="LOW">Low</SelectItem>
   506→                    <SelectItem value="MEDIUM">Medium</SelectItem>
   507→                    <SelectItem value="HIGH">High</SelectItem>
   508→                    <SelectItem value="CRITICAL">Critical</SelectItem>
   509→                  </SelectContent>
   510→                </Select>
   511→              </div>
   512→            </div>
   513→
   514→            <div className="grid grid-cols-2 gap-3">
   515→              <div className="grid gap-2">
   516→                <Label htmlFor="sp-idtype">ID Type</Label>
   517→                <Select value={form.idType} onValueChange={(v) => setForm((f) => ({ ...f, idType: v }))}>

   521→                  <SelectContent>

   541→            <div className="grid grid-cols-2 gap-3">

   677→            </div>

   682→      {/* Delete Confirmation */}
   683→      <Dialog open={deleteOpen} onOpenChange={setDeleteOpen}>
   684→        <DialogContent className="max-w-sm mx-4 sm:mx-0 w-[calc(100%-2rem)] sm:w-full">

   687→              <Trash2 className="h-5 w-5" />
   688→              Delete Suspected Person

   692→            </DialogDescription>

   694→          <DialogFooter className="gap-2">
   695→            <Button variant="outline" onClick={() => setDeleteOpen(false)}>Cancel</Button>
   696→            <Button
   697→              variant="destructive"
   698→              onClick={handleDelete}
   699→              disabled={deleteLoading}
   700→            >

   702→            </Button>

   704→        </DialogContent>
   705→      </Dialog>
   706→
   707→      {/* Pagination Controls */}
   708→      {!loading && persons.length > 0 && (

   711→          totalPages={pagination.totalPages}

   713→          pageSizeOptions={pagination.pageSizeOptions}
   714→          totalItems={persons.length}
   715→          rangeInfo={pagination.rangeInfo}
   716→          goToPage={pagination.goToPage}
