   522→   522→async function execViaPg(sql: string, params?: unknown[]): Promise<void> {
   523→   523→  const client = new Client({ connectionString: process.env.DATABASE_URL! });
   524→   524→  try {
   525→   525→    await client.connect();
   526→   526→    await client.query(sql, params);
   527→   527→  } finally {
   528→   528→    await client.end().catch(() => {});
   529→   529→  }
   530→   530→}

   533→   533→async function execViaPrisma(sql: string): Promise<void> {

   536→   536→  });

   539→   539→    await prisma.$executeRawUnsafe(sql);
   540→   540→  } finally {
   541→   541→    await prisma.$disconnect().catch(() => {});
   542→   542→  }
   543→   543→}
