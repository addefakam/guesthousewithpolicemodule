    96→  PENDING: "bg-yellow-100 text-yellow-800 hover:bg-yellow-100 border-yellow-200",

   101→
   102→function StatusBadge({ status }: { status: string }) {
   103→  const cfg = STATUS_CONFIG[status] || { label: status, variant: "secondary" as const };

   105→    <Badge variant={cfg.variant} className={STATUS_BADGE_CLASS[status] || ""}>
   106→      {cfg.label}
   107→    </Badge>
   108→  );

   147→export default function ProvidersPage() {
   148→  const { refreshKey, triggerRefresh, currentUser } = useAppStore();
   149→  const isSuperuser = currentUser?.role === "SUPERUSER";
   150→  const [providers, setProviders] = useState<Provider[]>([]);
   151→  const [loading, setLoading] = useState(true);
   152→  const [selectedProvider, setSelectedProvider] = useState<Provider | null>(null);
   153→  const [detailOpen, setDetailOpen] = useState(false);
   154→
   155→  // Registration form
   156→  const [registerOpen, setRegisterOpen] = useState(false);
   157→  const [registerForm, setRegisterForm] = useState<RegisterForm>(emptyRegisterForm);
   158→  const [registering, setRegistering] = useState(false);
   159→
   160→
   161→
   162→  // Action dialogs
   163→  const [rejectDialog, setRejectDialog] = useState<Provider | null>(null);
   164→  const [rejectReason, setRejectReason] = useState("");
   165→  const [confirmAction, setConfirmAction] = useState<{ provider: Provider; action: string } | null>(null);
   166→  const [actioning, setActioning] = useState(false);
   167→
   168→  // Suspension dialog (with reason + notification)
   169→  const [suspendDialog, setSuspendDialog] = useState<Provider | null>(null);
   170→  const [suspensionReason, setSuspensionReason] = useState("");
   171→  const [providerMessage, setProviderMessage] = useState("");
   172→  const [suspending, setSuspending] = useState(false);
   173→
   174→  const pagination = usePagination({ totalItems: providers.length, initialPageSize: 5, pageSizeOptions: [5, 10, 20, 50] });
   175→  const paginatedProviders = useMemo(() => pagination.paginate(providers), [providers, pagination]);
   176→
   177→  const fetchProviders = useCallback(async () => {
   178→    try {
   179→      setLoading(true);
   180→      const data = await apiGetProviders();
   181→      setProviders(Array.isArray(data) ? data : []);
   182→    } catch (err: unknown) {
   183→      const message = err instanceof Error ? err.message : "Failed to load providers";
   184→      toast.error(message);
   185→    } finally {
   186→      setLoading(false);
   187→    }
   188→  }, []);
   189→
   190→  useEffect(() => {
   191→    fetchProviders();
   192→  }, [fetchProviders, refreshKey]);
   193→
   194→  const openDetail = (provider: Provider) => {
   195→    setSelectedProvider(provider);
   196→    setDetailOpen(true);
   197→  };
   198→
   199→  const openReject = (provider: Provider) => {
   200→    setRejectDialog(provider);
   201→    setRejectReason("");
   202→  };
   203→
   204→
   205→
   206→  const handleReject = async () => {
   207→    if (!rejectDialog || !rejectReason.trim()) {
   208→      toast.error("Rejection reason is required");
   209→      return;
   210→    }
   211→    try {
   212→      setActioning(true);
   213→      await apiUpdateProvider(rejectDialog.id, { status: "REJECTED", rejectionReason: rejectReason.trim() });
   214→      toast.success("Provider rejected");
   215→      setRejectDialog(null);
   216→      setRejectReason("");
   217→      triggerRefresh();
   218→    } catch (err: unknown) {
   219→      const message = err instanceof Error ? err.message : "Failed to reject provider";
   220→      toast.error(message);
   221→    } finally {
   222→      setActioning(false);
   223→    }
   224→  };
   225→
   226→  const handleStatusAction = async (provider: Provider, status: string) => {
   227→    try {
   228→      setActioning(true);
   229→      if (status === "REACTIVATE") {
   230→        // Police reactivates: set back to APPROVED and clear suspension fields
   231→        await apiUpdateProvider(provider.id, { status: "APPROVED" });
   232→        toast.success(`"${provider.name}" has been reactivated`);
   233→      } else {
   234→        await apiUpdateProvider(provider.id, { status });
   235→        toast.success(`Provider ${status.toLowerCase()}`);
   236→      }
   237→      setConfirmAction(null);
   238→      triggerRefresh();
   239→    } catch (err: unknown) {
   240→      const message = err instanceof Error ? err.message : "Failed to update provider";
   241→      toast.error(message);
   242→    } finally {
   243→      setActioning(false);
   244→    }
   245→  };
   246→
   247→  const openSuspend = (provider: Provider) => {
   248→    setSuspendDialog(provider);
   249→    setSuspensionReason("");
   250→    setProviderMessage("");
   251→  };
   252→
   253→  const handleRegister = async () => {
   254→    if (!registerForm.name.trim() || !registerForm.ownerName.trim() || !registerForm.phone.trim()) {
   255→      toast.error("Guesthouse name, owner name, and phone are required");
   256→      return;
   257→    }
   258→    if (!registerForm.username.trim() || !registerForm.password.trim()) {
   259→      toast.error("Operator username and password are required");
   260→      return;
   261→    }
   262→    if (registerForm.password.trim().length < 4) {
   263→      toast.error("Password must be at least 4 characters");
   264→      return;
   265→    }
   266→    try {
   267→      setRegistering(true);
   268→      await apiSuperCreateProvider({
   269→        ...registerForm,
   270→        licenseFile: registerForm.licenseFileData || undefined,
   271→      });
   272→      toast.success(`"${registerForm.name}" registered and approved successfully`);
   273→      setRegisterOpen(false);
   274→      setRegisterForm(emptyRegisterForm);
   275→      triggerRefresh();
   276→    } catch (err: unknown) {
   277→      toast.error(err instanceof Error ? err.message : "Failed to register guesthouse");
   278→    } finally {
   279→      setRegistering(false);
   280→    }
   281→  };
   282→
   283→  const handleSuspend = async () => {
   284→    if (!suspendDialog) return;
   285→    if (!suspensionReason.trim()) {
   286→      toast.error("Please provide a reason for suspension");
   287→      return;
   288→    }

   290→      setSuspending(true);
   291→      await apiPoliceSuspendProvider({
   292→        providerId: suspendDialog.id,
   293→        suspensionReason: suspensionReason.trim(),
   294→        providerMessage: providerMessage.trim() || undefined,
   295→      });
   296→      toast.success(`"${suspendDialog.name}" has been suspended. Notification sent to provider.`);
   297→      setSuspendDialog(null);
   298→      setSuspensionReason("");
   299→      setProviderMessage("");
   300→      triggerRefresh();
   301→    } catch (err: unknown) {
   302→      toast.error(err instanceof Error ? err.message : "Failed to suspend provider");
   303→    } finally {
   304→      setSuspending(false);
   305→    }
   306→  };
   307→
   308→  const formatDate = (dateStr: string | null) => {
   309→    if (!dateStr) return "—";
   310→    return new Date(dateStr).toLocaleDateString("en-US", {
   311→      year: "numeric",
   312→      month: "short",
   313→      day: "numeric",
   314→    });
   315→  };
   316→
   317→  const approvedCount = providers.filter((p) => p.status === "APPROVED").length;
   318→  const pendingCount = providers.filter((p) => p.status === "PENDING").length;
   319→
   320→  return (

   327→              <Building2 className="h-4 w-4 sm:h-5 sm:w-5 text-slate-600" />
   328→            </div>
   329→            <div>
   330→              <p className="text-[10px] sm:text-sm text-muted-foreground">Total</p>
   331→              <p className="text-lg sm:text-2xl font-bold">{providers.length}</p>
   332→            </div>
   333→          </div>
   334→        </div>
   335→        <div className="rounded-xl border bg-card p-3 sm:p-4 shadow-sm">
   336→          <div className="flex items-center gap-2 sm:gap-3">
   337→            <div className="flex h-8 w-8 sm:h-10 sm:w-10 items-center justify-center rounded-lg bg-emerald-50">
   338→              <CheckCircle2 className="h-4 w-4 sm:h-5 sm:w-5 text-emerald-600" />
   339→            </div>
   340→            <div>
   341→              <p className="text-[10px] sm:text-sm text-muted-foreground">Approved</p>
   342→              <p className="text-lg sm:text-2xl font-bold text-emerald-600">{approvedCount}</p>
   343→            </div>
   344→          </div>
   345→        </div>
   346→        <div className="rounded-xl border bg-card p-3 sm:p-4 shadow-sm">
   347→          <div className="flex items-center gap-2 sm:gap-3">
   348→            <div className="flex h-8 w-8 sm:h-10 sm:w-10 items-center justify-center rounded-lg bg-yellow-50">
   349→              <ShieldCheck className="h-4 w-4 sm:h-5 sm:w-5 text-yellow-600" />
   350→            </div>

   363→          <p className="text-xs sm:text-sm text-muted-foreground">
   364→            {isSuperuser
   365→              ? "Register new guesthouses and manage existing ones."
   366→              : "Manage registrations and licensing"}
   367→          </p>
   368→        </div>
   369→        <div className="flex gap-2">
   370→          {isSuperuser && (

   375→        </div>

   377→
   378→      {/* Providers Table/Cards */}
   379→      <div className="rounded-xl border bg-card shadow-sm">
   380→        {loading ? (
   381→          <div className="space-y-3 p-4 sm:p-6">
   382→            {Array.from({ length: 5 }).map((_, i) => (
   383→              <Skeleton key={i} className="h-20 w-full sm:h-12" />

   385→          </div>
   386→        ) : providers.length === 0 ? (
   387→          <div className="flex flex-col items-center justify-center py-12 sm:py-16 text-center">
   388→            <Building2 className="mb-3 h-10 w-10 sm:h-12 sm:w-12 text-muted-foreground/40" />
   389→            <p className="text-xs sm:text-sm text-muted-foreground">No providers registered yet</p>
   390→          </div>
   391→        ) : (
   392→          <>
   393→            {/* Mobile/Tablet: Card layout */}
   394→            <div className="divide-y lg:hidden">
   395→              {paginatedProviders.map((provider) => (
   396→                <div key={provider.id} className={`p-3 sm:p-4 ${provider.status === "SUSPENDED" ? "bg-orange-50/60" : ""}`}>
   397→                  <div className="flex items-start justify-between gap-2">
   398→                    <button className="min-w-0 flex-1 text-left" onClick={() => openDetail(provider)}>
   399→                      <div className="flex items-center gap-2">
   400→                        <p className="truncate text-sm font-medium">{provider.name}</p>
   401→                        <StatusBadge status={provider.status} />
   402→                      </div>
   403→                      <div className="mt-1.5 flex flex-wrap items-center gap-x-3 gap-y-1 text-xs text-muted-foreground">

   405→                        <span className="flex items-center gap-1"><Phone className="h-3 w-3" /> {provider.phone}</span>
   406→                      </div>
   407→                      {provider.address && (
   408→                        <div className="mt-1 flex items-center gap-1 text-[10px] text-muted-foreground">
   409→                          <MapPin className="h-3 w-3" /> {provider.address}

   424→                        <CheckCircle2 className="h-3.5 w-3.5" /> Approve

   438→                      <button className="flex items-center gap-1 rounded-md px-2.5 py-1.5 text-xs font-medium text-orange-600 hover:bg-orange-50" onClick={() => openSuspend(provider)}>
   439→                        <Ban className="h-3.5 w-3.5" /> Suspend

   447→            {/* Desktop: Table layout */}
   448→            <div className="hidden lg:block overflow-x-auto">
   449→              <Table>
   450→                <TableHeader>

   452→                    <TableHead>Provider Name</TableHead>

   455→                    <TableHead>Phone</TableHead>
   456→                    <TableHead>Status</TableHead>
   457→                    <TableHead className="text-right">Actions</TableHead>
   458→                  </TableRow>
   459→                </TableHeader>
   460→                <TableBody>

   475→                          {provider.status !== "REJECTED" && provider.status !== "APPROVED" && provider.status !== "SUSPENDED" && (

   477→                              <XCircle className="mr-1 h-4 w-4" /> Reject

   499→      </div>
   500→
   501→      <PaginationControls

   512→      {/* Provider Detail Dialog */}
   513→      <Dialog open={detailOpen} onOpenChange={setDetailOpen}>
