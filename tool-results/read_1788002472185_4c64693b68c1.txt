    29→// ── Types ──
    30→interface Summary {
    31→  totalGuests: number; totalReservations: number; totalCheckIns: number;
    32→  totalCheckOuts: number; activeGuests: number; totalMatches: number;
    33→  totalProviders: number; totalRooms: number;
    34→}
    35→interface NameCount { name: string; count: number; }
    36→interface ProviderRow { name: string; guests: number; checkIns: number; checkOuts: number; matches: number; rooms: number; }
    37→interface OccupancyRow { name: string; total: number; occupied: number; available: number; reserved: number; maintenance: number; rate: number; }
    38→interface FreqStay { id: string; guestName: string; guestPhone: string; guestIdNumber: string; providerNames: string; stayCount: number; avgDaysBetween: number; riskLevel: string; isReviewed: boolean; createdAt: string; }
    39→
    40→interface ReportData {
    41→  period: string; date: string; label: string; startDate: string; endDate: string;
    42→  summary: Summary;
    43→  checkInTrend: { date: string; count: number }[];
    44→  checkOutTrend: { date: string; count: number }[];
    45→  nationalities: NameCount[];
    46→  idTypes: NameCount[];
    47→  providerBreakdown: ProviderRow[];
    48→  suspectSeverities: NameCount[];
    49→  occupancyByProvider: OccupancyRow[];
    50→  peakHours: { hour: string; count: number }[];
    51→  frequentStayAlerts: FreqStay[];
    52→  reservationStatuses: NameCount[];
    53→  providers: { id: string; name: string }[];
    54→}
    55→
    56→const COLORS = ["#1e3a5f", "#2563eb", "#7c3aed", "#dc2626", "#ea580c", "#ca8a04", "#16a34a", "#0d9488", "#6366f1", "#ec4899", "#8b5cf6", "#14b8a6", "#f59e0b", "#64748b", "#84cc16"];
    57→
    58→const STATUS_COLORS: Record<string, string> = { ACTIVE: "#16a34a", UPCOMING: "#2563eb", COMPLETED: "#64748b", CANCELLED: "#dc2626" };
    59→
    60→// ── KPI Card ──
    61→function KpiCard({ icon: Icon, label, value, sub, color = "text-primary" }: { icon: React.ElementType; label: string; value: number | string; sub?: string; color?: string }) {
    62→  return (
    63→    <Card className="relative overflow-hidden">
    64→      <CardContent className="p-4">
    65→        <div className="flex items-start justify-between">
    66→          <div className="space-y-1">
    67→            <p className="text-xs font-medium text-muted-foreground uppercase tracking-wider">{label}</p>
    68→            <p className={`text-2xl font-bold ${color}`}>{typeof value === "number" ? value.toLocaleString() : value}</p>
    69→            {sub && <p className="text-[11px] text-muted-foreground">{sub}</p>}
    70→          </div>
    71→          <div className="rounded-lg bg-primary/10 p-2.5">
    72→            <Icon className="size-5 text-primary" />
    73→          </div>
    74→        </div>
    75→      </CardContent>
    76→    </Card>
    77→  );
    78→}
    79→
    80→// ── Inline bar ──
    81→function InlineBar({ value, max, color = "bg-primary" }: { value: number; max: number; color?: string }) {
    82→  const pct = max > 0 ? Math.round((value / max) * 100) : 0;
    83→  return (
    84→    <div className="flex items-center gap-2">
    85→      <div className="h-2 w-24 rounded-full bg-muted overflow-hidden">
    86→        <div className={`h-full rounded-full ${color} transition-all duration-500`} style={{ width: `${pct}%` }} />
    87→      </div>
    88→      <span className="text-xs font-semibold tabular-nums min-w-[28px] text-right">{value}</span>
    89→    </div>
    90→  );
    91→}
    92→
    93→// ── Skeleton ──
    94→function SkeletonGrid() {
    95→  return (
    96→    <div className="space-y-4">

   101→      </div>
   102→      <Skeleton className="h-64 rounded-lg" />
   103→      <Skeleton className="h-64 rounded-lg" />
   104→    </div>
   105→  );
   106→}

   301→          {/* ═══════════════ OCCUPANCY TAB ═══════════════ */}
