   159→

   164→    name: "",
   165→    phone: "",
   166→    email: "",
   167→    idNumber: "",

   169→    nationality: "",
   170→    address: "",
   171→    notes: "",
   172→  });

   196→

   200→    reservation: Reservation;

   232→          status: r.status,
   233→          pricePerNight: r.pricePerNight,

   292→      list = list.filter((r) => r.status === statusFilter);

   310→  const paged = filtered.slice(page * pageSize, (page + 1) * pageSize);

   318→      return new Date(dateStr + "T00:00:00").toLocaleDateString("en-US", { month: "short", day: "numeric" });

   325→      return new Date(dateStr + "T00:00:00").toLocaleDateString("en-US", {
   326→        month: "short",
   327→        day: "numeric",
   328→        year: "numeric",
   329→      });
   330→    } catch {
   331→      return dateStr;
   332→    }

   341→      toast.error("Check-out must be after check-in");
   342→      return;

   367→        checkOut: createForm.checkOut,

   383→        return;
   384→      }
   385→      toast.error(parsed?.error || raw || "Failed to create reservation");
   386→    }
   387→  };
   388→

   390→    setCreateOpen(false);
   391→    setWizardStep(1);
   392→    setGuestMode("existing");
   393→    setSelectedGuestId("");

   407→      const message = err instanceof Error ? err.message : "Failed to delete reservation";
   408→      toast.error(message);
   409→    } finally {
   410→      setDeleting(false);
   411→    }

   418→      setActionLoading(true);
   419→      if (type === "checkin") {
   420→        await apiCheckin(reservation.id);
   421→        toast.success("Guest checked in successfully");

   423→        await apiCheckout(reservation.id);

   426→        await apiCancelReservation(reservation.id);

   435→      setActionLoading(false);

   470→      setPaying(false);

   494→

   500→            <Skeleton className="h-8 w-40" />

   507→        <div className="space-y-2">

   510→          ))}
   511→        </div>
   512→      </div>
   513→    );
   514→  }
   515→
   516→  return (
   517→    <div className="space-y-6 p-4 md:p-6">
   518→      {/* Header */}
   519→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   520→        <div>
   521→          <h1 className="text-2xl font-bold text-gray-900">Reservations</h1>
   522→          <p className="mt-1 text-sm text-gray-500">
   523→            {reservations.length} reservation{reservations.length !== 1 ? "s" : ""} total

   525→        </div>
   526→        <Button onClick={() => setCreateOpen(true)} className="gap-2">
   527→          <Plus className="h-4 w-4" />
   528→          New Reservation
   529→        </Button>
   530→      </div>
   531→
   532→      {/* Status Tabs + Search */}
   533→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">

   544→              </TabsTrigger>
   545→            ))}
   546→          </TabsList>
   547→        </Tabs>

   549→          <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-gray-400" />
   550→          <Input
   551→            placeholder="Search reservations..."
   552→            value={search}

   555→              setPage(0);
   556→              clearHighlight();
   557→            }}
   558→            className="pl-9"
   559→          />
   560→        </div>
   561→      </div>
   562→
   563→      {/* Table */}
   564→      <div className="rounded-xl border bg-white overflow-hidden">
   565→        <div className="overflow-x-auto">
   566→          <Table>

   570→                <TableHead className="min-w-[100px]">Room</TableHead>
   571→                <TableHead className="min-w-[100px]">Check-in</TableHead>
   572→                <TableHead className="min-w-[100px]">Check-out</TableHead>
   573→                <TableHead className="text-center">Nights</TableHead>
   574→                <TableHead className="text-right">Total</TableHead>
   575→                <TableHead className="text-right">Paid</TableHead>
   576→                <TableHead className="text-right">Balance</TableHead>
