   217→// ── Helper: get nav items based on role ──
   218→function getNavItems(user: CurrentUser, disabledPages: string[] = []): NavItem[] {
   219→  const disabledSet = new Set(disabledPages);
   220→
   221→  switch (user.role) {
   222→    case "POLICE": {
   223→      const rank = (user.policeRank || "OFFICER") as PoliceRank;
   224→      const allowedPages = POLICE_RANK_PERMISSIONS[rank] || POLICE_RANK_PERMISSIONS.OFFICER;
   225→      return POLICE_NAV_ITEMS.filter((item) => allowedPages.includes(item.page));
   226→    }
   227→
   228→    case "SUPERUSER": {
   229→      // Guest house owners (SUPERUSER with providerId) see the provider dashboard.
   230→      // System admins (SUPERUSER without providerId) see the admin dashboard.
   231→      if (user.providerId) {
   232→        return ALL_NAV_ITEMS.filter(
   233→          (item) => !OPERATOR_EXCLUDED.has(item.page) && !disabledSet.has(item.page)
   234→        );
   235→      }
   236→      return SUPERUSER_NAV_ITEMS;
   237→    }
   238→
   239→    case "OPERATOR":
   240→      return ALL_NAV_ITEMS.filter(
   241→        (item) => !OPERATOR_EXCLUDED.has(item.page) && !disabledSet.has(item.page)
   242→      );
   243→
   244→    case "STAFF": {
   245→      const items: NavItem[] = [];
   246→      // Always give staff access to dashboard
   247→      if (!disabledSet.has("dashboard")) {
   248→        items.push({
   249→          page: "dashboard",
   250→          label: "Dashboard",
   251→          icon: LayoutDashboard,
   252→        });
   253→      }
   254→      // Add items based on permissions (deduplicate by page key)
   255→      const seen = new Set<string>();
   256→      for (const perm of user.permissions) {
   257→        const mapped = PERMISSION_PAGE_MAP[perm];
   258→        if (mapped && !seen.has(mapped.page) && !disabledSet.has(mapped.page)) {
   259→          seen.add(mapped.page);
   260→          items.push(mapped);
   261→        }
   262→      }
   263→      return items;
   264→    }
   265→
   266→    default:
   267→      return [
   268→        { page: "dashboard", label: "Dashboard", icon: LayoutDashboard },
   269→      ];
   270→  }
   271→}

   273→// ── Helper: get initials from name ──
   274→function getInitials(name: string | undefined | null): string {
   275→  if (!name) return "?";
   276→  return name
   277→    .split(" ")
   278→    .map((w) => w[0])
   279→    .join("")
   280→    .toUpperCase()
   281→    .slice(0, 2);
   282→}

   284→// ── Helper: get role badge className (no label — use t() for that) ──
   285→function getRoleBadgeClass(role: string): string {
   286→  switch (role) {
   287→    case "SUPERUSER": return "bg-amber-100 text-amber-700 border-amber-200";
   288→    case "OPERATOR": return "bg-emerald-100 text-emerald-700 border-emerald-200";
   289→    case "STAFF": return "bg-sky-100 text-sky-700 border-sky-200";
   290→    case "POLICE": return "bg-rose-100 text-rose-700 border-rose-200";
   291→    default: return "bg-slate-100 text-slate-700 border-slate-200";
   292→  }
   293→}

   341→function SubscriptionStatusCard({ collapsed }: { collapsed: boolean }) {
   342→  const subscription = useAppStore((s) => s.subscription);
   343→
   344→  if (!subscription) return null;
   345→
   346→  const isActive = subscription.status === "ACTIVE";
   347→  const isWarning = subscription.status === "WARNING";
   348→  const isExpired = subscription.status === "EXPIRED";
   349→  const isSuspended = subscription.status === "SUSPENDED";
   350→
   351→  // Collapsed mode — icon indicator
   352→  if (collapsed) {
   353→    const bgClass = isActive
   354→      ? "bg-emerald-100"
   355→      : isSuspended || isExpired
   356→      ? "bg-rose-100"
   357→      : "bg-amber-100";
   358→    const iconClass = isActive
   359→      ? "text-emerald-600"
   360→      : isSuspended || isExpired
   361→      ? "text-rose-600"
   362→      : "text-amber-600";
   363→    return (
   364→      <div className="flex flex-col items-center gap-1 px-2 py-2">
   365→        <div className={`flex h-8 w-8 items-center justify-center rounded-full ${bgClass} animate-pulse shadow-sm`}> 
   366→          <Clock className={`size-4 ${iconClass}`} />
   367→        </div>
   368→      </div>
   369→    );
   370→  }
   371→
   372→  // Expanded card
   373→  const containerClass = isActive
   374→    ? "border-emerald-200 bg-emerald-50"
   375→    : isSuspended || isExpired
   376→    ? "border-rose-200 bg-rose-50"
   377→    : "border-amber-200 bg-amber-50";
   378→
   379→  const iconColorClass = isActive
   380→    ? "text-emerald-600"
   381→    : isSuspended || isExpired
   382→    ? "text-rose-600"
   383→    : "text-amber-600";
   384→
   385→  const titleClass = isActive
   386→    ? "text-emerald-800"
   387→    : isSuspended || isExpired
   388→    ? "text-rose-800"
   389→    : "text-amber-800";
   390→
   391→  const subtextClass = isActive
   392→    ? "text-emerald-700"
   393→    : isSuspended || isExpired
   394→    ? "text-rose-700"
   395→    : "text-amber-700";
   396→
   397→  const badgeClass = isActive
   398→    ? "bg-emerald-200/60 text-emerald-800"
   399→    : isSuspended || isExpired
   400→    ? "bg-rose-200/60 text-rose-800"
   401→    : "bg-amber-200/60 text-amber-800";
   402→
   403→  const Icon = isSuspended || isExpired ? AlertTriangle : Clock;
   404→
   405→  return (
   406→    <div className={`rounded-lg border p-3 ${containerClass} animate-subtle-pulse shadow-sm`}> 
   407→      <div className="flex items-center gap-2">
   408→        <Icon className={`h-4 w-4 shrink-0 ${iconColorClass}`} />
   409→        <p className={`text-xs font-semibold ${titleClass}`}>
   410→          {isActive
   411→            ? "Subscription Active"
   412→            : isSuspended
   413→            ? "Service Suspended"
   414→            : isExpired
   415→            ? "Subscription Expired"
   416→            : "Expiring Soon"}
   417→        </p>
   418→      </div>
   419→      <div className={`mt-1.5 flex items-center justify-between text-[11px] ${subtextClass}`}>
   420→        <span>{formatDaysRemaining(subscription.daysRemaining)}</span>
   421→        <span className={`inline-flex items-center gap-1 rounded-full px-1.5 py-0.5 font-semibold ${badgeClass}`}>
   422→          <Clock className="h-3 w-3" />
   423→          {Math.abs(subscription.daysRemaining)}d
   424→        </span>
   425→      </div>
   426→      {subscription.price > 0 ? (
   427→        <div className={`mt-1 text-[11px] ${subtextClass}`}>
   428→          Due: {subscription.price.toLocaleString()} ETB / {formatCycle(subscription.cycle)}
   429→        </div>
   430→      ) : (
   431→        <div className={`mt-1 text-[11px] ${subtextClass}`}>
   432→          {formatCycle(subscription.cycle)} cycle — Free trial
   433→        </div>
   434→      )}
   435→    </div>
   436→  );
   437→}

   673→        {/* Desktop collapse toggle */}

   680→            <span>{t("Collapse")}</span>
   681→          </button>
   682→        )}
   683→      </div>
   684→    </div>
   685→  );
