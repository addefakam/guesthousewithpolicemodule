   125→   125→

   127→   127→  const { t } = useTranslation("operations");
   128→   128→  const { refreshKey, triggerRefresh } = useAppStore();

   130→   130→  const [expenses, setExpenses] = useState<Expense[]>([]);
   131→   131→  const [categories, setCategories] = useState<Category[]>([]);
   132→   132→  const [loading, setLoading] = useState(true);
   133→   133→  const [categoryFilter, setCategoryFilter] = useState<string>("all");
   134→   134→  const [showCatMgmt, setShowCatMgmt] = useState(false);
   135→   135→  const [newCatName, setNewCatName] = useState("");
   136→   136→  const [newCatSaving, setNewCatSaving] = useState(false);
   137→   137→
   138→   138→  // Expense dialog
   139→   139→  const [dialogOpen, setDialogOpen] = useState(false);
   140→   140→  const [editing, setEditing] = useState<Expense | null>(null);
   141→   141→  const [form, setForm] = useState({
   142→   142→    date: new Date().toISOString().split("T")[0],
   143→   143→    category: "",
   144→   144→    description: "",
   145→   145→    amount: "",
   146→   146→    vendor: "",
   147→   147→    paymentMethod: "CASH",
   148→   148→    receiptNo: "",
   149→   149→    taxAmount: "0",
   150→   150→  });
   151→   151→  const [saving, setSaving] = useState(false);
   152→   152→  const [deleteTarget, setDeleteTarget] = useState<Expense | null>(null);
   153→   153→  const [deleting, setDeleting] = useState(false);
   154→   154→  const [deleteCatTarget, setDeleteCatTarget] = useState<Category | null>(null);
   155→   155→
   156→   156→  // ─── Data Fetching ────────────────────────────────────────────────────────
   157→   157→
   158→   158→  const fetchData = useCallback(async () => {

   160→   160→      setLoading(true);
   161→   161→      const [expData, catData] = await Promise.all([

   169→   169→    } finally {
   170→   170→      setLoading(false);
   171→   171→    }
   172→   172→  }, []);
   173→   173→
   174→   174→  useEffect(() => { fetchData(); }, [fetchData, refreshKey]);
   175→   175→

   177→   177→
   178→   178→  const filteredExpenses = useMemo(() => {
   179→   179→    if (categoryFilter === "all") return expenses;
   180→   180→    return expenses.filter((e) => e.category === categoryFilter);
   181→   181→  }, [expenses, categoryFilter]);
   182→   182→
   183→   183→  const totalExpenses = useMemo(() => expenses.reduce((s, e) => s + e.amount, 0), [expenses]);

   185→   185→  const thisMonthTotal = useMemo(() => {
   186→   186→    const now = new Date();

   188→   188→    return expenses.filter((e) => e.date.startsWith(ym)).reduce((s, e) => s + e.amount, 0);
   189→   189→  }, [expenses]);
   190→   190→
   191→   191→  const topCategory = useMemo(() => {
   192→   192→    const map: Record<string, number> = {};
   193→   193→    expenses.forEach((e) => { map[e.category] = (map[e.category] || 0) + e.amount; });
   194→   194→    const sorted = Object.entries(map).sort((a, b) => b[1] - a[1]);
   195→   195→    return sorted.length > 0 ? sorted[0] : ["None", 0];
   196→   196→  }, [expenses]);
   197→   197→
   198→   198→  const categoryBreakdown = useMemo(() => {
   199→   199→    const map: Record<string, number> = {};
   200→   200→    filteredExpenses.forEach((e) => { map[e.category] = (map[e.category] || 0) + e.amount; });
   201→   201→    const maxVal = Math.max(...Object.values(map), 1);
   202→   202→    return Object.entries(map)
   203→   203→      .sort((a, b) => b[1] - a[1])
   204→   204→      .map(([cat, val]) => ({
   205→   205→        category: cat,
   206→   206→        amount: val,
   207→   207→        percent: (val / maxVal) * 100,
   208→   208→        color: categories.find((c) => c.name === cat)?.color || DEFAULT_COLORS[Object.keys(map).indexOf(cat) % DEFAULT_COLORS.length],
   209→   209→      }));
   210→   210→  }, [filteredExpenses, categories]);
   211→   211→
   212→   212→  // ─── CRUD ─────────────────────────────────────────────────────────────────
   213→   213→
   214→   214→  const openCreate = () => {
   215→   215→    setEditing(null);
   216→   216→    setForm({
   217→   217→      date: new Date().toISOString().split("T")[0],
   218→   218→      category: categories[0]?.name || "",
   219→   219→      description: "",
   220→   220→      amount: "",
   221→   221→      vendor: "",
   222→   222→      paymentMethod: "CASH",
   223→   223→      receiptNo: "",
   224→   224→      taxAmount: "0",
   225→   225→    });
   226→   226→    setDialogOpen(true);
   227→   227→  };
   228→   228→
   229→   229→  const openEdit = (exp: Expense) => {
   230→   230→    setEditing(exp);
   231→   231→    setForm({
   232→   232→      date: exp.date,
   233→   233→      category: exp.category,
   234→   234→      description: exp.description,
   235→   235→      amount: String(exp.amount),
   236→   236→      vendor: exp.vendor,
   237→   237→      paymentMethod: exp.paymentMethod,
   238→   238→      receiptNo: exp.receiptNo,
   239→   239→      taxAmount: String(exp.taxAmount),
   240→   240→    });
   241→   241→    setDialogOpen(true);
   242→   242→  };
   243→   243→
   244→   244→  const handleSave = async () => {
   245→   245→    if (!form.date || !form.category || !form.description || !form.amount || !form.paymentMethod) {
   246→   246→      toast.error("Please fill in all required fields");
   247→   247→      return;
   248→   248→    }
   249→   249→    try {
   250→   250→      setSaving(true);
   251→   251→      const payload = {
   252→   252→        date: form.date,
   253→   253→        category: form.category,
   254→   254→        description: form.description,
   255→   255→        amount: Number(form.amount),
   256→   256→        vendor: form.vendor,
   257→   257→        paymentMethod: form.paymentMethod,
   258→   258→        receiptNo: form.receiptNo,
   259→   259→        taxAmount: Number(form.taxAmount) || 0,
   260→   260→      };
   261→   261→      if (editing) {
   262→   262→        await apiUpdateExpense(editing.id, payload);
   263→   263→        toast.success("Expense updated");
   264→   264→      } else {
   265→   265→        await apiCreateExpense(payload);
   266→   266→        toast.success("Expense created");
   267→   267→      }
   268→   268→      setDialogOpen(false);
   269→   269→      triggerRefresh();
   270→   270→    } catch (err: unknown) {
   271→   271→      toast.error(err instanceof Error ? err.message : "Failed to save expense");
   272→   272→    } finally {
   273→   273→      setSaving(false);
   274→   274→    }
   275→   275→  };
   276→   276→
   277→   277→  const handleDelete = async () => {
   278→   278→    if (!deleteTarget) return;
   279→   279→    try {
   280→   280→      setDeleting(true);
   281→   281→      await apiDeleteExpense(deleteTarget.id);
   282→   282→      toast.success("Expense deleted");
   283→   283→      setDeleteTarget(null);
   284→   284→      triggerRefresh();
   285→   285→    } catch (err: unknown) {
   286→   286→      toast.error(err instanceof Error ? err.message : "Failed to delete expense");
   287→   287→    } finally {
   288→   288→      setDeleting(false);
   289→   289→    }
   290→   290→  };
   291→   291→
   292→   292→  const handleAddCategory = async () => {
   293→   293→    if (!newCatName.trim()) return;
   294→   294→    const preset = CATEGORY_PRESETS.find((p) => p.name.toLowerCase() === newCatName.trim().toLowerCase());
   295→   295→    try {
   296→   296→      setNewCatSaving(true);
   297→   297→      await apiCreateExpenseCategory({
   298→   298→        name: newCatName.trim(),
   299→   299→        nameAm: "",
   300→   300→        color: preset?.color || DEFAULT_COLORS[categories.length % DEFAULT_COLORS.length],
   301→   301→        icon: preset?.icon || "📋",
   302→   302→      });
   303→   303→      toast.success("Category added");
   304→   304→      setNewCatName("");
   305→   305→      triggerRefresh();
   306→   306→    } catch (err: unknown) {
   307→   307→      toast.error(err instanceof Error ? err.message : "Failed to add category");
   308→   308→    } finally {
   309→   309→      setNewCatSaving(false);
   310→   310→    }
   311→   311→  };
   312→   312→
   313→   313→  const handleDeleteCategory = async () => {
   314→   314→    if (!deleteCatTarget) return;
   315→   315→    try {
   316→   316→      await apiDeleteExpenseCategory(deleteCatTarget.id);
   317→   317→      toast.success("Category deleted");
   318→   318→      setDeleteCatTarget(null);
   319→   319→      if (categoryFilter === deleteCatTarget.name) setCategoryFilter("all");
   320→   320→      triggerRefresh();
   321→   321→    } catch (err: unknown) {
   322→   322→      toast.error(err instanceof Error ? err.message : "Failed to delete category");
   323→   323→    }
   324→   324→  };
   325→   325→
   326→   326→  const getCategoryColor = (name: string) =>
   327→   327→    categories.find((c) => c.name === name)?.color || "#6b7280";
   328→   328→
   329→   329→  // ─── Render ───────────────────────────────────────────────────────────────
   330→   330→
   331→   331→  if (loading) {
   332→   332→    return (
   333→   333→      <div className="space-y-6 p-4 md:p-6">
   334→   334→        <div className="flex items-center justify-between">
   335→   335→          <div><Skeleton className="h-8 w-40" /><Skeleton className="mt-1 h-4 w-56" /></div>
   336→   336→          <Skeleton className="h-10 w-36" />
   337→   337→        </div>
   338→   338→        <div className="grid grid-cols-1 gap-4 md:grid-cols-3">
   339→   339→          {Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-28 rounded-xl" />)}
   340→   340→        </div>
   341→   341→        <Skeleton className="h-96 rounded-xl" />
   342→   342→      </div>
   343→   343→    );
   344→   344→  }
   345→   345→
   346→   346→  return (
   347→   347→    <div className="space-y-6 p-4 md:p-6">
   348→   348→      {/* Header */}
   349→   349→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   350→   350→        <div>
   351→   351→          <h1 className="text-2xl font-bold text-gray-900">{t("expensesTitle", "Expenses")}</h1>
   352→   352→          <p className="mt-1 text-sm text-gray-500">{t("expensesDesc", "Track and manage all business expenses")}</p>
   353→   353→        </div>
   354→   354→        <div className="flex gap-2">
   355→   355→          <Button variant="outline" onClick={() => setShowCatMgmt(!showCatMgmt)} className="gap-2">

   359→   359→          <Button onClick={openCreate} className="gap-2">

   361→   361→          </Button>
   362→   362→        </div>
   363→   363→      </div>
   364→   364→
   365→   365→      {/* Summary Cards */}
   366→   366→      <div className="grid grid-cols-1 gap-4 md:grid-cols-3">
   367→   367→        <Card className="gap-0 py-0">
   368→   368→          <CardContent className="p-4">
   369→   369→            <div className="flex items-center gap-3">
   370→   370→              <div className="flex h-11 w-11 items-center justify-center rounded-lg bg-emerald-50">
   371→   371→                <DollarSign className="h-5 w-5 text-emerald-600" />
   372→   372→              </div>
   373→   373→              <div>
   374→   374→                <p className="text-sm text-gray-500">Total Expenses</p>
   375→   375→                <p className="text-xl font-bold text-gray-900">{formatPrice(totalExpenses)}</p>
   376→   376→              </div>
   377→   377→            </div>
   378→   378→          </CardContent>
   379→   379→        </Card>
   380→   380→        <Card className="gap-0 py-0">
   381→   381→          <CardContent className="p-4">
   382→   382→            <div className="flex items-center gap-3">
   383→   383→              <div className="flex h-11 w-11 items-center justify-center rounded-lg bg-amber-50">
   384→   384→                <CalendarDays className="h-5 w-5 text-amber-600" />
   385→   385→              </div>
   386→   386→              <div>
   387→   387→                <p className="text-sm text-gray-500">This Month</p>
   388→   388→                <p className="text-xl font-bold text-gray-900">{formatPrice(thisMonthTotal)}</p>
   389→   389→              </div>
   390→   390→            </div>
   391→   391→          </CardContent>
   392→   392→        </Card>
   393→   393→        <Card className="gap-0 py-0">
   394→   394→          <CardContent className="p-4">
   395→   395→            <div className="flex items-center gap-3">
   396→   396→              <div className="flex h-11 w-11 items-center justify-center rounded-lg bg-violet-50">
   397→   397→                <TrendingDown className="h-5 w-5 text-violet-600" />
   398→   398→              </div>
   399→   399→              <div>
   400→   400→                <p className="text-sm text-gray-500">Top Category</p>
   401→   401→                <p className="text-xl font-bold text-gray-900">{topCategory[0]}</p>
   402→   402→                <p className="text-xs text-gray-400">{formatPrice(topCategory[1] as number)}</p>
   403→   403→              </div>
   404→   404→            </div>
   405→   405→          </CardContent>
   406→   406→        </Card>
   407→   407→      </div>
   408→   408→
   409→   409→      {/* Category Management */}
   410→   410→      {showCatMgmt && (
   411→   411→        <Card className="gap-0 py-0">
   412→   412→          <CardContent className="p-4">
   413→   413→            <h3 className="font-semibold text-gray-900 mb-3 flex items-center gap-2">
   414→   414→              <Tag className="h-4 w-4" /> Expense Categories
   415→   415→            </h3>
   416→   416→            <div className="flex flex-wrap gap-2 mb-4">
   417→   417→              {categories.map((cat) => (
   418→   418→                <Badge
   419→   419→                  key={cat.id}
   420→   420→                  variant="outline"
   421→   421→                  className="cursor-pointer hover:bg-red-50 hover:text-red-600 hover:border-red-200 transition-colors gap-1"
   422→   422→                  onClick={() => setDeleteCatTarget(cat)}
   423→   423→                >
   424→   424→                  <span>{cat.icon}</span>
   425→   425→                  {cat.name}
   426→   426→                  <span className="ml-1 text-gray-400 text-xs hover:text-red-500">×</span>
   427→   427→                </Badge>
   428→   428→              ))}
   429→   429→            </div>
   430→   430→            <Separator className="mb-3" />
   431→   431→            <div className="flex gap-2">
   432→   432→              <Input
   433→   433→                placeholder="New category name"
   434→   434→                value={newCatName}
   435→   435→                onChange={(e) => setNewCatName(e.target.value)}
   436→   436→                onKeyDown={(e) => e.key === "Enter" && handleAddCategory()}
   437→   437→                className="max-w-xs"
   438→   438→              />
   439→   439→              <Button onClick={handleAddCategory} disabled={!newCatName.trim() || newCatSaving} size="sm">
   440→   440→                {newCatSaving ? "Adding..." : "Add"}
   441→   441→              </Button>
   442→   442→            </div>
   443→   443→          </CardContent>
   444→   444→        </Card>
   445→   445→      )}
   446→   446→
   447→   447→      {/* Category Filter */}
   448→   448→      <div className="flex items-center gap-2 flex-wrap">
   449→   449→        <span className="text-sm text-gray-500 font-medium">Filter:</span>
   450→   450→        <Button
   451→   451→          variant={categoryFilter === "all" ? "default" : "outline"}
   452→   452→          size="sm"
   453→   453→          onClick={() => setCategoryFilter("all")}
   454→   454→        >
   455→   455→          All
   456→   456→        </Button>
   457→   457→        {categories.map((cat) => (
   458→   458→          <Button
   459→   459→            key={cat.id}
   460→   460→            variant={categoryFilter === cat.name ? "default" : "outline"}
