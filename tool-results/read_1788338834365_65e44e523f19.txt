   157→// ── Helpers ──
   158→function formatDate(d: string) {
   159→  if (!d) return "—";
   160→  try { return new Date(d + "T00:00:00").toLocaleDateString("en-US", { month: "short", day: "numeric" }); } catch { return d; }
   161→}
   162→
   163→function formatCurrency(v: number) {
   164→  return new Intl.NumberFormat("en-US", { style: "currency", currency: "ETB", maximumFractionDigits: 0 }).format(v);
   165→}
   166→
   167→function todayStr() { return new Date().toISOString().split("T")[0]; }
   168→
   169→function addDays(d: string, n: number) {
   170→  const dt = new Date(d + "T00:00:00");
   171→  dt.setDate(dt.getDate() + n);
   172→  return dt.toISOString().split("T")[0];
   173→}
   174→
   175→function getFloorFromNumber(num: string): number | null {
   176→  const match = num.match(/^\d/);
   177→  return match ? parseInt(match[0], 10) : null;
   178→}
   179→
   180→function parseAmenities(amenitiesStr: string | null | undefined): string[] {
   181→  if (!amenitiesStr) return [];
   182→  try {
   183→    const parsed = JSON.parse(amenitiesStr);
   184→    return Array.isArray(parsed) ? parsed : [];
   185→  } catch {
   186→    return amenitiesStr.split(",").map((s) => s.trim()).filter(Boolean);
   187→  }
   188→}
