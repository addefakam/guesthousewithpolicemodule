   122→  const { refreshKey } = useAppStore();

   124→  const monthAgo = new Date(Date.now() - 30 * 86400000).toISOString().split("T")[0];
   125→

   140→    setLoading(true);

   144→      if (to) params.set("to", to);

   147→        revenue: raw.revenue ?? 0,
   148→        expenses: raw.expenses ?? 0,
   149→        profit: raw.profit ?? 0,
   150→        occupancyRate: raw.occupancyRate ?? 0,
   151→        reservations: Array.isArray(raw.reservations) ? raw.reservations : [],
   152→        activeGuests: Array.isArray(raw.activeGuests) ? raw.activeGuests : [],
   153→        expenseBreakdown: Array.isArray(raw.expenseBreakdown) ? raw.expenseBreakdown : [],
   154→        dailyRevenue: Array.isArray(raw.dailyRevenue) ? raw.dailyRevenue : [],
   155→      };
   156→      setData(result);
   157→    } catch (err: unknown) {
   158→      toast.error(err instanceof Error ? err.message : "Failed to load reports");

   161→    }

   165→    setPreset(value);

   167→    const y = now.getFullYear();
   168→    const m = now.getMonth();
   169→    const d = now.getDate();
   170→    const dayOfWeek = now.getDay();

   172→    let newFrom = from;
   173→    let newTo = to;
   174→
   175→    if (value === "TODAY") {
   176→      newFrom = today;
   177→      newTo = today;
   178→    } else if (value === "THIS_WEEK") {
   179→      const monday = new Date(y, m, d - ((dayOfWeek + 6) % 7));
   180→      const sunday = new Date(monday);
   181→      sunday.setDate(monday.getDate() + 6);
   182→      newFrom = monday.toISOString().split("T")[0];
   183→      newTo = sunday.toISOString().split("T")[0];
   184→    } else if (value === "LAST_WEEK") {
   185→      const thisMonday = new Date(y, m, d - ((dayOfWeek + 6) % 7));
   186→      const lastMonday = new Date(thisMonday);
   187→      lastMonday.setDate(thisMonday.getDate() - 7);
   188→      const lastSunday = new Date(lastMonday);
   189→      lastSunday.setDate(lastMonday.getDate() + 6);
   190→      newFrom = lastMonday.toISOString().split("T")[0];
   191→      newTo = lastSunday.toISOString().split("T")[0];

   193→      const monthStr = value.replace("MONTH_", "");
   194→      const [my, mm] = monthStr.split("-").map(Number);
   195→      const first = new Date(my, mm - 1, 1);
   196→      const last = new Date(my, mm, 0);
   197→      newFrom = first.toISOString().split("T")[0];
   198→      newTo = last.toISOString().split("T")[0];
   199→    }
   200→    // CUSTOM does nothing — user picks from/to manually
   201→    if (value !== "CUSTOM") {
   202→      setFrom(newFrom);
   203→      setTo(newTo);
   204→    }
   205→  };

   213→      const val = `MONTH_${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;
   214→      const label = d.toLocaleDateString("en-US", { month: "long", year: "numeric" });

   368→
