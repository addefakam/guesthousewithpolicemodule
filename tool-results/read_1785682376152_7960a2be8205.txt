   486→/** Execute SQL via raw pg Client */
   487→async function execViaPg(sql: string, params?: unknown[]): Promise<void> {
   488→  const client = new Client({ connectionString: process.env.DATABASE_URL! });
   489→  try {
   490→    await client.connect();
   491→    await client.query(sql, params);
   492→  } finally {
   493→    await client.end().catch(() => {});
   494→  }
   495→}
   496→
   497→/** Execute DDL via Prisma $executeRawUnsafe (bypasses Prisma's table-existence check) */
   498→async function execViaPrisma(sql: string): Promise<void> {
   499→  const prisma = new PrismaClient({
   500→    log: ["warn", "error"],
   501→  });
   502→  try {
   503→    // $executeRawUnsafe sends SQL directly without schema validation
   504→    await prisma.$executeRawUnsafe(sql);
   505→  } finally {
   506→    await prisma.$disconnect().catch(() => {});
   507→  }
   508→}

   510→export async function ensureDatabase(): Promise<void> {
   511→  if (_initDone) return;
   512→  if (_initPromise) return _initPromise;
   513→
   514→  if (!process.env.DATABASE_URL) {
   515→    throw new Error("DATABASE_URL environment variable is not set.");
   516→  }
   517→
   518→  _initPromise = (async () => {
   519→    console.log("[init-db] Starting database initialization...");
   520→
   521→    let usePg = true;
   522→    let pgError: unknown = null;
   523→
   524→    // Strategy 1: Try using pg Client directly
   525→    try {
   526→      const client = new Client({ connectionString: process.env.DATABASE_URL });
   527→      await client.connect();
   528→      console.log("[init-db] pg Client connected successfully.");
   529→
   530→      // Check if User table already exists
   531→      const res = await client.query(
   532→        `SELECT EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema='public' AND table_name='User') as ok`
   533→      );
   534→      const tableExists = res.rows[0]?.ok === true;
   535→      console.log("[init-db] User table exists:", tableExists);
   536→
   537→      if (tableExists) {
   538→        console.log("[init-db] Tables already exist, running migrations to add any missing columns...");
   539→        try {
   540→          await client.query(MIGRATIONS_SQL);
   541→          console.log("[init-db] Migrations applied to existing tables.");
   542→          await client.query(INDEXES_SQL);
   543→          console.log("[init-db] Indexes verified.");
   544→        } catch (migrateErr) {
   545→          console.error("[init-db] Migration on existing tables failed:", migrateErr instanceof Error ? migrateErr.message : String(migrateErr));
   546→          // Don't throw — the tables exist, migrations are best-effort
   547→        }
   548→        await client.end().catch(() => {});
   549→        _initDone = true;
   550→        return;
   551→      }
   552→
   553→      // Create enums
   554→      console.log("[init-db] Creating enum types...");
   555→      await client.query(ENUMS_SQL);
   556→
   557→      // Create tables
   558→      console.log("[init-db] Creating tables...");
   559→      await client.query(TABLES_SQL);
   560→
   561→      // Verify User table was created
   562→      const verify = await client.query(
   563→        `SELECT EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema='public' AND table_name='User') as ok`
   564→      );
   565→      if (!verify.rows[0]?.ok) {
   566→        throw new Error("DDL executed but User table still not found!");
   567→      }
   568→      console.log("[init-db] Tables verified.");
   569→
   570→      // Create foreign keys
   571→      console.log("[init-db] Creating foreign keys...");
   572→      await client.query(FKS_SQL);
   573→
   574→      // Run migrations (add new columns to existing tables)
   575→      console.log("[init-db] Running migrations...");
   576→      await client.query(MIGRATIONS_SQL);
   577→
   578→      // Create indexes
   579→      console.log("[init-db] Creating indexes...");
   580→      await client.query(INDEXES_SQL);
   581→
   582→      // Seed SUPERUSER
   583→      console.log("[init-db] Seeding SUPERUSER...");
   584→      const hashed = await bcrypt.hash("Admin@2024", 12);
   585→      await client.query(
   586→        `INSERT INTO "User" ("id","username","password","name","role","permissions","policeRank","createdAt","updatedAt")
   587→         SELECT 'su-admin-001','admin',$1,'System Administrator','SUPERUSER','[]','',NOW(),NOW()
   588→         WHERE NOT EXISTS (SELECT 1 FROM "User" WHERE "username"='admin')`,
   589→        [hashed]
   590→      );
   591→
   592→      // Seed PoliceAlertConfig
   593→      await client.query(
   594→        `INSERT INTO "PoliceAlertConfig" ("id","createdAt","updatedAt")
   595→         SELECT 'default-alert-config',NOW(),NOW()
   596→         WHERE NOT EXISTS (SELECT 1 FROM "PoliceAlertConfig" WHERE "id"='default-alert-config')`
   597→      );
   598→
   599→      console.log("[init-db] Database initialization complete via pg.");
   600→      await client.end().catch(() => {});
   601→      _initDone = true;
   602→      return;
   603→    } catch (err) {
   604→      pgError = err;
   605→      console.error("[init-db] pg approach failed:", err instanceof Error ? err.message : String(err));
   606→      usePg = false;
   607→    }
   608→
   609→    // Strategy 2: Fallback — use Prisma $executeRawUnsafe
   610→    console.log("[init-db] Falling back to Prisma $executeRawUnsafe...");
   611→    try {
   612→      // First check if tables exist using Prisma raw query
   613→      const prisma = new PrismaClient({ log: ["warn", "error"] });
   614→      try {
   615→        const res: Array<{ ok: boolean }> = await prisma.$queryRawUnsafe(
   616→          `SELECT EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema='public' AND table_name='User') as ok`
   617→        );
   618→        if (res[0]?.ok) {
   619→          console.log("[init-db] Tables already exist (verified via Prisma), running migrations...");
   620→          try {
   621→            await execViaPrisma(MIGRATIONS_SQL);
   622→            console.log("[init-db] Migrations applied via Prisma.");
   623→            await execViaPrisma(INDEXES_SQL);
   624→            console.log("[init-db] Indexes verified via Prisma.");
   625→          } catch (migrateErr) {
   626→            console.error("[init-db] Prisma migration on existing tables failed:", migrateErr instanceof Error ? migrateErr.message : String(migrateErr));
   627→          }
   628→          _initDone = true;
   629→          return;
   630→        }
   631→      } catch {
   632→        // Table might not exist, which is expected
   633→        console.log("[init-db] Tables do not exist yet, creating via Prisma...");
   634→      } finally {
   635→        await prisma.$disconnect().catch(() => {});
   636→      }
   637→
   638→      // Execute DDL via Prisma (split into parts to avoid query size issues)
   639→      await execViaPrisma(ENUMS_SQL);
   640→      console.log("[init-db] Enums created via Prisma.");
   641→
   642→      await execViaPrisma(TABLES_SQL);
   643→      console.log("[init-db] Tables created via Prisma.");
   644→
   645→      await execViaPrisma(FKS_SQL);
   646→      console.log("[init-db] Foreign keys created via Prisma.");
   647→
   648→      await execViaPrisma(MIGRATIONS_SQL);
   649→      console.log("[init-db] Migrations ran via Prisma.");
   650→
   651→      await execViaPrisma(INDEXES_SQL);
   652→      console.log("[init-db] Indexes created via Prisma.");
   653→
   654→      // Seed SUPERUSER via Prisma
   655→      const prisma2 = new PrismaClient({ log: ["warn", "error"] });
   656→      try {
   657→        const hashed = await bcrypt.hash("Admin@2024", 12);
   658→        await prisma2.$executeRawUnsafe(
   659→          `INSERT INTO "User" ("id","username","password","name","role","permissions","policeRank","createdAt","updatedAt")
   660→           SELECT 'su-admin-001','admin',$1,'System Administrator','SUPERUSER','[]','',NOW(),NOW()
   661→           WHERE NOT EXISTS (SELECT 1 FROM "User" WHERE "username"='admin')`,
   662→          hashed
   663→        );
   664→        await prisma2.$executeRawUnsafe(
   665→          `INSERT INTO "PoliceAlertConfig" ("id","createdAt","updatedAt")
   666→           SELECT 'default-alert-config',NOW(),NOW()
   667→           WHERE NOT EXISTS (SELECT 1 FROM "PoliceAlertConfig" WHERE "id"='default-alert-config')`
   668→        );
   669→        console.log("[init-db] SUPERUSER seeded via Prisma.");
   670→      } finally {
   671→        await prisma2.$disconnect().catch(() => {});
   672→      }
   673→
   674→      console.log("[init-db] Database initialization complete via Prisma fallback.");
   675→      _initDone = true;
   676→    } catch (prismaErr) {
   677→      console.error("[init-db] Prisma fallback also failed:", prismaErr instanceof Error ? prismaErr.message : String(prismaErr));
   678→      _initDone = false;
   679→      _initPromise = null; // Allow retry
   680→      throw new Error(
   681→        `Database initialization failed. pg error: ${pgError instanceof Error ? pgError.message : String(pgError)} | Prisma error: ${prismaErr instanceof Error ? prismaErr.message : String(prismaErr)}`
   682→      );
   683→    }
   684→  })();
   685→
   686→  return _initPromise;
   687→}
