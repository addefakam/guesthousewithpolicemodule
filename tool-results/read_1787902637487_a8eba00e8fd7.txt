   282→    setResRoomId("");

   306→      return;

   308→    try {

   310→      await apiCreateGroupBooking({

   312→        contactName: contactName.trim(),
   313→        contactPhone: contactPhone.trim(),
   314→        contactEmail: contactEmail.trim(),
   315→        startDate, endDate,
   316→        notes: notes.trim(),
   317→      });
   318→      toast.success(t("toastCreated"));
   319→      setCreateOpen(false);

   321→      fetchGroupBookings();

   349→      toast.error(t("toastFailedDelete"));

   351→      setDeleting(false);
   352→    }

   358→      toast.error(t("toastInvalidGuestPhone"));

   372→      setShowNewGuest(false);

   375→      setNewGuestIdNumber("");

   379→      toast.error(t("toastFailedRegister"));
   380→    } finally {
   381→      setRegisteringGuest(false);

   383→  };
   384→

   387→      toast.error(t("toastSelectGuestRoom"));

   389→    }
   390→    if (!resCheckIn || !resCheckOut) {
   391→      toast.error(t("toastCheckinCheckoutRequired"));

   399→        toast.error(t("toastSecondGuestRequired"));

   402→      if (!isValidPhone(resSecondGuestPhone)) {
   403→        toast.error(t("toastInvalidSecondPhone"));
   404→        return;
   405→      }
   406→    }
   407→    if (resExceptionallyReserved && !resExceptionReason.trim()) {
   408→      toast.error(t("toastExceptionReasonRequired"));
   409→      return;
   410→    }

   412→      setAddingReservation(true);

   416→        checkIn: resCheckIn,

   418→        groupBookingId: addReservationGroupId,

   420→        secondGuestPhone: resSecondGuestPhone,
   421→        secondGuestIdNumber: resSecondGuestIdNumber,

   433→      setAddingReservation(false);
   434→    }

   491→          await apiCheckin(res.id);
   492→          checked++;
   493→        } catch {
   494→          /* skip failed ones */
   495→        }
   496→      }
   497→      toast.success(t("toastCheckedIn", { count: checked }));
   498→      fetchGroupBookings();
   499→    } catch {
   500→      toast.error(t("toastBulkCheckinFailed"));

   512→      fetchGroupBookings();
   513→    } catch (err: unknown) {
   514→      const msg = err instanceof Error ? err.message : t("toastCheckoutFailed");

   538→        method: paymentForm.method,
   539→        referenceNo: paymentForm.referenceNo,
   540→        notes: paymentForm.notes,
   541→      });

   544→      setPaymentOpen(false);
   545→      setPaymentGroupId(null);
   546→      fetchGroupBookings();
   547→    } catch (err: unknown) {
   548→      const msg = err instanceof Error ? err.message : t("toastPaymentFailed");
   549→      toast.error(msg);
   550→    } finally {
   551→      setPaying(false);
   552→    }

   554→

   565→

   569→      year: "numeric",
   570→      month: "short",
   571→      day: "numeric",
   572→    });
   573→  };
   574→
   575→  const selectedGroup = groupBookings.find((g) => g.id === addReservationGroupId);

   581→    g.reservations?.length ?? g._count?.reservations ?? 0;
   582→
   583→  return (
   584→    <div className="space-y-6 p-4 md:p-6">
   585→      {/* Header */}
   586→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
