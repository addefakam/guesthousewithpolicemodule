    39→export default function LoginPage() {
    40→  const { setCurrentUser, setCurrentPage } = useAppStore();
    41→
    42→  const [activeTab, setActiveTab] = useState("login");
    43→
    44→  // ── Login state ──
    45→  const [loginUsername, setLoginUsername] = useState("");
    46→  const [loginPassword, setLoginPassword] = useState("");
    47→  const [loginLoading, setLoginLoading] = useState(false);
    48→
    49→  // ── Register state ──
    50→  const [regName, setRegName] = useState("");
    51→  const [regPhone, setRegPhone] = useState("");
    52→  const [regEmail, setRegEmail] = useState("");
    53→  const [regGuestHouseName, setRegGuestHouseName] = useState("");
    54→  const [regSubcity, setRegSubcity] = useState("");
    55→  const [regWoreda, setRegWoreda] = useState("");
    56→  const [regAddress, setRegAddress] = useState("");
    57→  const [regLatitude, setRegLatitude] = useState("");
    58→  const [regLongitude, setRegLongitude] = useState("");
    59→  const [regGeocoding, setRegGeocoding] = useState(false);
    60→  const [regType, setRegType] = useState("");
    61→  const [regLicenseNo, setRegLicenseNo] = useState("");

    63→  const [regUsername, setRegUsername] = useState("");

    71→  // ── Login handler ──
    72→  async function handleLogin(e: FormEvent) {
    73→    e.preventDefault();
    74→    if (!loginUsername.trim() || !loginPassword.trim()) {
    75→      toast.error("Please enter both username and password.");
    76→      return;
    77→    }
    78→    setLoginLoading(true);
    79→    try {
    80→      const resp = await apiAuth({
    81→        username: loginUsername.trim(),
    82→        password: loginPassword,
    83→      });
    84→      const userData = resp.user;
    85→      setCurrentUser({ ...userData, providerName: resp.providerName });
    86→      // Route based on role
    87→      const page = userData.role === "POLICE" ? "police-dashboard" : userData.role === "SUPERUSER" ? "super-admin-dashboard" : "dashboard";
    88→      setCurrentPage(page);
    89→      toast.success(`Welcome back, ${userData.name}!`);
    90→    } catch (err: unknown) {
    91→      const message =
    92→        err instanceof Error ? err.message : "Login failed. Please try again.";
    93→      toast.error(message);
    94→    } finally {
    95→      setLoginLoading(false);
    96→    }
    97→  }
    98→
    99→  // ── Register handler ──
   100→  async function handleRegister(e: FormEvent) {
   101→    e.preventDefault();
   102→    if (
   103→      !regName.trim() ||
   104→      !regPhone.trim() ||
   105→      !regEmail.trim() ||
   106→      !regGuestHouseName.trim() ||
   107→      !regSubcity ||
   108→      !regWoreda ||
   109→      !regType ||
   110→      !regLicenseNo.trim() ||
   111→      !regUsername.trim() ||
   112→      !regPassword.trim()
   113→    ) {
   114→      toast.error("Please fill in all required fields.");
   115→      return;
   116→    }
   117→    setRegLoading(true);
   118→    try {
   119→      const formData = new FormData();
   120→      formData.append("name", regGuestHouseName.trim());   // backend 'name' = guest house name
   121→      formData.append("ownerName", regName.trim());         // backend 'ownerName' = owner full name
   122→      formData.append("phone", regPhone.trim());
   123→      formData.append("email", regEmail.trim());
   124→      // Compose a readable full address from the dropdowns + optional street detail
   125→      const fullAddress = composeBishoftuAddress(regSubcity, regWoreda) + (regAddress.trim() ? `, ${regAddress.trim()}` : "");
   126→      formData.append("address", fullAddress);
   127→      formData.append("subcity", regSubcity);
   128→      formData.append("woreda", regWoreda);
   129→      if (regLatitude) formData.append("latitude", regLatitude);
   130→      if (regLongitude) formData.append("longitude", regLongitude);
   131→      formData.append("type", regType);
   132→      formData.append("licenseNo", regLicenseNo.trim());
   133→      formData.append("username", regUsername.trim());
   134→      formData.append("password", regPassword);
   135→      if (regLicenseFile) {
   136→        formData.append("licenseFile", regLicenseFile);
   137→      }
   138→
   139→      await apiRegisterProvider(formData);
   140→      toast.success(
   141→        "Registration submitted successfully! An admin will review and activate your account."
   142→      );
   143→      // Reset form
   144→      setRegName("");
   145→      setRegPhone("");
   146→      setRegEmail("");
   147→      setRegGuestHouseName("");
   148→      setRegSubcity("");
   149→      setRegWoreda("");
   150→      setRegAddress("");
   151→      setRegLatitude("");
   152→      setRegLongitude("");
   153→      setRegType("");
   154→      setRegLicenseNo("");
   155→      setRegUsername("");
   156→      setRegPassword("");
   157→      setRegLicenseFile(null);
   158→      if (fileInputRef.current) fileInputRef.current.value = "";
   159→      setActiveTab("login");
   160→    } catch (err: unknown) {
   161→      const message =
   162→        err instanceof Error
   163→          ? err.message
   164→          : "Registration failed. Please try again.";
   165→      toast.error(message);
   166→    } finally {
   167→      setRegLoading(false);
   168→    }
   169→  }

   171→  // ── Geocode handler (Nominatim — free, no API key) ──
   172→  async function handleGeocode() {

   174→      ? `${regAddress.trim() || regGuestHouseName.trim()}, ${regSubcity}, Bishoftu, Ethiopia`

   182→      const res = await fetch(
   183→        `https://nominatim.openstreetmap.org/search?format=json&q=${encodeURIComponent(query)}&limit=1`,
   184→        { headers: { "User-Agent": "GHMS-Registration/1.0" } }
   185→      );
   186→      const data = await res.json();
   187→      if (data && data.length > 0) {
   188→        setRegLatitude(data[0].lat);
   189→        setRegLongitude(data[0].lon);
   190→        toast.success(`Location found: ${data[0].display_name.split(",").slice(0, 2).join(",")}`);
   191→      } else {
   192→        toast.error("Could not find coordinates. You can enter them manually.");
   193→      }
   194→    } catch {
   195→      toast.error("Geocoding failed. Please enter coordinates manually.");
   196→    } finally {
   197→      setRegGeocoding(false);
   198→    }
   199→  }

   270→                  disabled={loginLoading}
   271→                >
   272→                  {loginLoading ? (

   274→                      <span className="size-4 animate-spin rounded-full border-2 border-white/30 border-t-white" />
   275→                      Signing in...
