   183→   183→function ComposeTab({
   184→   184→  isViewer,
   185→   185→  isPolice,
   186→   186→  isAdmin,
   187→   187→}: {
   188→   188→  isViewer: boolean;
   189→   189→  isPolice: boolean;
   190→   190→  isAdmin: boolean;
   191→   191→}) {
   192→   192→  const [title, setTitle] = useState("");
   193→   193→  const [message, setMessage] = useState("");
   194→   194→  const [channel, setChannel] = useState("IN_APP");
   195→   195→  const [priority, setPriority] = useState("NORMAL");
   196→   196→  const [targetType, setTargetType] = useState<"ALL_PROVIDERS" | "SELECTED_PROVIDERS">("ALL_PROVIDERS");
   197→   197→  const [providers, setProviders] = useState<BroadcastProvider[]>([]);
   198→   198→  const [selectedProviders, setSelectedProviders] = useState<Set<string>>(new Set());
   199→   199→  const [loading, setLoading] = useState(false);
   200→   200→  const [sending, setSending] = useState(false);
   201→   201→  const [search, setSearch] = useState("");
   202→   202→
   203→   203→  const fetchProviders = useCallback(async () => {
   204→   204→    setLoading(true);
   205→   205→    try {
   206→   206→      const data = await apiGetBroadcastProviders();
   207→   207→      const arr = Array.isArray(data) ? data : [];
   208→   208→      setProviders(arr);
   209→   209→    } catch (err) {
   210→   210→      console.error("Failed to load providers:", err);
   211→   211→      toast.error("Failed to load provider list");
   212→   212→    } finally {
   213→   213→      setLoading(false);
   214→   214→    }
   215→   215→  }, []);
   216→   216→
   217→   217→  useEffect(() => {
   218→   218→    fetchProviders();
   219→   219→  }, [fetchProviders]);
   220→   220→
   221→   221→  // Filter providers by search
   222→   222→  const filteredProviders = providers.filter(
   223→   223→    (p) =>
   224→   224→      p.name.toLowerCase().includes(search.toLowerCase()) ||
   225→   225→      p.ownerName.toLowerCase().includes(search.toLowerCase()) ||
   226→   226→      p.address.toLowerCase().includes(search.toLowerCase())
   227→   227→  );
   228→   228→
   229→   229→  const toggleProvider = (id: string) => {
   230→   230→    setSelectedProviders((prev) => {
   231→   231→      const next = new Set(prev);
   232→   232→      if (next.has(id)) next.delete(id);
   233→   233→      else next.add(id);
   234→   234→      return next;
   235→   235→    });
   236→   236→  };
   237→   237→
   238→   238→  const toggleAll = () => {
   239→   239→    if (selectedProviders.size === filteredProviders.length) {
   240→   240→      setSelectedProviders(new Set());
   241→   241→    } else {
   242→   242→      setSelectedProviders(new Set(filteredProviders.map((p) => p.id)));
   243→   243→    }
   244→   244→  };
   245→   245→
   246→   246→  const canSend =
   247→   247→    !isViewer && title.trim() && message.trim() &&
   248→   248→    (targetType === "ALL_PROVIDERS" || selectedProviders.size > 0);
   249→   249→
   250→   250→  const handleSend = async () => {
   251→   251→    if (!canSend) return;
   252→   252→    setSending(true);
   253→   253→    try {
   254→   254→      const result = await apiSendBroadcast({
   255→   255→        title: title.trim(),
   256→   256→        message: message.trim(),
   257→   257→        channel,
   258→   258→        priority,
   259→   259→        targetType,
   260→   260→        providerIds: targetType === "SELECTED_PROVIDERS" ? Array.from(selectedProviders) : [],
   261→   261→      });
   262→   262→      toast.success(
   263→   263→        `Notification sent! ${result.sent} delivered, ${result.failed} failed`,
   264→   264→        { description: `Channel: ${channel} | Priority: ${priority}` }
   265→   265→      );
   266→   266→      // Reset form
   267→   267→      setTitle("");
   268→   268→      setMessage("");
   269→   269→      setPriority("NORMAL");
   270→   270→      setSelectedProviders(new Set());
   271→   271→      setTargetType("ALL_PROVIDERS");
   272→   272→    } catch (err) {
   273→   273→      toast.error(err instanceof Error ? err.message : "Failed to send broadcast");
   274→   274→    } finally {
   275→   275→      setSending(false);
   276→   276→    }
   277→   277→  };
   278→   278→
   279→   279→  // Check if selected channel is available for selected providers
   280→   280→  const getChannelAvailability = () => {
   281→   281→    if (targetType === "ALL_PROVIDERS") {
   282→   282→      const targetList = filteredProviders;
   283→   283→      if (channel === "SMS" || channel === "WHATSAPP") {
   284→   284→        const withPhone = targetList.filter((p) => p.hasPhone).length;
   285→   285→        return { available: withPhone, total: targetList.length, label: "phone number" };
   286→   286→      }
   287→   287→      if (channel === "TELEGRAM") {
   288→   288→        const withTelegram = targetList.filter((p) => p.hasTelegram).length;
   289→   289→        return { available: withTelegram, total: targetList.length, label: "Telegram chat ID" };
   290→   290→      }
   291→   291→      return { available: targetList.length, total: targetList.length, label: "" };
   292→   292→    }
   293→   293→    const selected = providers.filter((p) => selectedProviders.has(p.id));
   294→   294→    if (channel === "SMS" || channel === "WHATSAPP") {
   295→   295→      const withPhone = selected.filter((p) => p.hasPhone).length;
   296→   296→      return { available: withPhone, total: selected.length, label: "phone number" };
   297→   297→    }
   298→   298→    if (channel === "TELEGRAM") {
   299→   299→      const withTelegram = selected.filter((p) => p.hasTelegram).length;
   300→   300→      return { available: withTelegram, total: selected.length, label: "Telegram chat ID" };
   301→   301→    }
   302→   302→    return { available: selected.length, total: selected.length, label: "" };
   303→   303→  };
   304→   304→
   305→   305→  const availability = getChannelAvailability();
   306→   306→
   307→   307→  return (
   308→   308→    <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
   309→   309→      {/* Left column — Compose form */}
   310→   310→      <div className="lg:col-span-2 space-y-6">
   311→   311→        <Card>
   312→   312→          <CardHeader>
   313→   313→            <CardTitle className="flex items-center gap-2">
   314→   314→              <Radio className="size-5 text-primary" />
   315→   315→              Compose Notification
   316→   316→            </CardTitle>
   317→   317→            <CardDescription>
   318→   318→              Create a notification to dispatch to guest house service providers.
   319→   319→              {isViewer && (
   320→   320→                <span className="block mt-1 text-amber-600 font-medium">
   321→   321→                  Your rank (Viewer) is read-only. Contact an Officer or Admin to send notifications.
   322→   322→                </span>
   323→   323→              )}
   324→   324→            </CardDescription>
   325→   325→          </CardHeader>
   326→   326→          <CardContent className="space-y-5">
   327→   327→            {/* Title */}
   328→   328→            <div className="space-y-2">
   329→   329→              <Label htmlFor="title" className="text-sm font-medium">
   330→   330→                Title <span className="text-rose-500">*</span>
   331→   331→              </Label>
   332→   332→              <Input
   333→   333→                id="title"
   334→   334→                placeholder="e.g., Safety Inspection Notice, New Regulation Alert"
   335→   335→                value={title}
   336→   336→                onChange={(e) => setTitle(e.target.value)}
   337→   337→                disabled={isViewer}
   338→   338→              />
   339→   339→            </div>
   340→   340→
   341→   341→            {/* Message */}
   342→   342→            <div className="space-y-2">
   343→   343→              <Label htmlFor="message" className="text-sm font-medium">
   344→   344→                Message <span className="text-rose-500">*</span>
   345→   345→              </Label>
   346→   346→              <Textarea
   347→   347→                id="message"
   348→   348→                placeholder="Write the notification message content here...\n\nThis will be delivered to all selected service providers."
   349→   349→                value={message}
   350→   350→                onChange={(e) => setMessage(e.target.value)}
   351→   351→                rows={6}
   352→   352→                disabled={isViewer}
   353→   353→                className="resize-y"
   354→   354→              />
   355→   355→              <p className="text-xs text-slate-400">{message.length} characters</p>
   356→   356→            </div>
   357→   357→
   358→   358→            {/* Channel & Priority row */}
   359→   359→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   360→   360→              <div className="space-y-2">
   361→   361→                <Label>{t('lbldeliveryChannel', 'Delivery Channel')}</Label>
   362→   362→                <Select value={channel} onValueChange={setChannel} disabled={isViewer}>
   363→   363→                  <SelectTrigger>
   364→   364→                    <SelectValue />
   365→   365→                  </SelectTrigger>
   366→   366→                  <SelectContent>
   367→   367→                    {CHANNELS.map((ch) => {
   368→   368→                      const Icon = ch.icon;
   369→   369→                      return (
   370→   370→                        <SelectItem key={ch.value} value={ch.value}>
   371→   371→                          <span className="flex items-center gap-2">
   372→   372→                            <Icon className={`size-4 ${ch.color}`} />
   373→   373→                            {ch.label}
   374→   374→                          </span>
   375→   375→                        </SelectItem>
   376→   376→                      );
   377→   377→                    })}
   378→   378→                  </SelectContent>
   379→   379→                </Select>
   380→   380→                {/* Channel availability warning */}
   381→   381→                {channel !== "IN_APP" && availability.label && (
   382→   382→                  <div className={`text-xs mt-1.5 flex items-center gap-1.5 ${
   383→   383→                    availability.available === 0
   384→   384→                      ? "text-rose-600"
   385→   385→                      : availability.available < availability.total
   386→   386→                      ? "text-amber-600"
   387→   387→                      : "text-emerald-600"
   388→   388→                  }`}>
   389→   389→                    {availability.available === 0 ? (
   390→   390→                      <XCircle className="size-3" />
   391→   391→                    ) : availability.available < availability.total ? (
   392→   392→                      <AlertTriangle className="size-3" />
   393→   393→                    ) : (
   394→   394→                      <CheckCircle2 className="size-3" />
   395→   395→                    )}
   396→   396→                    {availability.available} of {availability.total} providers have a {availability.label} configured
   397→   397→                  </div>
   398→   398→                )}
   399→   399→              </div>
   400→   400→
   401→   401→              <div className="space-y-2">
   402→   402→                <Label>{t('lblpriorityLevel', 'Priority Level')}</Label>
   403→   403→                <Select value={priority} onValueChange={setPriority} disabled={isViewer}>
   404→   404→                  <SelectTrigger>
   405→   405→                    <SelectValue />
   406→   406→                  </SelectTrigger>
   407→   407→                  <SelectContent>
   408→   408→                    {PRIORITIES.map((pr) => (
   409→   409→                      <SelectItem key={pr.value} value={pr.value}>
   410→   410→                        <span className="flex items-center gap-2">
   411→   411→                          <span className={`inline-block h-2 w-2 rounded-full ${
   412→   412→                            pr.value === "URGENT"
   413→   413→                              ? "bg-rose-500"
   414→   414→                              : pr.value === "HIGH"
   415→   415→                              ? "bg-amber-500"
   416→   416→                              : pr.value === "NORMAL"
   417→   417→                              ? "bg-blue-500"
   418→   418→                              : "bg-slate-400"
   419→   419→                          }`} />
   420→   420→                          {pr.label}
   421→   421→                        </span>
   422→   422→                      </SelectItem>
   423→   423→                    ))}
   424→   424→                  </SelectContent>
   425→   425→                </Select>
   426→   426→              </div>
   427→   427→            </div>
   428→   428→
   429→   429→            {/* Target Type */}
   430→   430→            <div className="space-y-2">
   431→   431→              <Label>{t('lbltargetAudience', 'Target Audience')}</Label>
   432→   432→              <div className="flex items-center gap-4">
   433→   433→                <label className="flex items-center gap-2 cursor-pointer text-sm">
   434→   434→                  <input
   435→   435→                    type="radio"
   436→   436→                    name="targetType"
   437→   437→                    value="ALL_PROVIDERS"
   438→   438→                    checked={targetType === "ALL_PROVIDERS"}
   439→   439→                    onChange={() => setTargetType("ALL_PROVIDERS")}
   440→   440→                    disabled={isViewer}
   441→   441→                    className="accent-primary"
   442→   442→                  />
   443→   443→                  All Approved Providers
   444→   444→                </label>
   445→   445→                <label className="flex items-center gap-2 cursor-pointer text-sm">
   446→   446→                  <input
   447→   447→                    type="radio"
   448→   448→                    name="targetType"
   449→   449→                    value="SELECTED_PROVIDERS"
   450→   450→                    checked={targetType === "SELECTED_PROVIDERS"}
   451→   451→                    onChange={() => setTargetType("SELECTED_PROVIDERS")}
   452→   452→                    disabled={isViewer}
   453→   453→                    className="accent-primary"
   454→   454→                  />
   455→   455→                  Selected Providers Only
   456→   456→                </label>
   457→   457→              </div>
   458→   458→            </div>
   459→   459→
   460→   460→            {/* Send button */}
   461→   461→            <div className="flex items-center gap-3 pt-2">
   462→   462→              <Button
   463→   463→                onClick={handleSend}

   466→   466→              >
   467→   467→                {sending ? (
   468→   468→                  <>
   469→   469→                    <span className="mr-2 inline-block h-4 w-4 animate-spin rounded-full border-2 border-white border-t-transparent" />
   470→   470→                    Dispatching...

   475→   475→                    Dispatch to {" "}
