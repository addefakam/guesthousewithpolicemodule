    54→function buildYearOptions() {
    55→  const current = new Date().getFullYear();
    56→  const opts: { value: string; label: string }[] = [];
    57→  for (let y = current + 1; y >= current - 3; y--) {
    58→    opts.push({ value: String(y), label: String(y) });
    59→  }
    60→  return opts;
    61→}
    62→
    63→const YEAR_OPTIONS = buildYearOptions();
    64→
    65→// Severity helpers
    66→function getSeverity(p: { criticalCount: number; highCount: number; matchCount: number }) {
    67→  if (p.criticalCount > 0) return "CRITICAL";
    68→  if (p.highCount > 0) return "HIGH";
    69→  if (p.matchCount > 0) return "MEDIUM";
    70→  return "NONE";
    71→}
    72→
    73→function getRiskScore(p: { matchCount: number; criticalCount: number; highCount: number }) {
    74→  return p.matchCount * 1 + p.criticalCount * 5 + p.highCount * 3;
    75→}
