   783→export default function SuperSystemConfigPage() {
   784→  const [activeTab, setActiveTab] = useState<TabKey>("general");
   785→  const [loading, setLoading] = useState(true);
   786→  const [saving, setSaving] = useState(false);
   787→  const [dirty, setDirty] = useState(false);
   788→
   789→  const [config, setConfig] = useState<SystemConfig>(FULL_DEFAULTS);
   790→
   791→  // ── Fetch settings on mount ──
   792→
   793→  const fetchSettings = useCallback(async () => {
   794→    setLoading(true);
   795→    try {
   796→      const token = localStorage.getItem("ghms_token");
   797→      const headers: HeadersInit = { "Content-Type": "application/json" };
   798→      if (token) {
   799→        headers["Authorization"] = `Bearer ${token}`;
   800→      }
   801→
   802→      const res = await fetch("/api/settings", { headers });
   803→
   804→      if (!res.ok) {
   805→        throw new Error(`Failed to fetch settings (${res.status})`);
   806→      }
   807→
   808→      const data = await res.json();
   809→
   810→      // Merge fetched data with defaults to ensure all keys exist
   811→      setConfig({
   812→        general: { ...DEFAULT_GENERAL, ...data.general },
   813→        guesthouse: { ...DEFAULT_GUESTHOUSE, ...data.guesthouse },
   814→        security: { ...DEFAULT_SECURITY, ...data.security },
   815→        notifications: { ...DEFAULT_NOTIFICATIONS, ...data.notifications },
   816→      });
   817→    } catch (err) {
   818→      // API may not support this shape yet — use defaults
   819→      console.warn("Could not load system config, using defaults:", err);
   820→      setConfig(FULL_DEFAULTS);
   821→    } finally {
   822→      setLoading(false);
   823→    }
   824→  }, []);
   825→
   826→  useEffect(() => {
   827→    fetchSettings();
   828→  }, [fetchSettings]);
   829→
   830→  // ── Save settings ──
   831→
   832→  const handleSave = async () => {
   833→    setSaving(true);
   834→    try {
   835→      const token = localStorage.getItem("ghms_token");
   836→      const headers: HeadersInit = { "Content-Type": "application/json" };
   837→      if (token) {
   838→        headers["Authorization"] = `Bearer ${token}`;
   839→      }
   840→
   841→      const res = await fetch("/api/settings", {
   842→        method: "PUT",
   843→        headers,
   844→        body: JSON.stringify(config),
   845→      });
   846→
   847→      if (!res.ok) {
   848→        // Fallback to POST if PUT not supported
   849→        const resPost = await fetch("/api/settings", {
   850→          method: "POST",
   851→          headers,
   852→          body: JSON.stringify(config),
   853→        });
   854→        if (!resPost.ok) {
   855→          throw new Error(`Failed to save settings (${resPost.status})`);
   856→        }
   857→      }
   858→
   859→      setDirty(false);
   860→      toast.success("System configuration saved successfully");
   861→    } catch (err) {
   862→      console.error("Save failed:", err);
   863→      toast.error("Failed to save configuration. Please try again.");
   864→    } finally {
   865→      setSaving(false);
   866→    }
   867→  };
   868→
   869→  // ── Reset to defaults ──
   870→
   871→  const handleReset = () => {
   872→    setConfig(FULL_DEFAULTS);
   873→    setDirty(true);
   874→    toast.info("All settings reset to defaults. Save to apply.");
   875→  };
   876→
   877→  // ── Updaters ──
   878→
   879→  const updateGeneral = useCallback(
   880→    (partial: Partial<GeneralSettings>) => {
   881→      setConfig((prev) => ({ ...prev, general: { ...prev.general, ...partial } }));
   882→      setDirty(true);
   883→    },
   884→    []
   885→  );
   886→
   887→  const updateGuesthouse = useCallback(
   888→    (partial: Partial<GuesthouseSettings>) => {
   889→      setConfig((prev) => ({
   890→        ...prev,
   891→        guesthouse: { ...prev.guesthouse, ...partial },
   892→      }));
   893→      setDirty(true);
   894→    },
   895→    []
   896→  );
   897→
   898→  const updateSecurity = useCallback(
   899→    (partial: Partial<SecuritySettings>) => {
   900→      setConfig((prev) => ({
   901→        ...prev,
   902→        security: { ...prev.security, ...partial },
   903→      }));
   904→      setDirty(true);
   905→    },
   906→    []
   907→  );
   908→
   909→  const updateNotifications = useCallback(
   910→    (partial: Partial<NotificationSettings>) => {
   911→      setConfig((prev) => ({
   912→        ...prev,
   913→        notifications: { ...prev.notifications, ...partial },
   914→      }));
