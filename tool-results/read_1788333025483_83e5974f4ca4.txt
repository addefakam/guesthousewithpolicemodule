   471→                            <p className="text-[10px] text-muted-foreground">{t("staysAndTotal", { stays: g.totalStays, total: formatCurrency(g.totalSpent) })}</p>
   472→                          </div>
   473→                        </div>
   474→                      </TableCell>
   475→                      <TableCell>
   476→                        <p className="text-xs">{g.phone || "—"}</p>
   477→                        <p className="text-[10px] text-muted-foreground font-mono">{g.idNumber || "—"}</p>
   478→                      </TableCell>
   479→                      <TableCell>
   480→                        {g.activeReservation ? (
   481→                          <Badge variant="outline" className={`text-[10px] ${RES_STATUS[g.activeReservation.status]?.color || ""}`}>
   482→                            {resStatusLabel(g.activeReservation.status)}
   483→                          </Badge>
   484→                        ) : (
   485→                          <span className="text-[10px] text-muted-foreground">—</span>
   486→                        )}
   487→                      </TableCell>
   488→                      <TableCell className="text-sm">
   489→                        {g.activeReservation?.room ? (
   490→                          <span>{g.activeReservation.room.name ? t("roomWithName", { number: g.activeReservation.room.number, name: g.activeReservation.room.name }) : t("roomPrefix", { number: g.activeReservation.room.number })}</span>
   491→                        ) : "—"}
   492→                      </TableCell>
   493→                      <TableCell className="text-xs">
   494→                        {g.activeReservation?.secondGuestName ? (
   495→                          <div>
   496→                            <p className="font-medium">{g.activeReservation.secondGuestName}</p>
   497→                            <p className="text-[10px] text-muted-foreground">{g.activeReservation.secondGuestPhone || ""}</p>
   498→                          </div>
   499→                        ) : g.activeReservation?.exceptionallyReserved ? (
   500→                          <span className="text-[10px] text-amber-600">{t("naException")}</span>
   501→                        ) : (
   502→                          <span className="text-[10px] text-muted-foreground">—</span>

   505→                      <TableCell className="text-xs text-muted-foreground whitespace-nowrap">
   506→                        {g.activeReservation ? `${formatDate(g.activeReservation.checkIn)} → ${formatDate(g.activeReservation.checkOut)}` : "—"}
   507→                      </TableCell>
   508→                      <TableCell className="text-xs">
   509→                        {g.activeReservation ? formatCurrency(g.activeReservation.totalCost) : "—"}
   510→                      </TableCell>
   511→                      <TableCell className="text-right">
   512→                        <div className="flex items-center justify-end gap-1">
   513→                          {g.activeReservation?.status === "UPCOMING" && (
   514→                            <Button size="sm" variant="outline" className="h-7 text-[10px] gap-1 text-emerald-700 border-emerald-300 hover:bg-emerald-50" onClick={() => quickCheckin(g.activeReservation!)}>
   515→                              <LogIn className="h-3 w-3" /> {t("btnCheckIn", "Check In")}
   516→                            </Button>
   517→                          )}
   518→                          {g.activeReservation?.status === "ACTIVE" && (
   519→                            <Button size="sm" variant="outline" className="h-7 text-[10px] gap-1 text-sky-700 border-sky-300 hover:bg-sky-50" onClick={() => quickCheckout(g.activeReservation!)}>
   520→                              <LogOut className="h-3 w-3" /> {t("btnCheckOut", "Check Out")}
   521→                            </Button>
   522→                          )}
   523→                          {!g.activeReservation && (
   524→                            <Button size="sm" variant="outline" className="h-7 text-[10px] gap-1" onClick={() => { setResDialogOpen(true); setResForm({ ...emptyResForm, guestId: g.id }); setResGuestSearch(g.name); }}>
   525→                              <CalendarDays className="h-3 w-3" /> {t("btnReserve")}
   526→                            </Button>
   527→                          )}
   528→                        </div>
   529→                      </TableCell>
   530→                    </TableRow>
   531→                  ))}
   532→                </TableBody>
   533→              </Table>
   534→            </div>
   535→          </>
   536→        )}
   537→      </div>
   538→
   539→      {/* Pagination */}
   540→      {!loading && filtered.length > 0 && (
   541→        <PaginationControls
   542→          currentPage={pagination.currentPage} totalPages={pagination.totalPages}
   543→          pageSize={pagination.pageSize} pageSizeOptions={pagination.pageSizeOptions}
   544→          totalItems={pagination.rangeInfo.total} rangeInfo={pagination.rangeInfo}
   545→          goToPage={pagination.goToPage} setPageSize={pagination.setPageSize}

   547→      )}
   548→
   549→      {/* ── New Reservation Dialog ── */}
   550→      <Dialog open={resDialogOpen} onOpenChange={(open) => { if (!open) { setResDialogOpen(false); setResForm(emptyResForm); setResGuestSearch(""); } }}>
   551→        <DialogContent className="max-w-lg mx-4 w-[calc(100%-2rem)] max-h-[90vh] overflow-y-auto">
   552→          <DialogHeader>
   553→            <DialogTitle className="flex items-center gap-2"><CalendarDays className="h-5 w-5" /> {t("dlgNewReservationTitle")}</DialogTitle>
   554→            <DialogDescription>{t("dlgNewReservationDesc")}</DialogDescription>
   555→          </DialogHeader>
   556→          <div className="space-y-3">
   557→            {/* Guest Select */}
   558→            <div>
   559→              <Label>{t('lblguest', 'Guest')} *</Label>
   560→              {resForm.guestId ? (
   561→                <div className="flex items-center gap-2 mt-1 p-2 rounded-md border bg-muted/30">
   562→                  <span className="text-sm font-medium flex-1">{guests.find((g) => g.id === resForm.guestId)?.name || t("selected")}</span>

   564→                </div>
   565→              ) : (
   566→                <div className="relative mt-1">
   567→                  <Search className="absolute left-2.5 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-muted-foreground" />
   568→                  <Input value={resGuestSearch} onChange={(e) => setResGuestSearch(e.target.value)} placeholder={t("phSearchGuest")} className="h-9 pl-8 text-sm" />
   569→                </div>
   570→              )}
   571→              {!resForm.guestId && resGuestResults.length > 0 && (
   572→                <div className="mt-1 max-h-32 overflow-y-auto rounded-md border">
   573→                  {resGuestResults.map((g) => (
   574→                    <button key={g.id} className="w-full text-left px-3 py-2 text-xs hover:bg-muted/50 border-b last:border-b-0 flex justify-between items-center" onClick={() => { setResForm({ ...resForm, guestId: g.id }); setResGuestSearch(g.name); }}>
   575→                      <span className="font-medium">{g.name}</span>
   576→                      <span className="text-muted-foreground">{g.phone}{g.idNumber ? ` | ${g.idNumber}` : ""}</span>
   577→                    </button>
   578→                  ))}
   579→                </div>
   580→              )}
   581→            </div>
   582→
   583→            {/* Room Select */}
   584→            <div>
   585→              <Label>{t('lblroom', 'Room')} *</Label>
   586→              <Select value={resForm.roomId} onValueChange={(v) => setResForm({ ...resForm, roomId: v })}>
   587→                <SelectTrigger className="h-9 text-xs"><SelectValue placeholder={t("phSelectRoom")} /></SelectTrigger>
   588→                <SelectContent>
   589→                  {availableRooms.map((r) => (
   590→                    <SelectItem key={r.id} value={r.id}>{r.name ? t("roomWithName", { number: r.number, name: r.name }) : t("roomPrefix", { number: r.number })} — {r.type} — {formatCurrency(r.pricePerNight)}/night</SelectItem>
   591→                  ))}
   592→                </SelectContent>
   593→              </Select>
   594→            </div>
   595→
   596→            {/* ── Double/TWIN Room: Second Guest + Exception ── */}
   597→            {selectedRoomIsDouble && (
   598→              <div className="rounded-lg border border-amber-200 bg-amber-50/50 p-3 space-y-3">
   599→                <div className="flex items-center gap-2 text-amber-800">
   600→                  <BedDouble className="h-4 w-4" />
   601→                  <span className="text-xs font-semibold">{t("doubleRoomSecondGuestRequired")}</span>
   602→                </div>
   603→
   604→                {/* Exception toggle */}
   605→                <div className="flex items-center gap-3">
   606→                  <label className="flex items-center gap-2 cursor-pointer">
   607→                    <input
   608→                      type="radio"
   609→                      name="exceptionallyReserved"
   610→                      checked={!resForm.exceptionallyReserved}
   611→                      onChange={() => setResForm({ ...resForm, exceptionallyReserved: false, exceptionReason: "" })}
   612→                      className="h-3.5 w-3.5 text-emerald-600 accent-emerald-600"
   613→                    />
   614→                    <span className="text-xs font-medium">{t("twoGuests")}</span>
   615→                  </label>
   616→                  <label className="flex items-center gap-2 cursor-pointer">
   617→                    <input
   618→                      type="radio"
   619→                      name="exceptionallyReserved"
   620→                      checked={resForm.exceptionallyReserved}
   621→                      onChange={() => setResForm({ ...resForm, exceptionallyReserved: true, secondGuestName: "", secondGuestPhone: "", secondGuestIdNumber: "" })}
   622→                      className="h-3.5 w-3.5 text-amber-600 accent-amber-600"
   623→                    />
   624→                    <span className="text-xs font-medium text-amber-700">{t("exceptionallyReserved")}</span>
   625→                  </label>
   626→                </div>
   627→
   628→                {!resForm.exceptionallyReserved ? (
   629→                  /* Second guest fields */
   630→                  <div className="space-y-2">
   631→                    <p className="text-[10px] text-muted-foreground">{t("secondGuestDetailsHint")}</p>
   632→                    <div>
   633→                      <Label>{t('lblsecondGuestName', 'Second Guest Name')} *</Label>
   634→                      <Input
   635→                        value={resForm.secondGuestName}
   636→                        onChange={(e) => setResForm({ ...resForm, secondGuestName: e.target.value })}
   637→                        placeholder={t("phSecondGuestName")}
   638→                        className="h-9 text-sm"
   639→                      />
   640→                    </div>
   641→                    <div>
   642→                      <Label>{t('lblsecondGuestPhone', 'Second Guest Phone')} *</Label>
   643→                      <Input
   644→                        type="tel"
   645→                        value={resForm.secondGuestPhone}
   646→                        onChange={(e) => setResForm({ ...resForm, secondGuestPhone: e.target.value })}
   647→                        placeholder={t("phSecondGuestPhone")}
   648→                        className="h-9 text-sm"
   649→                      />
   650→                    </div>
   651→                    <div>
   652→                      <Label>{t('lblsecondGuestIdNumber', 'Second Guest ID Number')}</Label>
   653→                      <Input
   654→                        value={resForm.secondGuestIdNumber}
   655→                        onChange={(e) => setResForm({ ...resForm, secondGuestIdNumber: e.target.value })}
   656→                        placeholder={t("phSecondGuestId")}
   657→                        className="h-9 text-sm"
   658→                      />
   659→                    </div>
   660→                  </div>
   661→                ) : (
   662→                  /* Exception reason field */
   663→                  <div className="space-y-2">
   664→                    <div className="flex items-center gap-1.5 text-amber-700">
   665→                      <AlertTriangle className="h-3.5 w-3.5" />
   666→                      <p className="text-[10px] font-medium">{t("singleOccupancyException")}</p>
   667→                    </div>
   668→                    <div>
   669→                      <Label>{t('lblexceptionReason', 'Exception Reason')} *</Label>
   670→                      <Textarea
   671→                        value={resForm.exceptionReason}
   672→                        onChange={(e) => setResForm({ ...resForm, exceptionReason: e.target.value })}
   673→                        placeholder={t("phExceptionReason")}
   674→                        className="min-h-[60px] text-sm"
   675→                      />
   676→                    </div>

   679→              </div>

   681→
   682→            <div className="grid grid-cols-2 gap-3">
   683→              <div><Label>{t('lblcheckin', 'Check-in')} *</Label><Input type="date" value={resForm.checkIn} min={todayStr()} onChange={(e) => setResForm({ ...resForm, checkIn: e.target.value, checkOut: resForm.checkOut || addDays(e.target.value, 1) })} className="h-9 text-sm" /></div>

   685→            </div>
   686→            {resNights > 0 && resRate > 0 && (
   687→              <div className="rounded-md bg-muted/50 p-2 flex justify-between text-xs">
   688→                <span>{t("nightsTimes", { nights: resNights, rate: formatCurrency(resRate) })}</span>

   690→              </div>
   691→            )}
   692→            <div><Label>{t('lblnotes', 'Notes')}</Label><Input value={resForm.notes} onChange={(e) => setResForm({ ...resForm, notes: e.target.value })} placeholder={t("phNotes")} className="h-9 text-sm" /></div>
   693→          </div>
   694→          <DialogFooter>
   695→            <Button variant="outline" size="sm" onClick={() => { setResDialogOpen(false); setResForm(emptyResForm); setResGuestSearch(""); }}>{t("cancel")}</Button>
   696→            <Button size="sm" onClick={handleCreateRes} disabled={creatingRes || !resForm.guestId || !resForm.roomId}>{creatingRes ? t("btnCreating") : t("btnCreateReservation")}</Button>

   700→
   701→      {/* ── Check-in / Check-out Confirm ── */}

   709→            <AlertDialogFooter>

   712→            </AlertDialogFooter>
   713→          </AlertDialogContent>
   714→        </AlertDialog>

   716→    </div>
