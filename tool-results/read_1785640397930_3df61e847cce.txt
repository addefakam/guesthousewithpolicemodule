   404→
   405→function SuperuserSettings() {
   406→  const { currentUser, setCurrentUser, triggerRefresh } = useAppStore();
   407→  const [loading, setLoading] = useState(false);
   408→
   409→  // Profile form
   410→  const [name, setName] = useState(currentUser?.name || "");
   411→  const [username, setUsername] = useState(currentUser?.username || "");
   412→  const [email, setEmail] = useState("");
   413→  const [phone, setPhone] = useState("");
   414→  const [saving, setSaving] = useState(false);
   415→
   416→  // Password form
   417→  const [currentPassword, setCurrentPassword] = useState("");
   418→  const [newPassword, setNewPassword] = useState("");
   419→  const [confirmPassword, setConfirmPassword] = useState("");
   420→  const [showCurrent, setShowCurrent] = useState(false);
   421→  const [showNew, setShowNew] = useState(false);
   422→  const [changingPassword, setChangingPassword] = useState(false);
   423→
   424→  // System settings form
   425→  const [appName, setAppName] = useState("GHMS");
   426→  const [defaultCurrency, setDefaultCurrency] = useState("ETB");
   427→  const [defaultLanguage, setDefaultLanguage] = useState("en");
   428→  const [savingSystem, setSavingSystem] = useState(false);
   429→
   430→  useEffect(() => {
   431→    if (currentUser) {
   432→      setName(currentUser.name);
   433→      setUsername(currentUser.username);
   434→    }
   435→  }, [currentUser]);
   436→
   437→  const handleProfileSave = async () => {
   438→    if (!currentUser) return;
   439→    if (!name.trim()) {
   440→      toast.error("Name is required");
   441→      return;
   442→    }
   443→    if (!username.trim()) {
   444→      toast.error("Username is required");
   445→      return;
   446→    }
   447→    if (username.length < 3) {
   448→      toast.error("Username must be at least 3 characters");
   449→      return;
   450→    }
   451→
   452→    setSaving(true);
   453→    try {
   454→      const data: Record<string, unknown> = { name: name.trim(), username: username.trim() };
   455→      if (email.trim()) data.email = email.trim();
   456→      if (phone.trim()) data.phone = phone.trim();
   457→
   458→      await apiUpdateUser(currentUser.id, data);
   459→
   460→      // Update local session
   461→      setCurrentUser({
   462→        ...currentUser,
   463→        name: name.trim(),
   464→        username: username.trim(),
   465→      });
   466→
   467→      toast.success("Profile updated successfully");
   468→      triggerRefresh();
   469→    } catch (err: unknown) {
   470→      const msg = err instanceof Error ? err.message : "Failed to update profile";
   471→      toast.error(msg);
   472→    } finally {
   473→      setSaving(false);
   474→    }
   475→  };
   476→
   477→  const handlePasswordChange = async () => {
   478→    if (!currentUser) return;
   479→    if (!currentPassword) {
   480→      toast.error("Current password is required");
   481→      return;
   482→    }
   483→    if (!newPassword) {
   484→      toast.error("New password is required");
   485→      return;
   486→    }
   487→    if (newPassword.length < 6) {
   488→      toast.error("New password must be at least 6 characters");
   489→      return;
   490→    }
   491→    if (newPassword !== confirmPassword) {
   492→      toast.error("New passwords do not match");
   493→      return;
   494→    }
   495→    if (currentPassword === newPassword) {
   496→      toast.error("New password must be different from current password");
   497→      return;
   498→    }
   499→
   500→    setChangingPassword(true);
   501→    try {
   502→      await apiUpdateUser(currentUser.id, {
   503→        password: newPassword,
   504→      });
   505→      toast.success("Password changed successfully");
   506→      setCurrentPassword("");
   507→      setNewPassword("");
   508→      setConfirmPassword("");
   509→      setShowCurrent(false);
   510→      setShowNew(false);
   511→    } catch (err: unknown) {
   512→      const msg = err instanceof Error ? err.message : "Failed to change password";
   513→      toast.error(msg);
   514→    } finally {
   515→      setChangingPassword(false);
   516→    }
   517→  };
   518→
   519→  const handleSystemSave = async () => {
   520→    setSavingSystem(true);
   521→    try {
   522→      await apiUpdateSettings({
   523→        guestHouseName: appName,
   524→        currency: defaultCurrency,
   525→        language: defaultLanguage,
   526→      });
   527→      toast.success("System settings saved");
   528→      triggerRefresh();
   529→    } catch (err: unknown) {
   530→      const msg = err instanceof Error ? err.message : "Failed to save system settings";
   531→      toast.error(msg);
   532→    } finally {
   533→      setSavingSystem(false);
   534→    }
   535→  };
   536→
   537→  if (loading) {
   538→    return (
   539→      <div className="space-y-6 p-4 md:p-6">
   540→        <Skeleton className="h-8 w-40" />
   541→        <Skeleton className="h-96 rounded-xl" />
   542→      </div>
   543→    );
   544→  }
   545→
   546→  const getInitials = (n: string) =>
   547→    n.split(" ").map((w) => w[0]).join("").toUpperCase().slice(0, 2);
   548→
   549→  return (
   550→    <div className="flex justify-center p-4 md:p-6">
   551→      <div className="w-full max-w-2xl space-y-6">
   552→      <div>
   553→        <h1 className="text-2xl font-bold tracking-tight">My Profile</h1>
   554→        <p className="text-sm text-muted-foreground">
   555→          Manage your System Admin account and preferences.
   556→        </p>
   557→      </div>
   558→
   559→      {/* ── Profile Card ── */}
   560→      <Card>
   561→        <CardHeader>
   562→          <CardTitle className="flex items-center gap-2">
   563→            <User className="h-5 w-5" />
   564→            Profile Information
   565→          </CardTitle>
   566→          <CardDescription>
   567→            Update your personal details and contact information.
   568→          </CardDescription>
   569→        </CardHeader>
   570→        <CardContent className="space-y-5">
   571→          {/* Avatar + Role Badge */}
   572→          <div className="flex items-center gap-4">
   573→            <Avatar className="h-16 w-16 ring-2 ring-primary/20">
   574→              <AvatarFallback className="bg-primary/10 text-lg font-bold text-primary">
   575→                {getInitials(name || "SA")}
   576→              </AvatarFallback>
   577→            </Avatar>
   578→            <div>
   579→              <p className="font-semibold text-slate-900">{name || "System Admin"}</p>
   580→              <div className="mt-1 flex items-center gap-2">
   581→                <Badge
   582→                  variant="outline"
   583→                  className="bg-amber-100 text-amber-700 border-amber-200 text-xs font-semibold"
   584→                >
   585→                  <Shield className="mr-1 size-3" />
   586→                  Superuser
   587→                </Badge>
   588→                <Badge variant="outline" className="text-xs">
   589→                  <BadgeCheck className="mr-1 size-3 text-emerald-500" />
   590→                  System Administrator
   591→                </Badge>
   592→              </div>
   593→              <p className="mt-1 text-xs text-slate-400">
   594→                @username: {currentUser?.username}
   595→              </p>
   596→            </div>
   597→          </div>
   598→
   599→          <Separator />
   600→
   601→          {/* Name & Username */}
   602→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   603→            <div className="grid gap-2">
   604→              <Label htmlFor="su-name">

   835→          <div className="flex justify-end pt-1">

   840→                <Save className="mr-2 h-4 w-4" />
   841→              )}
   842→              Save System Settings
   843→            </Button>
   844→          </div>
   845→        </CardContent>
   846→      </Card>
   847→      </div>
   848→    </div>
   849→  );
