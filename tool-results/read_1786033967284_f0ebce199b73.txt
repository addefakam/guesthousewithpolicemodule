   739→/** Execute SQL via raw pg Client */
   740→async function execViaPg(sql: string, params?: unknown[]): Promise<void> {
   741→  const client = new Client({ connectionString: process.env.DATABASE_URL! });
   742→  try {
   743→    await client.connect();
   744→    await client.query(sql, params);
   745→  } finally {
   746→    await client.end().catch(() => {});
   747→  }
   748→}

   750→/** Execute DDL via Prisma $executeRawUnsafe — splits multi-statement SQL
   751→ *  because Prisma's driver does NOT support multiple statements per call. */
   752→async function execViaPrisma(sql: string): Promise<void> {
   753→  // Split on `DO $$` blocks and standalone statements
   754→  // Each `DO $$ ... END $$;` is one executable unit
   755→  const statements = sql
   756→    .split(/(?=DO \$\$)/)
   757→    .map(s => s.trim())
   758→    .filter(s => s.length > 0);
   759→
   760→  // If no DO blocks found, split by semicolons (for CREATE INDEX etc.)
   761→  const parts = statements.length > 1
   762→    ? statements
   763→    : sql.split(';').map(s => s.trim()).filter(s => s.length > 0);
   764→
   765→  const prisma = new PrismaClient({
   766→    log: ["warn", "error"],
   767→  });
   768→  try {
   769→    for (const part of parts) {
   770→      const stmt = part.endsWith(';') ? part : part + ';';
   771→      try {
   772→        await prisma.$executeRawUnsafe(stmt);
   773→      } catch (err) {
   774→        // Individual statement might fail (e.g. duplicate_object) — log but continue
   775→        console.error("[execViaPrisma] Statement failed (continuing):", stmt.slice(0, 80), err instanceof Error ? err.message : String(err));
   776→      }
   777→    }
   778→  } finally {
   779→    await prisma.$disconnect().catch(() => {});
   780→  }
   781→}

   802→export async function ensureDatabase(): Promise<void> {
   803→  if (_initDone) {
   804→    // Always run migrations synchronously to ensure columns exist
   805→    // before any query executes. Migrations are idempotent.
   806→    if (!_migrationsRan) {
   807→      _migrationsRan = true;
   808→      await runMigrationsOnly();
   809→    }
   810→    return;
   811→  }
   812→  if (_initPromise) return _initPromise;
   813→
   814→  if (!process.env.DATABASE_URL) {
   815→    throw new Error("DATABASE_URL environment variable is not set.");
   816→  }
   817→
   818→  _initPromise = (async () => {
   819→    console.log("[init-db] Starting database initialization...");
   820→
   821→    let usePg = true;
   822→    let pgError: unknown = null;
   823→
   824→    // Strategy 1: Try using pg Client directly
   825→    try {
   826→      const client = new Client({ connectionString: process.env.DATABASE_URL });
   827→      await client.connect();
   828→      console.log("[init-db] pg Client connected successfully.");
   829→
   830→      // Check if User table already exists
   831→      const res = await client.query(
   832→        `SELECT EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema='public' AND table_name='User') as ok`
   833→      );
   834→      const tableExists = res.rows[0]?.ok === true;
   835→      console.log("[init-db] User table exists:", tableExists);
   836→
   837→      if (tableExists) {
   838→        console.log("[init-db] Tables already exist, running migrations to add any missing columns...");
   839→        try {
   840→          await client.query(MIGRATIONS_SQL);
