   560→   560→/** Execute SQL via raw pg Client */
   561→   561→async function execViaPg(sql: string, params?: unknown[]): Promise<void> {
   562→   562→  const client = new Client({ connectionString: process.env.DATABASE_URL! });
   563→   563→  try {
   564→   564→    await client.connect();
   565→   565→    await client.query(sql, params);
   566→   566→  } finally {
   567→   567→    await client.end().catch(() => {});
   568→   568→  }
   569→   569→}

   571→   571→/** Execute DDL via Prisma $executeRawUnsafe — splits multi-statement SQL
   572→   572→ *  because Prisma's driver does NOT support multiple statements per call. */
   573→   573→async function execViaPrisma(sql: string): Promise<void> {
   574→   574→  // Split on `DO $$` blocks and standalone statements
   575→   575→  // Each `DO $$ ... END $$;` is one executable unit
   576→   576→  const statements = sql
   577→   577→    .split(/(?=DO \$\$)/)
   578→   578→    .map(s => s.trim())
   579→   579→    .filter(s => s.length > 0);
   580→   580→
   581→   581→  // If no DO blocks found, split by semicolons (for CREATE INDEX etc.)
   582→   582→  const parts = statements.length > 1
   583→   583→    ? statements
   584→   584→    : sql.split(';').map(s => s.trim()).filter(s => s.length > 0);
   585→   585→
   586→   586→  const prisma = new PrismaClient({
   587→   587→    log: ["warn", "error"],
   588→   588→  });
   589→   589→  try {
   590→   590→    for (const part of parts) {
   591→   591→      const stmt = part.endsWith(';') ? part : part + ';';
   592→   592→      try {
   593→   593→        await prisma.$executeRawUnsafe(stmt);
   594→   594→      } catch (err) {
   595→   595→        // Individual statement might fail (e.g. duplicate_object) — log but continue
   596→   596→        console.error("[execViaPrisma] Statement failed (continuing):", stmt.slice(0, 80), err instanceof Error ? err.message : String(err));
   597→   597→      }
   598→   598→    }
   599→   599→  } finally {
   600→   600→    await prisma.$disconnect().catch(() => {});
   601→   601→  }
   602→   602→}
