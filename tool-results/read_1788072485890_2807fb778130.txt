   385→  const handleBulkFileSelect = async (e: React.ChangeEvent<HTMLInputElement>) => {
   386→    const file = e.target.files?.[0];
   387→    if (!file) return;
   388→    setBulkFile(file);
   389→    setBulkErrors([]);
   390→    setBulkPreview([]);
   391→    try {
   392→      const XLSX = await import("xlsx");
   393→      const buffer = await file.arrayBuffer();
   394→      const wb = XLSX.read(buffer, { type: "array" });
   395→      const ws = wb.Sheets[wb.SheetNames[0]];
   396→      const rows = XLSX.utils.sheet_to_json<Record<string, string>>(ws, { defval: "" });
   397→
   398→      if (rows.length === 0) {
   399→        setBulkErrors([t('bulkNoDataRows')]);
   400→        return;
   401→      }
   402→
   403→      // Validate columns
   404→      const requiredCols = ["Full Name", "Phone", "Email", "Guesthouse Name", "Type", "License No", "Sub-City", "Woreda", "Username", "Password"];
   405→      const firstRowKeys = Object.keys(rows[0]);
   406→      const missingCols = requiredCols.filter((c) => !firstRowKeys.includes(c));
   407→      if (missingCols.length > 0) {
   408→        setBulkErrors([t('bulkMissingCols', { cols: missingCols.join(', ') })]);
   409→        return;
   410→      }
   411→
   412→      // Validate rows
   413→      const errors: string[] = [];
   414→      const validSubCities = Object.keys(SUB_CITY_WOREDAS);
   415→      const validTypes = GUESTHOUSE_TYPES.map((t) => t.value);
   416→      const validTypeLabels = GUESTHOUSE_TYPES.map((t) => t.label);
   417→
   418→      rows.forEach((row, idx) => {
   419→        const rowNum = idx + 2; // Excel row number (1-based + header)
   420→        const fullName = (row["Full Name"] || "").trim();
   421→        const phone = (row["Phone"] || "").trim();
   422→        const email = (row["Email"] || "").trim();
   423→        const ghName = (row["Guesthouse Name"] || "").trim();
   424→        const type = (row["Type"] || "").trim();
   425→        const licenseNo = (row["License No"] || "").trim();
   426→        const subCity = (row["Sub-City"] || "").trim();
   427→        const woreda = (row["Woreda"] || "").trim();
   428→        const username = (row["Username"] || "").trim();
   429→        const password = (row["Password"] || "").trim();
   430→
   431→        if (!fullName) errors.push(t('bulkRowFieldEmpty', { row: rowNum, field: "Full Name" }));
   432→        if (!phone) errors.push(t('bulkRowFieldEmpty', { row: rowNum, field: "Phone" }));
   433→        else if (!isValidPhone(phone)) errors.push(t('bulkRowInvalidPhone', { row: rowNum, value: phone }));
   434→        if (!email) errors.push(t('bulkRowFieldEmpty', { row: rowNum, field: "Email" }));
   435→        else if (!isValidEmail(email)) errors.push(t('bulkRowInvalidEmail', { row: rowNum, value: email }));
   436→        if (!ghName) errors.push(t('bulkRowFieldEmpty', { row: rowNum, field: "Guesthouse Name" }));
   437→        if (!type) errors.push(t('bulkRowFieldEmpty', { row: rowNum, field: "Type" }));
   438→        else if (!validTypes.includes(type) && !validTypeLabels.includes(type)) errors.push(t('bulkRowInvalidType', { row: rowNum, value: type, valid: validTypes.join(", ") }));
   439→        if (!licenseNo) errors.push(t('bulkRowFieldEmpty', { row: rowNum, field: "License No" }));
   440→        if (!subCity) errors.push(t('bulkRowFieldEmpty', { row: rowNum, field: "Sub-City" }));
   441→        else if (!validSubCities.includes(subCity)) errors.push(t('bulkRowInvalidSubCity', { row: rowNum, value: subCity, valid: validSubCities.join(", ") }));
   442→        if (!woreda) errors.push(t('bulkRowFieldEmpty', { row: rowNum, field: "Woreda" }));
   443→        else if (subCity && validSubCities.includes(subCity) && !SUB_CITY_WOREDAS[subCity]?.includes(woreda)) errors.push(t('bulkRowInvalidWoreda', { row: rowNum, woreda, subCity }));
   444→        if (!username) errors.push(t('bulkRowFieldEmpty', { row: rowNum, field: "Username" }));
   445→        if (!password) errors.push(t('bulkRowFieldEmpty', { row: rowNum, field: "Password" }));
   446→        else if (password.length < 4) errors.push(t('bulkRowPasswordMinLength', { row: rowNum }));
   447→      });
   448→
   449→      if (errors.length > 0) {
   450→        setBulkErrors(errors.slice(0, 20));
   451→        if (errors.length > 20) setBulkErrors((prev) => [...prev, t('bulkMoreErrors', { count: errors.length - 20 })]);
   452→      }
   453→
   454→      setBulkPreview(rows);
   455→    } catch {
   456→      setBulkErrors([t('bulkInvalidFile')]);
   457→    }
   458→  };

   460→  const handleBulkImport = async () => {
   461→    if (bulkPreview.length === 0) {
   462→      toast.error(t('bulkNoDataToImport'));
   463→      return;
   464→    }
   465→    if (bulkErrors.length > 0) {
   466→      toast.error(t('bulkFixErrorsBeforeImport'));
   467→      return;
   468→    }
   469→    try {
   470→      setBulkImporting(true);
   471→      const records = bulkPreview.map((row) => {
   472→        const subCity = (row["Sub-City"] || "").trim();
   473→        const woreda = (row["Woreda"] || "").trim();
   474→        let type = (row["Type"] || "").trim();
   475→        // Convert label to value if needed
   476→        const typeMatch = GUESTHOUSE_TYPES.find((t) => t.label === type);
   477→        if (typeMatch) type = typeMatch.value;
   478→        const address = ["Bishoftu", subCity, woreda].filter(Boolean).join(", ");
   479→        return {
   480→          ownerName: (row["Full Name"] || "").trim(),
   481→          phone: (row["Phone"] || "").trim(),
   482→          email: (row["Email"] || "").trim(),
   483→          name: (row["Guesthouse Name"] || "").trim(),
   484→          type,
   485→          licenseNo: (row["License No"] || "").trim(),
   486→          address,
   487→          username: (row["Username"] || "").trim(),
   488→          password: (row["Password"] || "").trim(),
   489→        };
   490→      });
   491→      const result = await apiSuperBulkImportProviders(records);
   492→      const data = result as { success: number; failed: number; errors: string[] };
   493→      toast.success(t('bulkImportSuccess', { success: data.success, failed: data.failed, failedPart: data.failed > 0 ? t('bulkImportFailed', { count: data.failed }) : '' }));
   494→      setBulkImportOpen(false);
   495→      setBulkFile(null);
   496→      setBulkPreview([]);
   497→      setBulkErrors([]);
   498→      if (bulkFileInputRef.current) bulkFileInputRef.current.value = "";
   499→      triggerRefresh();
   500→    } catch (err: unknown) {
   501→      toast.error(err instanceof Error ? err.message : t('bulkImportFailed'));
   502→    } finally {
   503→      setBulkImporting(false);
   504→    }
   505→  };

   519→  return (
