   217→// ── Helper: get nav items based on role ──
   218→function getNavItems(user: CurrentUser, disabledPages: string[] = []): NavItem[] {
   219→  const disabledSet = new Set(disabledPages);
   220→
   221→  switch (user.role) {
   222→    case "POLICE": {
   223→      const rank = (user.policeRank || "OFFICER") as PoliceRank;
   224→      const allowedPages = POLICE_RANK_PERMISSIONS[rank] || POLICE_RANK_PERMISSIONS.OFFICER;
   225→      return POLICE_NAV_ITEMS.filter((item) => allowedPages.includes(item.page));
   226→    }
   227→
   228→    case "SUPERUSER": {
   229→      // Guest house owners (SUPERUSER with providerId) see the provider dashboard.
   230→      // System admins (SUPERUSER without providerId) see the admin dashboard.
   231→      if (user.providerId) {
   232→        return ALL_NAV_ITEMS.filter(
   233→          (item) => !OPERATOR_EXCLUDED.has(item.page) && !disabledSet.has(item.page)
   234→        );
   235→      }
   236→      return SUPERUSER_NAV_ITEMS;
   237→    }
   238→
   239→    case "OPERATOR":
   240→      return ALL_NAV_ITEMS.filter(
   241→        (item) => !OPERATOR_EXCLUDED.has(item.page) && !disabledSet.has(item.page)
   242→      );
   243→
   244→    case "STAFF": {
   245→      const items: NavItem[] = [];
   246→      // Always give staff access to dashboard
   247→      if (!disabledSet.has("dashboard")) {
   248→        items.push({
   249→          page: "dashboard",
   250→          label: "Dashboard",
   251→          icon: LayoutDashboard,
   252→        });
   253→      }
   254→      // Add items based on permissions (deduplicate by page key)
   255→      const seen = new Set<string>();
   256→      for (const perm of user.permissions) {
   257→        const mapped = PERMISSION_PAGE_MAP[perm];
   258→        if (mapped && !seen.has(mapped.page) && !disabledSet.has(mapped.page)) {
   259→          seen.add(mapped.page);
   260→          items.push(mapped);
   261→        }
   262→      }
   263→      return items;
   264→    }
   265→
   266→    default:
   267→      return [
   268→        { page: "dashboard", label: "Dashboard", icon: LayoutDashboard },
   269→      ];
   270→  }
   271→}

   273→// ── Helper: get initials from name ──
   274→function getInitials(name: string | undefined | null): string {
   275→  if (!name) return "?";
   276→  return name
   277→    .split(" ")
   278→    .map((w) => w[0])
   279→    .join("")
   280→    .toUpperCase()
   281→    .slice(0, 2);
   282→}

   284→// ── Helper: get role badge className (no label — use t() for that) ──
   285→function getRoleBadgeClass(role: string): string {
   286→  switch (role) {
   287→    case "SUPERUSER": return "bg-amber-100 text-amber-700 border-amber-200";
   288→    case "OPERATOR": return "bg-emerald-100 text-emerald-700 border-emerald-200";
   289→    case "STAFF": return "bg-sky-100 text-sky-700 border-sky-200";
   290→    case "POLICE": return "bg-rose-100 text-rose-700 border-rose-200";
   291→    default: return "bg-slate-100 text-slate-700 border-slate-200";
   292→  }

   341→function SubscriptionStatusCard({ collapsed }: { collapsed: boolean }) {
   342→  const { t } = useTranslation("sidebar");
   343→  const subscription = useAppStore((s) => s.subscription);
   344→
   345→  if (!subscription) return null;
   346→
   347→  const isActive = subscription.status === "ACTIVE";
   348→  const isWarning = subscription.status === "WARNING";
   349→  const isExpired = subscription.status === "EXPIRED";
   350→  const isSuspended = subscription.status === "SUSPENDED";
   351→
   352→  // Collapsed mode — icon indicator
   353→  if (collapsed) {
   354→    const bgClass = isActive
   355→      ? "bg-emerald-100"
   356→      : isSuspended || isExpired
   357→      ? "bg-rose-100"
   358→      : "bg-amber-100";
   359→    const iconClass = isActive
   360→      ? "text-emerald-600"
   361→      : isSuspended || isExpired
   362→      ? "text-rose-600"
   363→      : "text-amber-600";
   364→    return (
   365→      <div className="flex flex-col items-center gap-1 px-2 py-2">
   366→        <div className={`flex h-8 w-8 items-center justify-center rounded-full ${bgClass} animate-pulse shadow-sm`}> 
   367→          <Clock className={`size-4 ${iconClass}`} />
   368→        </div>
   369→      </div>
   370→    );
   371→  }
   372→
   373→  // Expanded card
   374→  const containerClass = isActive
   375→    ? "border-emerald-200 bg-emerald-50"
   376→    : isSuspended || isExpired
   377→    ? "border-rose-200 bg-rose-50"
   378→    : "border-amber-200 bg-amber-50";
   379→
   380→  const iconColorClass = isActive
   381→    ? "text-emerald-600"
   382→    : isSuspended || isExpired
   383→    ? "text-rose-600"
   384→    : "text-amber-600";
   385→
   386→  const titleClass = isActive
   387→    ? "text-emerald-800"
   388→    : isSuspended || isExpired
   389→    ? "text-rose-800"
   390→    : "text-amber-800";
   391→
   392→  const subtextClass = isActive
   393→    ? "text-emerald-700"
   394→    : isSuspended || isExpired
   395→    ? "text-rose-700"
   396→    : "text-amber-700";
   397→
   398→  const badgeClass = isActive
   399→    ? "bg-emerald-200/60 text-emerald-800"
   400→    : isSuspended || isExpired
   401→    ? "bg-rose-200/60 text-rose-800"
   402→    : "bg-amber-200/60 text-amber-800";
   403→
   404→  const Icon = isSuspended || isExpired ? AlertTriangle : Clock;
   405→
   406→  return (
   407→    <div className={`rounded-lg border p-3 ${containerClass} animate-subtle-pulse shadow-sm`}> 
   408→      <div className="flex items-center gap-2">
   409→        <Icon className={`h-4 w-4 shrink-0 ${iconColorClass}`} />
   410→        <p className={`text-xs font-semibold ${titleClass}`}>
   411→          {isActive
   412→            ? t("Subscription Active")
   413→            : isSuspended
   414→            ? t("Service Suspended")
   415→            : isExpired
   416→            ? t("Subscription Expired")
   417→            : t("Expiring Soon")}
   418→        </p>
   419→      </div>
   420→      <div className={`mt-1.5 flex items-center justify-between text-[11px] ${subtextClass}`}>
   421→        <span>{formatDaysRemaining(subscription.daysRemaining)}</span>
   422→        <span className={`inline-flex items-center gap-1 rounded-full px-1.5 py-0.5 font-semibold ${badgeClass}`}>
   423→          <Clock className="h-3 w-3" />
   424→          {Math.abs(subscription.daysRemaining)}d
   425→        </span>
   426→      </div>
   427→      {subscription.price > 0 ? (
   428→        <div className={`mt-1 text-[11px] ${subtextClass}`}>
   429→          {t("Due: ")}{subscription.price.toLocaleString()} ETB / {formatCycle(subscription.cycle)}
   430→        </div>
   431→      ) : (
   432→        <div className={`mt-1 text-[11px] ${subtextClass}`}>
   433→          {formatCycle(subscription.cycle)} {t("cycle — Free trial")}
   434→        </div>
   435→      )}
   436→    </div>
   437→  );
   438→}

   674→        {/* Desktop collapse toggle */}

   682→          </button>

   684→      </div>
   685→    </div>
   686→  );
