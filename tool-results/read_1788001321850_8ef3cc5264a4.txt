   183→   183→function ComposeTab({
   184→   184→  isViewer,
   185→   185→  isPolice,
   186→   186→  isAdmin,
   187→   187→}: {
   188→   188→  isViewer: boolean;
   189→   189→  isPolice: boolean;
   190→   190→  isAdmin: boolean;
   191→   191→}) {
   192→   192→  const { t } = useTranslation("notificationDispatch");
   193→   193→
   194→   194→  const getChannelLabel = (val: string) => {
   195→   195→    const map: Record<string, string> = {
   196→   196→      IN_APP: t('channelsInApp'),
   197→   197→      SMS: t('channelsSms'),
   198→   198→      WHATSAPP: t('channelsWhatsapp'),
   199→   199→      TELEGRAM: t('channelsTelegram'),
   200→   200→    };
   201→   201→    return map[val] || val;
   202→   202→  };
   203→   203→  const getPriorityLabel = (val: string) => {
   204→   204→    const map: Record<string, string> = {
   205→   205→      LOW: t('prioritiesLow'),
   206→   206→      NORMAL: t('prioritiesNormal'),
   207→   207→      HIGH: t('prioritiesHigh'),
   208→   208→      URGENT: t('prioritiesUrgent'),
   209→   209→    };
   210→   210→    return map[val] || val;
   211→   211→  };
   212→   212→
   213→   213→  const [title, setTitle] = useState("");
   214→   214→  const [message, setMessage] = useState("");
   215→   215→  const [channel, setChannel] = useState("IN_APP");
   216→   216→  const [priority, setPriority] = useState("NORMAL");
   217→   217→  const [targetType, setTargetType] = useState<"ALL_PROVIDERS" | "SELECTED_PROVIDERS">("ALL_PROVIDERS");
   218→   218→  const [providers, setProviders] = useState<BroadcastProvider[]>([]);
   219→   219→  const [selectedProviders, setSelectedProviders] = useState<Set<string>>(new Set());
   220→   220→  const [loading, setLoading] = useState(false);
   221→   221→  const [sending, setSending] = useState(false);
   222→   222→  const [search, setSearch] = useState("");
   223→   223→
   224→   224→  const fetchProviders = useCallback(async () => {
   225→   225→    setLoading(true);
   226→   226→    try {
   227→   227→      const data = await apiGetBroadcastProviders();
   228→   228→      const arr = Array.isArray(data) ? data : [];
   229→   229→      setProviders(arr);
   230→   230→    } catch (err) {
   231→   231→      console.error("Failed to load providers:", err);
   232→   232→      toast.error(t('errorLoadProviders'));
   233→   233→    } finally {
   234→   234→      setLoading(false);
   235→   235→    }
   236→   236→  }, [t]);
   237→   237→
   238→   238→  useEffect(() => {
   239→   239→    fetchProviders();
   240→   240→  }, [fetchProviders]);
   241→   241→
   242→   242→  // Filter providers by search
   243→   243→  const filteredProviders = providers.filter(
   244→   244→    (p) =>
   245→   245→      p.name.toLowerCase().includes(search.toLowerCase()) ||
   246→   246→      p.ownerName.toLowerCase().includes(search.toLowerCase()) ||
   247→   247→      p.address.toLowerCase().includes(search.toLowerCase())
   248→   248→  );
   249→   249→
   250→   250→  const toggleProvider = (id: string) => {
   251→   251→    setSelectedProviders((prev) => {
   252→   252→      const next = new Set(prev);
   253→   253→      if (next.has(id)) next.delete(id);
   254→   254→      else next.add(id);
   255→   255→      return next;
   256→   256→    });
   257→   257→  };
   258→   258→
   259→   259→  const toggleAll = () => {
   260→   260→    if (selectedProviders.size === filteredProviders.length) {
   261→   261→      setSelectedProviders(new Set());
   262→   262→    } else {
   263→   263→      setSelectedProviders(new Set(filteredProviders.map((p) => p.id)));
   264→   264→    }
   265→   265→  };
   266→   266→
   267→   267→  const canSend =
   268→   268→    !isViewer && title.trim() && message.trim() &&
   269→   269→    (targetType === "ALL_PROVIDERS" || selectedProviders.size > 0);
   270→   270→
   271→   271→  const handleSend = async () => {
   272→   272→    if (!canSend) return;
   273→   273→    setSending(true);
   274→   274→    try {
   275→   275→      const result = await apiSendBroadcast({
   276→   276→        title: title.trim(),
   277→   277→        message: message.trim(),
   278→   278→        channel,
   279→   279→        priority,
   280→   280→        targetType,
   281→   281→        providerIds: targetType === "SELECTED_PROVIDERS" ? Array.from(selectedProviders) : [],
   282→   282→      });
   283→   283→      toast.success(
   284→   284→        t('successSent', { sent: result.sent, failed: result.failed }),
   285→   285→        { description: t('successSentDesc', { channel: getChannelLabel(channel), priority: getPriorityLabel(priority) }) }
   286→   286→      );
   287→   287→      // Reset form
   288→   288→      setTitle("");
   289→   289→      setMessage("");
   290→   290→      setPriority("NORMAL");
   291→   291→      setSelectedProviders(new Set());
   292→   292→      setTargetType("ALL_PROVIDERS");
   293→   293→    } catch (err) {
   294→   294→      toast.error(err instanceof Error ? err.message : t('errorSendBroadcast'));
   295→   295→    } finally {
   296→   296→      setSending(false);
   297→   297→    }
   298→   298→  };
   299→   299→
   300→   300→  // Check if selected channel is available for selected providers
   301→   301→  const getChannelAvailability = () => {
   302→   302→    if (targetType === "ALL_PROVIDERS") {
   303→   303→      const targetList = filteredProviders;
   304→   304→      if (channel === "SMS" || channel === "WHATSAPP") {
   305→   305→        const withPhone = targetList.filter((p) => p.hasPhone).length;
   306→   306→        return { available: withPhone, total: targetList.length, label: t('availabilityPhone') };
   307→   307→      }
   308→   308→      if (channel === "TELEGRAM") {
   309→   309→        const withTelegram = targetList.filter((p) => p.hasTelegram).length;
   310→   310→        return { available: withTelegram, total: targetList.length, label: t('availabilityTelegram') };
   311→   311→      }
   312→   312→      return { available: targetList.length, total: targetList.length, label: "" };
   313→   313→    }
   314→   314→    const selected = providers.filter((p) => selectedProviders.has(p.id));
   315→   315→    if (channel === "SMS" || channel === "WHATSAPP") {
   316→   316→      const withPhone = selected.filter((p) => p.hasPhone).length;
   317→   317→      return { available: withPhone, total: selected.length, label: t('availabilityPhone') };
   318→   318→    }
   319→   319→    if (channel === "TELEGRAM") {
   320→   320→      const withTelegram = selected.filter((p) => p.hasTelegram).length;
   321→   321→      return { available: withTelegram, total: selected.length, label: t('availabilityTelegram') };
   322→   322→    }
   323→   323→    return { available: selected.length, total: selected.length, label: "" };
   324→   324→  };
   325→   325→
   326→   326→  const availability = getChannelAvailability();
   327→   327→
   328→   328→  return (
   329→   329→    <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
   330→   330→      {/* Left column — Compose form */}
   331→   331→      <div className="lg:col-span-2 space-y-6">
   332→   332→        <Card>
   333→   333→          <CardHeader>
   334→   334→            <CardTitle className="flex items-center gap-2">
   335→   335→              <Radio className="size-5 text-primary" />
   336→   336→              {t('composeTitle')}
   337→   337→            </CardTitle>
   338→   338→            <CardDescription>
   339→   339→              {t('composeDescription')}
   340→   340→              {isViewer && (
   341→   341→                <span className="block mt-1 text-amber-600 font-medium">
   342→   342→                  {t('viewerWarning')}
   343→   343→                </span>
   344→   344→              )}
   345→   345→            </CardDescription>
   346→   346→          </CardHeader>
   347→   347→          <CardContent className="space-y-5">
   348→   348→            {/* Title */}
   349→   349→            <div className="space-y-2">
   350→   350→              <Label htmlFor="title" className="text-sm font-medium">
   351→   351→                {t('lblTitle')} <span className="text-rose-500">*</span>
   352→   352→              </Label>
   353→   353→              <Input
   354→   354→                id="title"
   355→   355→                placeholder={t('placeholderTitle')}
   356→   356→                value={title}
   357→   357→                onChange={(e) => setTitle(e.target.value)}
   358→   358→                disabled={isViewer}
   359→   359→              />
   360→   360→            </div>
   361→   361→
   362→   362→            {/* Message */}
   363→   363→            <div className="space-y-2">
   364→   364→              <Label htmlFor="message" className="text-sm font-medium">
   365→   365→                {t('lblMessage')} <span className="text-rose-500">*</span>
   366→   366→              </Label>
   367→   367→              <Textarea
   368→   368→                id="message"
   369→   369→                placeholder={t('placeholderMessage')}
   370→   370→                value={message}
   371→   371→                onChange={(e) => setMessage(e.target.value)}
   372→   372→                rows={6}
   373→   373→                disabled={isViewer}
   374→   374→                className="resize-y"
   375→   375→              />
   376→   376→              <p className="text-xs text-slate-400">{message.length} {t('characters')}</p>
   377→   377→            </div>
   378→   378→
   379→   379→            {/* Channel & Priority row */}
   380→   380→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   381→   381→              <div className="space-y-2">
   382→   382→                <Label>{t('lbldeliveryChannel', 'Delivery Channel')}</Label>
   383→   383→                <Select value={channel} onValueChange={setChannel} disabled={isViewer}>
   384→   384→                  <SelectTrigger>
   385→   385→                    <SelectValue />
   386→   386→                  </SelectTrigger>
   387→   387→                  <SelectContent>
   388→   388→                    {CHANNELS.map((ch) => {
   389→   389→                      const Icon = ch.icon;
   390→   390→                      return (
   391→   391→                        <SelectItem key={ch.value} value={ch.value}>
   392→   392→                          <span className="flex items-center gap-2">
   393→   393→                            <Icon className={`size-4 ${ch.color}`} />
   394→   394→                            {getChannelLabel(ch.value)}
   395→   395→                          </span>
   396→   396→                        </SelectItem>
   397→   397→                      );
   398→   398→                    })}
   399→   399→                  </SelectContent>
   400→   400→                </Select>
   401→   401→                {/* Channel availability warning */}
   402→   402→                {channel !== "IN_APP" && availability.label && (
   403→   403→                  <div className={`text-xs mt-1.5 flex items-center gap-1.5 ${
   404→   404→                    availability.available === 0
   405→   405→                      ? "text-rose-600"
   406→   406→                      : availability.available < availability.total
   407→   407→                      ? "text-amber-600"
   408→   408→                      : "text-emerald-600"
   409→   409→                  }`}>
   410→   410→                    {availability.available === 0 ? (
   411→   411→                      <XCircle className="size-3" />
   412→   412→                    ) : availability.available < availability.total ? (
   413→   413→                      <AlertTriangle className="size-3" />
   414→   414→                    ) : (
   415→   415→                      <CheckCircle2 className="size-3" />
   416→   416→                    )}
   417→   417→                    {t('availabilityWarning', { available: availability.available, total: availability.total, label: availability.label })}
   418→   418→                  </div>
   419→   419→                )}
   420→   420→              </div>
   421→   421→              <div className="space-y-2">
   422→   422→                <Label>{t('lblpriorityLevel', 'Priority Level')}</Label>
   423→   423→                <Select value={priority} onValueChange={setPriority} disabled={isViewer}>
   424→   424→                  <SelectTrigger>
   425→   425→                    <SelectValue />
   426→   426→                  </SelectTrigger>
   427→   427→                  <SelectContent>
   428→   428→                    {PRIORITIES.map((pr) => (
   429→   429→                      <SelectItem key={pr.value} value={pr.value}>
   430→   430→                        <span className="flex items-center gap-2">
   431→   431→                          <span className={`inline-block h-2 w-2 rounded-full ${
   432→   432→                            pr.value === "URGENT"
   433→   433→                              ? "bg-rose-500"
   434→   434→                              : pr.value === "HIGH"
   435→   435→                              ? "bg-amber-500"
   436→   436→                              : pr.value === "NORMAL"
   437→   437→                              ? "bg-blue-500"
   438→   438→                              : "bg-slate-400"
   439→   439→                          }`} />
   440→   440→                          {getPriorityLabel(pr.value)}
   441→   441→                        </span>
   442→   442→                      </SelectItem>
   443→   443→                    ))}
   444→   444→                  </SelectContent>
   445→   445→                </Select>
   446→   446→              </div>
   447→   447→            </div>
   448→   448→
   449→   449→            {/* Target Type */}
   450→   450→            <div className="space-y-2">
   451→   451→              <Label>{t('lbltargetAudience', 'Target Audience')}</Label>
   452→   452→              <div className="flex items-center gap-4">
   453→   453→                <label className="flex items-center gap-2 cursor-pointer text-sm">
   454→   454→                  <input
   455→   455→                    type="radio"
   456→   456→                    name="targetType"
   457→   457→                    value="ALL_PROVIDERS"
   458→   458→                    checked={targetType === "ALL_PROVIDERS"}
   459→   459→                    onChange={() => setTargetType("ALL_PROVIDERS")}
   460→   460→                    disabled={isViewer}
   461→   461→                    className="accent-primary"
   462→   462→                  />
   463→   463→                  {t('allApprovedProviders')}
   464→   464→                </label>
   465→   465→                <label className="flex items-center gap-2 cursor-pointer text-sm">
   466→   466→                  <input
   467→   467→                    type="radio"
   468→   468→                    name="targetType"
   469→   469→                    value="SELECTED_PROVIDERS"
   470→   470→                    checked={targetType === "SELECTED_PROVIDERS"}
   471→   471→                    onChange={() => setTargetType("SELECTED_PROVIDERS")}
   472→   472→                    disabled={isViewer}
   473→   473→                    className="accent-primary"
   474→   474→                  />
