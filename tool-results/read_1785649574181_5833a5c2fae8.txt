   202→// ── Helper: get nav items based on role ──
   203→function getNavItems(user: CurrentUser): NavItem[] {
   204→  switch (user.role) {
   205→    case "POLICE": {
   206→      const rank = (user.policeRank || "OFFICER") as PoliceRank;
   207→      const allowedPages = POLICE_RANK_PERMISSIONS[rank] || POLICE_RANK_PERMISSIONS.OFFICER;
   208→      return POLICE_NAV_ITEMS.filter((item) => allowedPages.includes(item.page));
   209→    }
   210→
   211→    case "SUPERUSER":
   212→      return SUPERUSER_NAV_ITEMS;
   213→
   214→    case "OPERATOR":
   215→      return ALL_NAV_ITEMS.filter((item) => !OPERATOR_EXCLUDED.has(item.page));
   216→
   217→    case "STAFF": {
   218→      const items: NavItem[] = [];
   219→      // Always give staff access to dashboard
   220→      items.push({
   221→        page: "dashboard",
   222→        label: "Dashboard",
   223→        icon: LayoutDashboard,
   224→      });
   225→      // Add items based on permissions
   226→      for (const perm of user.permissions) {
   227→        const mapped = PERMISSION_PAGE_MAP[perm];
   228→        if (mapped) {
   229→          items.push(mapped);
   230→        }
   231→      }
   232→      return items;
   233→    }
   234→
   235→    default:
   236→      return [
   237→        { page: "dashboard", label: "Dashboard", icon: LayoutDashboard },
   238→      ];
   239→  }
   240→}

   242→// ── Helper: get initials from name ──
   243→function getInitials(name: string): string {
   244→  return name
   245→    .split(" ")
   246→    .map((w) => w[0])
   247→    .join("")
   248→    .toUpperCase()
   249→    .slice(0, 2);
   250→}

   331→function SubscriptionStatusCard({ collapsed }: { collapsed: boolean }) {
   332→  const subscription = useAppStore((s) => s.subscription);
   333→
   334→  if (!subscription) return null;
   335→
   336→  const isActive = subscription.status === "ACTIVE";
   337→  const isWarning = subscription.status === "WARNING";
   338→  const isGrace = subscription.status === "GRACE";
   339→  const isSuspended = subscription.status === "SUSPENDED";
   340→
   341→  // Collapsed mode — icon indicator
   342→  if (collapsed) {
   343→    const bgClass = isActive
   344→      ? "bg-emerald-100"
   345→      : isSuspended || isGrace
   346→      ? "bg-rose-100"
   347→      : "bg-amber-100";
   348→    const iconClass = isActive
   349→      ? "text-emerald-600"
   350→      : isSuspended || isGrace
   351→      ? "text-rose-600"
   352→      : "text-amber-600";
   353→    return (
   354→      <div className="flex flex-col items-center gap-1 px-2 py-2">
   355→        <div className={`flex h-8 w-8 items-center justify-center rounded-full ${bgClass} animate-pulse shadow-sm`}> 
   356→          <Clock className={`size-4 ${iconClass}`} />
   357→        </div>
   358→      </div>
   359→    );
   360→  }
   361→
   362→  // Expanded card
   363→  const containerClass = isActive
   364→    ? "border-emerald-200 bg-emerald-50"
   365→    : isSuspended || isGrace
   366→    ? "border-rose-200 bg-rose-50"
   367→    : "border-amber-200 bg-amber-50";
   368→
   369→  const iconColorClass = isActive
   370→    ? "text-emerald-600"
   371→    : isSuspended || isGrace
   372→    ? "text-rose-600"
   373→    : "text-amber-600";
   374→
   375→  const titleClass = isActive
   376→    ? "text-emerald-800"
   377→    : isSuspended || isGrace
   378→    ? "text-rose-800"
   379→    : "text-amber-800";
   380→
   381→  const subtextClass = isActive
   382→    ? "text-emerald-700"
   383→    : isSuspended || isGrace
   384→    ? "text-rose-700"
   385→    : "text-amber-700";
   386→
   387→  const badgeClass = isActive
   388→    ? "bg-emerald-200/60 text-emerald-800"
   389→    : isSuspended || isGrace
   390→    ? "bg-rose-200/60 text-rose-800"
   391→    : "bg-amber-200/60 text-amber-800";
   392→
   393→  const Icon = isSuspended || isGrace ? AlertTriangle : Clock;
   394→
   395→  return (
   396→    <div className={`rounded-lg border p-3 ${containerClass} animate-subtle-pulse shadow-sm`}> 
   397→      <div className="flex items-center gap-2">
   398→        <Icon className={`h-4 w-4 shrink-0 ${iconColorClass}`} />
   399→        <p className={`text-xs font-semibold ${titleClass}`}>
   400→          {isActive
   401→            ? "Subscription Active"
   402→            : isSuspended
   403→            ? "Service Suspended"
   404→            : isGrace
   405→            ? "Subscription Expired"
   406→            : "Expiring Soon"}
   407→        </p>
   408→      </div>
   409→      <div className={`mt-1.5 flex items-center justify-between text-[11px] ${subtextClass}`}>
   410→        <span>{formatDaysRemaining(subscription.daysRemaining)}</span>
   411→        <span className={`inline-flex items-center gap-1 rounded-full px-1.5 py-0.5 font-semibold ${badgeClass}`}>
   412→          <Clock className="h-3 w-3" />
   413→          {Math.abs(subscription.daysRemaining)}d
   414→        </span>
   415→      </div>
   416→      {subscription.price > 0 ? (
   417→        <div className={`mt-1 text-[11px] ${subtextClass}`}>
   418→          Due: {subscription.price.toLocaleString()} ETB / {formatCycle(subscription.cycle)}
   419→        </div>
   420→      ) : (
   421→        <div className={`mt-1 text-[11px] ${subtextClass}`}>
   422→          {formatCycle(subscription.cycle)} cycle — Free trial
   423→        </div>
   424→      )}
   425→    </div>
   426→  );
   427→}

   580→              <button
   581→                onClick={() => setJointLoginDialogOpen(true)}
   582→                className="flex w-full items-center gap-3 rounded-lg border border-dashed border-amber-300 bg-amber-50/50 px-3 py-2.5 text-sm font-medium text-amber-700 transition-all hover:bg-amber-50 hover:border-amber-400 hover:text-amber-800 outline-none focus-visible:ring-2 focus-visible:ring-amber-400"
   583→              >
   584→                <ShieldCheck className="size-[18px] shrink-0" />
   585→                <span>Start Joint Session</span>
   586→              </button>
   587→            </>
   588→          )}
   589→          {canStartJoint && !jointSession.active && collapsed && (
   590→            <button
   591→              onClick={() => setJointLoginDialogOpen(true)}
   592→              className="flex items-center justify-center rounded-lg border border-dashed border-amber-300 p-2 text-amber-600 hover:bg-amber-50 transition-colors outline-none"
   593→              title="Start Joint Session"
   594→            >
   595→              <ShieldCheck className="size-[18px]" />
   596→            </button>
   597→          )}
   598→        </nav>
   599→      </ScrollArea>
   600→
   601→      {/* ── Subscription status for providers ── */}
   602→      {(user.role === "OPERATOR" || user.role === "STAFF") && (
   603→        <div className="px-3 py-2">
   604→          <SubscriptionStatusCard collapsed={collapsed} />
   605→        </div>
   606→      )}
   607→
   608→      <Separator className="bg-slate-200/60" />
   609→
   610→      {/* ── Bottom section: collapse toggle + logout ── */}
   611→      <div className="p-3">
   612→        {/* Desktop collapse toggle */}
   613→        {!collapsed && (
   614→          <button
   615→            onClick={onToggleCollapse}
   616→            className="mb-1 flex w-full items-center gap-3 rounded-lg px-3 py-2 text-sm font-medium text-slate-500 transition-colors hover:bg-slate-100 hover:text-slate-700 outline-none focus-visible:ring-2 focus-visible:ring-ring"
   617→          >
   618→            <ChevronLeft className="size-[18px] text-slate-400" />
   619→            <span>Collapse</span>
   620→          </button>
   621→        )}
   622→
   623→        {/* Logout — hidden for SUPERUSER (moved to name dropdown) */}
   624→        {user.role !== "SUPERUSER" && (
   625→          <button
   626→            onClick={onLogout}
   627→            className="flex w-full items-center gap-3 rounded-lg px-3 py-2.5 text-sm font-medium text-rose-600 transition-colors hover:bg-rose-50 outline-none focus-visible:ring-2 focus-visible:ring-ring"
   628→          >
   629→            <LogOut className="size-[18px]" />
   630→            {!collapsed && <span>Logout</span>}
   631→          </button>
   632→        )}
   633→      </div>
   634→    </div>
   635→  );
   636→}
