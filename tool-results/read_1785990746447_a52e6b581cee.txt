   661→   661→/** Execute SQL via raw pg Client */
   662→   662→async function execViaPg(sql: string, params?: unknown[]): Promise<void> {
   663→   663→  const client = new Client({ connectionString: process.env.DATABASE_URL! });
   664→   664→  try {
   665→   665→    await client.connect();
   666→   666→    await client.query(sql, params);
   667→   667→  } finally {
   668→   668→    await client.end().catch(() => {});
   669→   669→  }
   670→   670→}

   672→   672→/** Execute DDL via Prisma $executeRawUnsafe — splits multi-statement SQL
   673→   673→ *  because Prisma's driver does NOT support multiple statements per call. */
   674→   674→async function execViaPrisma(sql: string): Promise<void> {
   675→   675→  // Split on `DO $$` blocks and standalone statements
   676→   676→  // Each `DO $$ ... END $$;` is one executable unit
   677→   677→  const statements = sql
   678→   678→    .split(/(?=DO \$\$)/)
   679→   679→    .map(s => s.trim())
   680→   680→    .filter(s => s.length > 0);
   681→   681→
   682→   682→  // If no DO blocks found, split by semicolons (for CREATE INDEX etc.)
   683→   683→  const parts = statements.length > 1
   684→   684→    ? statements
   685→   685→    : sql.split(';').map(s => s.trim()).filter(s => s.length > 0);
   686→   686→
   687→   687→  const prisma = new PrismaClient({
   688→   688→    log: ["warn", "error"],
   689→   689→  });
   690→   690→  try {
   691→   691→    for (const part of parts) {
   692→   692→      const stmt = part.endsWith(';') ? part : part + ';';
   693→   693→      try {
   694→   694→        await prisma.$executeRawUnsafe(stmt);
   695→   695→      } catch (err) {
   696→   696→        // Individual statement might fail (e.g. duplicate_object) — log but continue
   697→   697→        console.error("[execViaPrisma] Statement failed (continuing):", stmt.slice(0, 80), err instanceof Error ? err.message : String(err));
   698→   698→      }
   699→   699→    }
   700→   700→  } finally {
   701→   701→    await prisma.$disconnect().catch(() => {});
   702→   702→  }
   703→   703→}
