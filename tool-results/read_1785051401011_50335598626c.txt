    10→import { Input } from "@/components/ui/input";

    33→

    37→

    40→  const [auditTotal, setAuditTotal] = useState(0);
    41→  const [auditLoading, setAuditLoading] = useState(true);

    57→    }
    58→    finally { setAuditLoading(false); }
    59→  }, [auditPag.currentPage, auditPag.pageSize, actionFilter]);
    60→
    61→  // Geofences
    62→  const [geofences, setGeofences] = useState<Geofence[]>([]);
    63→  const [geoLoading, setGeoLoading] = useState(true);
    64→  const [showGeoForm, setShowGeoForm] = useState(false);
    65→  const [geoForm, setGeoForm] = useState({ name: "", address: "", latitude: "", longitude: "", radius: "1000", severity: "HIGH" });
    66→  const [geoSaving, setGeoSaving] = useState(false);
    67→
    68→  // Alert Config

    71→  const [configSaving, setConfigSaving] = useState(false);
    72→

    79→
    80→  useEffect(() => { fetchAudit(); }, [fetchAudit, refreshKey]);

    85→    finally { setGeoLoading(false); }

    99→      setGeoSaving(true);
   100→      await apiPoliceCreateGeofence({ ...geoForm, latitude: parseFloat(geoForm.latitude), longitude: parseFloat(geoForm.longitude), radius: parseInt(geoForm.radius) });

   102→      setShowGeoForm(false);
   103→      setGeoForm({ name: "", address: "", latitude: "", longitude: "", radius: "1000", severity: "HIGH" });

   105→    } catch { toast.error("Failed to create geofence"); }
   106→    finally { setGeoSaving(false); }
   107→  };

   109→  const deleteGeofence = async (id: string) => {
   110→    try { await apiPoliceDeleteGeofence(id); toast.success("Geofence deleted"); fetchGeo(); }
   111→    catch { toast.error("Failed to delete"); }
   112→  };

   117→      setConfigSaving(true);
   118→      await apiPoliceUpdateAlertConfig(config);
   119→      toast.success("Alert config saved");
   120→    } catch { toast.error("Failed to save config"); }
   121→    finally { setConfigSaving(false); }

   125→    try { setOfficersLoading(true); const d = await apiPoliceOfficers(); setOfficers(Array.isArray(d) ? d : []); }

   127→    finally { setOfficersLoading(false); }

   133→      setOfficerSaving(true);
   134→      await apiPoliceCreateOfficer(officerForm);

   140→    finally { setOfficerSaving(false); }
   141→  };

   144→    try {
   145→      await apiPoliceUpdateOfficer(id, { policeRank: rank });
   146→      toast.success("Rank updated");
   147→      fetchOfficers();
   148→    } catch (e: unknown) { toast.error(e instanceof Error ? e.message : "Failed to update rank"); }

   150→
   151→  const deleteOfficer = async (id: string) => {
   152→    if (!confirm("Delete this officer?")) return;
   153→    try { await apiPoliceDeleteOfficer(id); toast.success("Officer deleted"); fetchOfficers(); }

   159→      const blob = await apiPoliceExport(`type=${type}&format=${format}`);
   160→      const url = window.URL.createObjectURL(blob as any);

   172→    { key: "alerts" as const, label: "Alert Settings", icon: Shield },

   204→                <SelectItem value="VIEW_MATCH">Viewed Match</SelectItem>

   207→                <SelectItem value="LOGIN">Login</SelectItem>
   208→              </SelectContent>
   209→            </Select>

   211→          <Card>
   212→            <CardContent className="p-0">
   213→              {auditLoading ? <div className="space-y-2 p-4"><Skeleton className="h-10 w-full" /><Skeleton className="h-10 w-full" /></div> : (
   214→                <div className="divide-y">
   215→                  {pagAudit.map((a) => (
   216→                    <div key={a.id} className="flex items-center justify-between px-3 sm:px-4 py-2.5">
   217→                      <div className="flex items-center gap-2 min-w-0">

   220→                        {a.targetId && <span className="text-[10px] text-muted-foreground hidden sm:inline">{a.targetType}: {a.targetId.slice(0, 8)}...</span>}

   223→                        {a.ipAddress && <span className="text-[10px] text-muted-foreground hidden sm:inline">{a.ipAddress}</span>}
   224→                        <span className="text-[10px] text-muted-foreground">{new Date(a.createdAt).toLocaleString()}</span>
   225→                      </div>
   226→                    </div>
   227→                  ))}
   228→                </div>

   230→            </CardContent>
   231→          </Card>
   232→          <PaginationControls currentPage={auditPag.currentPage} totalPages={auditPag.totalPages} pageSize={auditPag.pageSize} pageSizeOptions={auditPag.pageSizeOptions} totalItems={auditTotal} rangeInfo={auditPag.rangeInfo} goToPage={auditPag.goToPage} setPageSize={auditPag.setPageSize} />

   249→                <div className="space-y-1.5"><Label className="text-xs">Zone Name *</Label><Input value={geoForm.name} onChange={(e) => setGeoForm({ ...geoForm, name: e.target.value })} placeholder="e.g. Bole District" /></div>
   250→                <div className="space-y-1.5"><Label className="text-xs">Address</Label><Input value={geoForm.address} onChange={(e) => setGeoForm({ ...geoForm, address: e.target.value })} /></div>

   261→                      <SelectItem value="MEDIUM">Medium</SelectItem>

   268→                </div>
   269→              </CardContent>
   270→            </Card>

   272→
   273→          <Card>

   276→                <p className="py-8 text-center text-xs text-muted-foreground">No geofence zones. Add zones to get alerts when suspects check in nearby.</p>

   278→                <div className="divide-y">
   279→                  {geofences.map((g) => (
   280→                    <div key={g.id} className="flex items-center justify-between p-3 sm:px-4">
   281→                      <div className="flex items-center gap-3 min-w-0">

   286→                            <span>{g.latitude.toFixed(4)}, {g.longitude.toFixed(4)}</span>
   287→                            <span>Radius: {g.radius}m</span>

   314→                    <button onClick={() => setConfig({ ...config, emailEnabled: !config.emailEnabled })} className={"h-5 w-9 rounded-full transition-colors " + (config.emailEnabled ? "bg-emerald-600" : "bg-slate-200")}>

   347→                        <div className={"h-4 w-4 rounded-full bg-white shadow transition-transform " + (config.criticalImmediate ? "translate-x-4" : "translate-x-0.5")} />
