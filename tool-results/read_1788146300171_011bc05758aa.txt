   385→  const handleBulkFileSelect = async (e: React.ChangeEvent<HTMLInputElement>) => {
   386→    const file = e.target.files?.[0];
   387→    if (!file) return;
   388→    setBulkFile(file);
   389→    setBulkErrors([]);
   390→    setBulkPreview([]);
   391→    try {
   392→      const XLSX = await import("xlsx");
   393→      const buffer = await file.arrayBuffer();
   394→      const wb = XLSX.read(buffer, { type: "array" });
   395→      const ws = wb.Sheets[wb.SheetNames[0]];
   396→      const rows = XLSX.utils.sheet_to_json<Record<string, string>>(ws, { defval: "" });
   397→
   398→      if (rows.length === 0) {
   399→        setBulkErrors([t('bulkNoDataRows')]);
   400→        return;
   401→      }
   402→
   403→      // Validate columns
   404→      const requiredCols = ["Full Name", "Phone", "Email", "Guesthouse Name", "Type", "License No", "Sub-City", "Woreda", "Username", "Password"];
   405→      const firstRowKeys = Object.keys(rows[0]);
   406→      const missingCols = requiredCols.filter((c) => !firstRowKeys.includes(c));
   407→      if (missingCols.length > 0) {
   408→        setBulkErrors([t('bulkMissingCols', { cols: missingCols.join(', ') })]);
   409→        return;
   410→      }
   411→
   412→      // Validate rows
   413→      const errors: string[] = [];
   414→      const validSubCities = Object.keys(SUB_CITY_WOREDAS);
   415→      const validTypes = GUESTHOUSE_TYPES.map((t) => t.value);
   416→      rows.forEach((row, idx) => {
   417→        const rowNum = idx + 2; // Excel row number (1-based + header)
   418→        const fullName = (row["Full Name"] || "").trim();
   419→        const phone = (row["Phone"] || "").trim();
   420→        const email = (row["Email"] || "").trim();
   421→        const ghName = (row["Guesthouse Name"] || "").trim();
   422→        const type = (row["Type"] || "").trim();
   423→        const licenseNo = (row["License No"] || "").trim();
   424→        const subCity = (row["Sub-City"] || "").trim();
   425→        const woreda = (row["Woreda"] || "").trim();
   426→        const username = (row["Username"] || "").trim();
   427→        const password = (row["Password"] || "").trim();
   428→
   429→        if (!fullName) errors.push(t('bulkRowFieldEmpty', { row: rowNum, field: "Full Name" }));
   430→        if (!phone) errors.push(t('bulkRowFieldEmpty', { row: rowNum, field: "Phone" }));
   431→        else if (!isValidPhone(phone)) errors.push(t('bulkRowInvalidPhone', { row: rowNum, value: phone }));
   432→        if (!email) errors.push(t('bulkRowFieldEmpty', { row: rowNum, field: "Email" }));
   433→        else if (!isValidEmail(email)) errors.push(t('bulkRowInvalidEmail', { row: rowNum, value: email }));
   434→        if (!ghName) errors.push(t('bulkRowFieldEmpty', { row: rowNum, field: "Guesthouse Name" }));
   435→        if (!type) errors.push(t('bulkRowFieldEmpty', { row: rowNum, field: "Type" }));
   436→        else if (!validTypes.includes(type)) errors.push(t('bulkRowInvalidType', { row: rowNum, value: type, valid: validTypes.join(", ") }));
   437→        if (!licenseNo) errors.push(t('bulkRowFieldEmpty', { row: rowNum, field: "License No" }));
   438→        if (!subCity) errors.push(t('bulkRowFieldEmpty', { row: rowNum, field: "Sub-City" }));
   439→        else if (!validSubCities.includes(subCity)) errors.push(t('bulkRowInvalidSubCity', { row: rowNum, value: subCity, valid: validSubCities.join(", ") }));
   440→        if (!woreda) errors.push(t('bulkRowFieldEmpty', { row: rowNum, field: "Woreda" }));
   441→        else if (subCity && validSubCities.includes(subCity) && !SUB_CITY_WOREDAS[subCity]?.includes(woreda)) errors.push(t('bulkRowInvalidWoreda', { row: rowNum, woreda, subCity }));
   442→        if (!username) errors.push(t('bulkRowFieldEmpty', { row: rowNum, field: "Username" }));
   443→        if (!password) errors.push(t('bulkRowFieldEmpty', { row: rowNum, field: "Password" }));
   444→        else if (password.length < 4) errors.push(t('bulkRowPasswordMinLength', { row: rowNum }));
   445→      });
   446→
   447→      if (errors.length > 0) {
   448→        setBulkErrors(errors.slice(0, 20));
   449→        if (errors.length > 20) setBulkErrors((prev) => [...prev, t('bulkMoreErrors', { count: errors.length - 20 })]);
   450→      }
   451→
   452→      setBulkPreview(rows);
   453→    } catch {
   454→      setBulkErrors([t('bulkInvalidFile')]);
   455→    }
   456→  };

   458→  const handleBulkImport = async () => {

   462→    }
   463→    if (bulkErrors.length > 0) {
   464→      toast.error(t('bulkFixErrorsBeforeImport'));
   465→      return;
   466→    }
   467→    try {
   468→      setBulkImporting(true);
   469→      const records = bulkPreview.map((row) => {
   470→        const subCity = (row["Sub-City"] || "").trim();
   471→        const woreda = (row["Woreda"] || "").trim();
   472→        let type = (row["Type"] || "").trim();
   473→        // type is already the enum value from the Excel

   475→        return {
   476→          ownerName: (row["Full Name"] || "").trim(),
   477→          phone: (row["Phone"] || "").trim(),
   478→          email: (row["Email"] || "").trim(),
   479→          name: (row["Guesthouse Name"] || "").trim(),
   480→          type,
   481→          licenseNo: (row["License No"] || "").trim(),
   482→          address,
   483→          username: (row["Username"] || "").trim(),
   484→          password: (row["Password"] || "").trim(),
   485→        };
   486→      });
   487→      const result = await apiSuperBulkImportProviders(records);
   488→      const data = result as { success: number; failed: number; errors: string[] };
   489→      toast.success(t('bulkImportSuccess', { success: data.success, failed: data.failed, failedPart: data.failed > 0 ? t('bulkImportFailed', { count: data.failed }) : '' }));
   490→      setBulkImportOpen(false);
   491→      setBulkFile(null);
   492→      setBulkPreview([]);
   493→      setBulkErrors([]);
   494→      if (bulkFileInputRef.current) bulkFileInputRef.current.value = "";
   495→      triggerRefresh();
   496→    } catch (err: unknown) {
   497→      toast.error(err instanceof Error ? err.message : t('bulkImportFailed'));
   498→    } finally {
   499→      setBulkImporting(false);
   500→    }
   501→  };
