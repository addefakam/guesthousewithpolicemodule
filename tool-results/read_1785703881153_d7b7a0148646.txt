   543→/** Execute SQL via raw pg Client */
   544→async function execViaPg(sql: string, params?: unknown[]): Promise<void> {
   545→  const client = new Client({ connectionString: process.env.DATABASE_URL! });
   546→  try {
   547→    await client.connect();
   548→    await client.query(sql, params);
   549→  } finally {
   550→    await client.end().catch(() => {});
   551→  }
   552→}

   554→/** Execute DDL via Prisma $executeRawUnsafe — splits multi-statement SQL
   555→ *  because Prisma's driver does NOT support multiple statements per call. */
   556→async function execViaPrisma(sql: string): Promise<void> {
   557→  // Split on `DO $$` blocks and standalone statements
   558→  // Each `DO $$ ... END $$;` is one executable unit
   559→  const statements = sql
   560→    .split(/(?=DO \$\$)/)
   561→    .map(s => s.trim())
   562→    .filter(s => s.length > 0);
   563→
   564→  // If no DO blocks found, split by semicolons (for CREATE INDEX etc.)
   565→  const parts = statements.length > 1
   566→    ? statements
   567→    : sql.split(';').map(s => s.trim()).filter(s => s.length > 0);
   568→
   569→  const prisma = new PrismaClient({
   570→    log: ["warn", "error"],
   571→  });
   572→  try {
   573→    for (const part of parts) {
   574→      const stmt = part.endsWith(';') ? part : part + ';';
   575→      try {
   576→        await prisma.$executeRawUnsafe(stmt);
   577→      } catch (err) {
   578→        // Individual statement might fail (e.g. duplicate_object) — log but continue
   579→        console.error("[execViaPrisma] Statement failed (continuing):", stmt.slice(0, 80), err instanceof Error ? err.message : String(err));
   580→      }
   581→    }
   582→  } finally {
   583→    await prisma.$disconnect().catch(() => {});
   584→  }
   585→}

   606→export async function ensureDatabase(): Promise<void> {
   607→  if (_initDone) {
   608→    // Always run migrations synchronously to ensure columns exist
   609→    // before any query executes. Migrations are idempotent.
   610→    if (!_migrationsRan) {
   611→      _migrationsRan = true;
   612→      await runMigrationsOnly();
   613→    }
   614→    return;
   615→  }
   616→  if (_initPromise) return _initPromise;
   617→
   618→  if (!process.env.DATABASE_URL) {
   619→    throw new Error("DATABASE_URL environment variable is not set.");
   620→  }
   621→
   622→  _initPromise = (async () => {
   623→    console.log("[init-db] Starting database initialization...");
   624→
   625→    let usePg = true;
   626→    let pgError: unknown = null;
   627→
   628→    // Strategy 1: Try using pg Client directly
   629→    try {
   630→      const client = new Client({ connectionString: process.env.DATABASE_URL });
   631→      await client.connect();
   632→      console.log("[init-db] pg Client connected successfully.");
   633→
   634→      // Check if User table already exists
   635→      const res = await client.query(
   636→        `SELECT EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema='public' AND table_name='User') as ok`
   637→      );
   638→      const tableExists = res.rows[0]?.ok === true;
   639→      console.log("[init-db] User table exists:", tableExists);
   640→
   641→      if (tableExists) {
   642→        console.log("[init-db] Tables already exist, running migrations to add any missing columns...");
   643→        try {
   644→          await client.query(MIGRATIONS_SQL);
   645→          console.log("[init-db] Migrations applied to existing tables.");
   646→          await client.query(INDEXES_SQL);
   647→          console.log("[init-db] Indexes verified.");
   648→        } catch (migrateErr) {
   649→          console.error("[init-db] Migration on existing tables failed:", migrateErr instanceof Error ? migrateErr.message : String(migrateErr));
   650→          // Don't throw — the tables exist, migrations are best-effort
   651→        }
   652→        await client.end().catch(() => {});
   653→        _initDone = true;
   654→        return;
   655→      }
   656→
   657→      // Create enums
   658→      console.log("[init-db] Creating enum types...");
   659→      await client.query(ENUMS_SQL);
   660→
   661→      // Create tables
   662→      console.log("[init-db] Creating tables...");
   663→      await client.query(TABLES_SQL);
   664→
   665→      // Verify User table was created
   666→      const verify = await client.query(
   667→        `SELECT EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema='public' AND table_name='User') as ok`
   668→      );
   669→      if (!verify.rows[0]?.ok) {
   670→        throw new Error("DDL executed but User table still not found!");
   671→      }
   672→      console.log("[init-db] Tables verified.");
   673→
   674→      // Create foreign keys
   675→      console.log("[init-db] Creating foreign keys...");
   676→      await client.query(FKS_SQL);
   677→
   678→      // Run migrations (add new columns to existing tables)
   679→      console.log("[init-db] Running migrations...");
   680→      await client.query(MIGRATIONS_SQL);
   681→
   682→      // Create indexes
   683→      console.log("[init-db] Creating indexes...");
   684→      await client.query(INDEXES_SQL);
   685→
   686→      // Seed SUPERUSER
   687→      console.log("[init-db] Seeding SUPERUSER...");
   688→      const hashed = await bcrypt.hash("Admin@2024", 12);
   689→      await client.query(
   690→        `INSERT INTO "User" ("id","username","password","name","role","permissions","policeRank","createdAt","updatedAt")
   691→         SELECT 'su-admin-001','admin',$1,'System Administrator','SUPERUSER','[]','',NOW(),NOW()
   692→         WHERE NOT EXISTS (SELECT 1 FROM "User" WHERE "username"='admin')`,
   693→        [hashed]
   694→      );
   695→
   696→      // Seed PoliceAlertConfig
   697→      await client.query(
   698→        `INSERT INTO "PoliceAlertConfig" ("id","createdAt","updatedAt")
   699→         SELECT 'default-alert-config',NOW(),NOW()
   700→         WHERE NOT EXISTS (SELECT 1 FROM "PoliceAlertConfig" WHERE "id"='default-alert-config')`
   701→      );
   702→
   703→      // Seed default SubscriptionPlans
   704→      await client.query(`
   705→        INSERT INTO "SubscriptionPlan" ("id","name","cycle","price","isActive","createdAt","updatedAt")
   706→        SELECT 'plan-monthly-001','Monthly','MONTHLY',500,true,NOW(),NOW()
   707→        WHERE NOT EXISTS (SELECT 1 FROM "SubscriptionPlan" WHERE "id"='plan-monthly-001');
   708→        INSERT INTO "SubscriptionPlan" ("id","name","cycle","price","isActive","createdAt","updatedAt")
   709→        SELECT 'plan-quarterly-001','Quarterly','QUARTERLY',1400,true,NOW(),NOW()
   710→        WHERE NOT EXISTS (SELECT 1 FROM "SubscriptionPlan" WHERE "id"='plan-quarterly-001');
   711→        INSERT INTO "SubscriptionPlan" ("id","name","cycle","price","isActive","createdAt","updatedAt")
   712→        SELECT 'plan-semi-annual-001','Semi-Annual','SEMI_ANNUAL',2600,true,NOW(),NOW()
   713→        WHERE NOT EXISTS (SELECT 1 FROM "SubscriptionPlan" WHERE "id"='plan-semi-annual-001');
   714→        INSERT INTO "SubscriptionPlan" ("id","name","cycle","price","isActive","createdAt","updatedAt")
   715→        SELECT 'plan-yearly-001','Annual','YEARLY',4800,true,NOW(),NOW()
   716→        WHERE NOT EXISTS (SELECT 1 FROM "SubscriptionPlan" WHERE "id"='plan-yearly-001');
   717→      `);
   718→
   719→      console.log("[init-db] Database initialization complete via pg.");
   720→      await client.end().catch(() => {});
   721→      _initDone = true;
   722→      return;
   723→    } catch (err) {
   724→      pgError = err;
   725→      console.error("[init-db] pg approach failed:", err instanceof Error ? err.message : String(err));
   726→      usePg = false;
   727→    }
   728→
   729→    // Strategy 2: Fallback — use Prisma $executeRawUnsafe
   730→    console.log("[init-db] Falling back to Prisma $executeRawUnsafe...");
   731→    try {
   732→      // First check if tables exist using Prisma raw query
   733→      const prisma = new PrismaClient({ log: ["warn", "error"] });
   734→      try {
   735→        const res: Array<{ ok: boolean }> = await prisma.$queryRawUnsafe(
   736→          `SELECT EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema='public' AND table_name='User') as ok`
   737→        );
   738→        if (res[0]?.ok) {
   739→          console.log("[init-db] Tables already exist (verified via Prisma), running migrations...");
   740→          try {
   741→            await execViaPrisma(MIGRATIONS_SQL);
   742→            console.log("[init-db] Migrations applied via Prisma.");
   743→            await execViaPrisma(INDEXES_SQL);
   744→            console.log("[init-db] Indexes verified via Prisma.");
   745→          } catch (migrateErr) {
   746→            console.error("[init-db] Prisma migration on existing tables failed:", migrateErr instanceof Error ? migrateErr.message : String(migrateErr));
   747→          }
   748→          _initDone = true;
   749→          return;
   750→        }
   751→      } catch {
   752→        // Table might not exist, which is expected
   753→        console.log("[init-db] Tables do not exist yet, creating via Prisma...");
   754→      } finally {
   755→        await prisma.$disconnect().catch(() => {});
   756→      }
   757→
   758→      // Execute DDL via Prisma (split into parts to avoid query size issues)
   759→      await execViaPrisma(ENUMS_SQL);
   760→      console.log("[init-db] Enums created via Prisma.");
   761→
   762→      await execViaPrisma(TABLES_SQL);
   763→      console.log("[init-db] Tables created via Prisma.");
   764→
   765→      await execViaPrisma(FKS_SQL);
   766→      console.log("[init-db] Foreign keys created via Prisma.");
   767→
   768→      await execViaPrisma(MIGRATIONS_SQL);
   769→      console.log("[init-db] Migrations ran via Prisma.");
   770→
   771→      await execViaPrisma(INDEXES_SQL);
   772→      console.log("[init-db] Indexes created via Prisma.");
   773→
   774→      // Seed SUPERUSER via Prisma
   775→      const prisma2 = new PrismaClient({ log: ["warn", "error"] });
   776→      try {
   777→        const hashed = await bcrypt.hash("Admin@2024", 12);
   778→        await prisma2.$executeRawUnsafe(
   779→          `INSERT INTO "User" ("id","username","password","name","role","permissions","policeRank","createdAt","updatedAt")
   780→           SELECT 'su-admin-001','admin',$1,'System Administrator','SUPERUSER','[]','',NOW(),NOW()
   781→           WHERE NOT EXISTS (SELECT 1 FROM "User" WHERE "username"='admin')`,
   782→          hashed
   783→        );
   784→        await prisma2.$executeRawUnsafe(
   785→          `INSERT INTO "PoliceAlertConfig" ("id","createdAt","updatedAt")
   786→           SELECT 'default-alert-config',NOW(),NOW()
   787→           WHERE NOT EXISTS (SELECT 1 FROM "PoliceAlertConfig" WHERE "id"='default-alert-config')`
   788→        );
   789→        // Seed default SubscriptionPlans
   790→        await prisma2.$executeRawUnsafe(
   791→          `INSERT INTO "SubscriptionPlan" ("id","name","cycle","price","isActive","createdAt","updatedAt")
   792→           SELECT 'plan-monthly-001','Monthly','MONTHLY',500,true,NOW(),NOW()
   793→           WHERE NOT EXISTS (SELECT 1 FROM "SubscriptionPlan" WHERE "id"='plan-monthly-001')`
   794→        );
   795→        await prisma2.$executeRawUnsafe(
   796→          `INSERT INTO "SubscriptionPlan" ("id","name","cycle","price","isActive","createdAt","updatedAt")
   797→           SELECT 'plan-quarterly-001','Quarterly','QUARTERLY',1400,true,NOW(),NOW()
   798→           WHERE NOT EXISTS (SELECT 1 FROM "SubscriptionPlan" WHERE "id"='plan-quarterly-001')`
   799→        );
   800→        await prisma2.$executeRawUnsafe(
   801→          `INSERT INTO "SubscriptionPlan" ("id","name","cycle","price","isActive","createdAt","updatedAt")
   802→           SELECT 'plan-semi-annual-001','Semi-Annual','SEMI_ANNUAL',2600,true,NOW(),NOW()
   803→           WHERE NOT EXISTS (SELECT 1 FROM "SubscriptionPlan" WHERE "id"='plan-semi-annual-001')`
   804→        );
   805→        await prisma2.$executeRawUnsafe(
   806→          `INSERT INTO "SubscriptionPlan" ("id","name","cycle","price","isActive","createdAt","updatedAt")
   807→           SELECT 'plan-yearly-001','Annual','YEARLY',4800,true,NOW(),NOW()
   808→           WHERE NOT EXISTS (SELECT 1 FROM "SubscriptionPlan" WHERE "id"='plan-yearly-001')`
   809→        );
   810→        console.log("[init-db] SUPERUSER and plans seeded via Prisma.");
   811→      } finally {
   812→        await prisma2.$disconnect().catch(() => {});
   813→      }
   814→
   815→      console.log("[init-db] Database initialization complete via Prisma fallback.");
   816→      _initDone = true;
   817→    } catch (prismaErr) {
   818→      console.error("[init-db] Prisma fallback also failed:", prismaErr instanceof Error ? prismaErr.message : String(prismaErr));
   819→      _initDone = false;
   820→      _initPromise = null; // Allow retry
   821→      throw new Error(
   822→        `Database initialization failed. pg error: ${pgError instanceof Error ? pgError.message : String(pgError)} | Prisma error: ${prismaErr instanceof Error ? prismaErr.message : String(prismaErr)}`
   823→      );
   824→    }
   825→  })();
   826→
   827→  return _initPromise;
   828→}

   830→/**
   831→ * Force-run just the migrations + indexes on the live database.
   832→ * Useful after deploying schema changes to add new columns without full re-init.
   833→ * All migrations use EXCEPTION WHEN duplicate_column so they are safe to re-run.
   834→ */
   835→export async function forceMigrate(): Promise<void> {
   836→  console.log("[forceMigrate] Running migrations on existing database...");
   837→  try {
   838→    await execViaPrisma(MIGRATIONS_SQL);
   839→    console.log("[forceMigrate] Migrations applied.");
   840→  } catch (err) {
   841→    console.error("[forceMigrate] Migration failed:", err instanceof Error ? err.message : String(err));
   842→    throw err;
   843→  }
   844→  try {
   845→    await execViaPrisma(INDEXES_SQL);
   846→    console.log("[forceMigrate] Indexes verified.");
   847→  } catch (err) {
   848→    console.error("[forceMigrate] Index creation failed:", err instanceof Error ? err.message : String(err));
   849→    // Don't throw — indexes are best-effort
   850→  }
   851→  // Reset init flag so next ensureDatabase() will do a full check
   852→  _initDone = false;
   853→  _initPromise = null;
   854→}
