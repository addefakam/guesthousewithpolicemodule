   216→   216→// ── Helper: get nav items based on role ──
   217→   217→function getNavItems(user: CurrentUser): NavItem[] {
   218→   218→  switch (user.role) {
   219→   219→    case "POLICE": {
   220→   220→      const rank = (user.policeRank || "OFFICER") as PoliceRank;
   221→   221→      const allowedPages = POLICE_RANK_PERMISSIONS[rank] || POLICE_RANK_PERMISSIONS.OFFICER;
   222→   222→      return POLICE_NAV_ITEMS.filter((item) => allowedPages.includes(item.page));
   223→   223→    }
   224→   224→
   225→   225→    case "SUPERUSER":
   226→   226→      return SUPERUSER_NAV_ITEMS;
   227→   227→
   228→   228→    case "OPERATOR":
   229→   229→      return ALL_NAV_ITEMS.filter((item) => !OPERATOR_EXCLUDED.has(item.page));
   230→   230→
   231→   231→    case "STAFF": {
   232→   232→      const items: NavItem[] = [];
   233→   233→      // Always give staff access to dashboard
   234→   234→      items.push({
   235→   235→        page: "dashboard",
   236→   236→        label: "Dashboard",
   237→   237→        icon: LayoutDashboard,
   238→   238→      });
   239→   239→      // Add items based on permissions (deduplicate by page key)
   240→   240→      const seen = new Set<string>();
   241→   241→      for (const perm of user.permissions) {
   242→   242→        const mapped = PERMISSION_PAGE_MAP[perm];
   243→   243→        if (mapped && !seen.has(mapped.page)) {
   244→   244→          seen.add(mapped.page);
   245→   245→          items.push(mapped);
   246→   246→        }
   247→   247→      }
   248→   248→      return items;
   249→   249→    }
   250→   250→
   251→   251→    default:
   252→   252→      return [
   253→   253→        { page: "dashboard", label: "Dashboard", icon: LayoutDashboard },
   254→   254→      ];
   255→   255→  }
   256→   256→}

   258→   258→// ── Helper: get initials from name ──
   259→   259→function getInitials(name: string | undefined | null): string {
   260→   260→  if (!name) return "?";
   261→   261→  return name
   262→   262→    .split(" ")
   263→   263→    .map((w) => w[0])
   264→   264→    .join("")
   265→   265→    .toUpperCase()
   266→   266→    .slice(0, 2);
   267→   267→}
