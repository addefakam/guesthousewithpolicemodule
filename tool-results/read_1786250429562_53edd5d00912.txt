    88→const STATUS_CONFIG: Record<string, { label: string; variant: "default" | "secondary" | "destructive" | "outline" }> = {
    89→  PENDING: { label: "Pending", variant: "secondary" },
    90→  APPROVED: { label: "Approved", variant: "default" },

    92→  SUSPENDED: { label: "Suspended", variant: "outline" },
    93→};

    95→const STATUS_BADGE_CLASS: Record<string, string> = {
    96→  PENDING: "bg-yellow-100 text-yellow-800 hover:bg-yellow-100 border-yellow-200",
    97→  APPROVED: "bg-emerald-100 text-emerald-800 hover:bg-emerald-100 border-emerald-200",
    98→  REJECTED: "bg-red-100 text-red-800 hover:bg-red-100 border-red-200",
    99→  SUSPENDED: "bg-orange-100 text-orange-800 hover:bg-orange-100 border-orange-200",
   100→};
   101→
   102→function StatusBadge({ status }: { status: string }) {
   103→  const cfg = STATUS_CONFIG[status] || { label: status, variant: "secondary" as const };
   104→  return (
   105→    <Badge variant={cfg.variant} className={STATUS_BADGE_CLASS[status] || ""}>
   106→      {cfg.label}
   107→    </Badge>
   108→  );
   109→}

   147→export default function ProvidersPage() {
   148→  const { refreshKey, triggerRefresh, currentUser } = useAppStore();
   149→  const isSuperuser = currentUser?.role === "SUPERUSER";
   150→  const [providers, setProviders] = useState<Provider[]>([]);
   151→  const [loading, setLoading] = useState(true);
   152→  const [selectedProvider, setSelectedProvider] = useState<Provider | null>(null);
   153→  const [detailOpen, setDetailOpen] = useState(false);
   154→
   155→  // Registration form
   156→  const [registerOpen, setRegisterOpen] = useState(false);
   157→  const [registerForm, setRegisterForm] = useState<RegisterForm>(emptyRegisterForm);
   158→  const [registering, setRegistering] = useState(false);
   159→
   160→
   161→
   162→  // Action dialogs
   163→  const [rejectDialog, setRejectDialog] = useState<Provider | null>(null);
   164→  const [rejectReason, setRejectReason] = useState("");
   165→  const [confirmAction, setConfirmAction] = useState<{ provider: Provider; action: string } | null>(null);
   166→  const [actioning, setActioning] = useState(false);
   167→
   168→  // Suspension dialog (with reason + notification)
   169→  const [suspendDialog, setSuspendDialog] = useState<Provider | null>(null);
   170→  const [suspensionReason, setSuspensionReason] = useState("");
   171→  const [providerMessage, setProviderMessage] = useState("");
   172→  const [suspending, setSuspending] = useState(false);
   173→
   174→  const pagination = usePagination({ totalItems: providers.length, initialPageSize: 5, pageSizeOptions: [5, 10, 20, 50] });
   175→  const paginatedProviders = useMemo(() => pagination.paginate(providers), [providers, pagination]);
   176→
   177→  const fetchProviders = useCallback(async () => {
   178→    try {
   179→      setLoading(true);
   180→      const data = await apiGetProviders();
   181→      setProviders(Array.isArray(data) ? data : []);
   182→    } catch (err: unknown) {
   183→      const message = err instanceof Error ? err.message : "Failed to load providers";
   184→      toast.error(message);
   185→    } finally {
   186→      setLoading(false);
   187→    }
   188→  }, []);
   189→
   190→  useEffect(() => {
   191→    fetchProviders();
   192→  }, [fetchProviders, refreshKey]);
   193→
   194→  const openDetail = async (provider: Provider) => {
   195→    setSelectedProvider({ ...provider, licenseFile: "" });
   196→    setDetailOpen(true);
   197→    // Lazy-load license file only when viewing details
   198→    try {
   199→      const res = await req(`/api/providers/${provider.id}`);
   200→      if (res && res.licenseFile) {
   201→        setSelectedProvider(prev => prev ? { ...prev, licenseFile: res.licenseFile } : null);
   202→      }
   203→    } catch {
   204→      // License fetch failed — show detail without it
   205→    }
   206→  };
   207→
   208→  const openReject = (provider: Provider) => {
   209→    setRejectDialog(provider);
   210→    setRejectReason("");
   211→  };
   212→
   213→
   214→
   215→  const handleReject = async () => {
   216→    if (!rejectDialog || !rejectReason.trim()) {
   217→      toast.error("Rejection reason is required");
   218→      return;
   219→    }
   220→    try {
   221→      setActioning(true);
   222→      await apiUpdateProvider(rejectDialog.id, { status: "REJECTED", rejectionReason: rejectReason.trim() });
   223→      toast.success("Provider rejected");
   224→      setRejectDialog(null);
   225→      setRejectReason("");
   226→      triggerRefresh();
   227→    } catch (err: unknown) {
   228→      const message = err instanceof Error ? err.message : "Failed to reject provider";
   229→      toast.error(message);
   230→    } finally {
   231→      setActioning(false);
   232→    }
   233→  };
   234→
   235→  const handleStatusAction = async (provider: Provider, status: string) => {
   236→    try {
   237→      setActioning(true);
   238→      if (status === "REACTIVATE") {
   239→        // Police reactivates: set back to APPROVED and clear suspension fields
   240→        await apiUpdateProvider(provider.id, { status: "APPROVED" });
   241→        toast.success(`"${provider.name}" has been reactivated`);
   242→      } else {
   243→        await apiUpdateProvider(provider.id, { status });
   244→        toast.success(`Provider ${status.toLowerCase()}`);
   245→      }
   246→      setConfirmAction(null);
   247→      triggerRefresh();
   248→    } catch (err: unknown) {
   249→      const message = err instanceof Error ? err.message : "Failed to update provider";
   250→      toast.error(message);
   251→    } finally {
   252→      setActioning(false);
   253→    }
   254→  };
   255→
   256→  const openSuspend = (provider: Provider) => {
   257→    setSuspendDialog(provider);
   258→    setSuspensionReason("");
   259→    setProviderMessage("");
   260→  };
   261→
   262→  const handleRegister = async () => {
   263→    if (!registerForm.name.trim() || !registerForm.ownerName.trim() || !registerForm.phone.trim()) {
   264→      toast.error("Guesthouse name, owner name, and phone are required");
   265→      return;
   266→    }
   267→    if (!registerForm.username.trim() || !registerForm.password.trim()) {
   268→      toast.error("Operator username and password are required");
   269→      return;
   270→    }
   271→    if (registerForm.password.trim().length < 4) {
   272→      toast.error("Password must be at least 4 characters");
   273→      return;
   274→    }
   275→    try {
   276→      setRegistering(true);
   277→      await apiSuperCreateProvider({
   278→        ...registerForm,
   279→        licenseFile: registerForm.licenseFileData || undefined,
   280→      });
   281→      toast.success(`"${registerForm.name}" registered and approved successfully`);
   282→      setRegisterOpen(false);
   283→      setRegisterForm(emptyRegisterForm);
   284→      triggerRefresh();
   285→    } catch (err: unknown) {
   286→      toast.error(err instanceof Error ? err.message : "Failed to register guesthouse");
   287→    } finally {
   288→      setRegistering(false);
   289→    }
   290→  };
   291→
   292→  const handleSuspend = async () => {
   293→    if (!suspendDialog) return;
   294→    if (!suspensionReason.trim()) {
   295→      toast.error("Please provide a reason for suspension");
   296→      return;
   297→    }
   298→    try {
   299→      setSuspending(true);
   300→      await apiPoliceSuspendProvider({
   301→        providerId: suspendDialog.id,
   302→        suspensionReason: suspensionReason.trim(),
   303→        providerMessage: providerMessage.trim() || undefined,
   304→      });
   305→      toast.success(`"${suspendDialog.name}" has been suspended. Notification sent to provider.`);
   306→      setSuspendDialog(null);
   307→      setSuspensionReason("");
   308→      setProviderMessage("");
   309→      triggerRefresh();
   310→    } catch (err: unknown) {
   311→      toast.error(err instanceof Error ? err.message : "Failed to suspend provider");
   312→    } finally {
   313→      setSuspending(false);
   314→    }
   315→  };
   316→
   317→  const formatDate = (dateStr: string | null) => {
   318→    if (!dateStr) return "—";
   319→    return new Date(dateStr).toLocaleDateString("en-US", {
   320→      year: "numeric",
   321→      month: "short",
   322→      day: "numeric",
   323→    });
   324→  };
   325→
   326→  const approvedCount = providers.filter((p) => p.status === "APPROVED").length;
   327→  const pendingCount = providers.filter((p) => p.status === "PENDING").length;
   328→
   329→  return (
   330→    <div className="space-y-4 p-3 sm:p-4 md:p-6">
   331→      {/* Summary Cards */}
   332→      <div className="grid grid-cols-2 sm:grid-cols-3 gap-2 sm:gap-3">
   333→        <div className="rounded-xl border bg-card p-3 sm:p-4 shadow-sm">
   334→          <div className="flex items-center gap-2 sm:gap-3">
   335→            <div className="flex h-8 w-8 sm:h-10 sm:w-10 items-center justify-center rounded-lg bg-slate-100">
   336→              <Building2 className="h-4 w-4 sm:h-5 sm:w-5 text-slate-600" />
   337→            </div>
   338→            <div>
   339→              <p className="text-[10px] sm:text-sm text-muted-foreground">Total</p>
   340→              <p className="text-lg sm:text-2xl font-bold">{providers.length}</p>
   341→            </div>
   342→          </div>
   343→        </div>
   344→        <div className="rounded-xl border bg-card p-3 sm:p-4 shadow-sm">
   345→          <div className="flex items-center gap-2 sm:gap-3">
   346→            <div className="flex h-8 w-8 sm:h-10 sm:w-10 items-center justify-center rounded-lg bg-emerald-50">
   347→              <CheckCircle2 className="h-4 w-4 sm:h-5 sm:w-5 text-emerald-600" />
   348→            </div>
   349→            <div>
   350→              <p className="text-[10px] sm:text-sm text-muted-foreground">Approved</p>
   351→              <p className="text-lg sm:text-2xl font-bold text-emerald-600">{approvedCount}</p>
   352→            </div>
   353→          </div>
   354→        </div>
   355→        <div className="rounded-xl border bg-card p-3 sm:p-4 shadow-sm">
   356→          <div className="flex items-center gap-2 sm:gap-3">
   357→            <div className="flex h-8 w-8 sm:h-10 sm:w-10 items-center justify-center rounded-lg bg-yellow-50">
   358→              <ShieldCheck className="h-4 w-4 sm:h-5 sm:w-5 text-yellow-600" />
   359→            </div>
   360→            <div>
   361→              <p className="text-[10px] sm:text-sm text-muted-foreground">Pending</p>
   362→              <p className="text-lg sm:text-2xl font-bold text-yellow-600">{pendingCount}</p>
   363→            </div>
   364→          </div>
   365→        </div>
   366→      </div>
   367→      <div className="flex items-center justify-between">
   368→        <div>
   369→          <h2 className="text-base sm:text-lg font-semibold">
   370→            {isSuperuser ? "Guesthouses" : "Provider Applications"}
   371→          </h2>
   372→          <p className="text-xs sm:text-sm text-muted-foreground">
   373→            {isSuperuser
   374→              ? "Register new guesthouses and manage existing ones."
   375→              : "Manage registrations and licensing"}
   376→          </p>
   377→        </div>
   378→        <div className="flex gap-2">
   379→          {isSuperuser && (
   380→            <Button size="sm" className="gap-1.5" onClick={() => setRegisterOpen(true)}>
   381→              <UserPlus className="h-3.5 w-3.5" /> Register Guesthouse
   382→            </Button>
   383→          )}
   384→        </div>
   385→      </div>
   386→
   387→      {/* Providers Table/Cards */}
   388→      <div className="rounded-xl border bg-card shadow-sm">
   389→        {loading ? (
   390→          <div className="space-y-3 p-4 sm:p-6">
   391→            {Array.from({ length: 5 }).map((_, i) => (
   392→              <Skeleton key={i} className="h-20 w-full sm:h-12" />
   393→            ))}
   394→          </div>
   395→        ) : providers.length === 0 ? (
   396→          <div className="flex flex-col items-center justify-center py-12 sm:py-16 text-center">
   397→            <Building2 className="mb-3 h-10 w-10 sm:h-12 sm:w-12 text-muted-foreground/40" />
   398→            <p className="text-xs sm:text-sm text-muted-foreground">No providers registered yet</p>
   399→          </div>
   400→        ) : (
   401→          <>
   402→            {/* Mobile/Tablet: Card layout */}
   403→            <div className="divide-y lg:hidden">
   404→              {paginatedProviders.map((provider) => (
   405→                <div key={provider.id} className={`p-3 sm:p-4 ${provider.status === "SUSPENDED" ? "bg-orange-50/60" : ""}`}>
   406→                  <div className="flex items-start justify-between gap-2">
   407→                    <button className="min-w-0 flex-1 text-left" onClick={() => openDetail(provider)}>
   408→                      <div className="flex items-center gap-2">
   409→                        <p className="truncate text-sm font-medium">{provider.name}</p>
   410→                        <StatusBadge status={provider.status} />
   411→                      </div>
   412→                      <div className="mt-1.5 flex flex-wrap items-center gap-x-3 gap-y-1 text-xs text-muted-foreground">
   413→                        <span className="flex items-center gap-1"><User className="h-3 w-3" /> {provider.ownerName}</span>
   414→                        <span className="flex items-center gap-1"><Phone className="h-3 w-3" /> {provider.phone}</span>
   415→                      </div>
   416→                      {provider.address && (
   417→                        <div className="mt-1 flex items-center gap-1 text-[10px] text-muted-foreground">
   418→                          <MapPin className="h-3 w-3" /> {provider.address}
   419→                        </div>
   420→                      )}
   421→                      <div className="mt-1 flex items-center gap-2 text-[10px]">
   422→                        <span className="rounded bg-slate-100 px-1.5 py-0.5 font-medium">{provider.type.replace(/_/g, " ")}</span>
   423→                      </div>
   424→                    </button>
   425→                  </div>
   426→
   427→                  <div className="mt-2.5 flex items-center gap-1.5 border-t pt-2.5 flex-wrap">
   428→                    <button className="flex items-center gap-1 rounded-md px-2.5 py-1.5 text-xs font-medium text-slate-600 hover:bg-slate-100" onClick={() => openDetail(provider)}>
   429→                      <Eye className="h-3.5 w-3.5" /> Details
   430→                    </button>
   431→                    {provider.status !== "APPROVED" && provider.status !== "SUSPENDED" && (
   432→                      <button className="flex items-center gap-1 rounded-md px-2.5 py-1.5 text-xs font-medium text-emerald-600 hover:bg-emerald-50" onClick={() => setConfirmAction({ provider, action: "APPROVED" })}>
   433→                        <CheckCircle2 className="h-3.5 w-3.5" /> Approve
   434→                      </button>
   435→                    )}
   436→                    {provider.status === "SUSPENDED" && (
   437→                      <button className="flex items-center gap-1 rounded-md px-2.5 py-1.5 text-xs font-medium text-teal-600 hover:bg-teal-50" onClick={() => setConfirmAction({ provider, action: "REACTIVATE" })}>
   438→                        <RotateCcw className="h-3.5 w-3.5" /> Reactivate
   439→                      </button>
   440→                    )}
   441→                    {provider.status !== "REJECTED" && provider.status !== "APPROVED" && provider.status !== "SUSPENDED" && (
   442→                      <button className="flex items-center gap-1 rounded-md px-2.5 py-1.5 text-xs font-medium text-red-600 hover:bg-red-50" onClick={() => openReject(provider)}>
   443→                        <XCircle className="h-3.5 w-3.5" /> Reject
   444→                      </button>
   445→                    )}
   446→                    {provider.status === "APPROVED" && (
   447→                      <button className="flex items-center gap-1 rounded-md px-2.5 py-1.5 text-xs font-medium text-orange-600 hover:bg-orange-50" onClick={() => openSuspend(provider)}>
   448→                        <Ban className="h-3.5 w-3.5" /> Suspend
   449→                      </button>
   450→                    )}
   451→                  </div>
   452→                </div>
   453→              ))}
   454→            </div>
   455→
   456→            {/* Desktop: Table layout */}
   457→            <div className="hidden lg:block overflow-x-auto">
   458→              <Table>
   459→                <TableHeader>
   460→                  <TableRow>
   461→                    <TableHead>Provider Name</TableHead>
   462→                    <TableHead>Address</TableHead>
   463→                    <TableHead>Owner</TableHead>
   464→                    <TableHead>Phone</TableHead>
   465→                    <TableHead>Status</TableHead>
   466→                    <TableHead className="text-right">Actions</TableHead>
   467→                  </TableRow>
   468→                </TableHeader>
   469→                <TableBody>
   470→                  {paginatedProviders.map((provider) => (
   471→                    <TableRow key={provider.id} className={`cursor-pointer hover:bg-muted/50 ${provider.status === "SUSPENDED" ? "bg-orange-50/70 hover:bg-orange-100/60" : ""}`} onClick={() => openDetail(provider)}>
   472→                      <TableCell className="font-medium">{provider.name}</TableCell>
   473→                      <TableCell className="max-w-[150px] truncate text-xs">{provider.address || "—"}</TableCell>
   474→                      <TableCell>{provider.ownerName}</TableCell>
   475→                      <TableCell>{provider.phone}</TableCell>
   476→                      <TableCell><StatusBadge status={provider.status} /></TableCell>
   477→                      <TableCell className="text-right" onClick={(e) => e.stopPropagation()}>
   478→                        <div className="flex items-center justify-end gap-1">
   479→                          {provider.status !== "APPROVED" && provider.status !== "SUSPENDED" && (
   480→                            <Button size="sm" variant="ghost" className="h-8 text-emerald-600 hover:bg-emerald-50" onClick={() => setConfirmAction({ provider, action: "APPROVED" })}>
   481→                              <CheckCircle2 className="mr-1 h-4 w-4" /> Approve
   482→                            </Button>
   483→                          )}
   484→                          {provider.status !== "REJECTED" && provider.status !== "APPROVED" && provider.status !== "SUSPENDED" && (
   485→                            <Button size="sm" variant="ghost" className="h-8 text-red-600 hover:bg-red-50" onClick={() => openReject(provider)}>
   486→                              <XCircle className="mr-1 h-4 w-4" /> Reject
   487→                            </Button>
   488→                          )}
   489→                          {provider.status === "APPROVED" && (
   490→                            <Button size="sm" variant="ghost" className="h-8 text-orange-600 hover:bg-orange-50" onClick={() => openSuspend(provider)}>
   491→                              <Ban className="mr-1 h-4 w-4" /> Suspend
   492→                            </Button>
   493→                          )}
   494→                          {provider.status === "SUSPENDED" && (
   495→                            <Button size="sm" variant="ghost" className="h-8 text-teal-600 hover:bg-teal-50" onClick={() => setConfirmAction({ provider, action: "REACTIVATE" })}>
   496→                              <RotateCcw className="mr-1 h-4 w-4" /> Reactivate
   497→                            </Button>
   498→                          )}
   499→                        </div>
   500→                      </TableCell>
   501→                    </TableRow>
   502→                  ))}
   503→                </TableBody>
   504→              </Table>
   505→            </div>
   506→          </>
   507→        )}
   508→      </div>
   509→
   510→      <PaginationControls
   511→        currentPage={pagination.currentPage}
   512→        totalPages={pagination.totalPages}
   513→        pageSize={pagination.pageSize}
   514→        pageSizeOptions={pagination.pageSizeOptions}
