   107→   107→// ── Suspicious provider detection ──
   108→   108→interface SuspicionFlag {
   109→   109→  type: "high_maintenance" | "zero_available" | "all_non_available";
   110→   110→  message: string;
   111→   111→  severity: "HIGH" | "MEDIUM";
   112→   112→}
   113→   113→
   114→   114→function detectSuspicion(p: ProviderStats): SuspicionFlag[] {
   115→   115→  const flags: SuspicionFlag[] = [];
   116→   116→
   117→   117→  if (p.total === 0) return flags;
   118→   118→
   119→   119→  // More than 30% rooms in maintenance = suspicious hiding
   120→   120→  const maintPct = (p.maintenance / p.total) * 100;
   121→   121→  if (maintPct >= 30 && p.maintenance > 0) {
   122→   122→    flags.push({
   123→   123→      type: "high_maintenance",
   124→   124→      message: `${p.maintenance}/${p.total} rooms (${Math.round(maintPct)}%) in maintenance — possible room hiding`,
   125→   125→      severity: maintPct >= 50 ? "HIGH" : "MEDIUM",
   126→   126→    });
   127→   127→  }
   128→   128→
   129→   129→  // All rooms occupied/reserved/maintenance with zero available — suspicious during festivals
   130→   130→  if (p.available === 0 && p.total > 0) {
   131→   131→    flags.push({
   132→   132→      type: "zero_available",
   133→   133→      message: "No available rooms — all occupied, reserved, or in maintenance",
   134→   134→      severity: "MEDIUM",
   135→   135→    });
   136→   136→  }
   137→   137→
   138→   138→  return flags;
   139→   139→}

   152→   152→    default: return "bg-slate-100 text-slate-600 border-slate-200";
   153→   153→  }
