   195→function ToggleSwitch({
   196→  id,
   197→  checked,
   198→  onChange,
   199→  disabled = false,
   200→}: {
   201→  id: string;
   202→  checked: boolean;
   203→  onChange: (value: boolean) => void;
   204→  disabled?: boolean;
   205→}) {
   206→  return (
   207→    <button
   208→      id={id}
   209→      type="button"
   210→      role="switch"
   211→      aria-checked={checked}
   212→      disabled={disabled}
   213→      onClick={() => onChange(!checked)}
   214→      className={
   215→        `relative inline-flex h-6 w-11 shrink-0 cursor-pointer items-center rounded-full transition-colors duration-200 ease-in-out focus:outline-none focus-visible:ring-2 focus-visible:ring-primary/40 focus-visible:ring-offset-2 ` +
   216→        (checked
   217→          ? "bg-primary"
   218→          : "bg-slate-300") +
   219→        (disabled ? " opacity-50 cursor-not-allowed" : "")
   220→      }
   221→    >
   222→      <span
   223→        className={
   224→          `pointer-events-none inline-block h-4 w-4 transform rounded-full bg-white shadow-sm ring-0 transition-transform duration-200 ease-in-out ` +
   225→          (checked ? "translate-x-6" : "translate-x-1")
   226→        }
   227→      />
   228→    </button>
   229→  );
   230→}

   283→function NumberInput({
   284→  id,
   285→  value,
   286→  onChange,
   287→  min,
   288→  max,
   289→  step = 1,
   290→  suffix,
   291→  prefix,
   292→  disabled = false,
   293→}: {
   294→  id: string;
   295→  value: number;
   296→  onChange: (v: number) => void;
   297→  min?: number;
   298→  max?: number;
   299→  step?: number;
   300→  suffix?: string;
   301→  prefix?: string;
   302→  disabled?: boolean;
   303→}) {
   304→  return (
   305→    <div className="relative">
   306→      {prefix && (
   307→        <span className="absolute left-3 top-1/2 -translate-y-1/2 text-xs text-slate-500 pointer-events-none font-medium">
   308→          {prefix}
   309→        </span>
   310→      )}
   311→      <input
   312→        id={id}
   313→        type="number"
   314→        value={value}
   315→        min={min}
   316→        max={max}
   317→        step={step}
   318→        disabled={disabled}
   319→        onChange={(e) => onChange(Number(e.target.value))}
   320→        className={inputClass + (suffix ? " pr-10" : "") + (prefix ? " pl-8" : "")}
   321→      />
   322→      {suffix && (
   323→        <span className="absolute right-3 top-1/2 -translate-y-1/2 text-xs text-slate-400 pointer-events-none">
   324→          {suffix}
   325→        </span>
   326→      )}
   327→    </div>
   328→  );
   329→}
