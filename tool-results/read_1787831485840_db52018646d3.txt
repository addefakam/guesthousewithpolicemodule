   207→   207→function SubscriptionBadge() {
   208→   208→  const { t } = useTranslation("rooms");
   209→   209→  const { currentUser, subscription } = useAppStore();
   210→   210→
   211→   211→  // Only show for OPERATOR/STAFF
   212→   212→  if (!currentUser || (currentUser.role !== "OPERATOR" && currentUser.role !== "STAFF")) return null;
   213→   213→  if (!subscription) return null;
   214→   214→
   215→   215→  const isActive = subscription.status === "ACTIVE";
   216→   216→  const isWarning = subscription.status === "WARNING";
   217→   217→  const isExpired = subscription.status === "EXPIRED";
   218→   218→  const isSuspended = subscription.status === "SUSPENDED";
   219→   219→
   220→   220→  const badgeClass = isActive
   221→   221→    ? "border-emerald-300 bg-emerald-50 text-emerald-700"
   222→   222→    : isSuspended || isExpired
   223→   223→    ? "border-rose-300 bg-rose-50 text-rose-700"
   224→   224→    : "border-amber-300 bg-amber-50 text-amber-700";
   225→   225→
   226→   226→  const dotClass = isActive
   227→   227→    ? "bg-emerald-500"
   228→   228→    : isSuspended || isExpired
   229→   229→    ? "bg-rose-500"
   230→   230→    : "bg-amber-500";
   231→   231→
   232→   232→  const label = isActive
   233→   233→    ? t("subActive")
   234→   234→    : isSuspended
   235→   235→    ? t("subSuspended")
   236→   236→    : isExpired
   237→   237→    ? t("subExpired")
   238→   238→    : t("subExpiring");
   239→   239→
   240→   240→  return (
   241→   241→    <div className={`inline-flex items-center gap-1.5 rounded-full border px-2.5 py-1 text-xs font-semibold ${badgeClass} animate-subtle-pulse`}>
   242→   242→      <span className={`relative flex h-2 w-2`}>
   243→   243→        <span className={`absolute inline-flex h-full w-full animate-ping rounded-full opacity-75 ${dotClass}`} />
   244→   244→        <span className={`relative inline-flex h-2 w-2 rounded-full ${dotClass}`} />
   245→   245→      </span>
   246→   246→      <span>{label}</span>
   247→   247→      <span className="font-mono text-[11px] opacity-70">{Math.abs(subscription.daysRemaining)}d</span>
   248→   248→    </div>
   249→   249→  );
   250→   250→}
