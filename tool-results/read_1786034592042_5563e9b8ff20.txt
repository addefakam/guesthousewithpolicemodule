   241→  onToggleActive,

   249→  const roleConf = ROLE_CONFIG[user.role] || ROLE_CONFIG.STAFF;
   250→  const RoleIcon = roleConf.icon;
   251→  let perms: string[] = [];
   252→  try {
   253→    perms = user.permissions ? JSON.parse(user.permissions) : [];
   254→  } catch {
   255→    perms = [];
   256→  }
   257→
   258→  return (
   259→    <div className="group flex items-center gap-4 rounded-xl border border-slate-100 bg-white p-4 transition-all hover:border-slate-200 hover:shadow-sm">
   260→      {/* Avatar */}

   262→        <RoleIcon className={`h-5 w-5 ${roleConf.color}`} />

   272→          </span>

   274→            <Badge
   275→              variant="outline"
   276→              className={`text-[10px] font-semibold px-1.5 py-0 ${
   277→                user.policeRank === "ADMIN"
   278→                  ? "bg-amber-50 text-amber-700 border-amber-200"

   280→                  ? "bg-violet-50 text-violet-700 border-violet-200"
   281→                  : user.policeRank === "OFFICER"

   286→              {user.policeRank}

   290→        <div className="mt-1 flex items-center gap-3 text-xs text-slate-400">

   292→            <UserCircle className="h-3 w-3" />

   299→            </span>
   300→          )}

   302→            <span className="flex items-center gap-1">
   303→              <Phone className="h-3 w-3" />

   310→            <Building2 className="h-3 w-3 text-slate-400" />

   318→                  ? "bg-amber-50 text-amber-600 border-amber-200"

   385→// ── Main Page Component ──
   386→export default function SuperUserManagementPage() {
   387→  const { currentUser, refreshKey } = useAppStore();
   388→  const [users, setUsers] = useState<UserRecord[]>([]);
   389→  const [providers, setProviders] = useState<ProviderOption[]>([]);
   390→  const [stats, setStats] = useState<StatsInfo | null>(null);
   391→  const [pagination, setPagination] = useState<PaginationInfo | null>(null);
   392→  const [loading, setLoading] = useState(true);
   393→  const [search, setSearch] = useState("");
   394→  const [activeRoleTab, setActiveRoleTab] = useState("");
   395→  const [activeProvider, setActiveProvider] = useState("");
   396→  const [currentPage, setCurrentPage] = useState(1);
   397→
   398→  // Dialog states
   399→  const [formOpen, setFormOpen] = useState(false);
   400→  const [viewOpen, setViewOpen] = useState(false);
   401→  const [deleteTarget, setDeleteTarget] = useState<UserRecord | null>(null);
   402→  const [editingUser, setEditingUser] = useState<UserRecord | null>(null);
   403→  const [viewingUser, setViewingUser] = useState<UserRecord | null>(null);
   404→  const [form, setForm] = useState(emptyForm);
   405→  const [saving, setSaving] = useState(false);
   406→  const [deleting, setDeleting] = useState(false);
   407→
   408→  // Only approved providers can be assigned to operator/staff users
   409→  const approvedProviders = providers.filter((p) => p.status === "APPROVED");
   410→
   411→  const fetchData = useCallback(async () => {
   412→    setLoading(true);
   413→    try {
   414→      const [usersData, provDataRaw] = await Promise.all([
   415→        apiSuperGetUsers({
   416→          search: search || undefined,
   417→          role: activeRoleTab || undefined,
   418→          providerId: activeProvider || undefined,
   419→          page: currentPage,
   420→          pageSize: 12,
   421→        }),
   422→        apiSuperGetProviders(),
   423→      ]);
   424→      setUsers(usersData.users || []);
   425→      setPagination(usersData.pagination || null);
   426→      setStats(usersData.stats || null);
   427→      // Providers API returns an array directly
   428→      const provs = Array.isArray(provDataRaw)
   429→        ? provDataRaw
   430→        : (provDataRaw as Record<string, unknown>).providers || [];
   431→      setProviders(provs.map((p: Record<string, unknown>) => ({
   432→        id: p.id as string,
   433→        name: p.name as string,
   434→        status: p.status as string,
   435→      })));
   436→    } catch (err: unknown) {
   437→      const msg = err instanceof Error ? err.message : "Failed to load data";
   438→      toast.error(msg);
   439→    } finally {
   440→      setLoading(false);
   441→    }
   442→  }, [search, activeRoleTab, activeProvider, currentPage]);
   443→
   444→  useEffect(() => {
   445→    fetchData();
   446→  }, [fetchData, refreshKey]);
   447→
   448→  // Reset page when filters change
   449→  useEffect(() => {
   450→    setCurrentPage(1);
   451→  }, [search, activeRoleTab, activeProvider]);
   452→
   453→  // ── Form Handlers ──
   454→  const openCreate = () => {
   455→    setEditingUser(null);
   456→    setForm(emptyForm);
   457→    setFormOpen(true);
   458→  };
   459→
   460→  const openEdit = (user: UserRecord) => {
   461→    setEditingUser(user);
   462→    let perms: string[] = [];
   463→    try {
   464→      perms = user.permissions ? JSON.parse(user.permissions) : [];
   465→    } catch {
   466→      perms = [];
   467→    }
   468→    setForm({
   469→      username: user.username,
   470→      password: "",
   471→      name: user.name,
   472→      email: user.email || "",
   473→      phone: user.phone || "",
   474→      role: user.role,
   475→      policeRank: user.policeRank || "OFFICER",
   476→      permissions: perms,
   477→      providerId: user.providerId || "",
   478→      isActive: user.isActive,
   479→    });
   480→    setFormOpen(true);
   481→  };
   482→
   483→  const openView = (user: UserRecord) => {
   484→    setViewingUser(user);
   485→    setViewOpen(true);
   486→  };
   487→
   488→  const handleSave = async () => {
   489→    if (!form.username.trim() || !form.name.trim() || !form.role) {
   490→      toast.error("Username, name, and role are required");
   491→      return;
   492→    }
   493→    if (!editingUser && !form.password) {
   494→      toast.error("Password is required for new users");
   495→      return;
   496→    }
   497→    // Provider is optional — operator/staff can be created without one
   498→    // and assigned to a guesthouse later from the Providers page.
   499→    // No provider validation needed here.
   500→    if (form.role === "POLICE" && !form.policeRank) {
   501→      toast.error("Please select a police rank");
   502→      return;
   503→    }
   504→
   505→    setSaving(true);
   506→    try {
   507→      const payload: Record<string, unknown> = {
   508→        username: form.username.trim(),
   509→        name: form.name.trim(),
   510→        email: form.email.trim() || null,
   511→        phone: form.phone.trim() || null,
   512→        role: form.role,
   513→        policeRank: form.role === "POLICE" ? form.policeRank : "",
   514→        permissions: form.role === "STAFF" ? form.permissions : [],
   515→        providerId: (form.providerId && form.providerId !== "__none__") ? form.providerId : null,
   516→        isActive: form.isActive,
   517→      };
   518→      if (form.password) payload.password = form.password;
   519→
   520→      if (editingUser) {
   521→        await apiSuperUpdateUser(editingUser.id, payload);
   522→        toast.success("User updated successfully");
   523→      } else {
   524→        await apiSuperCreateUser(payload);
   525→        toast.success("User created successfully");
   526→      }
   527→      setFormOpen(false);
   528→      fetchData();
   529→    } catch (err: unknown) {
   530→      const msg = err instanceof Error ? err.message : "Failed to save user";
   531→      toast.error(msg);
   532→    } finally {
   533→      setSaving(false);
   534→    }
   535→  };
   536→
   537→  const handleDelete = async () => {
   538→    if (!deleteTarget) return;
   539→    setDeleting(true);
   540→    try {
   541→      await apiSuperDeleteUser(deleteTarget.id);
   542→      toast.success("User deleted successfully");
   543→      setDeleteTarget(null);
   544→      fetchData();
   545→    } catch (err: unknown) {
   546→      const msg = err instanceof Error ? err.message : "Failed to delete user";
   547→      toast.error(msg);
   548→    } finally {
   549→      setDeleting(false);
   550→    }
   551→  };
   552→
   553→  const handleToggleActive = async (user: UserRecord) => {
   554→    try {
   555→      await apiSuperUpdateUser(user.id, { isActive: !user.isActive });
   556→      toast.success(`User ${user.isActive ? "deactivated" : "activated"} successfully`);
   557→      fetchData();
   558→    } catch (err: unknown) {
   559→      const msg = err instanceof Error ? err.message : "Failed to update status";
   560→      toast.error(msg);
   561→    }
   562→  };
   563→
   564→  const togglePermission = (perm: string) => {
   565→    setForm((prev) => ({
   566→      ...prev,
   567→      permissions: prev.permissions.includes(perm)
   568→        ? prev.permissions.filter((p) => p !== perm)
   569→        : [...prev.permissions, perm],
   570→    }));
   571→  };
   572→
   573→  return (
   574→    <div className="space-y-6 p-4 md:p-6">
   575→      {/* ── Page Header ── */}
   576→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   577→        <div>
   578→          <h1 className="text-2xl font-bold tracking-tight text-slate-900">User Management</h1>
   579→          <p className="mt-1 text-sm text-slate-500">
   580→            Manage all user accounts, roles, and permissions across the system.
   581→          </p>
   582→        </div>
   583→        <Button onClick={openCreate} className="shadow-sm">
   584→          <UserPlus className="mr-2 h-4 w-4" />
   585→          Add New User
   586→        </Button>
   587→      </div>
   588→
   589→      {/* ── Stats Cards ── */}
   590→      {!loading && stats && (
   591→        <div className="grid grid-cols-2 gap-4 lg:grid-cols-4">
   592→          <StatCard
   593→            icon={Users}
   594→            label="Total Users"
   595→            value={stats.totalUsers}
   596→            sub="Across all providers"
   597→            color="text-indigo-600"
   598→            bgColor="bg-indigo-50"
   599→          />
   600→          <StatCard
   601→            icon={Building2}
   602→            label="Active Providers"
   603→            value={stats.activeProviderCount}
