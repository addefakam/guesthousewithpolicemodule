   145→   145→    CBE_BIRR: t("payMethodCBE_BIRR"),

   159→   159→  };

   168→   168→  }
   169→   169→

   171→   171→  const [loading, setLoading] = useState(true);
   172→   172→  const [showPayDialog, setShowPayDialog] = useState(false);
   173→   173→  const [selectedPlan, setSelectedPlan] = useState<SubData["plans"][0] | null>(null);
   174→   174→  const [submitting, setSubmitting] = useState(false);
   175→   175→  const [payMethod, setPayMethod] = useState("");
   176→   176→  const [payRef, setPayRef] = useState("");
   177→   177→  const [payNotes, setPayNotes] = useState("");

   179→   179→  const [showHistory, setShowHistory] = useState(false);

   182→   182→

   194→   194→  }, []);
   195→   195→
   196→   196→  const fetchData = useCallback(async () => {

   198→   198→      setLoading(true);
   199→   199→      const res = await apiMySubscription();

   201→   201→    } catch (err) {
   202→   202→      const msg = err instanceof Error ? err.message : t("toastFailedLoad");

   205→   205→      setLoading(false);
   206→   206→    }

   209→   209→  useEffect(() => {
   210→   210→    fetchData();

   218→   218→      setChapaVerifyResult(null);
   219→   219→

   230→   230→            setChapaVerifyResult("pending");

   232→   232→          }

   236→   236→        } finally {
   237→   237→          setChapaVerifying(false);
   238→   238→          fetchData();
   239→   239→        }

   241→   241→
   242→   242→      return () => clearTimeout(timer);

   265→   265→        cycle: selectedPlan.cycle,

   272→   272→      } else {
   273→   273→        toast.error(t("toastNoChapaUrl"));

   275→   275→    } catch (e: unknown) {
   276→   276→      toast.error(e instanceof Error ? e.message : t("toastChapaInitFailed"));

   278→   278→      setSubmitting(false);
   279→   279→    }

   314→   314→    }
   315→   315→  };
   316→   316→
   317→   317→  const sub = data?.subscription;

   331→   331→    );
   332→   332→  }
   333→   333→
   334→   334→  const isTrial = sub.price === 0 && data.payments.length === 0;
   335→   335→  const isExpired = sub.status === "EXPIRED";
   336→   336→  const penaltyPercent = data.config.latePaymentPenalty || 10;

   342→   342→

   364→   364→      {/* Pricing Info Banner */}
