   106→  PENDING: "bg-yellow-100 text-yellow-800 hover:bg-yellow-100 border-yellow-200",
   107→  APPROVED: "bg-emerald-100 text-emerald-800 hover:bg-emerald-100 border-emerald-200",

   111→
   112→function StatusBadge({ status }: { status: string }) {
   113→  const cfg = STATUS_CONFIG[status] || { label: status, variant: "secondary" as const };
   114→  return (
   115→    <Badge variant={cfg.variant} className={STATUS_BADGE_CLASS[status] || ""}>
   116→      {cfg.label}
   117→    </Badge>
   118→  );
   119→}

   157→export default function ProvidersPage() {
   158→  const { refreshKey, triggerRefresh, currentUser } = useAppStore();
   159→  const isSuperuser = currentUser?.role === "SUPERUSER";
   160→  const [providers, setProviders] = useState<Provider[]>([]);
   161→  const [loading, setLoading] = useState(true);
   162→  const [selectedProvider, setSelectedProvider] = useState<Provider | null>(null);
   163→  const [detailOpen, setDetailOpen] = useState(false);
   164→
   165→  // Registration form
   166→  const [registerOpen, setRegisterOpen] = useState(false);
   167→  const [registerForm, setRegisterForm] = useState<RegisterForm>(emptyRegisterForm);
   168→  const [registering, setRegistering] = useState(false);
   169→
   170→  // Coordinate editing
   171→  const [coordProvider, setCoordProvider] = useState<Provider | null>(null);
   172→  const [coordOpen, setCoordOpen] = useState(false);
   173→  const [editLat, setEditLat] = useState(9.02);
   174→  const [editLng, setEditLng] = useState(38.75);
   175→  const [savingCoord, setSavingCoord] = useState(false);
   176→  const [geocodingOne, setGeocodingOne] = useState(false);
   177→
   178→  // Batch geocode
   179→  const [geocodingAll, setGeocodingAll] = useState(false);
   180→  const [batchResult, setBatchResult] = useState<{ updated: number; failed: number; total: number; results?: any[] } | null>(null);
   181→  const [batchOpen, setBatchOpen] = useState(false);
   182→
   183→  // Action dialogs
   184→  const [rejectDialog, setRejectDialog] = useState<Provider | null>(null);
   185→  const [rejectReason, setRejectReason] = useState("");
   186→  const [confirmAction, setConfirmAction] = useState<{ provider: Provider; action: string } | null>(null);
   187→  const [actioning, setActioning] = useState(false);
   188→
   189→  // Suspension dialog (with reason + notification)
   190→  const [suspendDialog, setSuspendDialog] = useState<Provider | null>(null);
   191→  const [suspensionReason, setSuspensionReason] = useState("");
   192→  const [providerMessage, setProviderMessage] = useState("");
   193→  const [suspending, setSuspending] = useState(false);
   194→
   195→  const pagination = usePagination({ totalItems: providers.length, initialPageSize: 5, pageSizeOptions: [5, 10, 20, 50] });
   196→  const paginatedProviders = useMemo(() => pagination.paginate(providers), [providers, pagination]);
   197→
   198→  const fetchProviders = useCallback(async () => {
   199→    try {
   200→      setLoading(true);
   201→      const data = await apiGetProviders();
   202→      setProviders(Array.isArray(data) ? data : []);
   203→    } catch (err: unknown) {
   204→      const message = err instanceof Error ? err.message : "Failed to load providers";
   205→      toast.error(message);
   206→    } finally {
   207→      setLoading(false);
   208→    }
   209→  }, []);
   210→
   211→  useEffect(() => {
   212→    fetchProviders();
   213→  }, [fetchProviders, refreshKey]);
   214→
   215→  const openDetail = (provider: Provider) => {
   216→    setSelectedProvider(provider);
   217→    setDetailOpen(true);
   218→  };
   219→
   220→  const openReject = (provider: Provider) => {
   221→    setRejectDialog(provider);
   222→    setRejectReason("");
   223→  };
   224→
   225→  const openCoordPicker = (provider: Provider) => {
   226→    setCoordProvider(provider);
   227→    setEditLat(provider.latitude || 9.02);
   228→    setEditLng(provider.longitude || 38.75);
   229→    setCoordOpen(true);
   230→  };
   231→
   232→  const geocodeSingle = async () => {
   233→    if (!coordProvider || !coordProvider.address) {
   234→      toast.error("Provider has no address to geocode");
   235→      return;
   236→    }
   237→    try {
   238→      setGeocodingOne(true);
   239→      const result: any = await apiGeocodeAddress(coordProvider.address);
   240→      if (result.lat && result.lng) {
   241→        setEditLat(result.lat);
   242→        setEditLng(result.lng);
   243→        toast.success(`Found: ${result.lat.toFixed(4)}, ${result.lng.toFixed(4)}`);
   244→      } else {
   245→        toast.error("Could not find coordinates for this address");
   246→      }
   247→    } catch {
   248→      toast.error("Geocoding failed");
   249→    } finally {
   250→      setGeocodingOne(false);
   251→    }
   252→  };
   253→
   254→  const saveCoordinates = async () => {
   255→    if (!coordProvider) return;
   256→    try {
   257→      setSavingCoord(true);
   258→      await apiUpdateProvider(coordProvider.id, { status: coordProvider.status, latitude: editLat, longitude: editLng });
   259→      toast.success("Coordinates updated");
   260→      setCoordOpen(false);
   261→      triggerRefresh();
   262→    } catch {
   263→      toast.error("Failed to save coordinates");
   264→    } finally {
   265→      setSavingCoord(false);
   266→    }
   267→  };
   268→
   269→  const batchGeocode = async () => {
   270→    try {
   271→      setGeocodingAll(true);
   272→      const result: any = await apiGeocodeBatch();
   273→      setBatchResult(result);
   274→      toast.success(result.message || `Geocoded ${result.updated} providers`);
   275→      triggerRefresh();
   276→    } catch {
   277→      toast.error("Batch geocoding failed");
   278→    } finally {
   279→      setGeocodingAll(false);
   280→    }
   281→  };
   282→
   283→  const handleReject = async () => {
   284→    if (!rejectDialog || !rejectReason.trim()) {
   285→      toast.error("Rejection reason is required");
   286→      return;
   287→    }
   288→    try {
   289→      setActioning(true);
   290→      await apiUpdateProvider(rejectDialog.id, { status: "REJECTED", rejectionReason: rejectReason.trim() });
   291→      toast.success("Provider rejected");
   292→      setRejectDialog(null);
   293→      setRejectReason("");
   294→      triggerRefresh();
   295→    } catch (err: unknown) {
   296→      const message = err instanceof Error ? err.message : "Failed to reject provider";
   297→      toast.error(message);
   298→    } finally {
   299→      setActioning(false);
   300→    }
   301→  };
   302→
   303→  const handleStatusAction = async (provider: Provider, status: string) => {
   304→    try {
   305→      setActioning(true);
   306→      if (status === "REACTIVATE") {
   307→        await req(`/api/providers/${provider.id}/reactivate`, { method: "POST" });
   308→        toast.success(`"${provider.name}" has been reactivated and is fully active`);
   309→      } else {
   310→        await apiUpdateProvider(provider.id, { status });
   311→        toast.success(`Provider ${status.toLowerCase()}`);
   312→      }
   313→      setConfirmAction(null);
   314→      triggerRefresh();
   315→    } catch (err: unknown) {
   316→      const message = err instanceof Error ? err.message : "Failed to update provider";
   317→      toast.error(message);
   318→    } finally {
   319→      setActioning(false);
   320→    }
   321→  };
   322→
   323→  const openSuspend = (provider: Provider) => {
   324→    setSuspendDialog(provider);
   325→    setSuspensionReason("");
   326→    setProviderMessage("");
   327→  };
   328→
   329→  const handleRegister = async () => {
   330→    if (!registerForm.name.trim() || !registerForm.ownerName.trim() || !registerForm.phone.trim()) {
   331→      toast.error("Guesthouse name, owner name, and phone are required");
   332→      return;
   333→    }
   334→    if (!registerForm.username.trim() || !registerForm.password.trim()) {
   335→      toast.error("Operator username and password are required");
   336→      return;
   337→    }
   338→    if (registerForm.password.trim().length < 4) {
   339→      toast.error("Password must be at least 4 characters");
   340→      return;
   341→    }
   342→    try {
   343→      setRegistering(true);
   344→      await apiSuperCreateProvider({
   345→        ...registerForm,
   346→        licenseFile: registerForm.licenseFileData || undefined,
   347→      });
   348→      toast.success(`"${registerForm.name}" registered and approved successfully`);
   349→      setRegisterOpen(false);
   350→      setRegisterForm(emptyRegisterForm);
   351→      triggerRefresh();
   352→    } catch (err: unknown) {
   353→      toast.error(err instanceof Error ? err.message : "Failed to register guesthouse");
   354→    } finally {
   355→      setRegistering(false);
   356→    }
   357→  };
   358→
   359→  const handleSuspend = async () => {
   360→    if (!suspendDialog) return;
   361→    if (!suspensionReason.trim()) {
   362→      toast.error("Please provide a reason for suspension");
   363→      return;
   364→    }
   365→    try {
   366→      setSuspending(true);
   367→      await apiPoliceSuspendProvider({
   368→        providerId: suspendDialog.id,
   369→        suspensionReason: suspensionReason.trim(),
   370→        providerMessage: providerMessage.trim() || undefined,
   371→      });
   372→      toast.success(`"${suspendDialog.name}" has been suspended. Notification sent to provider.`);
   373→      setSuspendDialog(null);
   374→      setSuspensionReason("");
   375→      setProviderMessage("");
   376→      triggerRefresh();
   377→    } catch (err: unknown) {
   378→      toast.error(err instanceof Error ? err.message : "Failed to suspend provider");
   379→    } finally {
   380→      setSuspending(false);
   381→    }
   382→  };
   383→
   384→  const formatDate = (dateStr: string | null) => {
   385→    if (!dateStr) return "—";
   386→    return new Date(dateStr).toLocaleDateString("en-US", {
   387→      year: "numeric",
   388→      month: "short",
   389→      day: "numeric",
   390→    });
   391→  };
   392→
   393→  const approvedCount = providers.filter((p) => p.status === "APPROVED").length;
   394→  const pendingCount = providers.filter((p) => p.status === "PENDING").length;
   395→  const withCoords = providers.filter(
   396→    (p) => p.latitude !== 9.02 || p.longitude !== 38.75,
   397→  ).length;
   398→
   399→  return (
   400→    <div className="space-y-4 p-3 sm:p-4 md:p-6">
   401→      {/* Summary Cards */}
   402→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-2 sm:gap-3">
   403→        <div className="rounded-xl border bg-card p-3 sm:p-4 shadow-sm">
   404→          <div className="flex items-center gap-2 sm:gap-3">
   405→            <div className="flex h-8 w-8 sm:h-10 sm:w-10 items-center justify-center rounded-lg bg-slate-100">
   406→              <Building2 className="h-4 w-4 sm:h-5 sm:w-5 text-slate-600" />
   407→            </div>
   408→            <div>
   409→              <p className="text-[10px] sm:text-sm text-muted-foreground">Total</p>
   410→              <p className="text-lg sm:text-2xl font-bold">{providers.length}</p>
   411→            </div>
   412→          </div>
   413→        </div>
   414→        <div className="rounded-xl border bg-card p-3 sm:p-4 shadow-sm">
   415→          <div className="flex items-center gap-2 sm:gap-3">
   416→            <div className="flex h-8 w-8 sm:h-10 sm:w-10 items-center justify-center rounded-lg bg-emerald-50">
   417→              <CheckCircle2 className="h-4 w-4 sm:h-5 sm:w-5 text-emerald-600" />
   418→            </div>
   419→            <div>
   420→              <p className="text-[10px] sm:text-sm text-muted-foreground">Approved</p>
   421→              <p className="text-lg sm:text-2xl font-bold text-emerald-600">{approvedCount}</p>
   422→            </div>
   423→          </div>
   424→        </div>
   425→        <div className="rounded-xl border bg-card p-3 sm:p-4 shadow-sm">
   426→          <div className="flex items-center gap-2 sm:gap-3">
   427→            <div className="flex h-8 w-8 sm:h-10 sm:w-10 items-center justify-center rounded-lg bg-yellow-50">
   428→              <ShieldCheck className="h-4 w-4 sm:h-5 sm:w-5 text-yellow-600" />
   429→            </div>
   430→            <div>
   431→              <p className="text-[10px] sm:text-sm text-muted-foreground">Pending</p>
   432→              <p className="text-lg sm:text-2xl font-bold text-yellow-600">{pendingCount}</p>
   433→            </div>
   434→          </div>
   435→        </div>
   436→        <div className="rounded-xl border bg-card p-3 sm:p-4 shadow-sm">
   437→          <div className="flex items-center gap-2 sm:gap-3">
   438→            <div className="flex h-8 w-8 sm:h-10 sm:w-10 items-center justify-center rounded-lg bg-blue-50">
   439→              <MapPinned className="h-4 w-4 sm:h-5 sm:w-5 text-blue-600" />
   440→            </div>
   441→            <div>
   442→              <p className="text-[10px] sm:text-sm text-muted-foreground">With Coords</p>

   445→          </div>
   446→        </div>
   447→      </div>
   448→
   449→      {/* Geocode All Button */}
   450→      <div className="flex items-center justify-between">

   453→            {isSuperuser ? "Guesthouses" : "Provider Applications"}
   454→          </h2>
   455→          <p className="text-xs sm:text-sm text-muted-foreground">
   456→            {isSuperuser
   457→              ? "Register new guesthouses and manage existing ones."
   458→              : "Manage registrations, licensing, and map locations"}
   459→          </p>
   460→        </div>
   461→        <div className="flex gap-2">
   462→          {isSuperuser && (
   463→            <Button size="sm" className="gap-1.5" onClick={() => setRegisterOpen(true)}>
   464→              <UserPlus className="h-3.5 w-3.5" /> Register Guesthouse
   465→            </Button>
   466→          )}
   467→          {!isSuperuser && (
   468→            <Button variant="outline" size="sm" onClick={() => setBatchOpen(true)} disabled={geocodingAll}>
   469→              <Globe className={`mr-1 h-3.5 w-3.5 ${geocodingAll ? "animate-spin" : ""}`} />
   470→              {geocodingAll ? "Geocoding..." : "Geocode All"}

   472→          )}
   473→        </div>
   474→      </div>
   475→
   476→      {/* Providers Table/Cards */}
   477→      <div className="rounded-xl border bg-card shadow-sm">
   478→        {loading ? (
   479→          <div className="space-y-3 p-4 sm:p-6">
   480→            {Array.from({ length: 5 }).map((_, i) => (
   481→              <Skeleton key={i} className="h-20 w-full sm:h-12" />
   482→            ))}
   483→          </div>
   484→        ) : providers.length === 0 ? (
   485→          <div className="flex flex-col items-center justify-center py-12 sm:py-16 text-center">
   486→            <Building2 className="mb-3 h-10 w-10 sm:h-12 sm:w-12 text-muted-foreground/40" />
   487→            <p className="text-xs sm:text-sm text-muted-foreground">No providers registered yet</p>
   488→          </div>
   489→        ) : (
   490→          <>
   491→            {/* Mobile/Tablet: Card layout */}
   492→            <div className="divide-y lg:hidden">
   493→              {paginatedProviders.map((provider) => (
   494→                <div key={provider.id} className={`p-3 sm:p-4 ${provider.status === "SUSPENDED" ? "bg-orange-50/60" : ""}`}>
   495→                  <div className="flex items-start justify-between gap-2">
   496→                    <button className="min-w-0 flex-1 text-left" onClick={() => openDetail(provider)}>
   497→                      <div className="flex items-center gap-2">
   498→                        <p className="truncate text-sm font-medium">{provider.name}</p>
   499→                        <StatusBadge status={provider.status} />
   500→                      </div>
   501→                      <div className="mt-1.5 flex flex-wrap items-center gap-x-3 gap-y-1 text-xs text-muted-foreground">
   502→                        <span className="flex items-center gap-1"><User className="h-3 w-3" /> {provider.ownerName}</span>
   503→                        <span className="flex items-center gap-1"><Phone className="h-3 w-3" /> {provider.phone}</span>
   504→                      </div>
   505→                      {provider.address && (
   506→                        <div className="mt-1 flex items-center gap-1 text-[10px] text-muted-foreground">
   507→                          <MapPin className="h-3 w-3" /> {provider.address}
   508→                        </div>
   509→                      )}
   510→                      <div className="mt-1 flex items-center gap-2 text-[10px]">
   511→                        <span className="rounded bg-slate-100 px-1.5 py-0.5 font-medium">{provider.type.replace(/_/g, " ")}</span>
   512→                        {(provider.latitude !== 9.02 || provider.longitude !== 38.75) ? (
   513→                          <span className="flex items-center gap-0.5 text-emerald-600"><Crosshair className="h-2.5 w-2.5" /> {provider.latitude.toFixed(4)}, {provider.longitude.toFixed(4)}</span>
   514→                        ) : (
   515→                          <span className="text-muted-foreground">No coordinates</span>
   516→                        )}
   517→                      </div>
   518→                    </button>
   519→                  </div>
   520→
   521→                  <div className="mt-2.5 flex items-center gap-1.5 border-t pt-2.5 flex-wrap">
   522→                    <button className="flex items-center gap-1 rounded-md px-2.5 py-1.5 text-xs font-medium text-slate-600 hover:bg-slate-100" onClick={() => openDetail(provider)}>
   523→                      <Eye className="h-3.5 w-3.5" /> Details
   524→                    </button>
   525→                    {!isSuperuser && (
   526→                      <button className="flex items-center gap-1 rounded-md px-2.5 py-1.5 text-xs font-medium text-blue-600 hover:bg-blue-50" onClick={() => openCoordPicker(provider)}>
