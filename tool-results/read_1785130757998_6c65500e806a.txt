    92→const STATUS_BADGE: Record<string, string> = {
    93→  APPROVED: "bg-emerald-100 text-emerald-700 border-emerald-200",
    94→  PENDING: "bg-amber-100 text-amber-700 border-amber-200",
    95→  REJECTED: "bg-red-100 text-red-700 border-red-200",
    96→  SUSPENDED: "bg-slate-200 text-slate-700 border-slate-300",
    97→};
    98→
    99→const STATUS_ICON: Record<string, React.ReactNode> = {
   100→  APPROVED: <CheckCircle2 className="size-4 text-emerald-600" />,
   101→  PENDING: <Clock className="size-4 text-amber-600" />,
   102→  REJECTED: <Ban className="size-4 text-red-600" />,
   103→  SUSPENDED: <AlertTriangle className="size-4 text-slate-600" />,
   104→};
   105→
   106→const POLICE_RANK_BADGE: Record<string, string> = {
   107→  ADMIN: "bg-amber-100 text-amber-800 border-amber-200",
   108→  DETECTIVE: "bg-violet-100 text-violet-800 border-violet-200",
   109→  OFFICER: "bg-sky-100 text-sky-800 border-sky-200",
   110→  VIEWER: "bg-slate-100 text-slate-600 border-slate-200",
   111→};
   112→
   113→const POLICE_RANK_LABEL: Record<string, string> = {
   114→  ADMIN: "Police Admin",
   115→  DETECTIVE: "Detective",
   116→  OFFICER: "Officer",
   117→  VIEWER: "Viewer",
   118→};
   119→
   120→const POLICE_RANK_DESC: Record<string, string> = {
   121→  ADMIN: "Full management rights. Can create/edit/delete all police accounts below ADMIN.",
   122→  DETECTIVE: "Can create Officer & Viewer accounts. Can edit/delete officers below Detective rank.",
   123→  OFFICER: "Can create Viewer accounts. Can edit/delete viewers.",
   124→  VIEWER: "Basic read-only access. Cannot create or manage other police.",
   125→};
   126→
   127→// Rank hierarchy for permission check
   128→const RANK_ORDER: Record<string, number> = { VIEWER: 0, OFFICER: 1, DETECTIVE: 2, ADMIN: 3 };
   129→
   130→type TabType = "overview" | "owners" | "police";
   131→
   132→export default function OwnerAccountsPage() {
   133→  const { currentUser, refreshKey } = useAppStore();
   134→  const [providers, setProviders] = useState<ProviderWithOwner[]>([]);
   135→  const [policeUsers, setPoliceUsers] = useState<AccountUser[]>([]);
   136→  const [loading, setLoading] = useState(true);
   137→  const [search, setSearch] = useState("");
   138→  const [activeTab, setActiveTab] = useState<TabType>(
   139→    currentUser?.role === "POLICE" ? "police" : "overview"
   140→  );
   141→
   142→  // Determine if current user can manage police
   143→  const canManagePolice =
   144→    currentUser?.role === "SUPERUSER" ||
   145→    (currentUser?.role === "POLICE" &&
   146→      (RANK_ORDER[currentUser?.policeRank || ""] || 0) >= RANK_ORDER.OFFICER);
   147→
   148→  // Which ranks can this user create? (one level below their own)
   149→  const maxCreatableRank =
   150→    currentUser?.role === "SUPERUSER"
   151→      ? "ADMIN"
   152→      : currentUser?.role === "POLICE"
   153→        ? currentUser?.policeRank === "ADMIN"
   154→          ? "DETECTIVE"
   155→          : currentUser?.policeRank === "DETECTIVE"
   156→            ? "OFFICER"
   157→            : currentUser?.policeRank === "OFFICER"
   158→              ? "VIEWER"
   159→              : ""
   160→        : "";
   161→
   162→  // Ranks available in the dropdown for the Add/Edit dialog
   163→  const availableRanks = maxCreatableRank
   164→    ? Object.keys(RANK_ORDER).filter((r) => (RANK_ORDER[r] || 0) <= (RANK_ORDER[maxCreatableRank] || 0))
   165→    : [];
   166→
   167→  // ── Reset credentials dialog ──
   168→  const [resetOpen, setResetOpen] = useState(false);
   169→  const [resetUserId, setResetUserId] = useState("");
   170→  const [resetUsername, setResetUsername] = useState("");
   171→  const [resetPassword, setResetPassword] = useState("");
   172→  const [showPassword, setShowPassword] = useState(false);
   173→  const [saving, setSaving] = useState(false);
   174→  const [resetLabel, setResetLabel] = useState("");
   175→  const [resetSublabel, setResetSublabel] = useState("");
   176→
   177→  // ── Add/Edit police dialog ──
   178→  const [policeDialogOpen, setPoliceDialogOpen] = useState(false);
   179→  const [editMode, setEditMode] = useState(false);
   180→  const [editPoliceId, setEditPoliceId] = useState("");
   181→  const [policeName, setPoliceName] = useState("");
   182→  const [policeUsername, setPoliceUsername] = useState("");
   183→  const [policePassword, setPolicePassword] = useState("");
   184→  const [policeRank, setPoliceRank] = useState("OFFICER");
   185→  const [policeSaving, setPoliceSaving] = useState(false);
   186→  const [showPolicePassword, setShowPolicePassword] = useState(false);
   187→
   188→  // ── Delete confirmation ──
   189→  const [deleteOpen, setDeleteOpen] = useState(false);
   190→  const [deleteTarget, setDeleteTarget] = useState<AccountUser | null>(null);
   191→  const [deleting, setDeleting] = useState(false);
   192→
   193→  const fetchAccounts = useCallback(async () => {
   194→    setLoading(true);
   195→    try {
   196→      if (currentUser?.role === "POLICE") {
   197→        // POLICE users fetch from police-officers endpoint
   198→        const data = await apiPoliceOfficers();
   199→        setPoliceUsers(data);
   200→      } else {
   201→        const data: ApiResponse = await apiGetOwnerAccounts();
   202→        setProviders(data.providers);
   203→        setPoliceUsers(data.policeUsers);
   204→      }
   205→    } catch {
   206→      toast.error("Failed to load accounts");
   207→    } finally {
   208→      setLoading(false);
   209→    }
   210→  }, [currentUser?.role]);
   211→
   212→  const fetchPoliceOnly = useCallback(async () => {
   213→    try {
   214→      const data = await apiPoliceOfficers();
   215→      setPoliceUsers(data);
   216→    } catch {
   217→      // silent
   218→    }
   219→  }, []);
   220→
   221→  useEffect(() => {
   222→    fetchAccounts();
   223→  }, [fetchAccounts, refreshKey]);
   224→
   225→  // ── Owner reset handlers ──
   226→  const openOwnerReset = (provider: ProviderWithOwner) => {
   227→    const ownerUser = provider.users[0];
   228→    setResetUserId(ownerUser?.id || "");
   229→    setResetUsername(ownerUser?.username || "");
   230→    setResetPassword("");
   231→    setShowPassword(false);
   232→    setResetLabel(provider.name);
   233→    setResetSublabel(`${provider.ownerName}  ·  ${provider.phone}`);
   234→    setResetOpen(true);
   235→  };
   236→
   237→  const openPoliceReset = (police: AccountUser) => {
   238→    setResetUserId(police.id);
   239→    setResetUsername(police.username);
   240→    setResetPassword("");
   241→    setShowPassword(false);
   242→    setResetLabel("Police Account");
   243→    setResetSublabel(police.name);
   244→    setResetOpen(true);
   245→  };
   246→
   247→  const handleReset = async (e: FormEvent) => {
   248→    e.preventDefault();
   249→    if (!resetUserId) return;
   250→    if (!resetUsername.trim()) {
   251→      toast.error("Username is required");
   252→      return;
   253→    }
   254→    setSaving(true);
   255→    try {
   256→      const updateData: Record<string, unknown> = {};
   257→      if (resetUsername.trim()) updateData.username = resetUsername.trim();
   258→      if (resetPassword.trim()) updateData.password = resetPassword.trim();
   259→      if (Object.keys(updateData).length === 0) {
   260→        toast.error("No changes to save");
   261→        setSaving(false);
   262→        return;
   263→      }
   264→      await apiUpdateOwnerAccount(resetUserId, updateData);
   265→      toast.success("Credentials updated successfully");
   266→      setResetOpen(false);
   267→      fetchAccounts();
   268→    } catch (err: unknown) {
   269→      const msg = err instanceof Error ? err.message : "Failed to update credentials";
   270→      toast.error(msg);
   271→    } finally {
   272→      setSaving(false);
   273→    }
   274→  };
   275→
   276→  // ── Police CRUD handlers ──
   277→  const openAddPolice = () => {
   278→    setEditMode(false);
   279→    setEditPoliceId("");
   280→    setPoliceName("");
   281→    setPoliceUsername("");
   282→    setPolicePassword("");
   283→    setPoliceRank(
   284→      maxCreatableRank === "DETECTIVE"
   285→        ? "OFFICER"
   286→        : maxCreatableRank === "OFFICER"
   287→          ? "VIEWER"
   288→          : availableRanks[0] || "VIEWER"
   289→    );
   290→    setShowPolicePassword(false);
   291→    setPoliceDialogOpen(true);
   292→  };
   293→
   294→  const openEditPolice = (police: AccountUser) => {
   295→    setEditMode(true);
   296→    setEditPoliceId(police.id);
   297→    setPoliceName(police.name);
   298→    setPoliceUsername(police.username);
   299→    setPolicePassword("");
   300→    setPoliceRank(police.policeRank || "VIEWER");
   301→    setShowPolicePassword(false);
   302→    setPoliceDialogOpen(true);
   303→  };
   304→
   305→  const handlePoliceSubmit = async (e: FormEvent) => {
   306→    e.preventDefault();
   307→    if (!policeName.trim() || !policeUsername.trim()) {
   308→      toast.error("Name and username are required");
   309→      return;
   310→    }
   311→    if (!editMode && !policePassword.trim()) {
   312→      toast.error("Password is required for new accounts");
   313→      return;
   314→    }
   315→
   316→    setPoliceSaving(true);
   317→    try {
   318→      if (editMode) {
   319→        // Update existing officer
   320→        const updateData: Record<string, unknown> = {
   321→          id: editPoliceId,
   322→          name: policeName.trim(),
   323→          policeRank,
   324→        };
   325→        if (policePassword.trim()) {
   326→          updateData.password = policePassword.trim();
   327→        }
   328→        await apiPoliceUpdateOfficer(updateData);
   329→        toast.success("Officer updated successfully");
   330→      } else {
   331→        // Create new officer
   332→        await apiPoliceCreateOfficer({
   333→          username: policeUsername.trim(),
   334→          password: policePassword.trim(),
   335→          name: policeName.trim(),
   336→          policeRank,
   337→        });
   338→        toast.success("Officer created successfully");
   339→      }
   340→      setPoliceDialogOpen(false);
   341→      fetchPoliceOnly();
   342→      if (currentUser?.role !== "POLICE") fetchAccounts();
   343→    } catch (err: unknown) {
   344→      const msg = err instanceof Error ? err.message : "Failed to save officer";
   345→      toast.error(msg);
   346→    } finally {
   347→      setPoliceSaving(false);
   348→    }
   349→  };
   350→
   351→  const openDeleteConfirm = (police: AccountUser) => {
   352→    setDeleteTarget(police);
   353→    setDeleteOpen(true);
   354→  };
   355→
   356→  const handleDelete = async () => {
   357→    if (!deleteTarget) return;
   358→    setDeleting(true);
   359→    try {
   360→      await apiPoliceDeleteOfficer(deleteTarget.id);
   361→      toast.success("Officer deleted successfully");
   362→      setDeleteOpen(false);
   363→      setDeleteTarget(null);
   364→      fetchPoliceOnly();
   365→      if (currentUser?.role !== "POLICE") fetchAccounts();
   366→    } catch (err: unknown) {
   367→      const msg = err instanceof Error ? err.message : "Failed to delete officer";
   368→      toast.error(msg);
   369→    } finally {
   370→      setDeleting(false);
   371→    }
   372→  };
   373→
   374→  // ── Filtering ──
   375→  const filteredProviders = providers.filter(
   376→    (p) =>
   377→      p.name.toLowerCase().includes(search.toLowerCase()) ||
   378→      p.ownerName.toLowerCase().includes(search.toLowerCase()) ||
   379→      p.phone.includes(search) ||
   380→      p.users.some((u) => u.username.toLowerCase().includes(search.toLowerCase()))
   381→  );
   382→
   383→  const filteredPolice = policeUsers.filter(
   384→    (u) =>
   385→      u.name.toLowerCase().includes(search.toLowerCase()) ||
   386→      u.username.toLowerCase().includes(search.toLowerCase())
   387→  );
   388→
   389→  const hasNoResults =
   390→    (activeTab === "overview" && filteredProviders.length === 0 && filteredPolice.length === 0) ||
   391→    (activeTab === "owners" && filteredProviders.length === 0) ||
   392→    (activeTab === "police" && filteredPolice.length === 0);
   393→
   394→  const totalProviders = providers.length;
   395→  const approvedProviders = providers.filter((p) => p.status === "APPROVED").length;
   396→  const pendingProviders = providers.filter((p) => p.status === "PENDING").length;
   397→  const totalPolice = policeUsers.length;
   398→
   399→  if (loading) {
   400→    return (
   401→      <div className="space-y-6 p-4 md:p-6">
   402→        <Skeleton className="h-8 w-64" />
   403→        <Skeleton className="h-10 w-full max-w-sm" />
   404→        <div className="space-y-3">
   405→          {Array.from({ length: 4 }).map((_, i) => (
   406→            <Skeleton key={i} className="h-24 rounded-lg" />
   407→          ))}
   408→        </div>
   409→      </div>
   410→    );
   411→  }
   412→
   413→  // ── Render Police Table Row ──
   414→  const renderPoliceRow = (police: AccountUser, isMobile = false) => {
   415→    const rank = police.policeRank || "OFFICER";
   416→    const isSelf = police.id === currentUser?.id;
   417→
   418→    if (isMobile) {
   419→      return (
   420→        <div key={police.id} className="rounded-lg border p-4 space-y-3">
   421→          <div className="flex items-center gap-2">
   422→            <div className="flex h-9 w-9 shrink-0 items-center justify-center rounded-lg bg-rose-100">
   423→              <Shield className="h-4 w-4 text-rose-600" />
   424→            </div>
   425→            <div className="min-w-0 flex-1">
   426→              <div className="flex items-center gap-2">
   427→                <p className="font-semibold text-sm truncate">{police.name}</p>
   428→                {rank !== "OFFICER" && (
   429→                  <Badge variant="outline" className={`text-[10px] px-1.5 ${POLICE_RANK_BADGE[rank] || ""}`}>
   430→                    {POLICE_RANK_LABEL[rank] || rank}
   431→                  </Badge>
   432→                )}
   433→                {isSelf && (
   434→                  <Badge variant="secondary" className="text-[10px] px-1.5">You</Badge>
   435→                )}
   436→              </div>
   437→              <code className="rounded bg-muted px-1.5 py-0.5 text-xs font-mono">
   438→                {police.username}
   439→              </code>
   440→            </div>
   441→          </div>
   442→
   443→          <p className="text-xs text-muted-foreground">
   444→            Created: {new Date(police.createdAt).toLocaleDateString()}
   445→          </p>
   446→
   447→          <div className="flex gap-2">
   448→            {canManagePolice && !isSelf && (
   449→              <>
   450→                <Button variant="outline" size="sm" className="flex-1 gap-1.5" onClick={() => openEditPolice(police)}>
   451→                  <Pencil className="h-3.5 w-3.5" />
   452→                  Edit
   453→                </Button>
   454→                <Button variant="outline" size="sm" className="gap-1.5 text-red-600 hover:text-red-700" onClick={() => openDeleteConfirm(police)}>
   455→                  <Trash2 className="h-3.5 w-3.5" />
   456→                </Button>
   457→              </>
   458→            )}
   459→            <Button variant="outline" size="sm" className="flex-1 gap-1.5" onClick={() => openPoliceReset(police)}>
   460→              <KeyRound className="h-3.5 w-3.5" />
   461→              Reset
   462→            </Button>
   463→          </div>
   464→        </div>
   465→      );
   466→    }
   467→
   468→    // Desktop row
   469→    return (
   470→      <tr key={police.id} className="border-b last:border-0 hover:bg-muted/30 transition-colors">
   471→        <td className="px-4 py-3">
   472→          <div className="flex items-center gap-2">
   473→            <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-rose-100">
   474→              <Shield className="h-4 w-4 text-rose-600" />
   475→            </div>
   476→            <span className="font-medium">{police.name}</span>
   477→            {isSelf && (
   478→              <Badge variant="secondary" className="text-[10px] px-1.5">You</Badge>
   479→            )}
   480→          </div>
   481→        </td>
   482→        <td className="px-4 py-3">
   483→          <code className="rounded bg-muted px-1.5 py-0.5 text-xs font-mono">
   484→            {police.username}
   485→          </code>
   486→        </td>
   487→        <td className="px-4 py-3">
   488→          <Badge variant="outline" className={`text-xs ${POLICE_RANK_BADGE[rank] || "bg-rose-100 text-rose-700 border-rose-200"}`}>
   489→            {POLICE_RANK_LABEL[rank] || rank}
   490→          </Badge>
   491→        </td>
   492→        <td className="px-4 py-3 text-muted-foreground text-xs">
   493→          {new Date(police.createdAt).toLocaleDateString()}
   494→        </td>
   495→        <td className="px-4 py-3">
   496→          <div className="flex items-center justify-end gap-1.5">
   497→            {canManagePolice && !isSelf && (
   498→              <>
   499→                <Button variant="outline" size="sm" className="gap-1.5" onClick={() => openEditPolice(police)}>
   500→                  <Pencil className="h-3.5 w-3.5" />
   501→                </Button>
   502→                <Button variant="outline" size="sm" className="gap-1.5 text-red-600 hover:text-red-700" onClick={() => openDeleteConfirm(police)}>
   503→                  <Trash2 className="h-3.5 w-3.5" />
   504→                </Button>
   505→              </>
   506→            )}
   507→            <Button variant="outline" size="sm" className="gap-1.5" onClick={() => openPoliceReset(police)}>
   508→              <KeyRound className="h-3.5 w-3.5" />
   509→              Reset
   510→            </Button>
   511→          </div>
   512→        </td>
   513→      </tr>
   514→    );
   515→  };
   516→
   517→  return (
   518→    <div className="space-y-6 p-4 md:p-6">
   519→      {/* Header */}
   520→      <div>
   521→        <h1 className="text-2xl font-bold tracking-tight">
   522→          {currentUser?.role === "POLICE" ? "Manage Officers" : "Account Management"}
   523→        </h1>
   524→        <p className="text-sm text-muted-foreground">
   525→          {currentUser?.role === "POLICE"
   526→            ? "Create and manage police officers. You can manage ranks below your current rank."
   527→            : "Manage owner accounts and police officers."}
   528→        </p>
   529→      </div>
   530→
   531→      {/* Tabs — POLICE users only see the police tab */}
   532→      {currentUser?.role !== "POLICE" && (
   533→        <div className="flex gap-1 rounded-lg border bg-muted/50 p-1 w-fit">
   534→          {([
   535→            { key: "overview" as TabType, label: "Overview", icon: Users, count: totalProviders + totalPolice },
   536→            { key: "owners" as TabType, label: "Owners", icon: Building2, count: totalProviders },
   537→            { key: "police" as TabType, label: "Police", icon: Shield, count: totalPolice },
   538→          ]).map((tab) => (
   539→            <button
   540→              key={tab.key}
   541→              onClick={() => { setActiveTab(tab.key); setSearch(""); }}
   542→              className={`flex items-center gap-2 rounded-md px-4 py-2 text-sm font-medium transition-colors ${
   543→                activeTab === tab.key
   544→                  ? "bg-background text-foreground shadow-sm"
   545→                  : "text-muted-foreground hover:text-foreground"
   546→              }`}
   547→            >
   548→              <tab.icon className="h-4 w-4" />
   549→              {tab.label}
   550→              <Badge variant="secondary" className="ml-1 h-5 min-w-[20px] px-1.5 text-[10px]">
   551→                {tab.count}
   552→              </Badge>
   553→            </button>
   554→          ))}
   555→        </div>
   556→      )}
   557→
   558→      {/* Search + Add Officer */}
   559→      <div className="flex gap-2 max-w-lg">
   560→        <div className="relative flex-1 max-w-sm">
   561→          <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   562→          <Input
   563→            placeholder={
   564→              activeTab === "police" || currentUser?.role === "POLICE"
   565→                ? "Search by name or username..."
