   193→// ── Helper: get nav items based on role ──
   194→function getNavItems(user: CurrentUser): NavItem[] {
   195→  switch (user.role) {
   196→    case "POLICE": {
   197→      const rank = (user.policeRank || "OFFICER") as PoliceRank;
   198→      const allowedPages = POLICE_RANK_PERMISSIONS[rank] || POLICE_RANK_PERMISSIONS.OFFICER;
   199→      return POLICE_NAV_ITEMS.filter((item) => allowedPages.includes(item.page));
   200→    }
   201→
   202→    case "SUPERUSER":
   203→      return SUPERUSER_NAV_ITEMS;
   204→
   205→    case "OPERATOR":
   206→      return ALL_NAV_ITEMS.filter((item) => !OPERATOR_EXCLUDED.has(item.page));
   207→
   208→    case "STAFF": {
   209→      const items: NavItem[] = [];
   210→      // Always give staff access to dashboard
   211→      items.push({
   212→        page: "dashboard",
   213→        label: "Dashboard",
   214→        icon: LayoutDashboard,
   215→      });
   216→      // Add items based on permissions (deduplicate by page key)
   217→      const seen = new Set<string>();
   218→      for (const perm of user.permissions) {
   219→        const mapped = PERMISSION_PAGE_MAP[perm];
   220→        if (mapped && !seen.has(mapped.page)) {
   221→          seen.add(mapped.page);
   222→          items.push(mapped);
   223→        }
   224→      }
   225→      return items;
   226→    }
   227→
   228→    default:
   229→      return [
   230→        { page: "dashboard", label: "Dashboard", icon: LayoutDashboard },
   231→      ];
   232→  }
   233→}

   235→// ── Helper: get initials from name ──
   236→function getInitials(name: string): string {
   237→  return name
   238→    .split(" ")
   239→    .map((w) => w[0])
   240→    .join("")
   241→    .toUpperCase()
   242→    .slice(0, 2);
   243→}

   323→// ── Subscription status widget for provider sidebar ──
   324→function SubscriptionStatusCard({ collapsed }: { collapsed: boolean }) {
   325→  const subscription = useAppStore((s) => s.subscription);
   326→
   327→  if (!subscription) return null;
   328→
   329→  const isActive = subscription.status === "ACTIVE";
   330→  const isWarning = subscription.status === "WARNING";
   331→  const isGrace = subscription.status === "GRACE";
   332→  const isSuspended = subscription.status === "SUSPENDED";
   333→
   334→  // Collapsed mode — icon indicator
   335→  if (collapsed) {
   336→    const bgClass = isActive
   337→      ? "bg-emerald-100"
   338→      : isSuspended || isGrace
   339→      ? "bg-rose-100"
   340→      : "bg-amber-100";
   341→    const iconClass = isActive
   342→      ? "text-emerald-600"
   343→      : isSuspended || isGrace
   344→      ? "text-rose-600"
   345→      : "text-amber-600";
   346→    return (
   347→      <div className="flex flex-col items-center gap-1 px-2 py-2">
   348→        <div className={`flex h-8 w-8 items-center justify-center rounded-full ${bgClass} animate-pulse shadow-sm`}> 
   349→          <Clock className={`size-4 ${iconClass}`} />
   350→        </div>
   351→      </div>
   352→    );
   353→  }
   354→
   355→  // Expanded card
   356→  const containerClass = isActive
   357→    ? "border-emerald-200 bg-emerald-50"
   358→    : isSuspended || isGrace
   359→    ? "border-rose-200 bg-rose-50"
   360→    : "border-amber-200 bg-amber-50";
   361→
   362→  const iconColorClass = isActive
   363→    ? "text-emerald-600"
   364→    : isSuspended || isGrace
   365→    ? "text-rose-600"
   366→    : "text-amber-600";
   367→
   368→  const titleClass = isActive
   369→    ? "text-emerald-800"
   370→    : isSuspended || isGrace
   371→    ? "text-rose-800"
   372→    : "text-amber-800";
   373→
   374→  const subtextClass = isActive
   375→    ? "text-emerald-700"
   376→    : isSuspended || isGrace
   377→    ? "text-rose-700"
   378→    : "text-amber-700";
   379→
   380→  const badgeClass = isActive
   381→    ? "bg-emerald-200/60 text-emerald-800"
   382→    : isSuspended || isGrace
   383→    ? "bg-rose-200/60 text-rose-800"
   384→    : "bg-amber-200/60 text-amber-800";
   385→
   386→  const Icon = isSuspended || isGrace ? AlertTriangle : Clock;
   387→
   388→  return (
   389→    <div className={`rounded-lg border p-3 ${containerClass} animate-subtle-pulse shadow-sm`}> 
   390→      <div className="flex items-center gap-2">
   391→        <Icon className={`h-4 w-4 shrink-0 ${iconColorClass}`} />
   392→        <p className={`text-xs font-semibold ${titleClass}`}>
   393→          {isActive
   394→            ? "Subscription Active"
   395→            : isSuspended
   396→            ? "Service Suspended"
   397→            : isGrace
   398→            ? "Subscription Expired"
   399→            : "Expiring Soon"}
   400→        </p>
   401→      </div>
   402→      <div className={`mt-1.5 flex items-center justify-between text-[11px] ${subtextClass}`}>
   403→        <span>{formatDaysRemaining(subscription.daysRemaining)}</span>
   404→        <span className={`inline-flex items-center gap-1 rounded-full px-1.5 py-0.5 font-semibold ${badgeClass}`}>
   405→          <Clock className="h-3 w-3" />
   406→          {Math.abs(subscription.daysRemaining)}d
   407→        </span>
   408→      </div>
   409→      {subscription.price > 0 ? (
   410→        <div className={`mt-1 text-[11px] ${subtextClass}`}>
   411→          Due: {subscription.price.toLocaleString()} ETB / {formatCycle(subscription.cycle)}
   412→        </div>
   413→      ) : (
   414→        <div className={`mt-1 text-[11px] ${subtextClass}`}>
   415→          {formatCycle(subscription.cycle)} cycle — Free trial
   416→        </div>
   417→      )}
   418→    </div>
   419→  );
   420→}

   575→                className="flex w-full items-center gap-3 rounded-lg border border-dashed border-amber-300 bg-amber-50/50 px-3 py-2.5 text-sm font-medium text-amber-700 transition-all hover:bg-amber-50 hover:border-amber-400 hover:text-amber-800 outline-none focus-visible:ring-2 focus-visible:ring-amber-400"
   576→              >
   577→                <ShieldCheck className="size-[18px] shrink-0" />
   578→                <span>Start Joint Session</span>
   579→              </button>
   580→            </>
   581→          )}
   582→          {canStartJoint && !jointSession.active && collapsed && (
   583→            <button
   584→              onClick={() => setJointLoginDialogOpen(true)}
   585→              className="flex items-center justify-center rounded-lg border border-dashed border-amber-300 p-2 text-amber-600 hover:bg-amber-50 transition-colors outline-none"
   586→              title="Start Joint Session"
   587→            >
   588→              <ShieldCheck className="size-[18px]" />
   589→            </button>
   590→          )}
   591→        </nav>
   592→      </ScrollArea>
   593→
   594→      {/* ── Subscription status for providers ── */}
   595→      {(user.role === "OPERATOR" || user.role === "STAFF") && (
   596→        <div className="px-3 py-2">
   597→          <SubscriptionStatusCard collapsed={collapsed} />
   598→        </div>
   599→      )}
   600→
   601→      <Separator className="bg-slate-200/60" />
   602→
   603→      {/* ── Bottom section: collapse toggle + logout ── */}
   604→      <div className="p-3">
   605→        {/* Desktop collapse toggle */}
   606→        {!collapsed && (
   607→          <button
   608→            onClick={onToggleCollapse}
   609→            className="mb-1 flex w-full items-center gap-3 rounded-lg px-3 py-2 text-sm font-medium text-slate-500 transition-colors hover:bg-slate-100 hover:text-slate-700 outline-none focus-visible:ring-2 focus-visible:ring-ring"
   610→          >
   611→            <ChevronLeft className="size-[18px] text-slate-400" />
   612→            <span>Collapse</span>
   613→          </button>
   614→        )}
   615→
   616→        {/* Logout — hidden for SUPERUSER (moved to name dropdown) */}
   617→        {user.role !== "SUPERUSER" && (
   618→          <button
   619→            onClick={onLogout}
   620→            className="flex w-full items-center gap-3 rounded-lg px-3 py-2.5 text-sm font-medium text-rose-600 transition-colors hover:bg-rose-50 outline-none focus-visible:ring-2 focus-visible:ring-ring"
   621→          >
   622→            <LogOut className="size-[18px]" />
   623→            {!collapsed && <span>Logout</span>}
   624→          </button>
   625→        )}
   626→      </div>
   627→    </div>
   628→  );
   629→}
