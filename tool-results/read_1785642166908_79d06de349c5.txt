   201→   201→// ── Helper: get nav items based on role ──
   202→   202→function getNavItems(user: CurrentUser): NavItem[] {
   203→   203→  switch (user.role) {
   204→   204→    case "POLICE": {
   205→   205→      const rank = (user.policeRank || "OFFICER") as PoliceRank;
   206→   206→      const allowedPages = POLICE_RANK_PERMISSIONS[rank] || POLICE_RANK_PERMISSIONS.OFFICER;
   207→   207→      return POLICE_NAV_ITEMS.filter((item) => allowedPages.includes(item.page));
   208→   208→    }
   209→   209→
   210→   210→    case "SUPERUSER":
   211→   211→      return SUPERUSER_NAV_ITEMS;
   212→   212→
   213→   213→    case "OPERATOR":
   214→   214→      return ALL_NAV_ITEMS.filter((item) => !OPERATOR_EXCLUDED.has(item.page));
   215→   215→
   216→   216→    case "STAFF": {
   217→   217→      const items: NavItem[] = [];
   218→   218→      // Always give staff access to dashboard
   219→   219→      items.push({
   220→   220→        page: "dashboard",
   221→   221→        label: "Dashboard",
   222→   222→        icon: LayoutDashboard,
   223→   223→      });
   224→   224→      // Add items based on permissions
   225→   225→      for (const perm of user.permissions) {
   226→   226→        const mapped = PERMISSION_PAGE_MAP[perm];
   227→   227→        if (mapped) {
   228→   228→          items.push(mapped);
   229→   229→        }
   230→   230→      }
   231→   231→      return items;
   232→   232→    }
   233→   233→
   234→   234→    default:
   235→   235→      return [
   236→   236→        { page: "dashboard", label: "Dashboard", icon: LayoutDashboard },
   237→   237→      ];
   238→   238→  }
   239→   239→}

   361→   361→  // Expanded card
   362→   362→  const containerClass = isActive
   363→   363→    ? "border-emerald-200 bg-emerald-50"
   364→   364→    : isSuspended || isGrace
   365→   365→    ? "border-rose-200 bg-rose-50"
