   241→  onToggleActive,

   244→  onEdit: (u: UserRecord) => void;

   246→  onView: (u: UserRecord) => void;

   250→  const RoleIcon = roleConf.icon;
   251→  let perms: string[] = [];
   252→  try {
   253→    perms = user.permissions ? JSON.parse(user.permissions) : [];
   254→  } catch {
   255→    perms = [];
   256→  }
   257→
   258→  return (
   259→    <div className="group flex items-center gap-4 rounded-xl border border-slate-100 bg-white p-4 transition-all hover:border-slate-200 hover:shadow-sm">

   270→            <RoleIcon className="h-2.5 w-2.5" />

   272→          </span>

   274→            <Badge
   275→              variant="outline"

   280→                  ? "bg-violet-50 text-violet-700 border-violet-200"
   281→                  : user.policeRank === "OFFICER"

   286→              {user.policeRank}

   297→              <Mail className="h-3 w-3" />

   299→            </span>
   300→          )}
   301→          {user.phone && (
   302→            <span className="flex items-center gap-1">
   303→              <Phone className="h-3 w-3" />

   305→            </span>

   315→                user.provider.status === "APPROVED"

   385→// ── Main Page Component ──
   386→export default function SuperUserManagementPage() {
   387→  const { currentUser, refreshKey } = useAppStore();
   388→  const [users, setUsers] = useState<UserRecord[]>([]);
   389→  const [providers, setProviders] = useState<ProviderOption[]>([]);
   390→  const [stats, setStats] = useState<StatsInfo | null>(null);
   391→  const [pagination, setPagination] = useState<PaginationInfo | null>(null);
   392→  const [loading, setLoading] = useState(true);
   393→  const [search, setSearch] = useState("");
   394→  const [activeRoleTab, setActiveRoleTab] = useState("");
   395→  const [activeProvider, setActiveProvider] = useState("");
   396→  const [currentPage, setCurrentPage] = useState(1);
   397→
   398→  // Dialog states
   399→  const [formOpen, setFormOpen] = useState(false);
   400→  const [viewOpen, setViewOpen] = useState(false);
   401→  const [deleteTarget, setDeleteTarget] = useState<UserRecord | null>(null);
   402→  const [editingUser, setEditingUser] = useState<UserRecord | null>(null);
   403→  const [viewingUser, setViewingUser] = useState<UserRecord | null>(null);
   404→  const [form, setForm] = useState(emptyForm);
   405→  const [saving, setSaving] = useState(false);
   406→  const [deleting, setDeleting] = useState(false);
   407→
   408→  const fetchData = useCallback(async () => {
   409→    setLoading(true);
   410→    try {
   411→      const [usersData, provDataRaw] = await Promise.all([
   412→        apiSuperGetUsers({
   413→          search: search || undefined,
   414→          role: activeRoleTab || undefined,
   415→          providerId: activeProvider || undefined,
   416→          page: currentPage,
   417→          pageSize: 12,
   418→        }),
   419→        apiSuperGetProviders(),
   420→      ]);
   421→      setUsers(usersData.users || []);
   422→      setPagination(usersData.pagination || null);
   423→      setStats(usersData.stats || null);
   424→      // Providers API returns an array directly
   425→      const provs = Array.isArray(provDataRaw)
   426→        ? provDataRaw
   427→        : (provDataRaw as Record<string, unknown>).providers || [];
   428→      setProviders(provs.map((p: Record<string, unknown>) => ({
   429→        id: p.id as string,
   430→        name: p.name as string,
   431→        status: p.status as string,
   432→      })));
   433→    } catch (err: unknown) {
   434→      const msg = err instanceof Error ? err.message : "Failed to load data";
   435→      toast.error(msg);
   436→    } finally {
   437→      setLoading(false);
   438→    }
   439→  }, [search, activeRoleTab, activeProvider, currentPage]);
   440→
   441→  useEffect(() => {
   442→    fetchData();
   443→  }, [fetchData, refreshKey]);
   444→
   445→  // Reset page when filters change
   446→  useEffect(() => {
   447→    setCurrentPage(1);
   448→  }, [search, activeRoleTab, activeProvider]);
   449→
   450→  // ── Form Handlers ──
   451→  const openCreate = () => {
   452→    setEditingUser(null);
   453→    setForm(emptyForm);
   454→    setFormOpen(true);
   455→  };
   456→
   457→  const openEdit = (user: UserRecord) => {
   458→    setEditingUser(user);
   459→    let perms: string[] = [];
   460→    try {
   461→      perms = user.permissions ? JSON.parse(user.permissions) : [];
   462→    } catch {
   463→      perms = [];
   464→    }
   465→    setForm({
   466→      username: user.username,
   467→      password: "",
   468→      name: user.name,
   469→      email: user.email || "",
   470→      phone: user.phone || "",
   471→      role: user.role,
   472→      policeRank: user.policeRank || "OFFICER",
   473→      permissions: perms,
   474→      providerId: user.providerId || "",
   475→      isActive: user.isActive,
   476→    });
   477→    setFormOpen(true);
   478→  };
   479→
   480→  const openView = (user: UserRecord) => {
   481→    setViewingUser(user);
   482→    setViewOpen(true);
   483→  };
   484→
   485→  const handleSave = async () => {
   486→    if (!form.username.trim() || !form.name.trim() || !form.role) {
   487→      toast.error("Username, name, and role are required");
   488→      return;
   489→    }
   490→    if (!editingUser && !form.password) {
   491→      toast.error("Password is required for new users");
   492→      return;
   493→    }
   494→    if ((form.role === "OPERATOR" || form.role === "STAFF") && !form.providerId) {
   495→      toast.error("Please select a provider for this user");
   496→      return;
   497→    }
   498→    if (form.role === "POLICE" && !form.policeRank) {
   499→      toast.error("Please select a police rank");
   500→      return;
   501→    }
   502→
   503→    setSaving(true);
   504→    try {
   505→      const payload: Record<string, unknown> = {
   506→        username: form.username.trim(),
   507→        name: form.name.trim(),
   508→        email: form.email.trim() || null,
   509→        phone: form.phone.trim() || null,
   510→        role: form.role,
   511→        policeRank: form.role === "POLICE" ? form.policeRank : null,
   512→        permissions: form.role === "STAFF" ? form.permissions : [],
   513→        providerId: form.providerId || null,
   514→        isActive: form.isActive,
   515→      };
   516→      if (form.password) payload.password = form.password;
   517→
   518→      if (editingUser) {
   519→        await apiSuperUpdateUser(editingUser.id, payload);
   520→        toast.success("User updated successfully");
   521→      } else {
   522→        await apiSuperCreateUser(payload);
   523→        toast.success("User created successfully");
   524→      }
   525→      setFormOpen(false);
   526→      fetchData();
   527→    } catch (err: unknown) {
   528→      const msg = err instanceof Error ? err.message : "Failed to save user";
   529→      toast.error(msg);
   530→    } finally {
   531→      setSaving(false);
   532→    }
   533→  };
   534→
   535→  const handleDelete = async () => {
   536→    if (!deleteTarget) return;
   537→    setDeleting(true);
   538→    try {
   539→      await apiSuperDeleteUser(deleteTarget.id);
   540→      toast.success("User deleted successfully");
   541→      setDeleteTarget(null);
   542→      fetchData();
   543→    } catch (err: unknown) {
   544→      const msg = err instanceof Error ? err.message : "Failed to delete user";
   545→      toast.error(msg);
   546→    } finally {
   547→      setDeleting(false);
   548→    }
   549→  };
   550→
   551→  const handleToggleActive = async (user: UserRecord) => {
   552→    try {
   553→      await apiSuperUpdateUser(user.id, { isActive: !user.isActive });
   554→      toast.success(`User ${user.isActive ? "deactivated" : "activated"} successfully`);
   555→      fetchData();
   556→    } catch (err: unknown) {
   557→      const msg = err instanceof Error ? err.message : "Failed to update status";
   558→      toast.error(msg);
   559→    }
   560→  };
   561→
   562→  const togglePermission = (perm: string) => {
   563→    setForm((prev) => ({
   564→      ...prev,
   565→      permissions: prev.permissions.includes(perm)
   566→        ? prev.permissions.filter((p) => p !== perm)
   567→        : [...prev.permissions, perm],
   568→    }));
   569→  };
   570→
   571→  return (
   572→    <div className="space-y-6 p-4 md:p-6">
   573→      {/* ── Page Header ── */}
   574→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   575→        <div>
   576→          <h1 className="text-2xl font-bold tracking-tight text-slate-900">User Management</h1>
   577→          <p className="mt-1 text-sm text-slate-500">
   578→            Manage all user accounts, roles, and permissions across the system.
   579→          </p>
   580→        </div>
   581→        <Button onClick={openCreate} className="shadow-sm">
   582→          <UserPlus className="mr-2 h-4 w-4" />
   583→          Add New User
   584→        </Button>
   585→      </div>
   586→
   587→      {/* ── Stats Cards ── */}
   588→      {!loading && stats && (
   589→        <div className="grid grid-cols-2 gap-4 lg:grid-cols-4">
   590→          <StatCard
   591→            icon={Users}
   592→            label="Total Users"
   593→            value={stats.totalUsers}
   594→            sub="Across all providers"
   595→            color="text-indigo-600"
   596→            bgColor="bg-indigo-50"
   597→          />
   598→          <StatCard
   599→            icon={Building2}
   600→            label="Active Providers"
   601→            value={stats.activeProviderCount}
   602→            sub={`${stats.providerCount} total`}
   603→            color="text-emerald-600"
   604→            bgColor="bg-emerald-50"
   605→          />
   606→          <StatCard
