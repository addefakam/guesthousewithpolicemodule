    65→    65→// Severity helpers
    66→    66→function getSeverity(p: { criticalCount: number; highCount: number; matchCount: number }) {
    67→    67→  if (p.criticalCount > 0) return "CRITICAL";
    68→    68→  if (p.highCount > 0) return "HIGH";
    69→    69→  if (p.matchCount > 0) return "MEDIUM";
