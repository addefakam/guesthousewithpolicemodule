   718→/** Execute SQL via raw pg Client */
   719→async function execViaPg(sql: string, params?: unknown[]): Promise<void> {
   720→  const client = new Client({ connectionString: process.env.DATABASE_URL! });
   721→  try {
   722→    await client.connect();
   723→    await client.query(sql, params);
   724→  } finally {
   725→    await client.end().catch(() => {});
   726→  }
   727→}

   729→/** Execute DDL via Prisma $executeRawUnsafe — splits multi-statement SQL
   730→ *  because Prisma's driver does NOT support multiple statements per call. */
   731→async function execViaPrisma(sql: string): Promise<void> {
   732→  const prisma = new PrismaClient({ log: ["warn", "error"] });
   733→  try {
   734→    // Extract individual executable statements.
   735→    // Prisma $executeRawUnsafe cannot run DO $$ ... END $$ blocks,
   736→    // so we unwrap them and run the inner SQL, catching duplicate errors in JS.
   737→    const stmts = extractStatements(sql);
   738→
   739→    for (const stmt of stmts) {
   740→      const s = stmt.endsWith(';') ? stmt : stmt + ';';
   741→      try {
   742→        await prisma.$executeRawUnsafe(s);
   743→      } catch (err) {
   744→        // Ignore duplicate column / duplicate object / relation already exists errors
   745→        const msg = err instanceof Error ? err.message : String(err);
   746→        const ignorable =
   747→          /duplicate_column/i.test(msg) ||
   748→          /duplicate_object/i.test(msg) ||
   749→          /already exists/i.test(msg) ||
   750→          /relation .* already exists/i.test(msg);
   751→        if (!ignorable) {
   752→          console.error("[execViaPrisma] Statement failed (continuing):", s.slice(0, 100), msg);
   753→        }
   754→      }
   755→    }
   756→  } finally {
   757→    await prisma.$disconnect().catch(() => {});
   758→  }
   759→}

   761→/**
   762→ * Parse a SQL blob into individual executable statements.
   763→ * Handles:
   764→ *  - DO $$ BEGIN <stmt> EXCEPTION WHEN ... THEN null; END $$;  →  <stmt>
   765→ *  - Plain statements separated by ;
   766→ */
   767→function extractStatements(sql: string): string[] {
   768→  const results: string[] = [];
   769→
   770→  // 1. Extract inner statements from DO $$ blocks
   771→  //    Pattern: DO $$ BEGIN ... EXCEPTION WHEN <type> THEN null; END $$;
   772→  const doBlockRe = /DO\s+\$\$\s*BEGIN\s*([\s\S]*?)\s*EXCEPTION\s+WHEN\s+\w+\s+THEN\s+null;\s*END\s+\$\$/gi;
   773→  let remaining = sql;
   774→  let match: RegExpExecArray | null;
   775→
   776→  while ((match = doBlockRe.exec(remaining)) !== null) {
   777→    const inner = match[1].trim();
   778→    if (inner) {
   779→      // Inner content may have multiple statements separated by ;
   780→      const innerStmts = inner
   781→        .split(';')
   782→        .map(s => s.trim())
   783→        .filter(s => s.length > 0);
   784→      results.push(...innerStmts);
   785→    }
   786→  }
   787→
   788→  // 2. Remove DO blocks from remaining SQL
   789→  remaining = remaining.replace(doBlockRe, '');
   790→
   791→  // 3. Extract remaining plain statements (CREATE INDEX, CREATE TABLE, INSERT, etc.)
   792→  if (remaining.trim()) {
   793→    const plain = remaining
   794→      .split(';')
   795→      .map(s => s.trim())
   796→      .filter(s => s.length > 0 && !s.startsWith('--'));
   797→    results.push(...plain);
   798→  }
   799→
   800→  return results;
   801→}

   818→    // Indexes are best-effort
   819→  }
