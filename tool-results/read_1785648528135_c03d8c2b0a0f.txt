   385→// ── Main Page Component ──
   386→export default function SuperUserManagementPage() {
   387→  const { currentUser, refreshKey } = useAppStore();
   388→  const [users, setUsers] = useState<UserRecord[]>([]);
   389→  const [providers, setProviders] = useState<ProviderOption[]>([]);
   390→  const [stats, setStats] = useState<StatsInfo | null>(null);
   391→  const [pagination, setPagination] = useState<PaginationInfo | null>(null);
   392→  const [loading, setLoading] = useState(true);
   393→  const [search, setSearch] = useState("");
   394→  const [activeRoleTab, setActiveRoleTab] = useState("");
   395→  const [activeProvider, setActiveProvider] = useState("");
   396→  const [currentPage, setCurrentPage] = useState(1);
   397→
   398→  // Dialog states
   399→  const [formOpen, setFormOpen] = useState(false);
   400→  const [viewOpen, setViewOpen] = useState(false);
   401→  const [deleteTarget, setDeleteTarget] = useState<UserRecord | null>(null);
   402→  const [editingUser, setEditingUser] = useState<UserRecord | null>(null);
   403→  const [viewingUser, setViewingUser] = useState<UserRecord | null>(null);
   404→  const [form, setForm] = useState(emptyForm);
   405→  const [saving, setSaving] = useState(false);
   406→  const [deleting, setDeleting] = useState(false);
   407→
   408→  // Only approved providers can be assigned to operator/staff users
   409→  const approvedProviders = providers.filter((p) => p.status === "APPROVED");
   410→
   411→  const fetchData = useCallback(async () => {
   412→    setLoading(true);
   413→    try {
   414→      const [usersData, provDataRaw] = await Promise.all([
   415→        apiSuperGetUsers({
   416→          search: search || undefined,
   417→          role: activeRoleTab || undefined,
   418→          providerId: activeProvider || undefined,
   419→          page: currentPage,
   420→          pageSize: 12,
   421→        }),
   422→        apiSuperGetProviders(),
   423→      ]);
   424→      setUsers(usersData.users || []);
   425→      setPagination(usersData.pagination || null);
   426→      setStats(usersData.stats || null);
   427→      // Providers API returns an array directly
   428→      const provs = Array.isArray(provDataRaw)
   429→        ? provDataRaw
   430→        : (provDataRaw as Record<string, unknown>).providers || [];
   431→      setProviders(provs.map((p: Record<string, unknown>) => ({
   432→        id: p.id as string,
   433→        name: p.name as string,
   434→        status: p.status as string,
   435→      })));
   436→    } catch (err: unknown) {
   437→      const msg = err instanceof Error ? err.message : "Failed to load data";
   438→      toast.error(msg);
   439→    } finally {
   440→      setLoading(false);
   441→    }
   442→  }, [search, activeRoleTab, activeProvider, currentPage]);
   443→
   444→  useEffect(() => {
   445→    fetchData();
   446→  }, [fetchData, refreshKey]);
   447→
   448→  // Reset page when filters change
   449→  useEffect(() => {
   450→    setCurrentPage(1);
   451→  }, [search, activeRoleTab, activeProvider]);
   452→
   453→  // ── Form Handlers ──
   454→  const openCreate = () => {
   455→    setEditingUser(null);
   456→    setForm(emptyForm);
   457→    setFormOpen(true);
   458→  };
   459→
   460→  const openEdit = (user: UserRecord) => {
   461→    setEditingUser(user);
   462→    let perms: string[] = [];
   463→    try {
   464→      perms = user.permissions ? JSON.parse(user.permissions) : [];
   465→    } catch {
   466→      perms = [];
   467→    }
   468→    setForm({
   469→      username: user.username,
   470→      password: "",
   471→      name: user.name,
   472→      email: user.email || "",
   473→      phone: user.phone || "",
   474→      role: user.role,
   475→      policeRank: user.policeRank || "OFFICER",
   476→      permissions: perms,
   477→      providerId: user.providerId || "",
   478→      isActive: user.isActive,
   479→    });
   480→    setFormOpen(true);
   481→  };
   482→
   483→  const openView = (user: UserRecord) => {
   484→    setViewingUser(user);
   485→    setViewOpen(true);
   486→  };
   487→
   488→  const handleSave = async () => {
   489→    if (!form.username.trim() || !form.name.trim() || !form.role) {
   490→      toast.error("Username, name, and role are required");
   491→      return;
   492→    }
   493→    if (!editingUser && !form.password) {
   494→      toast.error("Password is required for new users");
   495→      return;
   496→    }
   497→    if (form.role === "OPERATOR" && !form.providerId) {
   498→      toast.error("Please select a guesthouse for this operator");
   499→      return;
   500→    }
   501→    if (form.role === "STAFF" && !form.providerId) {
   502→      toast.error("Please select a guesthouse for this staff member");
   503→      return;
   504→    }
   505→    if (form.role === "POLICE" && !form.policeRank) {
   506→      toast.error("Please select a police rank");
   507→      return;
   508→    }
   509→
   510→    setSaving(true);
   511→    try {
   512→      const payload: Record<string, unknown> = {
   513→        username: form.username.trim(),
   514→        name: form.name.trim(),
   515→        email: form.email.trim() || null,
   516→        phone: form.phone.trim() || null,
   517→        role: form.role,
   518→        policeRank: form.role === "POLICE" ? form.policeRank : null,
   519→        permissions: form.role === "STAFF" ? form.permissions : [],
   520→        providerId: form.providerId || null,
   521→        isActive: form.isActive,
   522→      };
   523→      if (form.password) payload.password = form.password;
   524→
   525→      if (editingUser) {
   526→        await apiSuperUpdateUser(editingUser.id, payload);
   527→        toast.success("User updated successfully");
   528→      } else {
   529→        await apiSuperCreateUser(payload);
   530→        toast.success("User created successfully");
   531→      }
   532→      setFormOpen(false);
   533→      fetchData();
   534→    } catch (err: unknown) {
   535→      const msg = err instanceof Error ? err.message : "Failed to save user";
   536→      toast.error(msg);
   537→    } finally {
   538→      setSaving(false);
   539→    }
   540→  };
   541→
   542→  const handleDelete = async () => {
   543→    if (!deleteTarget) return;
   544→    setDeleting(true);
   545→    try {
   546→      await apiSuperDeleteUser(deleteTarget.id);
   547→      toast.success("User deleted successfully");
   548→      setDeleteTarget(null);
   549→      fetchData();
   550→    } catch (err: unknown) {
   551→      const msg = err instanceof Error ? err.message : "Failed to delete user";
   552→      toast.error(msg);
   553→    } finally {
   554→      setDeleting(false);
   555→    }
   556→  };
   557→
   558→  const handleToggleActive = async (user: UserRecord) => {
   559→    try {
   560→      await apiSuperUpdateUser(user.id, { isActive: !user.isActive });
   561→      toast.success(`User ${user.isActive ? "deactivated" : "activated"} successfully`);
   562→      fetchData();
   563→    } catch (err: unknown) {
   564→      const msg = err instanceof Error ? err.message : "Failed to update status";
   565→      toast.error(msg);
   566→    }
   567→  };
   568→
   569→  const togglePermission = (perm: string) => {
   570→    setForm((prev) => ({
   571→      ...prev,
   572→      permissions: prev.permissions.includes(perm)
   573→        ? prev.permissions.filter((p) => p !== perm)
   574→        : [...prev.permissions, perm],
   575→    }));
   576→  };
   577→
   578→  return (
   579→    <div className="space-y-6 p-4 md:p-6">
   580→      {/* ── Page Header ── */}
   581→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   582→        <div>
   583→          <h1 className="text-2xl font-bold tracking-tight text-slate-900">User Management</h1>
   584→          <p className="mt-1 text-sm text-slate-500">
   585→            Manage all user accounts, roles, and permissions across the system.
   586→          </p>
   587→        </div>
   588→        <Button onClick={openCreate} className="shadow-sm">
   589→          <UserPlus className="mr-2 h-4 w-4" />
   590→          Add New User
   591→        </Button>
   592→      </div>
   593→
   594→      {/* ── Stats Cards ── */}
   595→      {!loading && stats && (
   596→        <div className="grid grid-cols-2 gap-4 lg:grid-cols-4">
   597→          <StatCard
   598→            icon={Users}
   599→            label="Total Users"
   600→            value={stats.totalUsers}
   601→            sub="Across all providers"
   602→            color="text-indigo-600"
   603→            bgColor="bg-indigo-50"
   604→          />
   605→          <StatCard
