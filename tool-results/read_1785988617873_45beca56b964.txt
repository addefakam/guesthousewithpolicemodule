    80→function getActionIcon(action: string) {
    81→  const lower = action.toLowerCase();
    82→  if (lower === "login" || lower === "login_failed") return <LogIn className="h-3.5 w-3.5" />;
    83→  if (lower === "logout") return <LogOut className="h-3.5 w-3.5" />;
    84→  if (lower === "create") return <Plus className="h-3.5 w-3.5" />;
    85→  if (lower === "update") return <Pencil className="h-3.5 w-3.5" />;
    86→  if (lower === "delete") return <Trash2 className="h-3.5 w-3.5" />;
    87→  if (lower === "view") return <Eye className="h-3.5 w-3.5" />;
    88→  if (lower === "export" || lower === "backup") return <FileDown className="h-3.5 w-3.5" />;
    89→  return <Activity className="h-3.5 w-3.5" />;
    90→}

    92→function getActionColor(action: string): string {
    93→  const lower = action.toLowerCase();
    94→  if (lower === "login") return "bg-emerald-100 text-emerald-700 border-emerald-200";
    95→  if (lower === "logout") return "bg-slate-100 text-slate-600 border-slate-200";
    96→  if (lower === "create" || lower === "approve") return "bg-sky-100 text-sky-700 border-sky-200";
    97→  if (lower === "update") return "bg-amber-100 text-amber-700 border-amber-200";
    98→  if (lower === "delete" || lower === "suspend") return "bg-rose-100 text-rose-700 border-rose-200";
    99→  if (lower === "view") return "bg-violet-100 text-violet-600 border-violet-200";
   100→  if (lower === "export" || lower === "backup") return "bg-cyan-100 text-cyan-700 border-cyan-200";
   101→  if (lower === "reject") return "bg-orange-100 text-orange-700 border-orange-200";
   102→  if (lower === "login_failed") return "bg-red-100 text-red-700 border-red-200";
   103→  if (lower === "config_change") return "bg-pink-100 text-pink-700 border-pink-200";
   104→  return "bg-slate-100 text-slate-600 border-slate-200";

   107→function formatTimestamp(dateStr: string): string {
   108→  try {
   109→    const date = new Date(dateStr);
   110→    return date.toLocaleDateString("en-US", {
   111→      year: "numeric",
   112→      month: "short",
   113→      day: "numeric",
   114→      hour: "2-digit",
   115→      minute: "2-digit",
   116→      second: "2-digit",
   117→      hour12: true,
   118→    });
   119→  } catch {
   120→    return dateStr;
   121→  }
   122→}
   123→
   124→function isToday(dateStr: string): boolean {
   125→  try {
   126→    const date = new Date(dateStr);
   127→    const now = new Date();
   128→    return (
   129→      date.getDate() === now.getDate() &&
   130→      date.getMonth() === now.getMonth() &&
   131→      date.getFullYear() === now.getFullYear()
   132→    );
   133→  } catch {
   134→    return false;
   135→  }
   136→}
   137→
   138→function isThisWeek(dateStr: string): boolean {
   139→  try {
   140→    const date = new Date(dateStr);
   141→    const now = new Date();
   142→    const startOfWeek = new Date(now);
   143→    startOfWeek.setDate(now.getDate() - now.getDay());
   144→    startOfWeek.setHours(0, 0, 0, 0);
   145→    return date >= startOfWeek;
   146→  } catch {
   147→    return false;
   148→  }
   149→}
