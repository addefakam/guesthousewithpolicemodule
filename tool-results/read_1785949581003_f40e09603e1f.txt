   626→/** Execute SQL via raw pg Client */
   627→async function execViaPg(sql: string, params?: unknown[]): Promise<void> {
   628→  const client = new Client({ connectionString: process.env.DATABASE_URL! });
   629→  try {
   630→    await client.connect();
   631→    await client.query(sql, params);
   632→  } finally {
   633→    await client.end().catch(() => {});
   634→  }
   635→}

   637→/** Execute DDL via Prisma $executeRawUnsafe — splits multi-statement SQL
   638→ *  because Prisma's driver does NOT support multiple statements per call. */
   639→async function execViaPrisma(sql: string): Promise<void> {
   640→  // Split on `DO $$` blocks and standalone statements
   641→  // Each `DO $$ ... END $$;` is one executable unit
   642→  const statements = sql
   643→    .split(/(?=DO \$\$)/)
   644→    .map(s => s.trim())
   645→    .filter(s => s.length > 0);
   646→
   647→  // If no DO blocks found, split by semicolons (for CREATE INDEX etc.)
   648→  const parts = statements.length > 1
   649→    ? statements
   650→    : sql.split(';').map(s => s.trim()).filter(s => s.length > 0);
   651→
   652→  const prisma = new PrismaClient({
   653→    log: ["warn", "error"],
   654→  });
   655→  try {
   656→    for (const part of parts) {
   657→      const stmt = part.endsWith(';') ? part : part + ';';
   658→      try {
   659→        await prisma.$executeRawUnsafe(stmt);
   660→      } catch (err) {
   661→        // Individual statement might fail (e.g. duplicate_object) — log but continue
   662→        console.error("[execViaPrisma] Statement failed (continuing):", stmt.slice(0, 80), err instanceof Error ? err.message : String(err));
   663→      }
   664→    }
   665→  } finally {
   666→    await prisma.$disconnect().catch(() => {});
   667→  }
   668→}

   674→async function runMigrationsOnly(): Promise<void> {

   684→  } catch {
   685→    // Indexes are best-effort
   686→  }

   918→export async function forceMigrate(): Promise<void> {
