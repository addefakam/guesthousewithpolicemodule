     1→     1→   718→### ⚠️ Cover Title Layout — calcTitleLayout() (Mandatory for ALL Recipes)
     2→     2→   719→
     3→     3→   720→**Every cover recipe MUST use `calcTitleLayout()` to determine title font size and line breaks.** Hardcoding font sizes or passing the full title as a single TextRun is FORBIDDEN.
     4→     4→   721→
     5→     5→   722→**Every paragraph with font size > body text MUST set explicit line spacing** to prevent top clipping:
     6→     6→   723→```js
     7→     7→   724→// ★ MANDATORY: prevent inherited small line spacing from clipping large fonts
     8→     8→   725→spacing: { line: Math.ceil(titlePt * 23), lineRule: "atLeast", after: 100 }
     9→     9→   726→// Example: 36pt → line: 828; 44pt → line: 1012
    10→    10→   727→```
    11→    11→   728→Without this, the paragraph inherits body text line spacing (e.g., 560tw), which is shorter than the font height → top of characters gets clipped.
    12→    12→   729→
    13→    13→   730→```js
    14→    14→   731→/**
    15→    15→   732→ * Calculate safe font size and smart line breaks for cover titles.
    16→    16→   733→ * MUST be called by every recipe before building title paragraphs.
    17→    17→   734→ *
    18→    18→   735→ * @param {string}  title          Full title string
    19→    19→   736→ * @param {number}  maxWidthTwips  Available width for title text (twips, after subtracting margins/padding)
    20→    20→   737→ * @param {number}  preferredPt    Desired max font size in pt (default 40)
    21→    21→   738→ * @param {number}  minPt          Minimum allowed font size in pt (default 24)
    22→    22→   739→ * @returns {{ titlePt: number, titleLines: string[] }}
    23→    23→   740→ */
    24→    24→   741→function calcTitleLayout(title, maxWidthTwips, preferredPt = 40, minPt = 24) {
    25→    25→   742→  // Each CJK character width ≈ pt × 20 twips
    26→    26→   743→  const charWidth = (pt) => pt * 20;
    27→    27→   744→  const charsPerLine = (pt) => Math.floor(maxWidthTwips / charWidth(pt));
    28→    28→   745→
    29→    29→   746→  // Try from preferredPt downward until title fits in ≤ 3 lines
    30→    30→   747→  let titlePt = preferredPt;
    31→    31→   748→  let lines;
    32→    32→   749→  while (titlePt >= minPt) {
    33→    33→   750→    const cpl = charsPerLine(titlePt);
    34→    34→   751→    if (cpl < 2) { titlePt -= 2; continue; }
    35→    35→   752→    lines = splitTitleLines(title, cpl);
    36→    36→   753→    if (lines.length <= 3) break;
    37→    37→   754→    titlePt -= 2;
    38→    38→   755→  }
    39→    39→   756→
    40→    40→   757→  // If still > 3 lines at minPt, force 3 lines
    41→    41→   758→  if (!lines || lines.length > 3) {
    42→    42→   759→    const cpl = charsPerLine(minPt);
    43→    43→   760→    lines = splitTitleLines(title, cpl);
    44→    44→   761→    titlePt = minPt;
    45→    45→   762→  }
    46→    46→   763→
    47→    47→   764→  return { titlePt, titleLines: lines };
    48→    48→   765→}
    49→    49→   766→
    50→    50→   767→/**
    51→    51→   768→ * Smart Chinese title line-breaking — breaks at semantic boundaries, never mid-word.
    52→    52→   769→ *
    53→    53→   770→ * Rules:
    54→    54→   771→ * 1. Prefer breaking after particles, punctuation, connectors, underscores, spaces
    55→    55→   772→ * 2. Never split a compound word (e.g., "管理规范" must not become "管理规" + "范")
    56→    56→   773→ * 3. No single-character orphan on the last line — merge into previous line
    57→    57→   774→ * 4. If no good break point found within 60-130% of charsPerLine, break at charsPerLine
    58→    58→   775→ *
    59→    59→   776→ * @param {string} title        Full title string
    60→    60→   777→ * @param {number} charsPerLine Max characters per line at current font size
    61→    61→   778→ * @returns {string[]}          Array of line strings
    62→    62→   779→ */
    63→    63→   780→function splitTitleLines(title, charsPerLine) {
    64→    64→   781→  if (title.length <= charsPerLine) return [title];
    65→    65→   782→
    66→    66→   783→  // Characters that are safe break points (break AFTER these)
    67→    67→   784→  const breakAfter = new Set([
    68→    68→   785→    ...'，。、；：！？',              // CJK punctuation
    69→    69→   786→    ...'的与和及之在于为',            // CJK particles/prepositions
    70→    70→   787→    ...'-_—–·/',                     // connectors
    71→    71→   788→    ...' \t',                         // whitespace
    72→    72→   789→  ]);
    73→    73→   790→
    74→    74→   791→  const lines = [];
    75→    75→   792→  let remaining = title;
    76→    76→   793→
    77→    77→   794→  while (remaining.length > charsPerLine) {
    78→    78→   795→    let breakAt = -1;
    79→    79→   796→
    80→    80→   797→    // Search backward from charsPerLine to 60% for a break point
    81→    81→   798→    for (let i = charsPerLine; i >= Math.floor(charsPerLine * 0.6); i--) {
    82→    82→   799→      if (i < remaining.length && breakAfter.has(remaining[i - 1])) {
    83→    83→   800→        breakAt = i;
    84→    84→   801→        break;
    85→    85→   802→      }
    86→    86→   803→    }
    87→    87→   804→
    88→    88→   805→    // If not found, search forward up to 130%
    89→    89→   806→    if (breakAt === -1) {
    90→    90→   807→      const limit = Math.min(remaining.length, Math.ceil(charsPerLine * 1.3));
    91→    91→   808→      for (let i = charsPerLine + 1; i < limit; i++) {
    92→    92→   809→        if (breakAfter.has(remaining[i - 1])) {
    93→    93→   810→          breakAt = i;
    94→    94→   811→          break;
    95→    95→   812→        }
    96→    96→   813→      }
    97→    97→   814→    }
    98→    98→   815→
    99→    99→   816→    // Last resort: break at charsPerLine, but avoid splitting compound CJK words
   100→   100→   817→    if (breakAt === -1) {
   101→   101→   818→      breakAt = charsPerLine;
   102→   102→   819→      // If both chars at the break boundary are CJK (likely a compound word),
   103→   103→   820→      // step back 1 char to keep the word together
   104→   104→   821→      const prevChar = remaining[breakAt - 1];
   105→   105→   822→      const nextChar = remaining[breakAt];
   106→   106→   823→      if (prevChar && nextChar &&
   107→   107→   824→          !breakAfter.has(prevChar) && !breakAfter.has(nextChar) &&
   108→   108→   825→          /[\u4e00-\u9fff]/.test(prevChar) && /[\u4e00-\u9fff]/.test(nextChar)) {
   109→   109→   826→        breakAt = breakAt - 1;
   110→   110→   827→      }
   111→   111→   828→    }
   112→   112→   829→
   113→   113→   830→    lines.push(remaining.slice(0, breakAt).trim());
   114→   114→   831→    remaining = remaining.slice(breakAt).trim();
   115→   115→   832→  }
   116→   116→   833→  if (remaining) lines.push(remaining);
   117→   117→   834→
   118→   118→   835→  // Prevent single-character orphan on last line — merge into previous
   119→   119→   836→  if (lines.length > 1 && lines[lines.length - 1].length <= 2) {
   120→   120→   837→    const last = lines.pop();
   121→   121→   838→    lines[lines.length - 1] += last;
   122→   122→   839→  }
   123→   123→   840→
   124→   124→   841→  return lines;
   125→   125→   842→}
   126→   126→   843→
   127→   127→   844→/**
   128→   128→   845→ * Calculate dynamic spacing values for cover elements to fit within page height.
   129→   129→   846→ *
   130→   130→   847→ * @param {object} params
   131→   131→   848→ * @param {number} params.titleLineCount   Number of title lines
   132→   132→   849→ * @param {number} params.titlePt          Title font size in pt
   133→   133→   850→ * @param {boolean} params.hasSubtitle     Whether subtitle exists
   134→   134→   851→ * @param {boolean} params.hasEnglishLabel Whether English label exists
   135→   135→   852→ * @param {number} params.metaLineCount    Number of meta info lines
   136→   136→   853→ * @param {number} params.fixedHeight      Sum of fixed-height elements (color strips, accent bars, footer) in twips
   137→   137→   854→ * @param {number} params.pageHeight       Total page height in twips (default 16838)
   138→   138→   855→ * @returns {{ topSpacing, midSpacing, bottomSpacing }}  Spacing values in twips
   139→   139→   856→ */
   140→   140→   857→function calcCoverSpacing(params) {
   141→   141→   858→  const {
   142→   142→   859→    titleLineCount = 1, titlePt = 36, hasSubtitle = false,
   143→   143→   860→    hasEnglishLabel = false, metaLineCount = 0,
   144→   144→   861→    fixedHeight = 800, pageHeight = 16838,
   145→   145→   862→    marginTop = 0, marginBottom = 0,   // ★ NEW: pass actual section margins
   146→   146→   863→  } = params;
   147→   147→   864→
   148→   148→   865→  // ★ Safety margin: 1200 twips (cross-engine: MS Office renders large fonts
   149→   149→   866→  // taller than calculated; extra 400tw buffer prevents footer clipping)
   150→   150→   867→  const SAFETY = 1200;
   151→   151→   868→  // ★ Subtract page margins from available height (cover section may have margins)
   152→   152→   869→  const usableHeight = pageHeight - marginTop - marginBottom - SAFETY;
   153→   153→   870→
   154→   154→   871→  // ★ Accurate height estimation per element:
   155→   155→   872→  const titleHeight = titleLineCount * (titlePt * 23 + 200);
   156→   156→   873→  const subtitleHeight = hasSubtitle ? (12 * 23 + 600) : 0;
   157→   157→   874→  const englishLabelHeight = hasEnglishLabel ? (9 * 23 + 600) : 0;
   158→   158→   875→  const metaHeight = metaLineCount * (10 * 23 + 100);
   159→   159→   876→
   160→   160→   877→  // ★ Account for implicit paragraph heights:
   161→   161→   878→  const implicitParaHeight = 3 * 300;
   162→   162→   879→
   163→   163→   880→  const contentHeight = titleHeight + subtitleHeight + englishLabelHeight +
   164→   164→   881→                        metaHeight + fixedHeight + implicitParaHeight;
   165→   165→   882→
   166→   166→   883→  const remainingSpace = usableHeight - contentHeight;
   167→   167→   884→  const safeRemaining = Math.max(remainingSpace, 400);
   168→   168→   885→
   169→   169→   886→  // ★ Footer protection: bottomSpacing must be ≥ FOOTER_MIN to prevent
   170→   170→   887→  // footer + accent line from being clipped at the cell bottom edge
   171→   171→   888→  const FOOTER_MIN = 800;
   172→   172→   889→  const rawTop = Math.floor(safeRemaining * 0.45);
   173→   173→   890→  const rawBottom = Math.floor(safeRemaining * 0.45);
   174→   174→   891→  const bottomSpacing = Math.max(rawBottom, FOOTER_MIN);
   175→   175→   892→  const topSpacing = Math.max(rawTop - Math.max(0, FOOTER_MIN - rawBottom), 400);
   176→   176→   893→  const midSpacing = Math.max(safeRemaining - topSpacing - bottomSpacing, 0);
   177→   177→   894→
   178→   178→   895→  return { topSpacing, midSpacing, bottomSpacing };
   179→   179→   896→}
   180→   180→   897→```
   181→   181→   898→
   182→   182→   899→**Usage in every recipe:**
   183→   183→   900→```js
   184→   184→   901→// Step 1: Calculate title layout
   185→   185→   902→const availableWidth = 11906 - leftPadding - rightPadding; // subtract margins
   186→   186→   903→const { titlePt, titleLines } = calcTitleLayout(config.title, availableWidth);
   187→   187→   904→const titleSize = titlePt * 2; // convert to half-points for docx-js
   188→   188→   905→
   189→   189→   906→// Step 2: Calculate spacing (pass actual section margins!)
   190→   190→   907→const spacing = calcCoverSpacing({
   191→   191→   908→  titleLineCount: titleLines.length,
   192→   192→   909→  titlePt,
   193→   193→   910→  hasSubtitle: !!config.subtitle,
   194→   194→   911→  hasEnglishLabel: !!config.englishLabel,
   195→   195→   912→  metaLineCount: (config.metaLines || []).length,
   196→   196→   913→  fixedHeight: 800, // sum of accent strips, footer table, etc.
   197→   197→   914→  marginTop: 0,     // ★ pass the cover section's actual top margin (twips)
   198→   198→   915→  marginBottom: 0,  // ★ pass the cover section's actual bottom margin (twips)
   199→   199→   916→});
   200→   200→   917→
   201→   201→   918→// Step 3: Use in paragraphs
   202→   202→   919→children.push(new Paragraph({ spacing: { before: spacing.topSpacing } }));
   203→   203→   920→// ... title paragraphs using titleLines and titleSize ...
   204→   204→   921→children.push(new Paragraph({ spacing: { before: spacing.bottomSpacing } }));
   205→   205→   922→```

   209→   209→   926→### ⚠️ CRITICAL — Cover Section Non-Negotiables (ALL Recipes)
   210→   210→   927→
   211→   211→   928→These 3 properties are MANDATORY for every cover implementation (R1–R7). Omitting ANY of them causes cover layout failure:
   212→   212→   929→
   213→   213→   930→1. **Cover section margin = 0**: The cover MUST be in its own section with `page.margin: { top: 0, bottom: 0, left: 0, right: 0 }`. Non-zero margins shrink the wrapper away from page edges → white gaps around the cover ("cover not filling the page"). This is the #1 cause of broken cover layouts.
   214→   214→   931→
   215→   215→   932→2. **Wrapper row exact height**: The outer wrapper table row MUST set `height: { value: 16838, rule: "exact" }`. Without this, content overflow pushes to page 2, or insufficient content leaves bottom whitespace.
   216→   216→   933→
   217→   217→   934→3. **Wrapper table borders = allNoBorders**: MUST explicitly set `borders: allNoBorders`. Default docx-js borders add ~8 twips per edge. MS Office includes border thickness in exact-height calculation → total exceeds 16838 → blank page 2 (WPS is lenient, MS Office is strict).
   218→   218→   935→
   219→   219→   936→**Cover section template (copy this for every Recipe):**
   220→   220→   937→```js
   221→   221→   938→sections: [
   222→   222→   939→  {

   224→   224→   941→    properties: {
   225→   225→   942→      page: {
   226→   226→   943→        size: { width: 11906, height: 16838 },
   227→   227→   944→        margin: { top: 0, bottom: 0, left: 0, right: 0 },
   228→   228→   945→      },
   229→   229→   946→    },
   230→   230→   947→    children: buildCoverRX(config), // ← replace with actual recipe function

   236→   236→   953→      page: {
   237→   237→   954→        size: { width: 11906, height: 16838 },
   238→   238→   955→        margin: { top: 1440, bottom: 1440, left: 1701, right: 1417 },
   239→   239→   956→      },

   248→   248→   965→### Recipe R1: Pure Paragraph Cover (Left-Aligned)
   249→   249→   966→
   250→   250→   967→**Visual:** Full-page dark background + left-aligned text + decoration via paragraph borders only
   251→   251→   968→**Use case:** Annual report, business report, tech proposal (most versatile premium recipe)
   252→   252→   969→**Nested tables: ZERO** — all decoration uses paragraph borders (bottom line, left accent bar, top separator)
   253→   253→   970→
   254→   254→   971→Visual hierarchy (top to bottom):
   255→   255→   972→1. Dynamic top whitespace (via `calcCoverSpacing`)

   728→   728→  1445→### Recipe R5: Clean White (Academic)
   729→   729→  1446→
   730→   730→  1447→**Visual:** Pure white background + school name + centered title + 2-column meta info table with underlines + footer
   731→   731→  1448→**Use case:** academic thesis, standards documents
   732→   732→  1449→**Architecture:** 16838 outer wrapper (white fill, invisible) + cell margins for page margin simulation. No top/bottom decorative lines.
   733→   733→  1450→
   734→   734→  1451→**Meta info table rules (cross-engine safe):**
   735→   735→  1452→- 2-column table: label + value, **percentage widths only** (`WidthType.PERCENTAGE`)
   736→   736→  1453→- **Table width is adaptive:** 55–75% of page, calculated by `calcR5MetaLayout()`. Table centered via `alignment: CENTER`.
   737→   737→  1454→- **Label column:** adaptive 25–45% of table, **LEFT aligned**, plain text with "：" appended. NO full-width space padding, NO right-alignment.
   738→   738→  1455→- **Label column borders:** none (no bottom border on label cells).
   739→   739→  1456→- **Value column:** remaining %, **LEFT aligned**, `bottom: single sz=4` border = fixed-length underline (consistent length for all rows regardless of text).
   740→   740→  1457→- No left/right/top borders on either column.
   741→   741→  1458→- ⚠️ Do NOT use DXA widths, full-width space padding (`\u3000`), spacer columns, or tab stops — WPS renders them inconsistently.
   742→   742→  1459→- ⚠️ Do NOT use `margins.top` on the wrapper cell — use `spacing.before` on first paragraph instead.
   743→   743→  1460→
   744→   744→  1461→**Known limitation:** When meta lines ≥ 6 AND title has 3 lines, MS Office may render content slightly taller than WPS, potentially clipping the footer line. Mitigate by reducing `midSpacing` or using a smaller title font.
   745→   745→  1462→
   746→   746→  1463→```js
   747→   747→  1464→// ── Width-aware title layout (handles mixed Chinese + English) ──
   748→   748→  1465→function estimateTextWidth(text, pt) {
   749→   749→  1466→  let width = 0;
   750→   750→  1467→  for (const ch of text) {
   751→   751→  1468→    const code = ch.codePointAt(0);
   752→   752→  1469→    const isCJK = (code >= 0x4E00 && code <= 0x9FFF) || (code >= 0x3400 && code <= 0x4DBF) ||
   753→   753→  1470→      (code >= 0x3000 && code <= 0x303F) || (code >= 0xFF00 && code <= 0xFFEF) ||
   754→   754→  1471→      (code >= 0x2E80 && code <= 0x2EFF);
   755→   755→  1472→    width += isCJK ? pt * 20 : pt * 11; // CJK: full-width, Latin: ~55% width
   756→   756→  1473→  }
   757→   757→  1474→  return width;
   758→   758→  1475→}
   759→   759→  1476→// Use estimateTextWidth() in calcTitleLayout() instead of simple char count
   760→   760→  1477→// to prevent mid-word breaks in mixed Chinese+English titles like "基于Transformer架构的..."
   761→   761→  1478→
   762→   762→  1479→// ── Meta info table ──
   763→   763→  1480→
   764→   764→  1481→// Calculate adaptive table and label column percentage based on longest label.
   765→   765→  1482→// Returns { tablePct, labelPct } — both as percentages.
   766→   766→  1483→// Uses ONLY WidthType.PERCENTAGE for cross-engine compatibility (MS Office + WPS).
   767→   767→  1484→function calcR5MetaLayout(metaEntries, fontPt = 12) {
   768→   768→  1485→  const maxLabelLen = Math.max(...metaEntries.map(e => [...e.label].length));
   769→   769→  1486→  // Label needs: maxLabelLen chars + "：" + 1 char padding
   770→   770→  1487→  const labelNeedTw = (maxLabelLen + 2) * fontPt * 20;
   771→   771→  1488→  // Value column: fixed ~5000tw for consistent underline length
   772→   772→  1489→  const valueNeedTw = 5000;
   773→   773→  1490→  const totalNeedTw = labelNeedTw + valueNeedTw;
   774→   774→  1491→  // Table width as % of page (11906tw), clamped to 55–75%
   775→   775→  1492→  const tablePct = Math.min(75, Math.max(55, Math.ceil(totalNeedTw / 11906 * 100)));
   776→   776→  1493→  // Label % within the table, clamped to 25–45%
   777→   777→  1494→  const rawLabelPct = Math.ceil(labelNeedTw / (tablePct / 100 * 11906) * 100);
   778→   778→  1495→  return { tablePct, labelPct: Math.max(25, Math.min(45, rawLabelPct)) };
   779→   779→  1496→}
   780→   780→  1497→
   781→   781→  1498→// Build R5 academic cover meta info table.
   782→   782→  1499→// ⚠️ CRITICAL cross-engine rules:
   783→   783→  1500→//   - Table width: WidthType.PERCENTAGE (NOT DXA — WPS breaks with DXA)
   784→   784→  1501→//   - Column widths: WidthType.PERCENTAGE
   785→   785→  1502→//   - Label column: LEFT aligned, plain text (NO full-width space padding)
   786→   786→  1503→//   - Value column: LEFT aligned, bottom border = fixed-length underline
   787→   787→  1504→//   - Table alignment: CENTER (visually centered on page)
   788→   788→  1505→function buildR5MetaTable(metaEntries) {
   789→   789→  1506→  // metaEntries: [{ label: "学院", value: "计算机科学与技术学院" }, ...]
   790→   790→  1507→  const { tablePct, labelPct } = calcR5MetaLayout(metaEntries);
   791→   791→  1508→  const valuePct = 100 - labelPct;
   792→   792→  1509→  const bottomBorder = { style: BorderStyle.SINGLE, size: 4, color: "000000" };
   793→   793→  1510→
   794→   794→  1511→  const rows = metaEntries.map(entry => new TableRow({
   795→   795→  1512→    children: [
   796→   796→  1513→      // Label cell: left-aligned, no bottom border
   797→   797→  1514→      new TableCell({
   798→   798→  1515→        width: { size: labelPct, type: WidthType.PERCENTAGE },
   799→   799→  1516→        borders: noBorders,
   800→   800→  1517→        margins: { top: 60, bottom: 60, left: 0, right: 0 },
   801→   801→  1518→        children: [new Paragraph({
   802→   802→  1519→          alignment: AlignmentType.LEFT,
   803→   803→  1520→          spacing: { before: 60, after: 60, line: 400 },
   804→   804→  1521→          children: [new TextRun({
   805→   805→  1522→            text: entry.label + "：",
   806→   806→  1523→            size: 24, font: { eastAsia: "SimSun", ascii: "Times New Roman" },
   807→   807→  1524→          })],
   808→   808→  1525→        })],
   809→   809→  1526→      }),
   810→   810→  1527→      // Value cell: left-aligned, bottom border = fixed-length underline
   811→   811→  1528→      new TableCell({
   812→   812→  1529→        width: { size: valuePct, type: WidthType.PERCENTAGE },
   813→   813→  1530→        borders: { top: NB, left: NB, right: NB, bottom: bottomBorder },
   814→   814→  1531→        margins: { top: 60, bottom: 60, left: 80, right: 0 },
   815→   815→  1532→        children: [new Paragraph({
   816→   816→  1533→          alignment: AlignmentType.LEFT,
   817→   817→  1534→          spacing: { before: 60, after: 60, line: 400 },
   818→   818→  1535→          children: [new TextRun({
   819→   819→  1536→            text: entry.value,
   820→   820→  1537→            size: 24, font: { eastAsia: "SimSun", ascii: "Times New Roman" },
   821→   821→  1538→          })],
   822→   822→  1539→        })],
   823→   823→  1540→      }),
   824→   824→  1541→    ],
   825→   825→  1542→  }));
   826→   826→  1543→
   827→   827→  1544→  return new Table({
   828→   828→  1545→    width: { size: tablePct, type: WidthType.PERCENTAGE },
   829→   829→  1546→    alignment: AlignmentType.CENTER,
   830→   830→  1547→    layout: TableLayoutType.FIXED,
   831→   831→  1548→    borders: allNoBorders,
   832→   832→  1549→    rows,
   833→   833→  1550→  });
   834→   834→  1551→}
   835→   835→  1552→
   836→   836→  1553→// ⚠️ MANDATORY: Cover section must use margin: 0. See "Cover Section Non-Negotiables" above.
   837→   837→  1554→function buildCoverR5(config) {
   838→   838→  1555→  const PAGE_H = 16838, SAFETY = 1200;
   839→   839→  1556→  const safeH = PAGE_H - SAFETY; // 15638
   840→   840→  1557→  const simMarginLR = 1701, simMarginT = 1200;
   841→   841→  1558→  const contentW = 11906 - simMarginLR * 2;
   842→   842→  1559→
   843→   843→  1560→  // ★ Width-aware title layout for mixed Chinese+English
   844→   844→  1561→  const { titlePt, titleLines } = calcTitleLayoutMixed(config.title, contentW, 36, 24);
   845→   845→  1562→  const titleSize = titlePt * 2;
   846→   846→  1563→
   847→   847→  1564→  // Parse meta entries
   848→   848→  1565→  const metaEntries = (config.metaLines || []).map(line => {
   849→   849→  1566→    const sep = line.indexOf("：") !== -1 ? "：" : ":";
   850→   850→  1567→    const idx = line.indexOf(sep);
   851→   851→  1568→    if (idx === -1) return { label: line, value: "" };
   852→   852→  1569→    return { label: line.slice(0, idx).trim(), value: line.slice(idx + sep.length).trim() };
   853→   853→  1570→  });
   854→   854→  1571→
   855→   855→  1572→  // Height budget (no margins.top — use spacing.before instead)
   856→   856→  1573→  const schoolNameH = config.schoolName ? (22 * 23 + 400) : 0;
   857→   857→  1574→  const titleTotalH = titleLines.length * (titlePt * 23 + 200);
   858→   858→  1575→  const subtitleH = config.subtitle ? (15 * 23 + 600) : 0;
   859→   859→  1576→  const metaRowH = 520; // 60+60 padding + ~400 line height
   860→   860→  1577→  const metaTableH = metaEntries.length * metaRowH;
   861→   861→  1578→  const footerH = config.footerRight ? (12 * 23 + 200) : 0;
   862→   862→  1579→  const spacerParas = 3 * 350;
   863→   863→  1580→  const fixedH = schoolNameH + titleTotalH + subtitleH + metaTableH + footerH + spacerParas;
   864→   864→  1581→  const remaining = Math.max(safeH - fixedH, 600);
   865→   865→  1582→
   866→   866→  1583→  // ★ topSpacing includes simulated top margin (simMarginT)
   867→   867→  1584→  const topSpacing = Math.min(Math.floor(remaining * 0.28) + simMarginT, 4200);
   868→   868→  1585→  const midSpacing = Math.min(Math.floor((remaining - simMarginT) * 0.18), 2000);
   869→   869→  1586→  const bottomSpacing = Math.min(remaining - topSpacing + simMarginT - midSpacing, 5500);
   870→   870→  1587→
   871→   871→  1588→  const children = [];
   872→   872→  1589→  children.push(new Paragraph({ spacing: { before: topSpacing } }));
   873→   873→  1590→
   874→   874→  1591→  // School name (optional)
   875→   875→  1592→  if (config.schoolName) {
   876→   876→  1593→    children.push(new Paragraph({
   877→   877→  1594→      alignment: AlignmentType.CENTER, spacing: { after: 400 },
   878→   878→  1595→      children: [new TextRun({ text: config.schoolName, size: 44, characterSpacing: 40,
   879→   879→  1596→        font: { eastAsia: "SimSun", ascii: "Times New Roman" } })],
   880→   880→  1597→    }));
   881→   881→  1598→  }
   882→   882→  1599→
   883→   883→  1600→  // Title
   884→   884→  1601→  for (let i = 0; i < titleLines.length; i++) {
   885→   885→  1602→    children.push(new Paragraph({
   886→   886→  1603→      alignment: AlignmentType.CENTER, spacing: { after: i < titleLines.length - 1 ? 120 : 300 },
   887→   887→  1604→      children: [new TextRun({ text: titleLines[i], size: titleSize, bold: true,
   888→   888→  1605→        font: { eastAsia: "SimHei", ascii: "Times New Roman" } })],
   889→   889→  1606→    }));
   890→   890→  1607→  }
   891→   891→  1608→
   892→   892→  1609→  // Subtitle
   893→   893→  1610→  if (config.subtitle) {
   894→   894→  1611→    children.push(new Paragraph({
   895→   895→  1612→      alignment: AlignmentType.CENTER, spacing: { after: 200 },
   896→   896→  1613→      children: [new TextRun({ text: config.subtitle, size: 30,
   897→   897→  1614→        font: { eastAsia: "SimSun", ascii: "Times New Roman" } })],
   898→   898→  1615→    }));
   899→   899→  1616→  }
   900→   900→  1617→
   901→   901→  1618→  children.push(new Paragraph({ spacing: { before: midSpacing } }));
   902→   902→  1619→
   903→   903→  1620→  // Meta info table
   904→   904→  1621→  if (metaEntries.length > 0) children.push(buildR5MetaTable(metaEntries));
   905→   905→  1622→
   906→   906→  1623→  children.push(new Paragraph({ spacing: { before: bottomSpacing } }));
   907→   907→  1624→
   908→   908→  1625→  // Footer
   909→   909→  1626→  if (config.footerRight) {
   910→   910→  1627→    children.push(new Paragraph({
   911→   911→  1628→      alignment: AlignmentType.CENTER,
   912→   912→  1629→      children: [new TextRun({ text: config.footerRight, size: 24, color: "404040",
   913→   913→  1630→        font: { eastAsia: "SimSun", ascii: "Times New Roman" } })],
   914→   914→  1631→    }));
   915→   915→  1632→  }
   916→   916→  1633→
   917→   917→  1634→  // ★ 16838 outer wrapper — only left/right margins, NO margins.top
   918→   918→  1635→  return [new Table({
   919→   919→  1636→    width: { size: 100, type: WidthType.PERCENTAGE },
   920→   920→  1637→    layout: TableLayoutType.FIXED,
   921→   921→  1638→    borders: allNoBorders,
   922→   922→  1639→    rows: [new TableRow({
   923→   923→  1640→      height: { value: PAGE_H, rule: "exact" },
   924→   924→  1641→      children: [new TableCell({
   925→   925→  1642→        shading: { type: ShadingType.CLEAR, fill: "FFFFFF" },
   926→   926→  1643→        borders: noBorders, verticalAlign: "top",
   927→   927→  1644→        margins: { left: simMarginLR, right: simMarginLR },
   928→   928→  1645→        children,
   929→   929→  1646→      })],
   930→   930→  1647→    })],
   931→   931→  1648→  })];
   932→   932→  1649→  // Height budget example (short title, 4 meta lines):
   933→   933→  1650→  // topSpacing(3818) + schoolName(906) + title(1028) + subtitle(945) + midSpacing(1467)
   934→   934→  1651→  // + metaTable(4×520=2080) + bottomSpacing(5268) + footer(476) = ~15988 < 15838 ✅
   935→   935→  1652→}
   936→   936→  1653→```

   940→   940→  1657→### blendColors Utility Function
   941→   941→  1658→
   942→   942→  1659→```js
   943→   943→  1660→function blendColors(hex1, hex2, ratio) {
   944→   944→  1661→  const p = (s, i) => parseInt(s.replace("#","").slice(i, i+2), 16);
   945→   945→  1662→  const mix = (c1, c2) => Math.round(c1 + (c2 - c1) * ratio);
   946→   946→  1663→  const r = mix(p(hex1,0), p(hex2,0)), g = mix(p(hex1,2), p(hex2,2)), b = mix(p(hex1,4), p(hex2,4));
   947→   947→  1664→  return [r, g, b].map(v => v.toString(16).padStart(2,"0")).join("");
   948→   948→  1665→}
   949→   949→  1666→```

   984→   984→  1701→
   985→   985→  1702→Default: **low-saturation (Morandi style)** palette to avoid flashy high-saturation. High-saturation palette only for explicitly energetic scenarios (events/education/creative).

   988→   988→  1705→const chartColors = {

   996→   996→  1713→// - report/whitepaper/consulting/academic/contract → default

  1000→  1000→  1717→**Color usage rules:**
  1001→  1001→  1718→- Max **5 colors** per chart; excess categories use depth variants of same hue
  1002→  1002→  1719→- Emphasis data uses document accent color; non-emphasis uses grey `#B0B0B0`
  1003→  1003→  1720→- Adjacent segments in pie/bar charts must have hue gap ≥ 60°
  1004→  1004→  1721→
  1005→  1005→  1722→1. **Anti-overlap**: If >6 x-axis labels, rotate 45° (`plt.xticks(rotation=45, ha='right')`)
  1006→  1006→  1723→2. **Anti-stretch**: Always set figure size explicitly (`fig, ax = plt.subplots(figsize=(10, 6))`)
  1007→  1007→  1724→3. **Aspect ratio (CRITICAL)**: When embedding in docx, MUST read actual image dimensions and calculate height proportionally. NEVER hardcode both width and height — pie charts become ellipses, radar charts become diamonds.
  1008→  1008→  1725→   ```js
  1009→  1009→  1726→   const sizeOf = require("image-size");
  1010→  1010→  1727→   const dims = sizeOf(chartBuffer);
  1011→  1011→  1728→   const displayWidth = 500;
  1012→  1012→  1729→   const displayHeight = Math.round(displayWidth * (dims.height / dims.width));
  1013→  1013→  1730→   // transformation: { width: displayWidth, height: displayHeight }
  1014→  1014→  1731→   ```
  1015→  1015→  1732→4. **DPI**: Save at 200+ DPI (`plt.savefig("chart.png", dpi=200, bbox_inches="tight")`)
  1016→  1016→  1733→5. **Colors**: Use palette accent color for primary data series
  1017→  1017→  1734→6. **Legend**: Place outside plot area if >4 series
  1018→  1018→  1735→7. **Square charts**: Pie and radar charts MUST use `figsize=(8, 8)` (equal width/height) to preserve circular/radial shape
  1019→  1019→  1736→7. **Grid**: Light gray grid (`ax.grid(True, alpha=0.3)`)
