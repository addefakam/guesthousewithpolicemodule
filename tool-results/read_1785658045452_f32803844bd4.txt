   452→/** Execute SQL via raw pg Client */
   453→async function execViaPg(sql: string, params?: unknown[]): Promise<void> {
   454→  const client = new Client({ connectionString: process.env.DATABASE_URL! });
   455→  try {
   456→    await client.connect();
   457→    await client.query(sql, params);
   458→  } finally {
   459→    await client.end().catch(() => {});
   460→  }
   461→}

   463→/** Execute DDL via Prisma $executeRawUnsafe (bypasses Prisma's table-existence check) */
   464→async function execViaPrisma(sql: string): Promise<void> {
   465→  const prisma = new PrismaClient({
   466→    log: ["warn", "error"],
   467→  });
   468→  try {
   469→    // $executeRawUnsafe sends SQL directly without schema validation
   470→    await prisma.$executeRawUnsafe(sql);
   471→  } finally {
   472→    await prisma.$disconnect().catch(() => {});
   473→  }
   474→}

   476→export async function ensureDatabase(): Promise<void> {
   477→  if (_initDone) return;
   478→  if (_initPromise) return _initPromise;
   479→
   480→  if (!process.env.DATABASE_URL) {
   481→    throw new Error("DATABASE_URL environment variable is not set.");
   482→  }
   483→
   484→  _initPromise = (async () => {
   485→    console.log("[init-db] Starting database initialization...");
   486→
   487→    let usePg = true;
   488→    let pgError: unknown = null;
   489→
   490→    // Strategy 1: Try using pg Client directly
   491→    try {
   492→      const client = new Client({ connectionString: process.env.DATABASE_URL });
   493→      await client.connect();
   494→      console.log("[init-db] pg Client connected successfully.");
   495→
   496→      // Check if User table already exists
   497→      const res = await client.query(
   498→        `SELECT EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema='public' AND table_name='User') as ok`
   499→      );
   500→      const tableExists = res.rows[0]?.ok === true;
   501→      console.log("[init-db] User table exists:", tableExists);
   502→
   503→      if (tableExists) {
   504→        console.log("[init-db] Tables already exist, skipping creation.");
   505→        await client.end().catch(() => {});
   506→        _initDone = true;
   507→        return;
   508→      }
   509→
   510→      // Create enums
   511→      console.log("[init-db] Creating enum types...");
   512→      await client.query(ENUMS_SQL);
   513→
   514→      // Create tables
   515→      console.log("[init-db] Creating tables...");
   516→      await client.query(TABLES_SQL);
   517→
   518→      // Verify User table was created
   519→      const verify = await client.query(
   520→        `SELECT EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema='public' AND table_name='User') as ok`
   521→      );
   522→      if (!verify.rows[0]?.ok) {
   523→        throw new Error("DDL executed but User table still not found!");
   524→      }
   525→      console.log("[init-db] Tables verified.");
   526→
   527→      // Create foreign keys
   528→      console.log("[init-db] Creating foreign keys...");
   529→      await client.query(FKS_SQL);
   530→
   531→      // Create indexes
   532→      console.log("[init-db] Creating indexes...");
   533→      await client.query(INDEXES_SQL);
   534→
   535→      // Seed SUPERUSER
   536→      console.log("[init-db] Seeding SUPERUSER...");
   537→      const hashed = await bcrypt.hash("Admin@2024", 12);
   538→      await client.query(
   539→        `INSERT INTO "User" ("id","username","password","name","role","permissions","policeRank","createdAt","updatedAt")
   540→         SELECT 'su-admin-001','admin',$1,'System Administrator','SUPERUSER','[]','',NOW(),NOW()
   541→         WHERE NOT EXISTS (SELECT 1 FROM "User" WHERE "username"='admin')`,
   542→        [hashed]
   543→      );
   544→
   545→      // Seed PoliceAlertConfig
   546→      await client.query(
   547→        `INSERT INTO "PoliceAlertConfig" ("id","createdAt","updatedAt")
   548→         SELECT 'default-alert-config',NOW(),NOW()
   549→         WHERE NOT EXISTS (SELECT 1 FROM "PoliceAlertConfig" WHERE "id"='default-alert-config')`
   550→      );
   551→
   552→      console.log("[init-db] Database initialization complete via pg.");
   553→      await client.end().catch(() => {});
   554→      _initDone = true;
   555→      return;
   556→    } catch (err) {
   557→      pgError = err;
   558→      console.error("[init-db] pg approach failed:", err instanceof Error ? err.message : String(err));
   559→      usePg = false;
   560→    }
   561→
   562→    // Strategy 2: Fallback — use Prisma $executeRawUnsafe
   563→    console.log("[init-db] Falling back to Prisma $executeRawUnsafe...");
   564→    try {
   565→      // First check if tables exist using Prisma raw query
   566→      const prisma = new PrismaClient({ log: ["warn", "error"] });
   567→      try {
   568→        const res: Array<{ ok: boolean }> = await prisma.$queryRawUnsafe(
   569→          `SELECT EXISTS (SELECT 1 FROM information_schema.tables WHERE table_schema='public' AND table_name='User') as ok`
   570→        );
   571→        if (res[0]?.ok) {
   572→          console.log("[init-db] Tables already exist (verified via Prisma).");
   573→          _initDone = true;
   574→          return;
   575→        }
   576→      } catch {
   577→        // Table might not exist, which is expected
   578→        console.log("[init-db] Tables do not exist yet, creating via Prisma...");
   579→      } finally {
   580→        await prisma.$disconnect().catch(() => {});
   581→      }
   582→
   583→      // Execute DDL via Prisma (split into parts to avoid query size issues)
   584→      await execViaPrisma(ENUMS_SQL);
   585→      console.log("[init-db] Enums created via Prisma.");
   586→
   587→      await execViaPrisma(TABLES_SQL);
   588→      console.log("[init-db] Tables created via Prisma.");
   589→
   590→      await execViaPrisma(FKS_SQL);
   591→      console.log("[init-db] Foreign keys created via Prisma.");
   592→
   593→      await execViaPrisma(INDEXES_SQL);
   594→      console.log("[init-db] Indexes created via Prisma.");
   595→
   596→      // Seed SUPERUSER via Prisma
   597→      const prisma2 = new PrismaClient({ log: ["warn", "error"] });
   598→      try {
   599→        const hashed = await bcrypt.hash("Admin@2024", 12);
   600→        await prisma2.$executeRawUnsafe(
   601→          `INSERT INTO "User" ("id","username","password","name","role","permissions","policeRank","createdAt","updatedAt")
   602→           SELECT 'su-admin-001','admin',$1,'System Administrator','SUPERUSER','[]','',NOW(),NOW()
   603→           WHERE NOT EXISTS (SELECT 1 FROM "User" WHERE "username"='admin')`,
   604→          hashed
   605→        );
   606→        await prisma2.$executeRawUnsafe(
   607→          `INSERT INTO "PoliceAlertConfig" ("id","createdAt","updatedAt")
   608→           SELECT 'default-alert-config',NOW(),NOW()
   609→           WHERE NOT EXISTS (SELECT 1 FROM "PoliceAlertConfig" WHERE "id"='default-alert-config')`
   610→        );
   611→        console.log("[init-db] SUPERUSER seeded via Prisma.");
   612→      } finally {
   613→        await prisma2.$disconnect().catch(() => {});
   614→      }
   615→
   616→      console.log("[init-db] Database initialization complete via Prisma fallback.");
   617→      _initDone = true;
   618→    } catch (prismaErr) {
   619→      console.error("[init-db] Prisma fallback also failed:", prismaErr instanceof Error ? prismaErr.message : String(prismaErr));
   620→      _initDone = false;
   621→      _initPromise = null; // Allow retry
   622→      throw new Error(
   623→        `Database initialization failed. pg error: ${pgError instanceof Error ? pgError.message : String(pgError)} | Prisma error: ${prismaErr instanceof Error ? prismaErr.message : String(prismaErr)}`
   624→      );
   625→    }
   626→  })();
   627→
   628→  return _initPromise;
   629→}
