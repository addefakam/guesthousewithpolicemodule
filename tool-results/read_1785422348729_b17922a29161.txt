    53→function buildYearOptions() {
    54→  const current = new Date().getFullYear();
    55→  const opts: { value: string; label: string }[] = [];
    56→  for (let y = current + 1; y >= current - 3; y--) {
    57→    opts.push({ value: String(y), label: String(y) });
    58→  }
    59→  return opts;
    60→}

    65→function getSeverity(p: { criticalCount: number; highCount: number; matchCount: number }) {
    66→  if (p.criticalCount > 0) return "CRITICAL";

    68→  if (p.matchCount > 0) return "MEDIUM";
    69→  return "NONE";
